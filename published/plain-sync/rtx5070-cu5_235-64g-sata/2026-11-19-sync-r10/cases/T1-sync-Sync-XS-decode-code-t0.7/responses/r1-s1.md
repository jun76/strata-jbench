word_freq.py — Count word occurrences in a text file and display the top N.

Usage:
    python word_freq.py --file <path> [--top N] [--case-sensitive]

Example:
    python word_freq.py --file sample.txt --top 10
"""

from __future__ import annotations

import argparse
import re
from collections import Counter
from pathlib import Path


def tokenize(text: str, *, case_sensitive: bool = False) -> list[str]:
    """
    Extract words (sequences of letters/digits/underscore) from text.

    Args:
        text: The input text string.
        case_sensitive: If False (default), words are lowercased.

    Returns:
        A list of tokens in the order they appear.
    """
    if not case_sensitive:
        text = text.lower()
    # Word = one or more alphanumeric or underscore characters
    pattern = r"[a-zA-Z0-9_]+"
    return re.findall(pattern, text)


def count_words(text: str, *, case_sensitive: bool = False) -> Counter[str]:
    """
    Count occurrences of each word in the given text.

    Args:
        text: Input text string.
        case_sensitive: If False (default), words are lowercased before counting.

    Returns:
        A collections.Counter mapping word -> count.
    """
    tokens = tokenize(text, case_sensitive=case_sensitive)
    return Counter(tokens)
