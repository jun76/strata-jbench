def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
                        False の場合、すべて小文字に変換する。

    Returns:
        単語のリスト（正規表現で抽出した英数字・日本語文字列）。

    Raises:
        ValueError: text が str でない場合。
    """
    if not isinstance(text, str):
        raise ValueError(f"text must be str, got {type(text).__name__}")

    # 英数字、ハイフン、アポストロフィ、日本語文字（ひらがな・カタカナ・漢字）を許容
    pattern = r"[a-zA-Z0-9'\-]+|[\u3040-\u30ff\u4e00-\u9fff]+"
    tokens = re.findall(pattern, text)

    if not case_sensitive:
        tokens = [t.lower() for t in tokens]

    return tokens
