def read_text(filepath: str) -> str:
    """Read text from a file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Split text into words, removing punctuation and converting to lowercase.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words with punctuation stripped.
    """
    words = []
    for word in text.split():
        # Strip common punctuation from the beginning and end of each word
        cleaned = word.strip(".,;:!?\"'()[]{}<>")
        if cleaned:
            words.append(cleaned.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to retrieve.

    Returns:
        A list of (word, count) tuples for the top N words.
    """
    return counter.most_common(n)
