"""
Text word frequency counter - command line tool.

Reads a text file, counts word occurrences (case-insensitive),
and displays the top N most frequent words in a table format.
"""

import argparse
import sys
import re
from pathlib import Path
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    Count word occurrences in the given text.
    
    Words are extracted using regex matching sequences of alphanumeric
    characters and underscores. Counting is case-insensitive.
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter object with word frequencies.
    """
    # Extract words: sequences of letters, digits, or underscores
    words = re.findall(r'\b\w+\b', text.lower())
    return Counter(words)


def get_top_words(counter: Counter, top_n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from a Counter.
    
    Args:
        counter: A Counter object containing word frequencies.
        top_n: Number of top words to return.
        
    Returns:
        A list of tuples (word, count), sorted by count descending,
        then by word alphabetically for ties.
    """
    # most_common returns sorted by count desc, but for ties we want alphabetical
    items = list(counter.items())
    # Sort by count descending, then word alphabetically
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:top_n]
