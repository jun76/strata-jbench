import argparse

def count_words(text: str) -> Counter:
    """Count word occurrences in text.
    
    Words are defined as sequences of alphanumeric characters and underscores,
    split by any other character. Counting is case-insensitive.
    
    Args:
        text: The input text to analyze.
    
    Returns:
        A Counter object containing word frequencies.
    """
    import re
    # Extract words: sequences of word characters (letters, digits, underscores)
    words = re.findall(r'\w+', text.lower())
    return Counter(words)


def read_file(filepath: Path) -> str:
    """Read text content from a file.
    
    Args:
        filepath: Path to the text file.
    
    Returns:
        The text content of the file.
    
    Raises:
        FileNotFoundError: If the file does not exist.
        UnicodeDecodeError: If the file cannot be decoded.
    """
    return filepath.read_text(encoding='utf-8')


def get_top_words(counter: Counter, top_n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        top_n: Number of top words to return.
    
    Returns:
        A list of tuples (word, count) sorted by descending frequency.
        For ties, words are sorted alphabetically.
    """
    # Sort by count descending, then by word ascending for tie-breaking
    items = counter.items()
    sorted_items = sorted(items, key=lambda item: (-item[1], item[0]))
    return sorted_items[:top_n]
