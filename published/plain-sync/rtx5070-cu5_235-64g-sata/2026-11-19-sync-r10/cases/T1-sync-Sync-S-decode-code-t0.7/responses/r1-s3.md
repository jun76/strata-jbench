以下は、テキストファイルから単語の出現頻度を数え、上位 N 件を表形式で表示する CLI ツールです。

テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

使い方:
    python word_freq.py input.txt
    python word_freq.py input.txt -n 10 --sort-by count
    python word_freq.py input.txt -n 5 --output-format csv
"""

@dataclass(frozen=True)
class WordEntry:
    """単語とその出現回数、および相対頻度（パーセンテージ）を表す不変オブジェクト。"""

    word: str
    count: int
    percentage: float = field(init=False)

    def __post_init__(self) -> None:
        # percentage は count から自動計算されるため、init=False として除外
        # ただし frozen dataclass では __post_init__ でフィールド更新ができないため、
        # percentage は外部で計算して渡す設計にする。
        pass

    # percentage は __post_init__ で設定できないので、factory メソッドで生成する


def make_word_entry(word: str, count: int, total_words: int) -> WordEntry:
    """WordEntry を生成し、percentage を計算して付与する。"""
    if total_words == 0:
        percentage = 0.0
    else:
        percentage = (count / total_words) * 100.0
    return WordEntry(word=word, count=count, percentage=percentage)
