def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    英数字以外の文字で区切り、大文字小文字を区別しない。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 小文字化された単語のリスト。
    """
    # 英数字のみを単語として抽出
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words

    """

    Returns:
        Dict[str, int]: 単語をキー、出現回数を値とする辞書。
    """
    words = extract_words(text)
    return dict(Counter(words))
