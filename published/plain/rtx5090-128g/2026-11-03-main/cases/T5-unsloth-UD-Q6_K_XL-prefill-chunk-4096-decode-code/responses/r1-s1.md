#!/usr/bin/env python3
"""
wordfreq.py - テキストファイルから単語頻度を集計し、上位 N 件を表示するツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストに分割する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        小文字化された単語のリスト
    """
    # 英数字とハイフン/アポストロフィを含むトークンを抽出
    # 例: "don't" -> "don't", "hello-world" -> "hello-world"
    return re.findall(r"[a-zA-Z0-9][a-zA-Z0-9'\-]*", text.lower())


def count_word_frequencies(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語をキー、出現回数を値とする Counter
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度が高い順に上位 N 件の単語を返す。
    頻度が同じ場合は辞書順でソートされる。
    
    Args:
        counter: 単語頻度カウンタ
        n: 取得する件数
        
    Returns:
        (単語, 頻度) のタプルのリスト
    """
    # most_common は頻度順だが、同頻度時の順序は保証されないため
    # 明示的に (-count, word) でソートする
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
