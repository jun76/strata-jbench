    Args:
        text (str): 解析対象のテキスト。
        
    Returns:
        List[Tuple[str, int]]: 頻度の高い順に並んだ (単語, 回数) のリスト。

        if char.isalnum() or char == '-':

                words.append(''.join(current_word))
                current_word = []
    if current_word:

    # 頻度降順、同頻度時は辞書順でソート
    sorted_words = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_words
