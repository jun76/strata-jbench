This command-line tool reads one or more text files, tokenizes the content

def extract_words(text: str) -> List[str]:
    """
    Extract words from a given text string.
    
    Words are defined as sequences of alphanumeric characters (letters and digits).
    The extraction is case-insensitive: all words are converted to lowercase.
    
    Args:
        text: The input text string to tokenize.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    # Find all sequences of alphanumeric characters
    words = re.findall(r'[a-zA-Z0-9]+', text)
    # Convert to lowercase for case-insensitive counting
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word in a list of words.
    
    Args:
        words: A list of lowercase words.
        
    Returns:
        A Counter object mapping each word to its frequency count.
    """
    return Counter(words)
