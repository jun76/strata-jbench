"""
word_freq_counter.py

A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    Extract words from the given text.

    Words are sequences of alphanumeric characters (including underscores),
    converted to lowercase for case-insensitive counting.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Use regex to find all sequences of word characters
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return [word.lower() for word in words]


def count_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words to count.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their frequencies.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top items to return.

    Returns:
        A list of tuples (word, count) for the top N most common words.
    """
    return counter.most_common(n)
