from __future__ import annotations

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語リストに分割する。

    単語は英数字、ハイフン、アポストロフィを含む連続文字列として扱う。
    例: "don't" → ["don't"], "hello-world" → ["hello-world"]

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
                        False の場合、すべて小文字に変換する。

    Returns:
        単語のリスト。
    """
    pattern = r"[a-zA-Z0-9'\-]+"
    words = re.findall(pattern, text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words
