import argparse

from collections import Counter
from typing import List, Tuple

def tokenize(text: str) -> List[str]:
    """
    テキストを単語にトークン化します。
    
    Args:
        text (str): 入力テキスト。
        
    Returns:
        List[str]: 小文字化され、句読点が除去された単語のリスト。
    """
    # 句読点や特殊文字を空白に置き換え、正規化
    cleaned = "".join(char if char.isalnum() or char.isspace() else " " for char in text)
    return [word.lower() for word in cleaned.split() if word]

def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントします。
    
    Args:
        text (str): 入力テキスト。
        
    Returns:
        Counter: 単語とその出現回数をキー・値とするカウンター。
    """
    tokens = tokenize(text)
    return Counter(tokens)

def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度上位N件の単語と回数を取り出します。
    同じ回数を持つ単語は、辞書順（アルファベット順）でソートされます。
    
    Args:
        counter (Counter): 単語頻度カウンター。
        n (int): 上位取得件数。
        
    Returns:
        List[Tuple[str, int]]: (単語, 回数) のタプルのリスト。
    """
    # 頻度降順、同じ頻度なら単語の昇順でソート
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
