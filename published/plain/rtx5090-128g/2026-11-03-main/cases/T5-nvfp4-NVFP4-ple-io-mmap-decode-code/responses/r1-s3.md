def tokenize(text: str, case_insensitive: bool = True) -> List[str]:
    """テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、単語を小文字に正規化する。

    Returns:
        抽出された単語のリスト。
    """
    if case_insensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    words = re.findall(r"[a-z0-9]+(?:['-][a-z0-9]+)*", text)
    return words


def count_frequencies(words: List[str]) -> Counter:
    """単語リストから頻度カウントを集計する。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
