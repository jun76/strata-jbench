#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントして上位N件を表示するツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def normalize_words(text: str) -> List[str]:
    """
    テキストから単語を抽出し、正規化を行う。

    - 小文字に変換
    - 英数字で構成されるトークンのみを抽出
    """
    # \w+ は [a-zA-Z0-9_] に一致する。
    # 必要に応じて [a-zA-Z]+ などに変更可だが、ここでは一般的な英数字単語を想定
    return [word.lower() for word in re.findall(r'\b[a-zA-Z0-9]+\b', text)]


def count_words(words: List[str]) -> Counter:
    """
    単語リストの出現頻度をカウントする。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    カウンターから最も出現頻度の高いN個の単語と回数を取り出す。
    """
    return counter.most_common(n)


def format_table(data: List[Tuple[str, int]]) -> str:
    """
    データを簡易なテキストテーブル形式に整形する。
    """
    if not data:
        return "No words found."

    # 列の最大長を計算
    max_word_len = max(len(word) for word, _ in data)
    max_count_len = max(len(str(count)) for _, count in data)

    # ヘッダー
    header = f"{'Word':<{max_word_len}} {'Count':>{max_count_len}}"
    separator = f"{'-' * max_word_len} {'-' * max_count_len}"

    lines = [header, separator]
    for word, count in data:
        line = f"{word:<{max_word_len}} {count:>{max_count_len}}"
