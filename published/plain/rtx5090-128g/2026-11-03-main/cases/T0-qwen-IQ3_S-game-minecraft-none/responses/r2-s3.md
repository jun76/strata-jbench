function genWorld(seed=1337){
  const r=mulberry32(seed);
  for(let x=0;x<SIZE;x++)for(let z=0;z<SIZE;z++){
    let h=4,amp=1,freq=0.05;
    for(let o=0;o<4;o++){h+=noise2(x*freq+seed,z*freq+seed)*amp*10;freq*=2;amp*=0.5;}
    const hill=Math.max(0,1-Math.hypot(x-SIZE*0.62,z-SIZE*0.38)/15);
    h+=hill*hill*13;
    h=Math.round(h);
    const top=clamp(h,1,HEIGHT-8);
    for(let y=0;y<=top;y++){
      let t=B.STONE;
      if(y===top) t = top<=SEA+1?B.SAND:B.GRASS;
      else if(y>=top-3) t=B.DIRT;
      setB(x,y,z,t);
    }
  }
  // trees
  let placed=0,tries=0;
  while(placed<20&&tries<400){
    tries++;
    const x=3+((r()*(SIZE-6))|0), z=3+((r()*(SIZE-6))|0);
    let top=-1;for(let y=HEIGHT-1;y>=0;y--)if(getB(x,y,z)){top=y;break;}
    if(top<0)continue;
    if(getB(x,top,z)!==B.GRASS)continue;
    const th=4+((r()*3)|0);
    for(let i=1;i<=th;i++)setB(x,top+i,z,B.LOG);
    const cy=top+th;
    for(let dy=-1;dy<=2;dy++)for(let dx=-2;dx<=2;dx++)for(let dz=-2;dz<=2;dz++){
      if(dx===0&&dz===0&&dy<=0)continue;
      if(Math.abs(dx)+Math.abs(dz)+Math.abs(dy)>4)continue;
      if(r()<0.18)continue;
      const X=x+dx,Y=cy+dy,Z=z+dz;
      if(inB(X,Y,Z)&&getB(X,Y,Z)===0)setB(X,Y,Z,B.LEAVES);
    }
    placed++;
  }
  // build all chunks
  for(let cx=0;cx<SIZE/8;cx++)for(let cy=0;cy<Math.ceil(HEIGHT/8);cy++)for(let cz=0;cz<SIZE/8;cz++)chunkMesh(cx,cy,cz);
}

function raycast(maxd=6.5){

  let tx=(sx>0?x+1-ox:ox-x)*ddx, ty=(sy>0?y+1-oy:oy-y)*ddy, tz=(sz>0?z+1-oz:oz-z)*ddz;
  let face=null,steps=0;
  while(steps++<120){
    if(getB(x,y,z))return{x,y,z,face};

    else {z+=sz;tz+=ddz;face=[0,0,-sz];}

  }
  return null;
}

## 実装のポイント

- 16×16 の**手描き風ピクセルテクスチャ**をコードで生成し、チャンクごとに**面カリング**（隣接面を消す）して描画。ブロックを置くとその場だけチャンクを再生成します。
- 丘・木・砂浜のある**プロシージャル地形**（Perlin風ノイズ）。掘ったブロックは素材としてインベントリに集まり、足場を組んで丘をのぼれます。
- 空のグラデーションシェーダー、漂う雲、ブロック面別のシェーディング、HUDのスキャンライン／グリッド／vignette、isometric ブロックアイコンのホットバー、スクロール表示のタイトルとロード用ローテキューブで、マイクラの雰囲気を出しつつUIは「箱庭のダッシュボード」風に仕上げています。
