function buildChunkMesh(cx,cz){
  const x0=cx*chunkSize,z0=cz*chunkSize;
  const buckets=new Map(); // matKey -> {pos,nor,uv,ind,mat}
  function bucket(mat){if(!buckets.has(mat)){buckets.set(mat,{p:[],n:[],u:[],i:[],vc:0,mat});}return buckets.get(mat);}
  function addFace(id,f,rot,x,y,z){
    const tex=faceTex(id,f);
    // choose material: for grass we want top=grass tex, side=grass_side, bottom=dirt. Use dedicated mats.
    let mat;
    if(id===GRASS){mat = f===2?newMatGrassTop:(f===3?newMatGrassBot:newMatGrassSide);}
    else mat=materials[id];
    const B=bucket(mat);
    for(let k=0;k<4;k++){const p=new THREE.Vector3(...FACES[f][k]);p.applyEuler(rot);
      const n=faceNormal[f];B.p.push(x+p.x,y+p.y,z+p.z);B.n.push(n.x,n.y,n.z);}
    for(let k=0;k<4;k++){const p=new THREE.Vector3(...FACES[f][k]);p.applyEuler(rot);
      let u,v;if(f===2||f===3){u=p.x+.5;v=p.z+.5;}else{u=(f===0||f===1)?p.z+.5:p.x+.5;v=p.y+.5;}B.u.push(u,v);}
    for(const k of QUAD)B.i.push(B.vc+k);B.vc+=4;
  }
  for(let x=x0;x<x0+chunkSize;x++)for(let z=z0;z<z0+chunkSize;z++){
    for(let y=1;y<WORLD.maxH+1;y++){
      const id=getB(x,y,z);if(!id)continue;const b=BLOCKS[id];if(b.liquid)continue;
      for(let f=0;f<6;f++){const d=faceDefs[f].dir,n=getB(x+d[0],y+d[1],z+d[2]);
        if(n===id)continue;if(n&&!BLOCKS[n].transparent)continue;addFace(id,f,faceRot[f],x,y,z);}
    }
  }
  const group=new THREE.Group();
  for(const {p,n,u,i,mat} of buckets.values()){
    const g=new THREE.BufferGeometry();
    g.setAttribute('position',new THREE.Float32BufferAttribute(p,3));
    g.setAttribute('normal',new THREE.Float32BufferAttribute(n,3));

    g.setIndex(i);g.computeBoundingSphere();
    const m=new THREE.Mesh(g,mat);m.castShadow=true;m.receiveShadow=true;group.add(m);
  }
  return group;
}

    const hit=[Math.floor(lx-nx.x*0.001+0.5),Math.floor(ly-nx.y*0.001+0.5),Math.floor(lz-nx.z*0.001+0.5)];
    const [a,b]=aabb(hit[0],hit[1],hit[2]);
    let best=null,bt=Infinity;
    const steps=24;const p0=camera.position.clone();
    const dirv=new THREE.Vector3(h.point.x-p0.x,h.point.y-p0.y,h.point.z-p0.z).normalize();
    for(let i=0;i<=steps;i++){const p=p0.clone().addScaledVector(dirv,i*max/steps);
      if(p.x>=a[0]&&p.x<b[0]&&p.y>=a[1]&&p.y<b[1]&&p.z>=a[2]&&p.z<b[2]){const t=p.distanceTo(p0);if(t<bt){bt=t;best=p;}break;}
    }
    if(!best)continue;
    const cell=[Math.floor(best.x),Math.floor(best.y),Math.floor(best.z)];
    const id=getB(cell[0],cell[1],cell[2]);
    if(!id||(BLOCKS[id]&&BLOCKS[id].liquid))continue;

    const place=[Math.floor(best.x+n.x*0.5),Math.floor(best.y+n.y*0.5),Math.floor(best.z+n.z*0.5)];
    return {cell,normal:n,place};
  }
  return null;
}
