"""Text frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words
in a formatted table.

Usage:
    python word_freq.py <file> [--top N] [--case-sensitive]
"""

import argparse

from collections import Counter
from typing import List, Tuple


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """Count word occurrences in the given text.

    Args:
        text: Input text string.
        case_sensitive: If True, words are case-sensitive. If False, all words
                        are lowercased before counting.

    Returns:
        A Counter mapping each word to its occurrence count.
    """
    if not case_sensitive:
        text = text.lower()
    # Split on whitespace and strip punctuation-like characters
    import re
    words = re.findall(r'\b[a-zA-Z0-9_\-\']+\b', text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words from a Counter.

    Args:
        counter: A Counter mapping words to counts.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples sorted by count descending,
        then by word ascending for ties.
    """
    items = counter.most_common()
    # Sort by count descending, then by word ascending for stable ordering
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
