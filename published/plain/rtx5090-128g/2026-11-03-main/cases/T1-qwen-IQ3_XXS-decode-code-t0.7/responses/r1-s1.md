#!/usr/bin/env python3
"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度を集計し、
上位 N 件を表形式で出力するコマンドラインツール。

使用例:
    python word_freq.py input.txt --top 10
    python word_freq.py input.txt --top 5 --case-sensitive
"""

import argparse

def read_text_file(path: Path) -> str:
    """
    指定されたパスのテキストファイルを読み込み、内容を文字列として返す。

    Args:
        path: 読み込むテキストファイルの Path オブジェクト。

    Returns:
        ファイルの内容を表す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        OSError: 読み取りエラーが発生した場合。
    """
    return path.read_text(encoding="utf-8")


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語リストを抽出する。

    単語は英数字とハイフンを含む文字列として定義される。
    句読点は単語から除外される。

    Args:
        text: トークン化対象の文字列。
        case_sensitive: True の場合、大文字・小文字を区別する。
                        False の場合、全て小文字に変換する。

    Returns:
        単語のリスト。
    """
    import re

    # 単語パターン: 英数字、ハイフン、アポストロフィを含む
    pattern = r"[a-zA-Z0-9]+(?:['-][a-zA-Z0-9]+)*"
    words = re.findall(pattern, text)

    if not case_sensitive:
        words = [w.lower() for w in words]

    return words
