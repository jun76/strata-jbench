word_freq.py: Count word frequencies in a text file and display the top N words.

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive] [--min-length L]

def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Extract words from the given text.

    Args:
        text: The input text string.
        case_sensitive: If True, preserve case; otherwise convert to lowercase.
        min_length: Minimum length of a word to include.

    Returns:
        A list of words extracted from the text.
    """
    if not case_sensitive:
        text = text.lower()

    # Use regex to extract word-like tokens (letters and digits, allowing apostrophes)
    words = re.findall(r"[a-zA-Z0-9']+", text)
    words = [w for w in words if len(w) >= min_length]
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top entries to return.

    Returns:
        A list of (word, frequency) tuples, sorted by frequency descending.
    """
    return counter.most_common(n)
