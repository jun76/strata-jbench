#!/usr/bin/env python3
"""
単語出現頻度集計コマンドラインツール

テキストファイルを読み込み、単語の出現頻度を計算し、上位N件を表形式で表示します。

Usage:
    python word_freq.py input.txt
    python word_freq.py input.txt -n 10
    cat input.txt | python word_freq.py - -n 5
"""

import argparse

import re
from collections import Counter

def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。

    正規表現を用いて、英字・数字・ハイフン・アポストロフィを単語として認識します。
    単語は小文字に正規化されます。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 抽出された単語のリスト（小文字）。
    """
    # 英字・数字・ハイフン・アポストロフィを含む単語パターン
    # 例: "don't", "well-known", "hello123"
    pattern = re.compile(r"[a-zA-Z0-9'\-]+")
    words = pattern.findall(text)
    return [word.lower() for word in words]


def count_frequencies(text: str) -> Counter[str]:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text (str): 入力テキスト。

    Returns:
        Counter[str]: 単語とその出現回数の辞書（Counter）。
    """
    words = tokenize(text)
    return Counter(words)
