"""
word_freq.py - Count word occurrences in text files and display top N results.

This command-line tool reads one or more text files, counts the frequency of
each word (case-insensitive, after normalizing by lowercasing and stripping
punctuation), and displays the top N most frequent words in a table format.

Usage:
    python word_freq.py [options] FILE1 [FILE2 ...]

Examples:
    python word_freq.py sample.txt
    python word_freq.py -n 5 sample.txt
    python word_freq.py --sort-order ascending -n 10 file1.txt file2.txt
"""

import argparse
import re

from collections import Counter
from dataclasses import dataclass
from typing import List, Optional, TextIO


@dataclass
class WordFrequency:
    """Represents a word and its frequency."""
    word: str
    count: int


def normalize_word(word: str) -> str:
    """
    Normalize a word by converting to lowercase and stripping non-alphabetic characters.
    
    Args:
        word: The raw word to normalize.
        
    Returns:
        The normalized word (lowercase, with non-alphabetic chars stripped).
    """
    return re.sub(r'[^a-z0-9]', '', word.lower())


def extract_words(text: str) -> List[str]:
    """
    Extract words from text by splitting on whitespace and normalizing.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of normalized words (non-empty ones only).
    """
    raw_words = text.split()
    normalized = [normalize_word(w) for w in raw_words]
    return [w for w in normalized if w]


def count_words(words: List[str]) -> Counter[str]:
    """
    Count the frequency of each word.
    
    Args:
        words: A list of words.
        
    Returns:
        A Counter mapping words to their counts.
    """
    return Counter(words)
