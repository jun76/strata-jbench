#!/usr/bin/env python3
"""
text_frequency.py

コマンドラインツール: テキストファイルを読み込んで単語の出現頻度を集計し、
上位 N 件を表形式で表示する。

Usage:
    python text_frequency.py <file> [--top N] [--case-sensitive] [--min-length L]
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple, Optional
import re


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。False の場合、全て小文字に変換する。
        min_length: 抽出する単語の最小文字数。これ未満の単語は除外される。

    Returns:
        抽出された単語トークンのリスト。
    """
    if not case_sensitive:
        text = text.lower()

    # 英数字とハイフン、アポストロフィを含むトークンを抽出
    tokens = re.findall(r"[a-zA-Z0-9]+(?:[\\-'][a-zA-Z0-9]+)*", text)

    # 最小長でフィルタリング
    if min_length > 1:
        tokens = [t for t in tokens if len(t) >= min_length]

    return tokens


def count_frequencies(tokens: List[str]) -> Counter:
    """
    トークンリストから単語出現頻度をカウントする。

    Args:
        tokens: 単語トークンのリスト。

    Returns:
        単語からカウントへの Counter。
    """
    return Counter(tokens)
