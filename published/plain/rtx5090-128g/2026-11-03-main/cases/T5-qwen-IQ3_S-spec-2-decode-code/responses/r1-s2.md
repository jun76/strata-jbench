#!/usr/bin/env python3
"""
word_freq.py - A command-line tool to count word frequencies in text files.

This tool reads one or more text files, counts the occurrence of each word,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """Extract words from text, converting to lowercase and keeping only alphanumeric characters.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    # Match sequences of alphanumeric characters (including underscores as part of words if desired,
    # but typically we want simple word boundaries). Using \b\w+\b to capture word-like tokens.
    # \w includes letters, digits, and underscores. We'll strip non-alphanumeric edges by using \b.
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text.lower())
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """Count the frequency of each word in the list.
    
    Args:
        words: A list of words.
        
    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(frequencies: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words.
    
    Args:
        frequencies: A Counter object with word frequencies.
        n: The number of top words to retrieve.
        
    Returns:
        A list of (word, count) tuples, sorted by count descending, then word ascending.
    """
    # Counter.most_common returns elements in sorted order by count descending.
    # For ties, Counter.most_common does not guarantee a secondary sort, so we sort manually.
    # We want descending count, then ascending word for deterministic output.
    items = list(frequencies.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
