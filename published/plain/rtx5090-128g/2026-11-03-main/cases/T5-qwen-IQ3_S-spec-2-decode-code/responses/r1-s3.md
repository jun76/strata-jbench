#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルから単語の出現頻度を集計し、
上位 N 件を表形式で表示します。

import argparse

from collections import Counter
from typing import List, Tuple, Dict


def normalize_word(word: str) -> str:
    """
    単語を正規化します。
    - 小文字に変換
    - 前後のホワイトスペースを除去
    - 非文字数字（句読点など）を除去
    
    Args:
        word: 単語文字列
        
    Returns:
        正規化された単語文字列。空文字列の場合は None を返す。
    """
    # 句読点や特殊文字を除き、英数字のみを残す
    cleaned = re.sub(r'[^a-zA-Z0-9]', '', word)
    if not cleaned:
        return None
    return cleaned.lower()


def count_words(text: str) -> Dict[str, int]:
    """
    テキストから単語の出現回数をカウントします。
    
    Args:
        text: 解析対象のテキスト
        
    Returns:
        {単語: 出現回数} の辞書
    """
    # 空白で分割し、正規化を適用
    raw_words = text.split()
    normalized_words = [normalize_word(w) for w in raw_words]
    # None (空文字列) をフィルタリング
    valid_words = [w for w in normalized_words if w]
    
    return dict(Counter(valid_words))
