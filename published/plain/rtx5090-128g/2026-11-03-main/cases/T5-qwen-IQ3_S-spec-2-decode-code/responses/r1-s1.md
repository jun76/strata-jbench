#!/usr/bin/env python3
"""
wordfreq.py - A command-line tool to count word frequencies in a text file
and display the top N most frequent words in a table format.

Usage:
    python wordfreq.py <input_file> [--top N] [--case-sensitive]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def read_text_file(filepath: str) -> str:
    """Read and return the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize_text(text: str, case_sensitive: bool = False) -> List[str]:
    """Tokenize text into words, extracting only alphanumeric sequences.

    Args:
        text: The input text to tokenize.
        case_sensitive: If True, preserve case. If False, convert to lowercase.

    Returns:
        A list of word tokens.
    """
    if not case_sensitive:
        text = text.lower()
    # Extract sequences of alphanumeric characters (including underscores)
    words = re.findall(r"\b[a-zA-Z0-9_]+\b", text)
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: A list of word tokens.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words and their counts.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) for the top N words.
    """
    return counter.most_common(n)
