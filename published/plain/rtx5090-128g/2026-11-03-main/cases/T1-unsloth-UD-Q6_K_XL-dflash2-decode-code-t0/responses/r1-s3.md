#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位N件表示する。

使い方:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt --top 20 --case-sensitive
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。

    Returns:
        抽出された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 N 件を取得する。

    Args:
        counter: 単語頻度の Counter。
        n: 取得する件数。

    Returns:
        (単語, 出現回数) のタプルのリスト。頻度が高い順にソートされ、
        頻度が同じ場合は単語の辞書順でソートされる。
    """
    # most_common は頻度順に返すが、頻度が同じ場合は挿入順になるため、
    # 明示的にソートする
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
