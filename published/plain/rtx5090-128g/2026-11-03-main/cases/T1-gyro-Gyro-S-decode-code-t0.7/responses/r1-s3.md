def read_text_from_file(path: str) -> str:
    """指定されたパスのテキストファイルを読み込み、内容文字列を返す。

    Args:
        path: テキストファイルのパス。

    Returns:
        ファイルの内容文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み取りに失敗した場合。
    """
    try:
        with open(path, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {path}")
    except IOError as e:
        raise IOError(f"ファイルの読み取りに失敗しました: {path} ({e})")


def tokenize(text: str) -> List[str]:
    """テキストを単語リストに分割する。

    単語は空白区切りで、句読点や特殊文字は除去される。
    大文字/小文字は区別しない（小文字に正規化）。

    Args:
        text: 入力テキスト。

    Returns:
        単語のリスト（小文字に正規化済み）。
    """
    # 単語を抽出：アルファベット・数字・ハイフン・アポストロフィを保持し、
    # それ以外の文字を空白に置換する
    import re
    # 単語境界パターン: 英字、数字、ハイフン、アポストロフィを含む連続部分
    words = re.findall(r"[a-zA-Z0-9]+(?:[-']?[a-zA-Z0-9]+)*", text)
    # 小文字に正規化
    return [w.lower() for w in words]
