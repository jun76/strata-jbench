def tokenize(text: str, case_insensitive: bool = True) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_insensitive: True の場合、単語を小文字に変換する

    Returns:
        単語のリスト
    """
    # 簡易的なトークナイザ: 空白と句読点で分割
    # 正規表現不使用のシンプルな実装
    words = []
    current_word = []
    for char in text:
        if char.isalnum() or char == '_':
            current_word.append(char)
        else:
            if current_word:
                word = ''.join(current_word)
                if case_insensitive:
                    word = word.lower()
                words.append(word)
                current_word = []
    if current_word:
        word = ''.join(current_word)
        if case_insensitive:
            word = word.lower()
        words.append(word)
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数の Counter
    """
    return Counter(words)
