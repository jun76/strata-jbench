```python
"""
コマンドラインツール: テキストファイルを読み込み、単語出現頻度を上位N件表示
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。"""
    return text.split()


def count_word_frequencies(words: List[str]) -> Counter:
    """単語の出現頻度をカウントする。"""
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """上位n件の単語と頻度を取得する。"""
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """表形式で結果をフォーマットする。"""
    if not items:
        return "単語が見つかりません。"
    lines = []
    for word, count in items:
        lines.append(f"{word:10s} {count:6d}")
    return "\n".join(lines)


def main() -> None:
    """コマンドラインツールを起動する。"""
    parser = argparse.ArgumentParser(
        description="テキストファイルから単語出現頻度の上位N件を表示します"
    )
    parser.add_argument(
        "file",
        help="テキストファイルのパス"
    )
    parser.add_argument(
        "-n", "--top",
        type=int,
        default=10,
        help="表示する上位単語の数 (デフォルト: 10)"
    )
    parser.add_argument(
        "--ignore-case",
        action="store_true",
        help="大文字・小文字を区別しない"
    )
    
    args = parser.parse_args()
    
    try:
        with open(args.file, "r", encoding="utf-8") as f:
            text = f.read()
    except FileNotFoundError:
        print(f"エラー: ファイル '{args.file}' が見つかりません", file=sys.stderr)
        sys.exit(1)
    except UnicodeDecodeError:
        print(f"エラー: ファイル '{args.file}' をUTF-8として読み込めません", file=sys.stderr)
        sys.exit(1)
    
    words = tokenize(text)
