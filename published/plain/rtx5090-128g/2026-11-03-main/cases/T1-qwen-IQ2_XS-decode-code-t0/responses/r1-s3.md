#!/usr/bin/env python3
"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_file(filepath: Path) -> str:
    """Read the contents of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        PermissionError: If the file cannot be read.
    """
    return filepath.read_text(encoding="utf-8")


def count_words(text: str) -> Counter:
    """Count word occurrences in the given text.

    Words are split by whitespace and converted to lowercase.
    Punctuation is stripped from each word.

    Args:
        text: The input text.

    Returns:
        A Counter mapping each word to its occurrence count.
    """
    words = []
    for raw_word in text.split():
        # Strip punctuation from the word
        cleaned = raw_word.strip(".,;:!?\"'()[]{}<>")
        if cleaned:
            words.append(cleaned.lower())
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
