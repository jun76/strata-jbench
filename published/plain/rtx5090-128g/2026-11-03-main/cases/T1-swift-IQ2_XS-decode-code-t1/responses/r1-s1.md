wordfreq: Count word frequencies in a text file and display top N words.

Usage:
    python wordfreq.py <file> [-n N] [--case-sensitive]
"""

import argparse

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Split text into words, optionally preserving case.
    
    Words are sequences of alphanumeric characters (letters, digits, underscores).
    Non-alphanumeric characters are treated as separators.
    
    Args:
        text: Input text string.
        case_sensitive: If True, preserve case; if False, lowercase.
    
    Returns:
        List of tokens (words).
    """
    # Use regex to find sequences of word characters
    words = re.findall(r'\w+', text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_words(tokens: List[str]) -> Counter:
    """
    Count occurrences of each token.
    
    Args:
        tokens: List of word tokens.
    
    Returns:
        Counter mapping tokens to their frequencies.
    """
    return Counter(tokens)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top n most common words.
    
    Args:
        counter: Counter of word frequencies.
        n: Number of top words to return.
    
    Returns:
        List of (word, count) tuples, sorted by count descending.
        Ties in count are broken by alphabetical order (ascending).
    """
    # Counter.most_common() sorts by count descending.
    # For ties, Counter's most_common does not guarantee order, so we need
    # to sort explicitly by (-count, word) to ensure deterministic output.
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
