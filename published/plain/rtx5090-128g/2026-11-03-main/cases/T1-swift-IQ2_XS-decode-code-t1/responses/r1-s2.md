"""word_freq.py – Count word frequencies and display top N results.

Command-line tool that reads a text file, counts word frequencies,
and prints the top N most frequent words in a formatted table.
"""

import argparse

from collections import Counter
from typing import List, Tuple

def count_words(text: str) -> Counter:
    """Count the frequency of each word in the given text.
    
    Words are split by whitespace, stripped of punctuation, and
    converted to lowercase. Empty tokens are ignored.
    
    Args:
        text: Input text string.
        
    Returns:
        A Counter mapping lowercase words to their occurrence count.
    """
    # Split by whitespace and clean each token
    tokens = text.split()
    cleaned = []
    for token in tokens:
        # Strip leading/trailing punctuation from each token
        stripped = token.strip(".,;:!?\"'()[]{}")
        if stripped:
            cleaned.append(stripped.lower())
    return Counter(cleaned)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words.
    
    Args:
        counter: A Counter mapping words to counts.
        n: Number of top items to return.
        
    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then alphabetically.
    """
    # Counter.most_common(n) already sorts by count descending
    # For ties, we need to sort alphabetically. most_common does not
    # guarantee alphabetical order for ties, so we sort manually.
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
