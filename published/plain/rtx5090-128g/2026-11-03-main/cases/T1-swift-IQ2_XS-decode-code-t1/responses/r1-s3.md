Usage:
    python word_count.py <file> [--top N] [--case-sensitive]

Example:
    python word_count.py sample.txt --top 10

from typing import List, Tuple, Dict

def count_words(text: str, case_sensitive: bool = False) -> Dict[str, int]:
    """Count word frequencies in the given text.
    
    Args:
        text: The input text string.
        case_sensitive: If False, words are normalized to lowercase.
    
    Returns:
        A dictionary mapping each word to its count.
    """
    if not case_sensitive:
        text = text.lower()
    
    # Split on whitespace and strip punctuation from each word
    words: List[str] = []
    for raw_word in text.split():
        # Strip leading and trailing punctuation characters
        stripped = raw_word.strip(".,;:!?\"'()[]{}<>")
        if stripped:
            words.append(stripped)
    
    return dict(Counter(words))

def top_n_words(counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """Return the top n most frequent words.
    
    Args:
        counts: Dictionary mapping words to their counts.
        n: Number of top entries to return.
    
    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then alphabetically for ties.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(counts.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
