以下は、テキストファイルを読み込んで単語出現頻度の上位 N 件を表形式で表示するコマンドラインツールです。型ヒント、docstring、argparse、および pytest テストを含みます。

```python
#!/usr/bin/env python3
"""
wordfreq.py — テキストファイルの単語出現頻度を上位 N 件を表形式で表示する CLI ツール。

Usage:
    python wordfreq.py <input_file> [-n N] [--case-sensitive]
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合は大文字小文字を区別し、False の場合は小文字化して集計する。

    Returns:
        単語のリスト（句読点は除外し、英数字・日本語文字のみを対象）。
    """
    # 句読点・記号を除去して単語を抽出
    # 日本語を含むため、正規表現で英数字と日本語文字を抽出
    import re
    # 英単語・数字・日本語文字を抽出
    pattern = r'[A-Za-z0-9\u3040-\u309F\u30A0-\u30FF\u4E00-\u9FFF]+'
    tokens = re.findall(pattern, text)
    if not case_sensitive:
        tokens = [t.lower() for t in tokens]
    return tokens


def count_words(tokens: List[str]) -> Counter:
    """単語リストから出現頻度を集計する。

    Args:
        tokens: 単語のリスト。

    Returns:
        Counter オブジェクト（単語 -> 出現回数のマッピング）。
    """
    return Counter(tokens)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """出現頻度の上位 N 件を返す。

    Args:
        counter: Counter オブジェクト。
        n: 取得したい上位件数。

    Returns:
        (単語, 出現回数) のタプルを降順に並べたリスト。
    """
    return counter.most_common(n)
