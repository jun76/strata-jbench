#!/usr/bin/env python3
"""wordcount: Count word frequencies in a text file and display top N entries as a table."""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def read_file(path: str) -> str:
    """Read the contents of a text file.

    Args:
        path: Path to the text file.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        PermissionError: If the file cannot be read due to permissions.
    """
    with open(path, 'r', encoding='utf-8') as f:
        return f.read()


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.

    Words are split by whitespace and normalized to lowercase.
    Non-alphabetic tokens (e.g., punctuation attached to words) are stripped.

    Args:
        text: The input text string.

    Returns:
        A Counter mapping each word to its frequency.
    """
    words = text.split()
    # Strip punctuation from each word
    cleaned = []
    for word in words:
        # Strip leading and trailing non-alphanumeric characters
        stripped = word.strip('.,;:!?\"\'()[]{}<>-_/\\|~`@#$%^&*+=')
        if stripped:
            cleaned.append(stripped.lower())
    return Counter(cleaned)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top entries to return.

    Returns:
        A list of tuples (word, count) sorted by frequency descending.
    """
    return counter.most_common(n)
