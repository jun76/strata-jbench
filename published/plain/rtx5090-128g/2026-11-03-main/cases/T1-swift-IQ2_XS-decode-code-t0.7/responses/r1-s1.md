#!/usr/bin/env python3
"""word_freq.py – Count word frequencies in a text file and display top N results."""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def read_words(filepath: str) -> List[str]:
    """Read all words from a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        A list of lowercased words (split on whitespace).
    """
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            content = f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    words = content.split()
    return [w.lower() for w in words]


def count_frequencies(words: List[str]) -> Counter:
    """Count word frequencies.

    Args:
        words: List of words.

    Returns:
        A Counter mapping each word to its frequency.
    """
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top items to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
