"""テキストファイルから単語の出現頻度を上位N件集計するコマンドラインツール。"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from typing import List, Tuple


@dataclass
class WordCount:
    """単語とその出現回数を保持するデータクラス。"""

    word: str
    count: int


def tokenize(text: str) -> List[str]:
    """テキストから小文字化された単語トークンを抽出する。

    Args:
        text: 入力テキスト。

    Returns:
        正規化された単語トークンのリスト。
    """
    # 英数字とハイフンを含むトークンを抽出（例: "don't" -> "don't" ではなく "don", "t" となる仕様）
    # ここでは一般的な単語分割として英数字シーケンスを取得
    tokens = re.findall(r"[A-Za-z0-9]+", text)
    return [token.lower() for token in tokens]


def count_words(text: str) -> Counter:
    """テキスト内の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト。

    Returns:
        単語 -> 出現回数の Counter。
    """
    tokens = tokenize(text)
    return Counter(tokens)


def top_n_words(counter: Counter, n: int) -> List[WordCount]:
    """Counter から上位 N 件の単語と出現数を取得する。

    Args:
        counter: 単語頻度の Counter。
        n: 取得する上位件数。

    Returns:
        WordCount のリスト（降順）。
    """
    most_common = counter.most_common(n)
    return [WordCount(word=word, count=count) for word, count in most_common]
