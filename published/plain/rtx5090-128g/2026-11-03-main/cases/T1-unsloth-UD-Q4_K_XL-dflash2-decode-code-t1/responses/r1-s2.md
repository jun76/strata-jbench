#!/usr/bin/env python3
"""
テキストファイルから単語出現頻度を計算し、上位N件を表形式で出力するコマンドラインツール。

Usage:
    python word_counter.py input.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def clean_and_tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出・正規化するための関数。
    - 英数字のみを単語と見なす。
    - 小文字に変換する。
    - 1文字以上の英数字列を抽出する。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 正規化された単語のリスト。
    """
    # 正規表現で英数字のみを抽出し、小文字化
    words = re.findall(r'\b[a-z0-9]+\b', text.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words (List[str]): 単語のリスト。

    Returns:
        Counter: 単語ごとの出現回数を持つCounterオブジェクト。
    """
    return Counter(words)


def get_top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度が上位N件の単語と回数を取り出す。

    Args:
        counter (Counter): 単語カウントのCounter。
        n (int): 取得する上位件数。

    Returns:
        List[Tuple[str, int]]: (単語, 回数) のタプルのリスト。
    """
    return counter.most_common(n)
