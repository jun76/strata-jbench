"""
コマンドラインツール: テキストファイルの単語頻度を上位N件で表示する。

Usage:
    python wordfreq.py <filename> [--top N]

import argparse

import re


def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストに分割する。
    
    Args:
        text: 解析対象のテキスト
        
    Returns:
        小文字化された単語のリスト
    """
    # 英数字のみを単語とみなす（ハイフンやアポストロフィは除外）
    words = re.findall(r'[a-z0-9]+', text.lower())
    return words


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 解析対象のテキスト
        
    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度が高い順に上位 N 件の単語と出現回数を取得する。
    
    Args:
        counter: 単語頻度を保持する Counter オブジェクト
        n: 取得する件数
        
    Returns:
        (単語, 出現回数) のタプルのリスト（降順）
    """
    return counter.most_common(n)
