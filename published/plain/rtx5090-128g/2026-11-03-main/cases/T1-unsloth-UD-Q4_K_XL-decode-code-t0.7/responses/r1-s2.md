    python word_freq.py sample.txt --min-length 3

def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別する場合は True。
        min_length: 単語の最小文字数（この長さ未満の単語は無視）。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    
    # 英数字とハイフン、アポストロフィを単語の一部として扱い、それ以外を区切りとする
    words = []
    current_word = []
    for char in text:
        if char.isalnum() or char in ("'", "-"):
            current_word.append(char)
        else:
            if current_word:
                word = "".join(current_word)
                # 先頭・末尾のハイフンやアポストロフィを除去
                word = word.strip("-'")
                if len(word) >= min_length:
                    words.append(word)
                current_word = []
    if current_word:
        word = "".join(current_word)
        word = word.strip("-'")
        if len(word) >= min_length:
            words.append(word)
    
    return words

    """

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
