"""
word_freq.py - A command-line tool to count word frequencies in text files.

This module provides functionality to read text files, count word occurrences,
and display the top N most frequent words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    Extract words from the given text.
    
    Words are defined as sequences of alphanumeric characters. The extraction
    is case-insensitive (all words are converted to lowercase).
    
    Args:
        text: The input text from which to extract words.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    # Convert to lowercase and extract alphanumeric words
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word in the list.
    
    Args:
        words: A list of words to count.
        
    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top items to return.
        
    Returns:
        A list of tuples, each containing a word and its count,
        sorted by count in descending order. If counts are equal,
        words are sorted alphabetically.
    """
    # Get the most common items, breaking ties alphabetically
    items = counter.items()
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(items, key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
