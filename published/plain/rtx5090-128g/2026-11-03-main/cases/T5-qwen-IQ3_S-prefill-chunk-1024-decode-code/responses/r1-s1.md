#!/usr/bin/env python3
"""
word_freq: A command-line tool to count word frequencies in text files
and display the top N most frequent words in a table format.
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    Extract words from text, treating them as case-insensitive.
    
    Words are sequences of alphanumeric characters and underscores.
    Returns a list of lowercase words.
    
    Args:
        text: Input text string.
        
    Returns:
        List of lowercase words extracted from the text.
    """
    return re.findall(r'\b[a-zA-Z0-9_]+\b', text.lower())


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word in the given list.
    
    Args:
        words: List of words.
        
    Returns:
        Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.
        
    Returns:
        List of tuples (word, count) for the top N words.
    """
    return counter.most_common(n)
