"""
word_freq: コマンドラインツールでテキストファイルを読み込み、
単語の出現頻度を上位 N 件を表形式で表示する。
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    アルファベットと数字からなる連続部分列を単語とみなす。
    """
    return re.findall(r'\b[a-zA-Z0-9]+\b', text.lower())


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度を Counter で返す。
    """
    tokens = tokenize(text)
    return Counter(tokens)


def top_n_words(word_counts: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度上位 n 件の (word, count) リストを返す。
    頻度が同じ場合は辞書順でソート。
    """
    # most_common は頻度順だが、同頻度時の順序は不定。
    # 辞書順で安定させるためにカスタムソートを使う。
    items = list(word_counts.items())
    # 頻度の降順、同頻度なら単語の昇順
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(rows: List[Tuple[str, int]]) -> str:
    """
    (word, count) のリストを表形式の文字列に変換する。
    """
    if not rows:
        return ""
    
    # 列幅を決める
    word_width = max(len(word) for word, _ in rows)
    count_width = max(len(str(count)) for _, count in rows)
    
    # ヘッダー
    header = f"{'Word':<{word_width}}  {'Count':>{count_width}}"
    separator = f"{'-' * word_width}  {'-' * count_width}"
    
    lines = [header, separator]
    for word, count in rows:
        lines.append(f"{word:<{word_width}}  {count:>{count_width}}")
    
    return "\n".join(lines)
