#!/usr/bin/env python3
"""テキストファイルから単語出現頻度をカウントし、上位 N 件を表形式で表示するコマンドラインツール。"""

def tokenize(text: str) -> List[str]:
    """テキストから単語（アルファベットと数字のシーケンス）を取り出す。

    Args:
        text: 入力テキスト文字列。

    Returns:
        小文字化された単語のリスト。
    """
    return re.findall(r"[a-zA-Z0-9]+", text.lower())


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト文字列。

    Returns:
        単語とその出現回数を保持する Counter オブジェクト。
    """
    words = tokenize(text)
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Counter から出現回数の上位 n 件を返す。

    Args:
        counter: 単語出現頻度を保持する Counter。
        n: 取得する上位件数。

    Returns:
        (単語, 出現回数) のタプルを、出現回数降順で並べたリスト。
    """
    return counter.most_common(n)
