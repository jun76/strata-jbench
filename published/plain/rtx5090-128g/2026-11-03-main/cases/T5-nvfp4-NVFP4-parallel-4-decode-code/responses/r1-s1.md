"""
word_freq.py - テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

Usage:
    python word_freq.py <file> [--top N] [--ignore-case/--case-sensitive] [--min-len MIN]
"""

import argparse

from collections import Counter
from typing import List, Tuple


def tokenize(text: str, ignore_case: bool = True, min_len: int = 1) -> List[str]:
    """
    テキストを単語リストに分割する。

    Args:
        text: 入力テキスト文字列。
        ignore_case: True の場合、単語を小文字に変換する。
        min_len: この長さ未満の単語は除外する。

    Returns:
        単語のリスト。
    """
    if ignore_case:
        text = text.lower()

    words = text.split()
    # 句読点や記号を除去
    cleaned_words: List[str] = []
    for word in words:
        # 前後の句読点を除去
        cleaned = word.strip(".,;:!?\"'()[]{}<>-")
        if len(cleaned) >= min_len and cleaned:
            cleaned_words.append(cleaned)

    return cleaned_words


def count_words(text: str, ignore_case: bool = True, min_len: int = 1) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト文字列。
        ignore_case: 大文字小文字を区別しない場合 True。
        min_len: 最小単語長。

    Returns:
        単語 -> 出現回数 の Counter オブジェクト。
    """
    words = tokenize(text, ignore_case=ignore_case, min_len=min_len)
    return Counter(words)
