def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 解析対象のテキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。

    Returns:
        抽出された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    # 例: "don't", "state-of-the-art", "hello"
    pattern = r"[a-zA-Z0-9][a-zA-Z0-9'-]*"
    words = re.findall(pattern, text)
    return words


def count_words(
    text: str,
    case_sensitive: bool = False,
    min_length: int = 1,
) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 解析対象のテキスト。
        case_sensitive: 大文字・小文字を区別するかどうか。
        min_length: 最小単語長。これ未満の単語は除外。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = tokenize(text, case_sensitive=case_sensitive)
    filtered_words = [w for w in words if len(w) >= min_length]
    return Counter(filtered_words)
