"""
Word frequency counter command-line tool.

This module provides a command-line tool that reads a text file,
counts word occurrences, and displays the top N most frequent words
in a formatted table.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize_text(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.
    
    Args:
        text: Input text string.
        
    Returns:
        A list of lowercase word tokens.
    """
    # Extract words using regex (sequences of alphanumeric characters)
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_word_frequencies(text: str) -> Counter:
    """
    Count word frequencies in the given text.
    
    Args:
        text: Input text string.
        
    Returns:
        A Counter object mapping each word to its frequency.
    """
    tokens = tokenize_text(text)
    return Counter(tokens)


def get_top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: Number of top words to return.
        
    Returns:
        A list of (word, count) tuples sorted by frequency descending.
        For ties, words are sorted alphabetically.
    """
    # Sort by count descending, then by word ascending for tie-breaking
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
