def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """
    単語を正規化します。

    Args:
        word: 正規化する単語。
        case_sensitive: 大文字小文字を区別する場合は True。デフォルトは False。

    Returns:
        正規化された単語。
    """
    if not case_sensitive:
        word = word.lower()
    return word
