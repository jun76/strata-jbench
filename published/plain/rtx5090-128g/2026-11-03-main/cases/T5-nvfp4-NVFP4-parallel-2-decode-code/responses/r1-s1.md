import argparse

from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出してリストとして返す。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        抽出された単語のリスト（順序は出現順）。
    """
    if not case_sensitive:
        text = text.lower()
    # 単語として英数字とアンダースコアのみを抽出
    words = re.findall(r'[a-z0-9_]+', text, re.IGNORECASE)
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト中の単語出現頻度を Counter として返す。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        単語をキー、出現回数を値とする collections.Counter。
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)
