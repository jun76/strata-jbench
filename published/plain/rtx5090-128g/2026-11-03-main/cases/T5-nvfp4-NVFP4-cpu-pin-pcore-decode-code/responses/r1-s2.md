## `wordfreq/cli.py`

```python
"""
wordfreq CLI: テキストファイルを読み込み、単語の出現頻度を上位 N 件で表形式表示する。

使い方:
    python -m wordfreq.cli <file> [-n N] [--case-sensitive] [--min-len L]

例:
    python -m wordfreq.cli sample.txt -n 10
    python -m wordfreq.cli sample.txt --case-sensitive --min-len 3
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Iterable


# 英単語を抽出するための正規表現パターン
# 英字・数字・ハイフン・アポストロフィを含む連続した文字列を単語とみなす
_WORD_PATTERN = re.compile(r"[A-Za-z0-9'\-]+")


def extract_words(text: str, *, case_sensitive: bool = False, min_length: int = 1) -> list[str]:
    """テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。
                        False の場合、すべて小文字に変換する。
        min_length: この長さ未満の単語は除外する。

    Returns:
        抽出された単語のリスト（出現順）。

    Examples:
        >>> extract_words("Hello world hello", case_sensitive=False)
        ['hello', 'world', 'hello']
        >>> extract_words("Hello world hello", case_sensitive=True)
        ['Hello', 'world', 'hello']
        >>> extract_words("a bb ccc", min_length=2)
