import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを単語リストにトークン化します。
    
    Args:
        text: 入力テキスト文字列。
        
    Returns:
        小文字化され、アルファベットと数字のみを含む単語のリスト。
    """
    # 正規表現で単語を抽出 (アルファベット、数字、ハイフン、アポストロフィを許可)
    # 例: "don't" -> "don't", "state-of-the-art" -> "state-of-the-art"
    return re.findall(r"[a-zA-Z0-9'’-]+", text.lower())


def count_words(text: str) -> Counter:
    """テキスト内の単語出現頻度をカウントします。
    
    Args:
        text: 入力テキスト文字列。
        
    Returns:
        単語をキー、出現回数を値とするCounterオブジェクト。
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """上位 N 件の単語と頻度を取得します。
    
    Args:
        counter: 単語頻度カウンター。
        n: 上位何件を取得するか。
        
    Returns:
        (単語, 頻度) のタプルのリスト。頻度降順、頻度が同じ場合は辞書順。
    """
    # most_common は頻度順に返すが、頻度同士の順序は不定なため、
    # 明示的にソートして再現性を担保する
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
