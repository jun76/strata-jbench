def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: Trueなら大文字小文字を区別、Falseなら小文字に変換。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む連続文字列を単語とみなす
    words = re.findall(r"[a-z0-9'-]+", text, flags=re.IGNORECASE if case_sensitive else 0)
    return words

    """

    Returns:
        単語をキー、出現回数を値とするCounterオブジェクト。
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)
