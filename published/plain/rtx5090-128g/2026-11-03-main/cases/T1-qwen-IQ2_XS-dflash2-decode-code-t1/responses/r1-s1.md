#!/usr/bin/env python3
"""
Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words in a table format.
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def read_text_file(path: str) -> str:
    """Read the contents of a text file."""
    try:
        with open(path, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {path}")
    except Exception as e:
        raise RuntimeError(f"Error reading file '{path}': {e}")


def tokenize_words(text: str) -> List[str]:
    """Extract words from text, normalizing to lowercase and stripping punctuation."""
    words = []
    for word in text.split():
        # Strip punctuation from edges
        cleaned = word.strip(".,;:!?\"'()[]{}")
        if cleaned:
            words.append(cleaned.lower())
    return words


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """Count occurrences of each word."""
    return Counter(words)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top N words by frequency, sorted descending."""
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]]) -> str:
    """Format the top entries as a table string."""
    if not entries:
        return "No words found."

    max_word_len = max(len(word) for word, _ in entries)
    max_count_len = max(len(str(count)) for _, count in entries)

    header_word = "Word".ljust(max_word_len + 2)
    header_count = "Count".rjust(max_count_len)
    separator = "-" * (max_word_len + max_count_len + 2)

    lines = [f"{header_word}{header_count}", separator]
    for word, count in entries:
        lines.append(f"{word.ljust(max_word_len)} {str(count).rjust(max_count_len)}")

    return "\n".join(lines)
