#!/usr/bin/env python3
"""
Text word frequency counting tool.

Reads a text file, counts word occurrences, and displays the top N results
in a table format.
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path


def extract_words(text: str) -> list[str]:
    """
    Extract words from text by splitting on non-word characters.

    Args:
        text: Input text string.

    Returns:
        List of lowercase words extracted from the text.
    """
    # Match sequences of word characters (letters, digits, underscores)
    words = re.findall(r'\w+', text.lower())
    return words


def count_words(text: str) -> Counter[str]:
    """
    Count word frequencies in text.

    Args:
        text: Input text string.

    Returns:
        Counter mapping words to their frequencies.
    """
    words = extract_words(text)
    return Counter(words)


def format_table(counter: Counter[str], top_n: int) -> str:
    """
    Format the top N word frequencies as a table.

    Args:
        counter: Counter mapping words to frequencies.
        top_n: Number of top results to display.

    Returns:
        Formatted table string.
    """
    top_words = counter.most_common(top_n)

    if not top_words:
        return "No words found."

    # Determine column widths
    word_width = max(len(word) for word, _ in top_words)
    count_width = max(len(str(count)) for _, count in top_words)

    # Header
    header_line = f"{'Word':<{word_width}}  {'Count':<{count_width}}"
    separator = '-' * word_width + '  ' + '-' * count_width

    lines = [header_line, separator]

    for word, count in top_words:
        lines.append(f"{word:<{word_width}}  {count:<{count_width}}")

    return '\n'.join(lines)
