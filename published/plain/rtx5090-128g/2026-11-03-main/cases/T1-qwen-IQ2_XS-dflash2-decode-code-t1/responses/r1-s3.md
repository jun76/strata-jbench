#!/usr/bin/env python3
"""コマンドラインツール：テキストファイルから単語の出現頻度を集計し、上位N件を表形式で表示する。"""

import argparse
import sys
from collections import Counter
from pathlib import Path


def count_words(text: str) -> Counter[str]:
    """テキストから単語（空白・改行で区切られたトークン）の出現頻度を返す。

    Args:
        text: 読み込まれたテキスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(text.split())


def top_words(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """Counter の上位 n 件の (word, count) を返す。

    Args:
        counter: 単語と出現回数の Counter。
        n: 上位件数。

    Returns:
        上位 n 件の (word, count) のリスト（高頻度順）。
    """
    return counter.most_common(n)


def format_table(rows: list[tuple[str, int]]) -> str:
    """(word, count) の行を ASCII 表形式の文字列にする。

    Args:
        rows: (word, count) のリスト。

    Returns:
        表形式の文字列。
    """
    if not rows:
        return "(empty)"

    max_word_len = max(len(word) for word, _ in rows)
    max_count_len = max(len(str(count)) for _, count in rows)

    header_word = "word"
    header_count = "count"

    # 列幅はヘッダーより狭くなることはない
    word_width = max(max_word_len, len(header_word))
    count_width = max(max_count_len, len(header_count))

    lines = [
        f"{'word'.ljust(word_width)} | {'count'.rjust(count_width)}",
        "-" * word_width + " | " + "-" * count_width,
    ]
    for word, count in rows:
        lines.append(f"{word.ljust(word_width)} | {str(count).rjust(count_width)}")

    return "\n".join(lines)
