"""word_count.py - Count word frequencies from a text file and display top N words in a table."""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def normalize_word(word: str) -> str:
    """Normalize a word by converting to lowercase and stripping non-alphanumeric characters."""
    return word.lower()


def extract_words(text: str) -> List[str]:
    """Extract words from text, treating sequences of alphanumeric characters as words.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of normalized words (lowercase).
    """
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [normalize_word(w) for w in words]


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter mapping normalized words to their frequencies.
    """
    words = extract_words(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words from the counter.
    
    Args:
        counter: A Counter mapping words to frequencies.
        n: The number of top items to return.
        
    Returns:
        A list of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]], header: str = "Word\tCount") -> str:
    """Format word-count pairs as a tab-separated table string.
    
    Args:
        items: A list of (word, count) tuples.
        header: The header line. Default is "Word\\tCount".
        
    Returns:
        A string representing the formatted table.
    """
    lines = [header]
    for word, count in items:
        lines.append(f"{word}\t{count}")
    return "\n".join(lines)
