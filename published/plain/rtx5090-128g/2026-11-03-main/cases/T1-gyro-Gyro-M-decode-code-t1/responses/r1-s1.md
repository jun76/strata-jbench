"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a formatted table.

Usage:
    python wordfreq.py <file> [-n N]
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    Count word occurrences in the given text.

    Words are extracted by splitting on non-alphanumeric characters,
    normalized to lowercase.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    # Split on any non-alphanumeric character, normalize to lowercase
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from a Counter.

    Args:
        counter: A Counter mapping words to frequencies.
        n: The number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by frequency descending,
        then alphabetically for ties.
    """
    # Sort by frequency descending, then alphabetically ascending for ties
    sorted_items = sorted(
        counter.items(),
        key=lambda x: (-x[1], x[0])
    )
    return sorted_items[:n]
