import argparse

from collections import Counter

def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。

    - 英数字を単語として認識する
    - 大文字/小文字は区別しない（小文字に正規化）
    - 句読点や記号は除外する

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト（小文字化済み）
    """
    # 英数字の連続を単語として抽出し、小文字に変換
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_frequencies(text: str) -> Counter:
    """
    テキスト内の単語出現頻度を集計する。

    Args:
        text: 入力テキスト

    Returns:
        単語 -> 出現回数の Counter
    """
    words = tokenize(text)
    return Counter(words)
