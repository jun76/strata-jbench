"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度を集計し、
上位 N 件の頻出単語を表形式で表示するコマンドラインツール。

Usage:
    python word_freq.py <input_file> [-n N]

Example:
    python word_freq.py sample.txt -n 5

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    英数字の連続した文字列を単語とみなし、小文字に変換する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語のリスト（小文字化済み）
    """
    # 英数字の連続を単語として抽出
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_frequencies(text: str) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語 -> 出現回数 の Counter オブジェクト
    """
    words = tokenize(text)
    return Counter(words)

    """

        n: 取得する件数
        
    Returns:
        上位 n 件の (単語, 出現回数) のリスト
    """
    return counter.most_common(n)
