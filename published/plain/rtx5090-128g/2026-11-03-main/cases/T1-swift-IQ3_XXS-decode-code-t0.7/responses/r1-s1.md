#!/usr/bin/env python3
"""
word_freq.py - 単語の出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツール。

使い方:
    python word_freq.py <file> [--top N]

例:
    python word_freq.py sample.txt --top 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    英数字のみの単語を抽出し、大文字小文字を区別しない（小文字化する）。
    """
    words: List[str] = []
    current: List[str] = []
    for char in text:
        if char.isalnum():
            current.append(char.lower())
        else:
            if current:
                words.append("".join(current))
                current = []
    if current:
        words.append("".join(current))
    return words


def count_words(text: str) -> Counter:
    """
    テキスト中の単語出現頻度を Counter で返す。
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (word, count) を返す。
    頻度が同じ場合は辞書順で安定ソートする。
    """
    # Counter.most_common は頻度順で返すが、同頻度時の順序は保証されないため
    # 明示的にソートする: (-count, word) の順
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(rows: List[Tuple[str, int]]) -> str:
    """
    (word, count) のリストを ASCII 表形式でフォーマットする。
    """
    if not rows:
        return ""

    # 列幅を計算
    word_col_width = max(len(w) for w, _ in rows)
    count_col_width = max(len(str(c)) for _, c in rows)

    # ヘッダー
    header_word = "Word".ljust(word_col_width
