"""
コマンドラインツール: テキストファイルから単語出現頻度を計算し、上位 N 件を表形式で表示する。

使い方例:
    python wordfreq.py input.txt --top 10
    python wordfreq.py input.txt --top 20 --case-sensitive --output csv
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from typing import Iterable


# --- 型定義とデータ構造 ---

@dataclass
class WordFrequency:
    """単語とその出現回数、および順位を表すデータクラス。"""
    rank: int
    word: str
    count: int


def tokenize(text: str, case_sensitive: bool = False) -> list[str]:
    """
    テキストを単語リストに分割する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: True の場合、大文字・小文字を区別する。

    Returns:
        単語（トークン）のリスト。正規表現 \\w+ を用いて抽出する。
    """
    if not case_sensitive:
        text = text.lower()
    # Unicode 対応の \\w は日本語等も含むが、ここでは英単語・数字を想定
    tokens = re.findall(r'\w+', text)
    return tokens


def count_words(tokens: Iterable[str]) -> Counter[str]:
    """
    トークンリストから単語出現頻度を Counter で集計する。

    Args:
        tokens: 単語（トークン）の iterable。

    Returns:
        Counter[str]: 単語をキー、出現回数を値とする辞書。
    """
    return Counter(tokens)
