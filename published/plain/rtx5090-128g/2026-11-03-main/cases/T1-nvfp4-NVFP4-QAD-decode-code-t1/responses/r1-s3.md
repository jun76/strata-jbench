#!/usr/bin/env python3
"""Top word frequency command-line tool.

Reads a text file, tokenizes it into words, counts word frequencies,
and displays the top N most frequent words in a formatted table.
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from pathlib import Path
from typing import List


@dataclass
class WordFrequency:
    """Represents a word and its occurrence count."""

    word: str
    count: int


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.

    Extracts words consisting of alphanumeric characters (including underscores)
    and converts them to lowercase.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase word tokens.
    """
    return re.findall(r'\b[a-zA-Z0-9_]+\b', text.lower())


def count_word_frequencies(text: str) -> Counter:
    """Count the frequency of each word in the given text.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter mapping each word to its frequency count.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_words(counter: Counter, n: int) -> List[WordFrequency]:
    """Get the top N most frequent words from a counter.

    Args:
        counter: A Counter mapping words to their frequencies.
        n: The number of top words to return.

    Returns:
        A list of WordFrequency objects for the top N words,
        sorted by count (descending), then alphabetically (ascending)
        for ties.
    """
    # Sort by count descending, then by word ascending for stability
    sorted_items = sorted(
        counter.items(),
        key=lambda item: (-item[1], item[0])
    )
    return [WordFrequency(word=word, count=count) for word, count in sorted_items[:n]]
