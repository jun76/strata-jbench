"""
コマンドラインツール: 単語の出現頻度を集計し上位N件を表示するツール。

使用方法:
    python word_freq.py <input_file> [-n N]

引数:
    input_file: 解析対象のテキストファイルパス (必須)
    -n, --top-n: 表示する上位単語数 (デフォルト: 10)

例:
    python word_freq.py sample.txt -n 5
"""

import argparse

def extract_words(text: str) -> List[str]:
    """
    与えられたテキストから単語を抽出する。

    単語は小文字に正規化され、アルファベットと数字のみで構成される。
    句読点や記号は除外される。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 小文字に正規化された単語のリスト。
    """
    # 正規表現: アルファベット(a-z)と数字(0-9)の連続した文字列を抽出
    # \b は単語境界、\w+ は単語文字（アルファベット、数字、アンダースコア）
    # ここではアンダースコアを除いて純粋な単語を抽出するため、[a-z0-9]+ を使用
    words = re.findall(r"[a-z0-9]+", text.lower())
    return words
