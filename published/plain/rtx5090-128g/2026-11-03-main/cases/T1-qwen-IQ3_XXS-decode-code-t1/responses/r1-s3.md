"""word_freq.py - 単語出現頻度集計ツール

テキストファイルを読み込み、単語の出現頻度を集計して上位N件を表形式で表示するコマンドラインツール。

Usage:
    python word_freq.py <file> [-n N] [-w W] [-s] [--sort-freq/--sort-alpha] [--json]

Example:
    python word_freq.py sample.txt -n 10 --sort-freq
"""

from __future__ import annotations

# ---------------------------------------------------------------------------

WordCount = Tuple[str, int]  # (word, count)


def tokenize(text: str, *, ignore_case: bool = True) -> List[str]:
    """テキストを単語リストに分割する。

    Args:
        text: 入力テキスト。
        ignore_case: True の場合、単語を小文字に正規化する。

    Returns:
        単語のリスト（句読点は除外される）。
    """
    if ignore_case:
        text = text.lower()
    # 英数字・ハイフン・アポストロフィ・Unicode 文字を単語として抽出
    # 例: "don't" -> ["don't"], "hello-world" -> ["hello-world"]
    words: List[str] = re.findall(r"[a-z0-9]+(?:['\-]?[a-z0-9]+)*", text, re.IGNORECASE)
    return words


def count_words(
    words: List[str],
    *,
    min_length: int = 1,
) -> Dict[str, int]:
    """単語リストから頻度辞書を構築する。

    Args:
        words: 単語のリスト。
        min_length: この長さ未満の単語は無視する（デフォルト: 1）。

    Returns:
        単語 -> 出現回数 の辞書。
    """
    filtered = [w for w in words if len(w) >= min_length]
    counter: Counter[str] = Counter(filtered)
    return dict(counter)
