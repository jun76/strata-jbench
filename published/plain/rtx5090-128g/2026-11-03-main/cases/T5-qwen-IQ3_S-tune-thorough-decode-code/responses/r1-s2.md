#!/usr/bin/env python3
"""
word_freq.py

コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を計算し、
上位 N 件を表形式で出力します。

Usage:
    python word_freq.py <file> [--top N] [--case-sensitive]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text (str): 入力テキスト。
        case_sensitive (bool): 大文字・小文字を区別するか。

    Returns:
        List[str]: 単語リスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とアポストロフィを含む単語を抽出 (例: "don't", "it's")
    words = re.findall(r"[a-zA-Z0-9']+", text)
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから頻度カウントを取得する。

    Args:
        words (List[str]): 単語リスト。

    Returns:
        Counter: 単語と出現回数の対応。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の単語と出現回数を取得する。

    Args:
        counter (Counter): 単語頻度カウンター。
        n (int): 上位件数。

    Returns:
        List[Tuple[str, int]]: (単語, 出現回数) のリスト。
    """
    return counter.most_common(n)
