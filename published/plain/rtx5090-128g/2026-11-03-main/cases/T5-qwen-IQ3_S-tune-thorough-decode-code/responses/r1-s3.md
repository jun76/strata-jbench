word_counter.py - Text file word frequency counter.

Reads a text file, counts word occurrences, and displays the top N words
in a formatted table.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.
    
    Extracts words consisting of alphanumeric characters and underscores,
    converted to lowercase.
    
    Args:
        text: The input text to tokenize.
        
    Returns:
        A list of lowercase word tokens.
    """
    return re.findall(r'\b\w+\b', text.lower())


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.
    
    Args:
        text: The input text.
        
    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a counter.
    
    Args:
        counter: A Counter object with word counts.
        n: The number of top words to return.
        
    Returns:
        A list of tuples, each containing a word and its count,
        ordered by descending count. Ties are broken by alphabetical order.
    """
    # Sort by count descending, then by word ascending for consistent tie-breaking
    sorted_items = sorted(counter.items(), key=lambda item: (-item[1], item[0]))
    return sorted_items[:n]
