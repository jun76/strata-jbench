def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    if not intervals:
        return []
    
    # Create a copy to avoid modifying the input
    sorted_intervals = sorted(intervals, key=lambda x: x[0])
    
    merged = []
    current_start, current_end = sorted_intervals[0][0], sorted_intervals[0][1]
    
    for interval in sorted_intervals[1:]:
        start, end = interval[0], interval[1]
        # Check if overlapping or touching
        if start <= current_end:
            # Merge: update current_end to the max of current_end and end
            current_end = max(current_end, end)
        else:
            # No overlap: add current interval to merged and start new
            merged.append([current_start, current_end])
            current_start, current_end = start, end
    
    # Add the last interval
    merged.append([current_start, current_end])
    
    return merged
