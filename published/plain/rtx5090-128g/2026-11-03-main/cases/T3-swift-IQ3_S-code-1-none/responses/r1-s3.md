def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    if not intervals:
        return []
    
    # Sort by start, then by end
    sorted_intervals = sorted(intervals, key=lambda x: (x[0], x[1]))
    
    merged = [sorted_intervals[0][:]]  # copy of the first interval
    
    for current in sorted_intervals[1:]:
        last = merged[-1]
        # Check if current overlaps or touches the last merged interval
        # Touching: current[0] <= last[1] means overlap (including touching at endpoint)
        if current[0] <= last[1]:
            # Merge: extend the end if current's end is larger
            last[1] = max(last[1], current[1])
        else:
            merged.append(current[:])
    
    return merged
