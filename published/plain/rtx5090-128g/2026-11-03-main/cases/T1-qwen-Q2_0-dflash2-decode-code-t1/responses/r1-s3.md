#!/usr/bin/env python3

Word frequency counter: Read a text file, count word occurrences, and display
the top N words in a table format.

Usage:
    python word_freq.py <filename> [-n NUM] [--min-length MIN]
"""

import argparse

from collections import Counter
from typing import Dict, List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Split text into words.
    
    Words are sequences of alphanumeric characters, underscores, and hyphens.
    Words are converted to lowercase for case-insensitive matching.
    
    Args:
        text: Input text string.
        
    Returns:
        List of normalized words.
    """
    # Normalize: extract sequences of letters/digits/underscores/hyphens
    words = re.findall(r"[A-Za-z0-9_\-]+", text)
    return [word.lower() for word in words]


def count_words(words: List[str], min_length: int = 1) -> Dict[str, int]:
    """
    Count occurrences of each word, filtering by minimum length.
    
    Args:
        words: List of normalized words.
        min_length: Minimum word length to include (default 1).
        
    Returns:
        Dictionary mapping word to its count.
    """
    filtered = [w for w in words if len(w) >= min_length]
    return dict(Counter(filtered))


def get_top_n(word_counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    Return the top N most frequent words.
    
    Args:
        word_counts: Dictionary of word -> count.
        n: Number of top words to return.
        
    Returns:
        List of (word, count) tuples sorted by count descending.
    """
    sorted_items = sorted(word_counts.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
