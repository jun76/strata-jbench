以下に、要件を満たすコマンドラインツールと単体テストを作成します。

まず、メインのツールファイルを作成します。

```python
#!/usr/bin/env python3
"""
word_freq: テキストファイルを読み込んで単語の出現頻度を集計し、
上位 N 件を表形式で表示するコマンドラインツール。
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    テキストから単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト文字列

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    # 句読点や記号を除去し、小文字化して単語を分割
    import re
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text.lower())
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 N 件の (単語, 回数) を取得する。

    Args:
        counter: 単語の出現頻度 Counter
        n: 取得する上位件数

    Returns:
        (単語, 回数) のタプルのリスト（回数の降順）
    """
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]], word_width: int = 20) -> str:
    """
    (単語, 回数) のリストを表形式の文字列に整形する。

    Args:
        rows: (単語, 回数) のタプルのリスト
        word_width: 単語列の幅

    Returns:
        整形された表文字列
    """
    if not rows:
        return "単語が見つかりませんでした。"

    # ヘッダー
    header = f"{'単語':<{word_width}} {'回数':>10}"
    separator = f"{'-' * word_width} {'-' * 10}"

    lines = [header, separator]
    for word, count in rows:
        lines.append(f"{word:<{word_width}} {count:>10}")

    return "\n".join(lines)
