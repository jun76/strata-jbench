#!/usr/bin/env python3
"""
Text word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words
in a tabular format.
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into words.

    Words are sequences of alphanumeric characters. All words are lowercased.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase word tokens.
    """
    return [word.lower() for word in re.findall(r'[a-zA-Z0-9]+', text)]


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of word tokens.

    Returns:
        A Counter mapping each word to its occurrence count.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words.

    Args:
        counter: A Counter mapping words to counts.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by count descending, then by word ascending for consistent ordering
    return sorted(counter.items(), key=lambda x: (-x[1], x[0]))[:n]
