"""Word frequency counter CLI tool.

Reads one or more text files, counts word occurrences (case-insensitive,
punctuation stripped), and displays the top N words in a table format.
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words, stripping punctuation.

    Args:
        text: Input text string.

    Returns:
        List of lowercase words with punctuation removed.
        Empty strings are filtered out.
    """
    # Convert to lowercase and split on non-alphanumeric characters
    # This handles hyphens, apostrophes, quotes, etc.
    words = re.findall(r"[a-zA-Z0-9']+", text.lower())
    # Filter out empty strings (shouldn't occur with the regex above, but safe)
    return [w.strip("'") for w in words if w.strip("'")]


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        Counter mapping words to their occurrence counts.
    """
    return Counter(tokenize(text))


def read_file(path: Path) -> str:
    """Read text content from a file.

        path: Path to the text file.

    Returns:
        The file's text content.

    Raises:
        FileNotFoundError: If the file does not exist.
        PermissionError: If the file cannot be read.
        OSError: For other I/O errors.
    """
