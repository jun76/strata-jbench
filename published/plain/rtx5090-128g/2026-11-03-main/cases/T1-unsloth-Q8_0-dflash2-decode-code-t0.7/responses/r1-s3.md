import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト文字列
        case_sensitive: 大文字・小文字を区別するかどうか。
                        False の場合、すべて小文字に変換する。

    Returns:
        抽出された単語のリスト
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とアンダースコアのみを単語として扱う
    words = re.findall(r'\b[a-z0-9_]+\b', text)
    return words


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    return Counter(words)

    """

    Args:
        counter: 単語頻度を格納した Counter
        n: 取得する件数

    Returns:
        (単語, 出現回数) のタプルのリスト。回数の降順、同順位の場合は辞書順。
    """
    # most_common は回数の降順で返すが、同順位の場合は挿入順になるため
    # 辞書順でソートするために一旦ソートしてから上位 n 件を取る
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
