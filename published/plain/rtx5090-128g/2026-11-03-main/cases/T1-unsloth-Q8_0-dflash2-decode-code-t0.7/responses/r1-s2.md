def extract_words(text: str) -> List[str]:
    """
    テキストから単語リストを抽出する。

    単語は英数字とハイフンで構成される文字列として認識される。
    大文字・小文字は無視される（すべて小文字に変換）。

    Args:
        text (str): 入力テキスト

    Returns:
        List[str]: 抽出された単語のリスト
    """
    # 英数字とハイフンを許可するパターン。
    # 例: "don't" -> "don't", "hello-world" -> "hello-world"
    # 純粋なアルファベット/数字の単語のみを抽出する場合の簡易版
    words = re.findall(r'[a-zA-Z0-9]+(?:[-\'][a-zA-Z0-9]+)*', text)
    return [word.lower() for word in words]


def count_words(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text (str): 入力テキスト

    Returns:
        Counter[str]: 単語をキー、出現回数を値とする Counter オブジェクト
    """
    words = extract_words(text)
    return Counter(words)
