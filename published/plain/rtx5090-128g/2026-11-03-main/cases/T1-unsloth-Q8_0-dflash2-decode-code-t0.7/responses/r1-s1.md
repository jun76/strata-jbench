#!/usr/bin/env python3
"""
word_count.py: テキストファイルから単語の出現頻度をカウントし、上位N件を表形式で表示するコマンドラインツール。
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple, Dict


def normalize_word(word: str) -> str:
    """
    単語を正規化する。
    小文字に変換し、前後の空白と句読点を除去する。
    
    Args:
        word (str): 元の単語
        
    Returns:
        str: 正規化された単語。空文字列の場合は空文字列を返す。
    """
    # 句読点や記号を除去（英数字とハイフン、アポストロフィは保持）
    import re
    cleaned = re.sub(r"[^\w'-]", "", word)
    return cleaned.lower()


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text (str): 入力テキスト
        
    Returns:
        Counter: 単語と出現回数をまとめた Counter オブジェクト
    """
    words = text.split()
    normalized_words = [normalize_word(w) for w in words]
    # 空文字列を除外
    filtered_words = [w for w in normalized_words if w]
    return Counter(filtered_words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の単語とカウントを取得する。
    
    Args:
        counter (Counter): 単語とカウントの Counter
        n (int): 取得する件数
        
    Returns:
        List[Tuple[str, int]]: [(単語, カウント), ...] のリスト。カウント降順。
    """
    if n <= 0:
        return []
    return counter.most_common(n)
