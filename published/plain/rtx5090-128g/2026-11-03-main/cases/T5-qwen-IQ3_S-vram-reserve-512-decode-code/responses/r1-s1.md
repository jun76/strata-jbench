#!/usr/bin/env python3
"""
Word Frequency Counter CLI Tool

This script reads a text file, counts the occurrence of each word,
and displays the top N most frequent words in a formatted table.
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.
    
    Words are defined as sequences of alphanumeric characters.
    Non-alphanumeric characters (punctuation, etc.) are treated as separators.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase word tokens.
    """
    # Replace non-alphanumeric characters with spaces to separate tokens
    # This is a simple approach; for production, consider using re.findall(r'\b\w+\b', text.lower())
    import re
    return re.findall(r'\b[a-z0-9]+\b', text.lower())


def count_words(text: str) -> Counter:
    """
    Count the frequency of each word in the text.
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter object mapping words to their frequencies.
    """
    tokens = tokenize(text)
    return Counter(tokens)
