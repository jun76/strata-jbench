"""
テキストファイルを読み込み、単語の出現頻度を計算して上位 N 件を表示するツール。
"""

from collections import Counter

def tokenize(text: str) -> List[str]:
    """
    テキストを単語トークンに分割します。
    英数字のみを単語として扱い、小文字化します。
    
    Args:
        text (str): 入力テキスト
        
    Returns:
        List[str]: 単語のリスト
    """
    # 正規表現で英数字のみを含むトークンを抽出し、小文字化
    return [word.lower() for word in re.findall(r'[a-zA-Z0-9]+', text)]


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度を計算します。
    
    Args:
        text (str): 入力テキスト
        
    Returns:
        Counter: 単語 -> 出現回数 の辞書
    """
    tokens = tokenize(text)
    return Counter(tokens)
