#!/usr/bin/env python3
"""
Word Frequency Counter

A command-line tool that reads a text file, counts word frequencies,
and displays the top N words in a tabular format.
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.
    
    Words are tokenized by splitting on non-alphanumeric characters
    and converted to lowercase for case-insensitive counting.
    
    Args:
        text: The input text to analyze.
        
    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    # Normalize to lowercase and extract words (alphanumeric sequences)
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word counts.
        n: The number of top entries to return.
        
    Returns:
        A list of tuples (word, count) for the top N words,
        sorted by count descending, then by word ascending for ties.
    """
    # Get the most common N items
    most_common = counter.most_common(n)
    # For ties, sort by word alphabetically
    # Counter.most_common already sorts by count descending,
    # but for equal counts, we want alphabetical order
    sorted_items = sorted(most_common, key=lambda x: (-x[1], x[0]))
    return sorted_items
