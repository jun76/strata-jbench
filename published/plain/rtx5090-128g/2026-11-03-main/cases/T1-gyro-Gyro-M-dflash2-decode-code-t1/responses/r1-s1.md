        抽出された単語のリスト。順序は出現順。

        ['hello', 'world', 'hello']
        >>> tokenize_text("Hello World hello", case_sensitive=True)
        ['Hello', 'World', 'hello']
    """
    tokens = [m.group() for m in pattern.finditer(text)]

        tokens = [tok.lower() for tok in tokens]
