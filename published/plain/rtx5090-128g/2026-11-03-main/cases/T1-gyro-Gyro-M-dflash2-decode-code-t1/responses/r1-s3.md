#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込み、
単語出現頻度を集計して上位 N 件を表形式で表示する。
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> list[str]:
    """
    テキストを単語リストに分割する。

    英単語のみを抽出し、正規表現 \\w+ に一致する連続した文字列を
    小文字化して返す。

    Args:
        text: 入力テキスト

    Returns:
        小文字化された単語のリスト
    """
    return [word.lower() for word in re.findall(r"\w+", text)]


def count_word_frequencies(words: list[str]) -> Counter[str]:
    """
    単語リストから出現頻度を集計する。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数 の Counter
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の単語と出現回数を取得する。

    Args:
        counter: 単語 -> 出現回数 の Counter
        n: 取得する件数 (0 または正の整数)

    Returns:
        (単語, 回数) のリスト。回数の降順、同回数ならアルファベット順。
    """
    # Counter.most_common() は回数の降順で返すが、
    # 同回数時の順序は不定。アルファベット順に安定させるため手動ソートする。
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
