word_freq: Command-line tool to count word frequencies in a text file.

Usage:
    python word_freq.py <input_file> [--top N] [--ignore-case] [--min-length L] [--output FORMAT]
"""

import argparse

def tokenize(text: str, ignore_case: bool = True, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words, optionally lowercasing and filtering by minimum length.

    Args:
        text: The input text string.
        ignore_case: If True, convert words to lowercase.
        min_length: Minimum character length for a word to be counted.

    Returns:
        A list of tokens (words) extracted from the text.
    """
    # Extract words using a regex that matches sequences of letters, digits, and apostrophes
    words = re.findall(r"\b[a-zA-Z'0-9]+\b", text)
    if ignore_case:
        words = [w.lower() for w in words]
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    return words


def count_frequencies(tokens: List[str]) -> Counter:
    """
    Count the frequency of each token.

    Args:
        tokens: A list of word tokens.

    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    return Counter(tokens)


def get_top_words(counter: Counter, top_n: int) -> List[Tuple[str, int]]:
    """
    Return the top N words by frequency.

    Args:
        counter: A Counter object with word frequencies.
        top_n: Number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by frequency descending, then alphabetically for ties.
    """
    # Sort by count descending, then by word ascending for ties
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:top_n]
