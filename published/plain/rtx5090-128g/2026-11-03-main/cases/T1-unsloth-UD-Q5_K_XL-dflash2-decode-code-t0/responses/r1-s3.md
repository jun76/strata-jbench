#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位 N 件の単語とその頻度を表形式で出力するコマンドラインツール。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    正規表現を用いて、英数字とハイフンを含む連続した文字列を単語として抽出する。
    大文字・小文字は無視して小文字に変換する。
    
    Args:
        text (str): 入力テキスト
        
    Returns:
        List[str]: 小文字化された単語のリスト
    """
    # 英数字、ハイフン、アポストロフィを含む単語を抽出
    # 例: "don't" -> "don't", "state-of-the-art" -> "state-of-the-art"
    words = re.findall(r"[a-zA-Z0-9'-]+", text)
    return [word.lower() for word in words]


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text (str): 入力テキスト
        
    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト
    """
    words = tokenize(text)
    return Counter(words)
