#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語出現頻度を上位N件表示するCLIツール

Usage:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt --top 20 --case-sensitive
"""

import argparse
import re
import sys
from collections import Counter

from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか

    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフンを含む単語を抽出（日本語は未対応）
    words = re.findall(r"[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*", text)
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか

    Returns:
        単語をキー、出現回数を値とするCounter
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位N件を取得する。

    Args:
        counter: 単語頻度のCounter
        n: 取得する件数

    Returns:
        (単語, 出現回数) のタプルのリスト（降順）
    """
    return counter.most_common(n)
