#!/usr/bin/env python3
"""
word_count CLI tool: Count word frequencies in a text file and display top N entries.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def parse_args() -> argparse.Namespace:
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and display top N entries."
    )
    parser.add_argument(
        "file",
        help="Path to the text file to analyze",
    )
    parser.add_argument(
        "-n", "--top",
        type=int,
        default=10,
        help="Number of top words to display (default: 10)",
    )
    parser.add_argument(
        "-c", "--case-insensitive",
        action="store_true",
        default=False,
        help="Count words case-insensitively (default: False)",
    )
    return parser.parse_args()


def extract_words(text: str, case_insensitive: bool = False) -> List[str]:
    """
    Extract words from text using a simple regex that matches alphanumeric sequences.

    Args:
        text: The input text string.
        case_insensitive: If True, convert all words to lowercase before extraction.

    Returns:
        A list of extracted words.
    """
    if case_insensitive:
        text = text.lower()
    # Match sequences of alphanumeric characters (words)
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    return words


def count_words(words: List[str]) -> Counter:
    """
    Count occurrences of each word in the list.

    Args:
        words: A list of words.

    Returns:
        A Counter object mapping words to their counts.
    """
    return Counter(words)
