def read_words_from_file(filepath: str) -> List[str]:
    """Read words from a text file, splitting on whitespace and lowercasing.

    Args:
        filepath: Path to the text file.

    Returns:
        List of lowercase words found in the file.
    """
    words: List[str] = []
    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            # Split on whitespace and convert to lowercase
            for word in line.split():
                # Strip punctuation from word boundaries
                cleaned = word.strip(".,;:!?\"'()[]{}")
                if cleaned:
                    words.append(cleaned.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """Count occurrences of each word in the list.

    Args:
        words: List of words to count.

    Returns:
        Counter object mapping words to their counts.
    """
    return Counter(words)


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N words by frequency, with ties broken alphabetically.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        List of (word, count) tuples sorted by count descending, then word ascending.
    """
    # Sort by count descending, then word ascending for stable ties
    sorted_items = sorted(counter.items(), key=lambda item: (-item[1], item[0]))
    return sorted_items[:n]
