"""
word_freq.py – Command-line tool to count word frequencies in a text file.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words using regex."""
    return re.findall(r'\b[a-zA-Z]+\b', text.lower())


def count_words(text: str) -> Counter[str]:
    """Count word occurrences in the given text."""
    words = tokenize(text)
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top n most common words with their counts."""
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """Format word-count pairs as a table string."""
    if not items:
        return "No words found."
    max_word_len = max(len(w) for w, _ in items)
    header = f"{'Word':<{max_word_len}} | Count"
    separator = '-' * (max_word_len + 8)
    lines = [header, separator]
    for word, count in items:
        lines.append(f"{word:<{max_word_len}} | {count}")
    return '\n'.join(lines)
