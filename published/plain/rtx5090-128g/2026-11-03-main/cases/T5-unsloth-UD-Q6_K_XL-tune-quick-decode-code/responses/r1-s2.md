"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度をカウントして上位 N 件を表形式で表示するコマンドラインツール。

Usage:

from collections import Counter

def normalize_word(word: str) -> str:
    """
    単語を正規化する。

    小文字に変換し、前後の空白および句読点を除去する。

    Args:
        word: 正規化する元の単語文字列。

    Returns:
        正規化された単語文字列。
    """
    return word.strip().strip(".,;:!?\"'()[]{}").lower()


def count_words(text: str) -> Counter:
    """
    テキスト内の単語の出現頻度をカウントする。

    Args:
        text: カウント対象のテキスト文字列。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    # 空白および改行で分割
    words = text.split()
    normalized_words = [normalize_word(word) for word in words]
    # 空文字列を除外
    filtered_words = [word for word in normalized_words if word]
    return Counter(filtered_words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から出現頻度の高い上位 N 件を取得する。

    Args:
        counter: 単語と出現回数の Counter。
        n: 取得する上位件数。

    Returns:
        (単語, 出現回数) のタプルのリスト。頻度が高い順に並ぶ。
    """
    return counter.most_common(n)
