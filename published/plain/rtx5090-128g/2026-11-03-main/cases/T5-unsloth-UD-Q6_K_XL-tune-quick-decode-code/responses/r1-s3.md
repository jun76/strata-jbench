#!/usr/bin/env python3
"""
Text frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a tabular format.
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def read_file(file_path: str) -> str:
    """
    Read the contents of a text file.

    Args:
        file_path: Path to the text file to read.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(file_path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Extracts sequences of alphanumeric characters (including underscores)
    as words, converted to lowercase.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words.
    """
    # Match sequences of word characters (letters, digits, underscores)
    return [word.lower() for word in re.findall(r"\w+", text)]


def count_words(text: str) -> Counter:
    """
    Count word occurrences in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a counter.

    Args:
        counter: A Counter object mapping words to counts.
        n: The number of top entries to return.

    Returns:
        A list of tuples (word, count) sorted by count descending.
    """
    return counter.most_common(n)
