"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位N件表示する。

使い方:
    python word_freq.py <file> [-n N] [--case-sensitive] [--min-length MIN] [--top TOP]
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
        min_length: 最小単語長。これ未満の単語は無視される。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    # min_length 以上の単語のみ残す
    words = [w for w in words if len(w) >= min_length]
    return words


def count_words(words: List[str]) -> Counter:
    """単語リストの出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (word, count) タプルを返す。

    Args:
        counter: 単語頻度の Counter。
        n: 上位の件数。

    Returns:
        上位 n 件の (word, count) のリスト。
    """
    return counter.most_common(n)
