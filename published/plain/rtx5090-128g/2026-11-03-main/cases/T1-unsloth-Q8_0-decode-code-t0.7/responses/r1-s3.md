#!/usr/bin/env python3
"""
word_count.py - テキストファイルから単語の出現頻度をカウントし、上位N件を表示するツール
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def preprocess_text(text: str) -> List[str]:
    """
    テキストを正規化し、単語のリストに変換する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        小文字化され、非英数字が除去された単語のリスト
    """
    # 英数字のみを残し、それ以外をスペースに置き換える
    # 正規表現: 英数字[a-zA-Z0-9]以外の文字をスペースに置換
    cleaned = re.sub(r'[^a-zA-Z0-9\s]', ' ', text)
    # 小文字化
    cleaned = cleaned.lower()
    # スペースで分割し、空文字列を除去
    words = [word for word in cleaned.split() if word]
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。
    
    Args:
        words: 単語のリスト
        
    Returns:
        単語をキー、出現回数を値とするCounterオブジェクト
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counterから出現頻度の高い上位n件を取得する。
    
    Args:
        counter: 単語の出現頻度を保持するCounter
        n: 取得する件数
        
    Returns:
        (単語, 出現回数) のタプルのリスト (頻度順)
    """
    # most_commonは頻度順に並んでいる
    return counter.most_common(n)
