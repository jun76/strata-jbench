#!/usr/bin/env python3
"""
wordfreq.py - テキストファイルから単語出現頻度を上位 N 件表示する CLI ツール。

使い方:
    python wordfreq.py input.txt -n 10
    cat input.txt | python wordfreq.py --stdin -n 5
"""

from __future__ import annotations

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import Iterable, TextIO


def extract_words(text: str) -> list[str]:
    """テキストから単語を抽出する。

    アルファベット、数字、ハイフン、アポストロフィを含む連続文字列を単語とみなす。
    大文字・小文字は区別せず、小文字に変換する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト（出現順、重複を含む）。
    """
    import re
    # 英数字、ハイフン、アポストロフィを含む連続文字列を単語として抽出
    return [m.group().lower() for m in re.finditer(r"[a-zA-Z0-9'\-]+", text)]


def count_words(words: Iterable[str]) -> Counter[str]:
    """単語リストの出現頻度をカウントする。

    Args:
        words: 単語のイテラブル。

    Returns:
        単語ごとの出現回数を格納した Counter。
    """
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """Counter から最も頻出する上位 n 件の (word, count) を返す。

    Args:
        counter: 単語頻度 Counter。
        n: 取得する件数。

    Returns:
        頻度降順の (word, count) タプルのリスト。
    """
    return counter.most_common(n)
