"""
word_freq.py - コマンドラインツール: テキストファイルから単語出現頻度を上位N件表示
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    英数字の連続を単語とし、それ以外を区切りとする。
    大文字・小文字は区別しない（すべて小文字に変換）。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト（すべて小文字）
    """
    # 英数字の連続を抽出し、小文字に変換
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [w.lower() for w in words]


def count_word_frequencies(text: str) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (word, count) を取得する。
    同数の場合は辞書順（アルファベット順）でソートする。

    Args:
        counter: 単語頻度の Counter
        n: 取得する件数

    Returns:
        (word, count) のリスト、上位順
    """
    # Counter.most_common は同数時の順序が不定なので、
    # 明示的に (-count, word) でソートして安定性を確保
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
