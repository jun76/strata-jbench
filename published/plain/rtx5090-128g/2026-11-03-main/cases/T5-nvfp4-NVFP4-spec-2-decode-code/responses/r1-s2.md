word_freq.py - テキストファイルから単語の出現頻度を算出し、上位 N 件を表形式で表示する CLI ツール。

Usage:
    python word_freq.py <file> [-n TOP_N] [--case-sensitive]

import argparse
import re

from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。

    Returns:
        抽出された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とアンダースコアの連続を単語とみなす
    words = re.findall(r"\w+", text)
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語と出現回数をマッピングした Counter オブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の単語と出現回数を取得する。

    Args:
        counter: 単語と出現回数をマッピングした Counter オブジェクト。
        n: 取得する件数。

    Returns:
        (単語, 出現回数) のタプルのリスト。出現回数の降順。
    """
    return counter.most_common(n)
