def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Extract words from the given text.

    Words are defined as sequences of alphanumeric characters and underscores
    (i.e., matching the regex pattern \\w+).

    Args:
        text: The input text to extract words from.
        case_sensitive: If False, words are converted to lowercase.

    Returns:
        A list of extracted words.
    """
    if not case_sensitive:
        text = text.lower()
    # Use regex to find all word-like tokens
    words = re.findall(r'\w+', text)
    return words

    Args:
        words: A list of words.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)
