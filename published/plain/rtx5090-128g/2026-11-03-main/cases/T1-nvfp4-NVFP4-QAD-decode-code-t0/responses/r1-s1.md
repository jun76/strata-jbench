def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。Falseの場合、すべて小文字に変換。
        min_length: 単語の最小長。これより短い単語は除外。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    # 例: "don't", "state-of-the-art" なども単語として扱う
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    
    # 最小長でフィルタリング
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    
    return words

    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    return Counter(words)
