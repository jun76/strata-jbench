#!/usr/bin/env python3
"""
word_count.py - テキストファイルの単語出現頻度を集計し、上位 N 件を表示するツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    単語は英数字とアンダースコアで構成される連続文字列として定義される。
    すべて小文字に変換される。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語のリスト
    """
    # \w+ は [a-zA-Z0-9_]+ に一致する
    return re.findall(r'\w+', text.lower())


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    words = extract_words(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から最も多く出現した n 個の単語を返す。
    
    Args:
        counter: 単語頻度 Counter
        n: 取得する上位件数
        
    Returns:
        (単語, 出現回数) のタプルのリスト、出現回数の多い順
    """
    return counter.most_common(n)
