def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストに分割する。

    大文字小文字を区別せず、アルファベットと数字のみからなる単語を抽出する。
    正規表現を使用せず、str.translate と isalnum で処理する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    # 文字を小文字に変換し、非英数字をスペースに置き換える
    cleaned_chars = []
    for ch in text.lower():
        if ch.isalnum():
            cleaned_chars.append(ch)
        else:
            cleaned_chars.append(' ')
    cleaned_text = ''.join(cleaned_chars)
    words = cleaned_text.split()
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)
