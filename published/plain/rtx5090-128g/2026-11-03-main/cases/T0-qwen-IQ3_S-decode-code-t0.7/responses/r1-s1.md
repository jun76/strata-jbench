    """
    テキストから単語を抽出し、正規化してリストとして返す。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 単語のリスト（小文字化、非英数字除去済み）。
    """
    # 英数字のみに分割し、小文字に変換
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text.lower())
    return words
