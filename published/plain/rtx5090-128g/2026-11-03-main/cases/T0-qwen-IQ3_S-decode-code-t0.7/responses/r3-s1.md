word_freq_counter.py

A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a table format.
"""

import argparse

from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    Count word occurrences in the given text.
    
    Words are defined as sequences of alphanumeric characters and apostrophes.
    The counting is case-insensitive.
    
    Args:
        text: The input text to analyze.
        
    Returns:
        A Counter object mapping each word to its frequency count.
    """
    # Normalize to lowercase and extract words
    # Words are defined as sequences of alphanumeric characters and apostrophes
    words = re.findall(r"[a-z0-9']+", text.lower())
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top items to return.
        
    Returns:
        A list of tuples (word, count) sorted by count descending,
        then by word ascending for ties.
    """
    # Counter.most_common returns items sorted by count descending.
    # For ties, we need to sort by word ascending as secondary key.
    # We can do this by using most_common and then re-sorting the ties,
    # but a cleaner approach is to sort explicitly.
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
