def count_words(text: str, case_sensitive: bool = False) -> Dict[str, int]:
    """
    テキストから単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するか (False の場合、すべて小文字に変換)

    Returns:
        単語をキー、出現回数を値とする辞書
    """
    if not case_sensitive:
        text = text.lower()

    # 非英数字を空白に置換して単語分割
    cleaned: str = ""
    for ch in text:
        if ch.isalnum() or ch in ("'", "-", "_"):
            cleaned += ch
        else:
            cleaned += " "

    words: List[str] = [w for w in cleaned.split() if w]
    freq: Dict[str, int] = {}
    for word in words:
        freq[word] = freq.get(word, 0) + 1

    return freq


def get_top_n(freq: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    頻度辞書から上位N件の単語と回数を取得する。

    Args:
        freq: 単語と出現回数の辞書
        n: 上位N件

    Returns:
        (単語, 回数)のタプルのリスト。頻度降順、同率時は辞書順。
    """
    # 頻度降順、同率時は単語の辞書順でソート
    sorted_items: List[Tuple[str, int]] = sorted(
        freq.items(),
        key=lambda x: (-x[1], x[0])
    )
    return sorted_items[:n]
