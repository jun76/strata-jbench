word_counter.py - Text file word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N most frequent
words in a table format.
"""

import argparse
import sys

def parse_args(argv: List[str] | None = None) -> argparse.Namespace:
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and display top N results."
    )
    parser.add_argument(
        "input_file",
        type=str,
        help="Path to the input text file.",
    )
    parser.add_argument(
        "-n", "--top",
        type=int,
        default=10,
        help="Number of top frequent words to display (default: 10).",
    )
    parser.add_argument(
        "--ignore-case",
        action="store_true",
        help="Ignore case when counting words (default: False).",
    )
    parser.add_argument(
        "--min-length",
        type=int,
        default=1,
        help="Minimum word length to include (default: 1).",
    )
    return parser.parse_args(argv)


def read_file(file_path: str) -> str:
    """Read and return the contents of a text file.

    Args:
        file_path: Path to the text file.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    path = Path(file_path)
    if not path.exists():
        raise FileNotFoundError(f"File not found: {file_path}")
    try:
        with open(path, "r", encoding="utf-8") as f:
            return f.read()
    except Exception as e:
        raise IOError(f"Error reading file {file_path}: {e}") from e
