テキストファイルを読み込み、単語の出現頻度をカウントし、上位N件を表形式で表示するツール。
"""

import argparse

def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    英数字とハイフンで構成されるトークンを単語とみなし、小文字化して返す。
    
    Args:
        text: 入力テキスト文字列
        
    Returns:
        単語のリスト（小文字）
    """
    # 英数字とハイフンを単語文字として扱う
    words = re.findall(r"[a-z0-9]+(?:-[a-z0-9]+)*", text.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。
    
    Args:
        words: 単語のリスト
        
    Returns:
        単語からカウントへのCounter辞書
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位N件の単語とカウントを返す。
    カウントが同じ場合は辞書順でソートする。
    
    Args:
        counter: 単語カウントのCounter
        n: 取得する上位件数
        
    Returns:
        (単語, カウント) タプルのリスト

    # 明示的にキーでソートする

    return items[:n]
