#!/usr/bin/env python3
"""テキストファイルの単語出現頻度を上位N件表示するコマンドラインツール。

使い方:
    python wordfreq.py input.txt -n 10
    cat input.txt | python wordfreq.py - -n 5
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Iterable


def tokenize(text: str) -> list[str]:
    """テキストを単語リストに分割する。

    単語はアルファベット、数字、ハイフン、アポストロフィを含む連続文字列とし、
    大文字小文字は無視して小文字に正規化する。

    Args:
        text: 分割対象のテキスト。

    Returns:
        正規化された単語のリスト。
    """
    return re.findall(r"[a-zA-Z0-9'-]+", text.lower())


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 対象テキスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(tokenize(text))


def format_table(
    counter: Counter[str],
    top_n: int,
    word_width: int = 20,
    count_width: int = 10,
) -> str:
    """Counter を表形式の文字列に整形する。

    Args:
        counter: 単語頻度の Counter。
        top_n: 表示する上位件数。
        word_width: 単語列の幅。
        count_width: 回数列の幅。

    Returns:
        表形式の文字列。
    """

        f"{'-' * word_width} {'-' * word_width} {'-' * count_width}",
    ]
