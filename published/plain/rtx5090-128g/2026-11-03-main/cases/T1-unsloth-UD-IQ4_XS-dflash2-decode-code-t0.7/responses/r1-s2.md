def load_text(filepath: str) -> str:
    """
    指定されたファイルからテキストを読み込む。

    Args:
        filepath: 読み込むファイルのパス。

    Returns:
        読み込んだテキスト文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイル読み取りに失敗した場合。
    """
    path = Path(filepath)
    if not path.exists():
        raise FileNotFoundError(f"ファイルが見つかりません: {filepath}")
    try:
        return path.read_text(encoding="utf-8")
    except Exception as e:
        raise IOError(f"ファイル読み取りに失敗しました: {e}")


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    単語は英数字とハイフンを含む連続文字列として抽出される。
    例: "hello-world" は ["hello-world"] として扱われる。

    Args:
        text: 分割対象のテキスト。
        case_sensitive: Trueの場合、大文字小文字を区別する。デフォルトはFalse。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフンを許容する単語パターン
    words = re.findall(r"[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*", text)
    return words
