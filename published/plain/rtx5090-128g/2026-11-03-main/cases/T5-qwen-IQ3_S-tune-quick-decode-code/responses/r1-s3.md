def tokenize(text: str, *, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するか。
        min_length: 最小単語長（この長さ未満のトークンは除外）。

    Returns:
        正規化された単語トークンのリスト。
    """
    # 英数字とハイフン・アポストロフィを含むトークンを抽出
    # 例: "don't" -> "don't", "e-mail" -> "e-mail"
    pattern = r"[a-zA-Z0-9]+(?:['\-][a-zA-Z0-9]+)*"
    tokens = re.findall(pattern, text)

    if not case_sensitive:
        tokens = [t.lower() for t in tokens]

    # 最小長フィルタ
    tokens = [t for t in tokens if len(t) >= min_length]

    return tokens
