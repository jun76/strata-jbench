# 単語パターン: 英数字とアンダースコアで構成される文字列
WORD_PATTERN = re.compile(r'\b[a-zA-Z0-9_]+\b')


def read_file(filepath: str) -> str:
    """指定されたファイルを読み込んで、その内容を文字列として返す。

    Args:
        filepath: 読み込むファイルのパス。

    Returns:
        ファイルの内容を表す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み取り中にエラーが発生した場合。
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
                        False の場合、すべて小文字に変換する。

    Returns:
        抽出された単語のリスト。
    """
    words = WORD_PATTERN.findall(text)
    if not case_sensitive:
        words = [word.lower() for word in words]
    return words

    """単語リストから出現頻度をカウントする。

    """
    return Counter(words)
