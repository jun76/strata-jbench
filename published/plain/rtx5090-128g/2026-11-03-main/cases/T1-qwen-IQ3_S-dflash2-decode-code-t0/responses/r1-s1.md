#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位N件表示する。

使い方:
    python word_freq.py <file> [-n N]

例:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    単語は英数字とハイフンで構成される文字列とし、大文字小文字は無視する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        抽出された単語のリスト（小文字化済み）
    """
    # 英数字とハイフンを含む単語を抽出
    words = re.findall(r'[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*', text)
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。
    
    Args:
        words: 単語のリスト
        
    Returns:
        単語をキー、出現回数を値とするCounter
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位N件を取得する。
    頻度が同じ場合は辞書順でソートする。
    
    Args:
        counter: 単語頻度のCounter
        n: 取得する件数
        
    Returns:
        (単語, 頻度)のタプルのリスト、上位n件
    """
    # 頻度降順、同頻度なら単語の辞書順
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
