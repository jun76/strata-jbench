word_freq.py – Count word frequencies in a text file and display the top N words.

Usage:
    python word_freq.py <input_file> [-n N]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse

from collections import Counter
from typing import List, Tuple


def read_file(path: str) -> str:
    """Read the entire content of a text file.

    Args:
        path: Path to the text file.

    Returns:
        The full text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Split text into lowercase words, stripping punctuation.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words with punctuation removed.
    """
    # Remove common punctuation characters
    import string
    translator = str.maketrans("", "", string.punctuation)
    cleaned = text.translate(translator)
    words = cleaned.split()
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter mapping each word to its frequency.
    """
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
