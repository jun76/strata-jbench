word_freq.py – Count word frequencies in a text file and display the top N words.

Usage:
    python word_freq.py <input_file> [-n N]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse

from collections import Counter

def read_file(path: str) -> str:
    """Read the entire content of a text file.

    Args:
        path: Path to the text file.

    Returns:
        The full text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.

    Words are split by whitespace and non-alphanumeric characters.
    Only alphabetic words are kept (no numbers, no punctuation).

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens.
    """
    # Split by non-alphanumeric characters, keep only alphabetic tokens
    import re
    tokens = re.findall(r"[a-zA-Z]+", text)
    return [token.lower() for token in tokens]


def count_words(tokens: List[str]) -> Counter:
    """Count the frequency of each word token.

    Args:
        tokens: A list of word tokens.

    Returns:
        A Counter mapping each word to its frequency.
    """
    return Counter(tokens)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
