"""wordfreq: Count word frequencies in a text file and display top N results.

Usage:
    python wordfreq.py input.txt [--top N] [--case-sensitive] [--min-len LENGTH]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """Extract words from the given text.

    Args:
        text: The input text to process.
        case_sensitive: If False, words are lowercased before counting.
        min_length: Minimum word length to include (words shorter than this are ignored).

    Returns:
        A list of extracted words (after filtering and normalization).
    """
    # Extract sequences of word characters (letters, digits, underscores)
    words = re.findall(r'\b\w+\b', text)

    if not case_sensitive:
        words = [word.lower() for word in words]

    words = [word for word in words if len(word) >= min_length]

    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: A list of words to count.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(freqs: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words.

    Args:
        freqs: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    if n <= 0:
        return []
    return freqs.most_common(n)
