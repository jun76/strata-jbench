#!/usr/bin/env python3
"""テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示するツール。

Usage:
    python wordfreq.py INPUT_FILE [--top N] [--case-sensitive]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Optional


# 単語を抽出するための正規表現パターン
# 英数字、ハイフン、アポストロフィを含む連続文字列を1単語とみなす
WORD_PATTERN = re.compile(r"[A-Za-z0-9'\-]+")


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。

    Returns:
        抽出された単語のリスト。case_sensitive が False の場合、
        すべての単語は小文字に変換される。
    """
    words = WORD_PATTERN.findall(text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
