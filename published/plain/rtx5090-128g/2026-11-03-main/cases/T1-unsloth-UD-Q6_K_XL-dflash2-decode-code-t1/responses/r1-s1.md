#!/usr/bin/env python3
"""
Text frequency counter: reads text files, counts word occurrences,
and displays the top N words in a table format.
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def read_text_file(filepath: str) -> str:
    """
    Read the content of a text file.

    Args:
        filepath: Path to the text file to read.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        OSError: If there is an error reading the file.
    """
    return Path(filepath).read_text(encoding="utf-8")


def extract_words(text: str) -> List[str]:
    """
    Extract words from text, converting to lowercase and keeping only alphabetic
    characters (including accented letters) and apostrophes within words.

    Args:
        text: The input text to process.

    Returns:
        A list of lowercased words found in the text.
    """
    # Match words: sequences of letters and apostrophes (e.g., "don't")
    # This regex finds sequences of Unicode letters and apostrophes
    words: List[str] = re.findall(r"[a-zA-ZÀ-ÿ]+(?:'[a-zA-ZÀ-ÿ]+)*", text.lower())
    return words


def count_word_frequencies(text: str) -> Dict[str, int]:
    """
    Count the frequency of each word in the given text.

    Args:
        text: The input text to analyze.

    Returns:
        A dictionary mapping each word to its occurrence count.
    """
    words = extract_words(text)
    counter = Counter(words)
    return dict(counter)
