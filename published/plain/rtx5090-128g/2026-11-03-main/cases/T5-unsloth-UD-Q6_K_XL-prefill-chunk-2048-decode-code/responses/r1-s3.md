"""
word_frequency.py - コマンドラインツール: テキストファイルの単語出現頻度を集計し、上位 N 件を表形式で表示する。

Usage:
    python word_frequency.py <input_file> [-n TOP_N] [--case-sensitive] [--min-length MIN_LEN]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Optional


def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字・小文字を区別するかどうか
        min_length: 単語の最小長（デフォルト: 1）

    Returns:
        抽出された単語のリスト
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    # 最小長でフィルタ
    words = [w for w in words if len(w) >= min_length]
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の単語と出現回数を取得する。

    Args:
        counter: 単語頻度の Counter
        n: 取得する件数

    Returns:
        (単語, 出現回数) のタプルのリスト、頻度順（降順）
    """
    return counter.most_common(n)
