テキストファイルを読み込み、単語の出現頻度を計算し、

def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    英数字とハイフンを含むトークンを単語とみなし、小文字化する。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 小文字化された単語のリスト。
    """
    # 正規表現で単語を抽出。英数字、ハイフン、アポストロフィを含むトークンを対象とする。
    # 例: "don't" -> "don't", "state-of-the-art" -> "state-of-the-art"
    words = re.findall(r"[a-z0-9][a-z0-9'-]*[a-z0-9]", text.lower())
    return words


def count_word_frequencies(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text (str): 入力テキスト。

    Returns:
        Counter[str]: 単語をキー、出現回数を値とするCounterオブジェクト。
    """
    words = extract_words(text)
    return Counter(words)
