#!/usr/bin/env python3
"""
Text file word frequency counter.

Reads a text file, counts word occurrences, and displays the top N most frequent words
in a tabular format.
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase alphanumeric words.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase alphanumeric tokens.
    """
    # Match sequences of alphanumeric characters (words)
    return [match.lower() for match in re.findall(r'\b[a-zA-Z0-9]+\b', text)]


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter object mapping words to their frequencies.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.
        
    Returns:
        A list of tuples (word, count), sorted by count descending, then word ascending.
    """
    # Most common words, ties broken alphabetically
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(entries: List[Tuple[str, int]]) -> str:
    """Format word frequency entries as a table string.
    
    Args:
        entries: A list of tuples (word, count).
        
    Returns:
        A formatted table string.
    """
    if not entries:
        return "No words found."

    word_width = max(len(word) for word, _ in entries)
    count_width = max(len(str(count)) for _, count in entries)
