import argparse

def read_file(filepath: str) -> str:
    """
    指定されたファイルを読み込み、その内容を文字列として返す。

    Args:
        filepath: 読み込むファイルのパス

    Returns:
        ファイルの内容

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        IOError: ファイルの読み取りに失敗した場合
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    単語は英数字およびハイフンを含む文字列として抽出される。
    非英数字の文字は区切りとして扱われる。

    Args:
        text: 分割するテキスト
        case_sensitive: 大文字小文字を区別するかどうか

    Returns:
        単語のリスト
    """
    # 英数字、ハイフン、アポストロフィを含む単語を抽出
    # 例: "don't" -> "don't", "state-of-the-art" -> "state-of-the-art"
    pattern = r"[a-zA-Z0-9][a-zA-Z0-9'\-]*"
    tokens = re.findall(pattern, text)

    if not case_sensitive:
        tokens = [token.lower() for token in tokens]

    return tokens


def count_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    return Counter(words)
