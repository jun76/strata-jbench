def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。

    単語は英数字およびハイフンで構成される連続した文字列として定義される。
    すべて小文字に変換して返す。

    Args:
        text (str): 解析対象のテキスト。

    Returns:
        List[str]: 小文字化された単語のリスト。
    """
    # 英数字、ハイフンを含む連続文字列を単語とみなす
    # 例: "don't" -> ["don", "t"] よりも "don't" を単語として扱いたい場合の調整も可能だが、
    # 標準的な単語抽出では [a-zA-Z0-9'-]+ などがよく使われる。
    # ここでは、英数字とハイフン、アポストロフィを許可する。
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return [word.lower() for word in words]
