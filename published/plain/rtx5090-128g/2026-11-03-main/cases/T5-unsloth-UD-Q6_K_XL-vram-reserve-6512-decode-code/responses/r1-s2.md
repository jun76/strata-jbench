#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルから単語の出現頻度を計算し、上位 N 件を表形式で表示します。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出します。
    正規表現を用いて英数字とハイフンを含む連続した文字列を単語として扱います。
    日本語などの文字は現状の正規表現では抽出されません。
    """
    # 正規表現: 1文字以上の英数字またはハイフン
    # \w は [a-zA-Z0-9_] に一致しますが、日本語は含まれません。
    # より厳密に英単語のみを抽出するなら [a-zA-Z0-9'-]+ などを使うこともできますが、
    # ここでは一般的な \w+ を使用します。
    return re.findall(r'\w+', text)


def calculate_word_frequencies(words: List[str]) -> Counter[str]:
    """
    単語リストから出現頻度を計算します。
    単語は小文字に変換されます。
    """
    return Counter(word.lower() for word in words)


def get_top_n_words(frequencies: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    頻度上位 N 件の (単語, 頻度) のリストを返します。
    頻度が同じ場合は辞書順でソートされます。
    """
    # most_common は頻度順、同頻度時は挿入順（ここでは辞書順を保証しないため追加ソート）
    items = frequencies.most_common()
    # 頻度降順、同頻度時は単語の昇順でソート
    sorted_items = sorted(items, key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
