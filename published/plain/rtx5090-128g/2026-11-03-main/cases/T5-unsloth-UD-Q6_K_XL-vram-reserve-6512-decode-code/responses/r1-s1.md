#!/usr/bin/env python3
"""Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a tabular format.
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path

def normalize_word(word: str) -> str:
    """Normalize a word by converting to lowercase and stripping non-alphanumeric chars.

    Args:
        word: The raw word string.

    Returns:
        The normalized word string.
    """
    return word.lower().strip(".,;:!?\"'()[]{}")


def tokenize(text: str) -> List[str]:
    """Tokenize text into words.

    Args:
        text: The input text string.

    Returns:
        A list of normalized words.
    """
    # Split on whitespace and handle punctuation
    raw_tokens = re.split(r'\s+', text.strip())
    words: List[str] = []
    for token in raw_tokens:
        # Further split on non-alphanumeric boundaries while preserving words
        parts = re.findall(r"[a-zA-Z0-9']+", token)
        for part in parts:
            cleaned = part.strip(".,;:!?\"'()[]{}")
            if cleaned:
                words.append(cleaned.lower())
    return words


def count_frequencies(words: List[str]) -> Counter[str]:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) for the top N words.
    """
    return counter.most_common(n)
