def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 単語の最小長（これ未満の単語は除外）。

    Returns:
        単語のリスト。
    """
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    # 例: "don't", "well-known" なども単語として扱う
    pattern = r"[a-zA-Z0-9]+(?:['-][a-zA-Z0-9]+)*"
    words = re.findall(pattern, text)

    if not case_sensitive:
        words = [word.lower() for word in words]

    # 最小長でフィルタリング
    if min_length > 1:
        words = [word for word in words if len(word) >= min_length]

    return words


def count_frequencies(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    return Counter(words)
