def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    """
    Merge all overlapping or touching closed intervals and return them sorted by start.
    
    Args:
        intervals: A list of intervals, each represented as [start, end].
                   Intervals may be unsorted and may be empty.
    
    Returns:
        A new list of merged intervals, sorted by start time.
        Does not modify the input.
    """
    if not intervals:
        return []
    
    # Create a sorted copy of the intervals (by start, then by end)
    sorted_intervals = sorted(intervals, key=lambda x: (x[0], x[1]))
    
    merged = []
    for interval in sorted_intervals:
        if not merged:
            merged.append([interval[0], interval[1]])
        else:
            last = merged[-1]
            # If overlapping or touching, merge
            if interval[0] <= last[1]:
                last[1] = max(last[1], interval[1])
            else:
                merged.append([interval[0], interval[1]])
    
    return merged
