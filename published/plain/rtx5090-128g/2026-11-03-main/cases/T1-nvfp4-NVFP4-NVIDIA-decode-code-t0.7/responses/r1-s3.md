Word Frequency Counter CLI Tool.

This module provides a command-line interface to read a text file,
count the frequency of each word, and display the top N most frequent

Usage:

def normalize_words(text: str, ignore_case: bool = True, min_length: int = 1) -> List[str]:
    """
    Extract words from text, normalize them, and filter by length.

    Args:
        text: The input text string.
        ignore_case: If True, convert words to lowercase.
        min_length: Minimum length of a word to be counted.

    Returns:
        A list of normalized words.
    """
    # Simple tokenization: split by whitespace and strip punctuation from edges
    # For a more robust solution, one might use `re.findall(r'\b\w+\b', text)`
    import re
    words = re.findall(r'\b\w+\b', text, flags=re.UNICODE)
    
    processed_words = []
    for word in words:
        if ignore_case:
            word = word.lower()
        if len(word) >= min_length:
            processed_words.append(word)
    return processed_words


def count_word_frequencies(words: List[str]) -> Dict[str, int]:
    """
    Count the frequency of each word in the list.

    Args:
        words: A list of words.

    Returns:
        A dictionary mapping each word to its count.
    """
    return dict(Counter(words))
