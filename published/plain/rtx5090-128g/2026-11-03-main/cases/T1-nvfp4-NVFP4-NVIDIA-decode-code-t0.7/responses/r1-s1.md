def tokenize(text: str, case_sensitive: bool = False, delimiter: Optional[str] = None) -> List[str]:
    """
    テキストを単語のリストに分割する。

    Args:
        text (str): 入力テキスト。
        case_sensitive (bool): 大文字小文字を区別するかどうか。False の場合、小文字に変換する。
        delimiter (Optional[str]): 区切り文字。None の場合、空白および一般的な句読点を区切りとして扱う。

    Returns:
        List[str]: 単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()

    if delimiter is not None:
        # 指定された区切り文字で分割
        words = text.split(delimiter)
    else:
        # 単語としてアルファベット、数字、ハイフン、アポストロフィを含む文字列を抽出
        # 日本語などの文字も考慮する場合、より複雑な正規表現が必要だが、
        # ここでは一般的な英単語と数字を抽出する。
        words = re.findall(r"[a-zA-Z0-9'\-]+", text)

    # 空文字列を除去
    return [word for word in words if word]
