def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """単語を正規化する。

    Args:
        word: 正規化対象の単語。
        case_sensitive: True の場合、大文字/小文字を区別する。False の場合、小文字に変換する。

    Returns:
        正規化された単語。
    """
    if not case_sensitive:
        return word.lower()
    return word


def extract_words(text: str, min_length: int = 1) -> List[str]:
    """テキストから単語を抽出する。

    単語はアルファベット・数字・ハイフン・アポストロフィで構成される連続文字列として定義される。
    指定した最小長未満の単語は除外される。

    Args:
        text: 解析対象のテキスト。
        min_length: 単語として認める最小文字数。

    Returns:
        抽出された単語のリスト（正規化前）。
    """
    # アルファベット、数字、ハイフン、アポストロフィを含む単語を抽出
    pattern = r"[a-zA-Z0-9'-]+"
    words = re.findall(pattern, text)
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    return words
