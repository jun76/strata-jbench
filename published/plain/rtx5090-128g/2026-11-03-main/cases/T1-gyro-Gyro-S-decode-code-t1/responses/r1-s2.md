import argparse

import re
from collections import Counter
from typing import List, Tuple

def count_words(file_path: str, case_sensitive: bool = False) -> List[Tuple[str, int]]:
    """テキストファイルから単語を抽出し、出現頻度をカウントする。

    Args:
        file_path: テキストファイルのパス
        case_sensitive: True の場合、大文字小文字を区別して集計する

    Returns:
        (単語, 出現回数) のタプルリスト。出現回数の降順、同回数時は単語の辞書順でソート。
    """
    with open(file_path, "r", encoding="utf-8") as f:
        text = f.read()

    if case_sensitive:
        # 大文字小文字を区別するため、原文のまま単語を抽出
        words = re.findall(r"[A-Za-z]+", text)
    else:
        # 小文字に統一してから単語を抽出
        words = re.findall(r"[A-Za-z]+", text.lower())

    counter: Counter[str] = Counter(words)
    # 頻度降順、同頻度時は辞書順でソート
    items: List[Tuple[str, int]] = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items
