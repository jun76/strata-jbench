"""
Top-N word frequency counter CLI tool.

Reads a text file, counts word occurrences (case-insensitive, splitting on non-alphanumeric),
and displays the top N most frequent words in a tabular format.
"""

import argparse
import sys
from collections import Counter
from typing import Dict, List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into words.

    Words are defined as sequences of alphanumeric characters.
    The tokens are converted to lowercase for case-insensitive counting.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase word tokens.
    """
    # Split on non-alphanumeric characters and filter empty strings
    import re
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_word_frequencies(text: str) -> Dict[str, int]:
    """
    Count the frequency of each word in the given text.

    Args:
        text: The input text to analyze.

    Returns:
        A dictionary mapping each word to its occurrence count.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n_frequencies(freq: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    Return the top N words by frequency, sorted by count (descending) then by word (ascending).

    Args:
        freq: A dictionary mapping words to their counts.
        n: The number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending, then word ascending.
    """
    items = list(freq.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
