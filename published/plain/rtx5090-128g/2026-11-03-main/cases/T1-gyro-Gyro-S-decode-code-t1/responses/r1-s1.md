def count_words(text: str) -> Counter:
    """Count words in *text* case-insensitively.

    A "word" is any token produced by ``str.split()`` (whitespace-delimited).
    Punctuation attached to words is stripped via ``str.strip(".,;:?!")``.

    Args:
        text: The input string to analyse.

    Returns:
        A ``Counter`` mapping lower-cased words to their occurrence counts.
    """
    words: Iterable[str] = text.split()
    cleaned: list[str] = []
    for word in words:
        stripped = word.strip(".,;:?!")
        if stripped:
            cleaned.append(stripped.lower())
    return Counter(cleaned)


def top_n(counter: Counter, n: int) -> list[tuple[str, int]]:
    """Return the top *n* entries of *counter*, sorted by descending count.

    Ties are resolved alphabetically.

    Args:
        counter: A ``Counter`` of word frequencies.
        n: How many entries to return. Must be >= 0.

    Returns:
        List of ``(word, count)`` tuples, highest count first.
    """
    if n < 0:
        raise ValueError("n must be non-negative")
    return counter.most_common(n)
