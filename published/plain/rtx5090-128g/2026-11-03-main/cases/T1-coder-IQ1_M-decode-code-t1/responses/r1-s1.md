"""
Text file word frequency counting CLI tool.

Usage:
    python wordfreq.py --file example.txt --top-n 10

This tool reads a text file, counts the frequency of each word,
and displays the top N most frequent words along with their counts.
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> List[Tuple[str, int]]:
    """
    Count word frequencies in the given text.

    Parameters:
        text: The input text string.

    Returns:
        A list of tuples (word, count) sorted by count in descending order.
        Words with the same count are sorted alphabetically.
    """
    if not text:
        return []
    words = text.split()
    counter = Counter(words)
    # Sort by count descending, then by word ascending
    sorted_words = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_words


def get_top_n(words_counts: List[Tuple[str, int]], top_n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from the sorted word count list.

    Parameters:
        words_counts: List of (word, count) tuples.
        top_n: Number of top entries to return.

    Returns:
        List of top N (word, count) tuples.
    """
    if top_n <= 0:
        return []
    return words_counts[:top_n]

    """
    Read the content of a text file.

    Parameters:
        file_path: Path to the text file.

    Returns:
        The content of the file as a string.
    """
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        raise
    except Exception as e:
