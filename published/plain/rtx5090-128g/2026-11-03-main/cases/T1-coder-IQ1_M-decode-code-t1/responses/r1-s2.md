"""
Command-line tool to analyze word frequency in a text file.

Usage example:
    python word_freq.py --file example.txt --top 10

"""
import argparse
import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """Extract words from text.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase alphanumeric words.
    """
    # Only match alphanumeric sequences (including underscores) as words
    words = re.findall(r'[a-zA-Z0-9_]+', text)
    return [word.lower() for word in words]


def count_frequencies(words: List[str]) -> Counter:
    """Count word frequencies.

    Args:
        words: List of words.

    Returns:
        Counter object with word frequencies.
    """
    return Counter(words)


def top_n_words(freq_counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get top N most common words.

    Args:
        freq_counter: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples sorted by frequency descending,
        then alphabetically for ties.
    """
    return sorted(freq_counter.items(), key=lambda x: (-x[1], x[0]))[:n]
