#!/usr/bin/env python3
"""word_freq: 計算 Word Frequency in Text File and Show Top-N Words"""

import argparse
from collections import Counter
import sys


def count_words(file_path: str) -> dict[str, int]:
    """
    Reads a text file and counts the occurrences of each word.
    
    Args:
        file_path: Path to the text file.
    
    Returns:
        Dictionary mapping each word (lowercase) to its frequency count.
    """
    with open(file_path, 'r', encoding='utf-8') as f:
        text = f.read()
    
    words = []
    for token in text.split():
        word = token.strip().lower()
        if word:
            words.append(word)
    
    return dict(Counter(words))


def top_n_frequencies(word_counts: dict[str, int], n: int = 10) -> list[tuple[str, int]]:
    """
    Returns the top-N most frequent words along with their counts.
    
    Args:
        word_counts: Dictionary mapping words to their frequency counts.
        n: Number of top words to return.
    
    Returns:
        List of tuples (word, count), sorted by frequency in descending order.
    """
    sorted_words = sorted(word_counts.items(), key=lambda x: x[1], reverse=True)
    return sorted_words[:n]


def format_output(top_words: list[tuple[str, int]]) -> str:
    """
    Formats the top-N word frequency results as a table string.
    
    Args:
        top_words: List of tuples (word, count).
    
    Returns:
        Formatted table string.
    """
    if not top_words:
        return "No words found."
    
    max_word_len = max(len(word) for word, _ in top_words)
    header = f"{'Word':<{max_word_len}} Count"
    separator = "-" * (max_word_len + 6)
    
    lines = [header, separator]
    for word, count in top_words:
        lines.append(f"{word:<{max_word_len}} {count}")
    
    return "\n".join(lines)
