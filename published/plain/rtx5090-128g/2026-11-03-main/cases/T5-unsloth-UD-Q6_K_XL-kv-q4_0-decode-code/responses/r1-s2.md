#!/usr/bin/env python3
"""Text word frequency counter command-line tool.

Reads a text file, counts word occurrences (case-insensitive, alphanumeric-only words),
and displays the top N most frequent words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_file(filepath: Path) -> str:
    """Read and return the contents of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The file contents as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """Extract lowercase alphanumeric words from text.

    Words are sequences of alphanumeric characters (including underscores).
    All words are converted to lowercase.

    Args:
        text: Input text string.

    Returns:
        List of lowercase words.
    """
    return re.findall(r"\b[a-zA-Z0-9_]+\b", text.lower())


def count_words(words: List[str]) -> Counter[str]:
    """Count occurrences of each word.

    Args:
        words: List of words to count.

    Returns:
        Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        List of (word, count) tuples, sorted by frequency descending,
        then alphabetically ascending for ties.
    """
    # Get all items and sort by count descending, then word ascending
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
