"""
wordfreq: コマンドラインツールでテキストファイルを読み込み、
単語の出現頻度を上位 N 件を表形式で表示する。

Usage:
    python wordfreq.py [OPTIONS] FILENAME

Options:
    -n, --top N       表示する上位単語の数 (default: 10)
    -c, --case-sensitive  大文字小文字を区別する (default: False)
"""

import argparse
import sys
import re

from typing import List, Tuple, Dict


def parse_args(argv: List[str] = None) -> argparse.Namespace:
    """
    コマンドライン引数を解析する。

    Args:
        argv: 引数のリスト。None の場合、sys.argv[1:] を使用。

    Returns:
        解析された引数を含む argparse.Namespace オブジェクト。
    """
    parser = argparse.ArgumentParser(
        description="テキストファイルから単語頻度を上位 N 件表示するツール。"
    )
    parser.add_argument(
        "filename",
        help="読み込むテキストファイルのパス",
    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=10,
        help="表示する上位単語の数 (default: 10)",
    )
    parser.add_argument(
        "-c",
        "--case-sensitive",
        action="store_true",
        help="大文字小文字を区別する",
    )
    return parser.parse_args(argv)
