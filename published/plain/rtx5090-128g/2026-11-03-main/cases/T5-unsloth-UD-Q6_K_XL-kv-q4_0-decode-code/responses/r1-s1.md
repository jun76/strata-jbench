#!/usr/bin/env python3
"""
Text word frequency counter and top-N display tool.

Reads a text file, counts word occurrences, and displays the top N most frequent
words in a tabular format.

Usage:
    python word_counter.py input.txt --top 10
"""

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from typing import List, Tuple


@dataclass
class WordFrequency:
    """Represents a word and its frequency."""
    word: str
    count: int


def extract_words(text: str) -> List[str]:
    """
    Extract words from text, converting to lowercase and splitting on non-alphanumeric characters.
    
    Args:
        text: Input text string.
        
    Returns:
        List of lowercase words extracted from the text.
    """
    # Convert to lowercase and extract words (sequences of alphanumeric characters)
    words = re.findall(r'[a-z0-9]+', text.lower())
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.
    
    Args:
        words: List of words.
        
    Returns:
        Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n_frequencies(counter: Counter, n: int) -> List[WordFrequency]:
    """
    Get the top N most frequent words.
    
    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.
        
    Returns:
        List of WordFrequency objects for the top N words.
    """
    top_items = counter.most_common(n)
    return [WordFrequency(word=word, count=count) for word, count in top_items]
