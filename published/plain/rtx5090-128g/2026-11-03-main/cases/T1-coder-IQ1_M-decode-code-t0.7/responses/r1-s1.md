"""Word frequency counter command-line tool.

Reads a text file, counts word occurrences (case-insensitive), and displays
the top N most frequent words in a table format.
"""

import argparse
import re
from collections import Counter
from collections import OrderedDict


def clean_word(word: str) -> str:
    """Strip punctuation and convert to lowercase."""
    return re.sub(r'[^a-z0-9]', '', word.lower())


def count_words(text: str) -> Counter:
    """Count word frequencies from raw text.

    Args:
        text: Raw text string.

    Returns:
        A Counter mapping cleaned words to their frequencies.
    """
    words = text.split()
    cleaned = [clean_word(w) for w in words if clean_word(w)]
    return Counter(cleaned)


def get_top_n(counter: Counter, n: int) -> list[tuple[str, int]]:
    """Return the top N most frequent words as (word, count) pairs.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)
