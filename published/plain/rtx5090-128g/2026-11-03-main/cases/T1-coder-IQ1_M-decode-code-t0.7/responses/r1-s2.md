#!/usr/bin/env python3
"""
Command-line tool to count word frequencies in a text file and display the top N most frequent words.

Usage:
    word_counter.py <input_file> [-n N] [--case-sensitive]

Examples:
    word_counter.py sample.txt
    word_counter.py sample.txt -n 10
"""

import argparse
import sys
from collections import Counter


def extract_words(text: str, case_sensitive: bool = False) -> list[str]:
    """
    Extract words from text. Words are defined as sequences of alphanumeric characters
    (including underscores for identifiers). By default, words are lowercased unless
    case_sensitive is True.

    Parameters:
        text: The input text to tokenize.
        case_sensitive: If True, preserve original case; otherwise, convert to lowercase.

    Returns:
        A list of extracted words.
    """
    words: list[str] = []
    current_word: list[str] = []
    for char in text:
        if char.isalnum() or char == '_':
            current_word.append(char)
        else:
            if current_word:
                word = ''.join(current_word)
                if not case_sensitive:
                    word = word.lower()
                words.append(word)
                current_word = []
    if current_word:
        word = ''.join(current_word)
        if not case_sensitive:
            word = word.lower()
        words.append(word)
    return words


def count_word_frequencies(text: str, case_sensitive: bool = False) -> Counter:
    """
    Count the frequency of each word in the given text.

    Parameters:
        text: The input text.
        case_sensitive: Whether to treat words as case-sensitive.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    words = extract_words(text, case_sensitive)
    return Counter(words)
