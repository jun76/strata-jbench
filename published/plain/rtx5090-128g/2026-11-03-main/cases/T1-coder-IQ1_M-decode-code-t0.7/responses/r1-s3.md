import argparse

def get_word_frequency(text: str) -> Counter:
    """
    Given a string of text, return a Counter object containing the frequency of each word.
    Words are normalized to lowercase and stripped of non-alphanumeric characters.
    
    Args:
        text (str): The input text string.
        
    Returns:
        Counter: A collections.Counter object with word frequencies.
    """
    # Normalize text to lowercase
    text = text.lower()
    
    # Remove non-alphanumeric characters (except spaces) and split into words
    cleaned_text = "".join(
        char if char.isalnum() or char.isspace() else " " for char in text
    )
    
    # Split into words and filter out empty strings
    words = [word for word in cleaned_text.split() if word]
    
    return Counter(words)
