def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。

    単語は、Unicode 文字・数字・アンダースコアからなる連続部分列と定義する。
    大文字小文字は無視して小文字に変換する。

    Args:
        text: 入力テキスト。

    Returns:
        単語のリスト（出現順）。
    """
    # \w は Unicode 対応（Python 3 標準）
    pattern = re.compile(r"[\w]+", re.UNICODE)
    return [word.lower() for word in pattern.findall(text)]
