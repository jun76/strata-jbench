#!/usr/bin/env python3
"""テキストファイルの単語出現頻度を上位 N 件で表示するコマンドラインツール."""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出し、小文字化してリストで返す.

    Args:
        text: 入力テキスト文字列.

    Returns:
        抽出された単語のリスト（すべて小文字）.
    """
    return re.findall(r"[a-zA-Z]+", text.lower())


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を Counter で返す.

    Args:
        text: 入力テキスト文字列.

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト.
    """
    words = extract_words(text)
    return Counter(words)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (単語, 回数) を返す.

    Args:
        counter: 単語出現頻度の Counter.
        n: 取得する件数.

    Returns:
        (単語, 出現回数) のタプルのリスト. 回数の降順.
    """
    if n <= 0:
        return []
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """(単語, 回数) のリストを表形式の文字列で返す.

    Args:
        rows: 表示する行のリスト.

    Returns:
        整形された表の文字列.
    """
    if not rows:
        return "No words found."

    word_width = max(len(word) for word, _ in rows)
