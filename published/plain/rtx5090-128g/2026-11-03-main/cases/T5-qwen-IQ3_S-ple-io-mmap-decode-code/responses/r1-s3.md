#!/usr/bin/env python3
"""
word_count.py

テキストファイルを読み込み、単語の出現頻度を計算し、上位 N 件を表形式で出力するコマンドラインツール。
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割し、小文字化して返す。
    非アルファベット・数字文字は区切り文字として扱う。

    Args:
        text: 入力テキスト

    Returns:
        正規化された単語のリスト
    """
    import re
    # 英数字の連続を単語として抽出し、小文字に変換
    words = re.findall(r'\w+', text.lower())
    return words


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト

    Returns:
        単語 -> 出現回数 の Counter
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (word, count) を返す。
    同点数の場合は辞書順で安定してソートする。

    Args:
        counter: 単語頻度の Counter
        n: 取得する件数

    Returns:
        上位 n 件の (word, count) のリスト
    """
    # Counter.most_common は同点数時の順序を保証しないため、
    # 明示的に (-count, word) でソートして安定性を確保
    items = counter.items()
    sorted_items = sorted(items, key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
