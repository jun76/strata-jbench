def count_word_frequencies(text: str) -> Counter:
    """テキストから単語の出現頻度をカウントする。

    Args:
        text: 解析対象のテキスト文字列

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    # 小文字化し、非英数字の文字をスペースで区切ることで単語分割
    cleaned = text.lower()
    # 単語抽出: 英数字の連続部分のみ抽出
    words: List[str] = []
    current_word: List[str] = []
    for char in cleaned:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    if current_word:
        words.append(''.join(current_word))
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counterから出現回数の上位N件を取得する。

    Args:
        counter: 単語頻度を表す Counter
        n: 取得する上位件数

    Returns:
        (単語, 出現回数) のタプルのリスト、出現回数の降順
    """
    return counter.most_common(n)
