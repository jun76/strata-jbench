#!/usr/bin/env python3
"""wordfreq: 単語出現頻度を集計し上位 N 件を表形式で表示する CLI ツール。

Usage:
    python wordfreq.py [-n TOP_N] [-d DELIMITERS] input.txt
"""
from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from typing import Iterable, List, Sequence, Tuple

# デフォルトの区切り文字（空白 + 区切り文字）
DEFAULT_DELIMITERS: str = r'[\s,.;:!?(){}\[\]<>"\'`/\-–—]+'


def tokenize(text: str, delimiters: str = DEFAULT_DELIMITERS) -> List[str]:
    """テキストを単語トークンに分割する。

    Args:
        text: 入力テキスト。
        delimiters: 区切り文字の正規表現パターン。

    Returns:
        単語リスト（アルファベット/数字のみ抽出、小文字化）。
    """
    # 区切り文字で分割し、空でない要素を抽出
    tokens = re.split(delimiters, text)
    return [tok.strip() for tok in tokens if tok.strip()]


def count_words(tokens: Iterable[str]) -> Counter:
    """トークン列から出現頻度を計算する。

    Args:
        tokens: 単語のイテラブル。

    Returns:
        単語 -> 出現回数の Counter。
    """
    return Counter(tokens)
