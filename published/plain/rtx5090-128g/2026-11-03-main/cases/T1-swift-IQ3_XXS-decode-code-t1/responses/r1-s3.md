# 単語出現頻度集計ツール (wordfreq)

以下に、Python 製の CLI ツール本体と pytest 用の単体テストを示します。

## 1. `wordfreq.py` — 本体

```python
#!/usr/bin/env python3
"""
wordfreq: 単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

使用例:
    $ python wordfreq.py data.txt
    $ python wordfreq.py data.txt -n 20 --min-count 3 --format json
"""

from __future__ import annotations

import argparse
import json
import re
import sys
from collections import Counter
from dataclasses import dataclass
from pathlib import Path
from typing import Literal


@dataclass(frozen=True)
class WordFreq:
    """単語と出現回数を保持する不変データクラス。"""
    word: str
    count: int


def tokenize(text: str) -> list[str]:
    """テキストを英字・数字・ハイフンを含むトークンに分割する。

    Args:
        text: 入力テキスト。

    Returns:
        正規化（小文字化）済みの単語リスト。
    """
    return re.findall(r"[a-zA-Z0-9'-]+", text.lower())


def compute_frequencies(text: str) -> Counter[str]:
    """テキスト内の単語出現回数を Counter で返す。"""
    return Counter(tokenize(text))


def top_n(
    freq: Counter[str],
    n: int,
    min_count: int = 1,
) -> list[WordFreq]:
    """出現回数が多い順に上位 N 件を返す。

    Args:
        freq: 単語ごとの出現回数 Counter。
        n: 返す最大件数。
        min_count: 表示対象の最小出現回数。

    Returns:
        WordFreq のリスト（出現回数降順、同数時は辞書順）。
    """
    items = [
        WordFreq(word=word, count=count)
        for word, count in freq.items()
        if count >= min_count
    ]
    # 出現回数降順 → 辞書順
    items.sort(key=lambda wf: (-wf.count, wf.word))
    return items[:n]
