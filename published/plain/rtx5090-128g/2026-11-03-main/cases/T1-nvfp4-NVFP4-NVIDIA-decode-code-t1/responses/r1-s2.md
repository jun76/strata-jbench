WORD_PATTERN = re.compile(r"[A-Za-z0-9]+(?:['-][A-Za-z0-9]+)*")


def normalize_word(word: str, ignore_case: bool = True, min_len: int = 1) -> str:
    """
    単語を正規化する。

    Args:
        word: 元の単語文字列。
        ignore_case: True の場合、小文字に変換する。
        min_len: この長さに満たない単語は空文字を返して除外対象とする。

    Returns:
        正規化された単語文字列。条件を満たさない場合は空文字。
    """
    result = word.lower() if ignore_case else word
    if len(result) < min_len:
        return ""
    return result
