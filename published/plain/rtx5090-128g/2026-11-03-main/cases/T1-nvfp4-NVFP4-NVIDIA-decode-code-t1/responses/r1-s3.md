<content>"""
pytest 単体テスト。

実行方法:
    python -m pytest wordfreq_tests.py -v
    （テストファイル名は wordfreq_tests.py で、実行用スクリプトは word_freq.py。）
"""

from __future__ import annotations

import pytest

import word_freq


# --------------------------------------------------------------------------- #
# extract_words のテスト
# --------------------------------------------------------------------------- #
def test_extract_words_basic() -> None:
    """句読点や大文字・小文字を正規化して単語を取り出すこと。"""
    assert word_freq.extract_words("Hello, world! Hello again.") == [
        "hello",
        "world",
        "hello",
        "again",
    ]


def test_extract_words_punctuation_only() -> None:
    """英数字以外が全て除去され、空リストになること。"""
    assert word_freq.extract_words("!!! ??? --- ...") == []


def test_extract_words_empty_string() -> None:
    """空文字列でも例外にならず空リストを返すこと。"""
    assert word_freq.extract_words("") == []


def test_extract_words_numeric_tokens() -> None:
    """数字のみで構成される語（例: '2024'）や '2024th' のような混在語も抽出されること。"""
    assert word_freq.extract_words("Year 2024, 2024th edition") == [
        "year",
        "2024",
        "2024th",
        "edition",
    ]


# --------------------------------------------------------------------------- #
# count_words のテスト
# --------------------------------------------------------------------------- #
def test_count_words_basic() -> None:
    """出現回数が正しく集計されること。"""
    assert word_freq.count_words("a b a C a b") == {"a": 3, "b": 2, "c": 1}


def test_count_words_empty_list() -> None:
    """空リストに対しては空の dict を返すこと。"""
    assert word_freq.count_words([]) == {}
