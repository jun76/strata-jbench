@dataclass(frozen=True)
class WordCount:
    """単一の単語とその出現回数。"""

    word: str
    count: int


def tokenize(text: str) -> list[str]:
    """
    テキストから英数字からなる単語を抽出する。

    正規表現 \\b[a-zA-Z0-9]+\\b を使用し、大文字は小文字に変換する。
    日本語や非ASCII文字列は考慮しない。

    Args:
        text: 対象テキスト。

    Returns:
        小文字化された単語リスト（重複あり）。
    """
    return re.findall(r"\b[a-zA-Z0-9]+\b", text.lower())


def count_words(text: str) -> Counter[str]:
    """
    テキスト内の単語出現回数をCounterで返す。

    Args:
        text: 対象テキスト。

    Returns:
        単語 -> 出現回数の Counter。
    """
    return Counter(tokenize(text))


def read_text(path: Path) -> str:
    """
    ファイルから UTF-8 テキストを読み込む。

    Args:
        path: 読み込み対象ファイルパス。

    Returns:
        ファイル内容（文字列）。

    Raises:
        FileNotFoundError: 存在しないパスの場合。
        UnicodeDecodeError: UTF-8 でデコードできない場合。
    """
    return path.read_text(encoding="utf-8")
