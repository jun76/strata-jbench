"""
コマンドラインツール: テキストファイルの単語出現頻度上位 N 件を表示
"""

from __future__ import annotations

import argparse
import sys
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    非アルphanumeric 文字を区切りとして使用し, 単語を lowercase 化.
    """
    # 単語を抽出: 連続する alphanumeric 文字
    words = re.findall(r"[a-zA-Z0-9]+", text)
    return [w.lower() for w in words]


def count_words(text: str) -> Counter:
    """
    テキスト中の単語出現頻度を Counter として返す。
    """
    return Counter(tokenize(text))


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位 n 件の (word, count) 一覧を返す。
    同頻度の場合, 単語の alphabetical order として安定.
    """
    # Counter.most_common(n) は同頻度時に任意順.
    # 安定性のため, 頻度降順 + 単語昇順で sort.
    items = sorted(
        counter.items(),
        key=lambda x: (-x[1], x[0])
    )
    return items[:n]
