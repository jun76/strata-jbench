#!/usr/bin/env python3
"""
Command-line tool to count word frequencies in a text file and display the top N results.

Usage:
    python word_counter.py <file_path> [--top N]

Example:
    python word_counter.py sample.txt --top 10
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple, Dict


def read_text_file(file_path: str) -> str:
    """
    Read the contents of a text file.

    Args:
        file_path: Path to the text file.

    Returns:
        The text content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If there is an error reading the file.
    """
    with open(file_path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize_words(text: str) -> List[str]:
    """
    Tokenize text into words, ignoring punctuation and case.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Extract sequences of alphabetic characters (letters only)
    words = re.findall(r"[a-zA-Z]+", text.lower())
    return words


def count_word_frequencies(words: List[str]) -> Dict[str, int]:
    """
    Count the frequency of each word in the list.

    Args:
        words: A list of words.

    Returns:
        A dictionary mapping each word to its count.
    """
    return Counter(words)


def get_top_n_words(freq: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words.

    Args:
        freq: A dictionary mapping words to their counts.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) sorted by count descending, then by word ascending.
    """
    sorted_items = sorted(freq.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
