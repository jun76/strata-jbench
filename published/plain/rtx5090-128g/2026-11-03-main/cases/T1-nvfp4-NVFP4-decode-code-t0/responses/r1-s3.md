word_freq.py - テキストファイルから単語の出現頻度を集計し、上位N件を表形式で表示するCLIツール。

Usage:
    python word_freq.py <file> [-n N] [--case-sensitive] [--min-length LEN]
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: Trueの場合、大文字小文字を区別する。Falseの場合、すべて小文字に変換する。
        min_length: この長さ未満の単語は除外する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()

    # 英数字とハイフン、アポストロフィを含む単語を抽出
    # 例: "don't" -> "don't", "state-of-the-art" -> "state-of-the-art"
    words = re.findall(r"[a-z0-9]+(?:['-][a-z0-9]+)*", text, re.IGNORECASE if case_sensitive else 0)

    # 最小長でフィルタ
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]

    return words
