word_freq.py - A command-line tool to count word frequencies in a text file.

This tool reads a text file, counts the occurrence of each word (case-insensitive,
punctuation stripped), and displays the top N most frequent words in a table format.

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive] [--min-length LENGTH]

Example:
    python word_freq.py sample.txt -n 10

def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Extract words from the given text.

    Words are defined as sequences of alphanumeric characters and underscores.
    Punctuation and other non-word characters are treated as separators.

    Args:
        text: The input text to extract words from.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.
        min_length: Minimum length of a word to be included.

    Returns:
        A list of extracted words.
    """
    # Use regex to find all word-like tokens (alphanumeric + underscore)
    words = re.findall(r'\b\w+\b', text)
    
    if not case_sensitive:
        words = [word.lower() for word in words]
    
    if min_length > 1:
        words = [word for word in words if len(word) >= min_length]
    
    return words
