#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイル内の単語出現頻度を上位N件で表示する。
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple, Dict


def normalize_word(word: str) -> str:
    """
    単語を正規化する。
    
    Args:
        word: 元の単語文字列
        
    Returns:
        正規化された単語（小文字化、句読点除去）
    """
    # 句読点や非英数字を除去し、小文字に変換
    cleaned = ''.join(char.lower() for char in word if char.isalnum())
    return cleaned


def count_words(text: str) -> Dict[str, int]:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 分析対象のテキスト文字列
        
    Returns:
        単語をキー、出現回数を値とする辞書
    """
    # 空白区切りで単語に分割
    raw_words = text.split()
    # 正規化して空でない単語のみをカウント
    normalized_words = [normalize_word(w) for w in raw_words]
    filtered_words = [w for w in normalized_words if w]
    
    return dict(Counter(filtered_words))
