#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルから単語出現頻度を集計し、
上位N件を表形式で表示する。

Usage:
    python word_freq.py <input_file> [-n TOP_N] [--case-sensitive]
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def load_and_count_words(
    filepath: str,
    case_sensitive: bool = False
) -> Counter[str]:
    """
    指定されたファイルからテキストを読み込み、単語の出現頻度をカウントする。

    Args:
        filepath: 読み込むテキストファイルのパス
        case_sensitive: 大文字小文字を区別するかどうか

    Returns:
        単語をキー、出現回数を値とするCounterオブジェクト
    """
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            text = f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except IOError as e:
        raise IOError(f"Error reading file: {filepath}: {e}")

    # 単語を抽出（空白や改行で分割し、前後の空白を除く）
    # 単語として扱うため、空白文字で分割する
    words = text.split()

    if not case_sensitive:
        words = [word.lower() for word in words]

    counter = Counter(words)
    return counter


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counterオブジェクトから出現回数の上位n件の単語と回数をリストで返す。

    Args:
        counter: 単語出現回数を保持するCounterオブジェクト
        n: 取得したい上位件数

    Returns:
        (単語, 出現回数) のタプルのリスト。出現回数が多い順に並ぶ。
    """
    if n <= 0:
        return []
    # most_common(n) は出現回数が多い順にn個返す
    return counter.most_common(n)
