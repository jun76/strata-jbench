#!/usr/bin/env python3
"""
word_count.py - Count word frequencies in a text file and display top N.

This command-line tool reads a text file, counts the occurrences of each word,
and displays the top N most frequent words in a formatted table.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase word tokens.
    """
    # Convert to lowercase and extract words (sequences of alphanumeric characters)
    return re.findall(r'[a-z0-9]+', text.lower())


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(tokenize(text))


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a counter.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
