"""
Command-line tool to count word frequencies from a text file and display the top N words.

Usage:
    python word_freq.py <input_file> [--top N]

Example:
    python word_freq.py sample.txt --top 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.
    
    Words are extracted using a regex that matches sequences of alphanumeric characters
    and underscores, then lowercased for case-insensitive counting.
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter object mapping each word to its frequency.
    """
    # Extract words: sequences of word characters (letters, digits, underscore)
    words = re.findall(r'\b\w+\b', text.lower())
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.
        
    Returns:
        A list of tuples (word, count), sorted by count descending.
    """
    return counter.most_common(n)
