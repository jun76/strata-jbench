# 単語頻度集計コマンドラインツール

以下に、Python でテキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する CLI ツールと、対応する pytest 単体テストを提供します。

---

## 1. メインスクリプト: `word_freq.py`

#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語の出現頻度を上位N件表示するCLIツール.

使い方:
    python word_freq.py input.txt
    python word_freq.py input.txt --top 10
    python word_freq.py input.txt --top 5 --case-sensitive
    python word_freq.py input.txt --format json
"""

from __future__ import annotations

import argparse

import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語トークンを抽出する.

    Args:
        text: 解析対象のテキスト.
        case_sensitive: 大文字小文字を区別するかどうか.

    Returns:
        単語トークンのリスト.
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    tokens = re.findall(r"[a-zA-Z0-9]+(?:['\-][a-zA-Z0-9]+)*", text)
    return tokens


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする.

    Args:
        text: 解析対象のテキスト.
        case_sensitive: 大文字小文字を区別するかどうか.

    Returns:
        単語 -> 出現回数の Counter.
    """
    tokens = tokenize(text, case_sensitive=case_sensitive)
    return Counter(tokens)
