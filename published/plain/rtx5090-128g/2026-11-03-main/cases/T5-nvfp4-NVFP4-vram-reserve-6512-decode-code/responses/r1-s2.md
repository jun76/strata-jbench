#!/usr/bin/env python3
"""単語出現頻度解析ツール.

テキストファイルを読み込み、単語の出現頻度を集計し、
上位 N 件を表形式で表示するコマンドラインツール。
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def load_text(filepath: str | Path) -> str:
    """ファイルからテキストを読み込む.

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        PermissionError: ファイルを読み取る権限がない場合。
    """
    path = Path(filepath)
    return path.read_text(encoding="utf-8")


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する.

    英数字のみを単語として抽出し、すべて小文字に変換する。

    Args:
        text: 分割対象のテキスト。

    Returns:
        小文字化された単語のリスト。
    """
    words = re.findall(r"[a-zA-Z0-9]+", text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter[str]:
    """単語の出現頻度をカウントする.

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
