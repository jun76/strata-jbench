#!/usr/bin/env python3
"""
wordfreq - Count word frequencies in text files and display top N results.

This command-line tool reads one or more text files, tokenizes the content
into words (lowercased, stripped of punctuation), counts occurrences, and
displays the top N most frequent words in a formatted table.
"""

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from typing import List, Tuple


@dataclass(frozen=True)
class WordCount:
    """Represents a word and its frequency count."""
    word: str
    count: int


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.
    
    Words are defined as sequences of alphanumeric characters (including
    underscores). Consecutive non-word characters act as separators.
    
    Args:
        text: The input text to tokenize.
        
    Returns:
        A list of lowercase words found in the text.
    """
    # Extract words: sequences of word characters (letters, digits, underscores)
    words = re.findall(r'\w+', text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word.
    
    Args:
        words: A list of words.
        
    Returns:
        A Counter mapping words to their counts.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[WordCount]:
    """
    Get the top N most frequent words.
    
    If multiple words have the same count, they are sorted alphabetically.
    
    Args:
        counter: A Counter of word frequencies.
        n: The number of top entries to return.
        
    Returns:
        A list of WordCount objects, sorted by count descending, then word ascending.
    """
    # Sort by count descending, then word ascending for deterministic output
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return [WordCount(word=word, count=count) for word, count in items[:n]]
