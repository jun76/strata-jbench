#!/usr/bin/env python3
"""コマンドラインツール：テキストファイルの単語出現頻度を集計し、上位 N 件を表形式で表示する。"""

import argparse
import re
import sys
from collections import Counter
from typing import Dict, List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを小文字の単語に分割する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    return re.findall(r'\b[a-zA-Z]+\b', text.lower())


def count_words(text: str) -> Dict[str, int]:
    """テキスト中の単語出現頻度を集計する。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とする辞書。
    """
    return Counter(tokenize(text))


def top_n_words(counter: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """上位 N 件の単語と出現回数を取り出す。

    出現回数の降順でソートし、同じ頻度の場合は辞書順で昇順になる。

    Args:
        counter: 単語と出現回数の辞書。
        n: 上位件数。

    Returns:
        (単語, 出現回数) のリスト。
    """
    return sorted(counter.items(), key=lambda x: (-x[1], x[0]))[:n]
