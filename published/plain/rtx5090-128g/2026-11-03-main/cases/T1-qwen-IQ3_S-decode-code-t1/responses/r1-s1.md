import argparse
import re
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """テキストから単語を抽出し、出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    # 単語を抽出: アルファベット、数字、ハイフンを含む連続文字列（ただし先頭/末尾のハイフンは除く）
    words = re.findall(r'[a-zA-Z0-9]+(?:[-\''][a-zA-Z0-9]+)*', text.lower())
    return Counter(words)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (単語, 回数) のリストを返す。

    Args:
        counter: 単語頻度の Counter
        n: 返す上位件数

    Returns:
        (単語, 出現回数) のタプルのリスト。頻度降順、同頻度時は辞書順でソート。
    """
    # most_commonは頻度降順。同頻度時は挿入順だが、我々は辞書順にしたい
    items = counter.items()
    sorted_items = sorted(items, key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
