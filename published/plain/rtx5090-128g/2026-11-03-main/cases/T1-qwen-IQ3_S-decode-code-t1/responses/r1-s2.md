import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def read_text(filepath: str) -> str:
    """指定されたファイルからテキストを読み込む。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。

    単語は英数字のみを含む文字列として定義される。
    大文字小文字は区別せず、小文字に変換される。

    Args:
        text: 入力テキスト。

    Returns:
        抽出された単語のリスト。
    """
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_words(words: List[str]) -> Counter:
    """単語リストから頻度を計算する。

    Args:
        words: 単語のリスト。

    Returns:
        単語と頻度をキーと値とするCounterオブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """頻度が上位N件の単語とその頻度を返す。

    Args:
        counter: 単語頻度を表すCounterオブジェクト。
        n: 上位件数。

    Returns:
        (単語, 頻度) のタプルのリスト。
    """
    return counter.most_common(n)
