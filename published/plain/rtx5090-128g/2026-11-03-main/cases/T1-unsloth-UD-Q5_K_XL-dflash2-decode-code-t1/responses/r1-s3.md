_WORD_PATTERN = re.compile(r"[A-Za-z0-9']+")


def tokenize(text: str, *, ignore_case: bool = True) -> list[str]:
    """テキストを単語に分割し、正規化する。

    Args:
        text: 入力テキスト。
        ignore_case: True の場合、単語を小文字に変換する。

    Returns:
        単語のリスト（正規化済み）。
    """
    words = _WORD_PATTERN.findall(text)
    if ignore_case:
        return [w.lower() for w in words]
    return words
