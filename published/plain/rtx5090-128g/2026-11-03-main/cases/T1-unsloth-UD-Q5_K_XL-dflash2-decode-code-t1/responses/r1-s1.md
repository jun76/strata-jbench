#!/usr/bin/env python3

Top Words CLI - Count word frequencies in a text file and display top N results.

Usage:
    python top_words.py <input_file> [-n N] [--min-length M] [--ignore-case]
"""

import argparse
import re

from collections import Counter
from typing import List, Tuple


def tokenize(text: str, ignore_case: bool = True, min_length: int = 1) -> List[str]:
    """Tokenize text into words.
    
    Args:
        text: Input text to tokenize.
        ignore_case: If True, convert all characters to lowercase.
        min_length: Minimum length of a token to include.
    
    Returns:
        List of tokens extracted from the text.
    """
    if ignore_case:
        text = text.lower()
    # Extract sequences of alphanumeric characters (including underscores)
    tokens = re.findall(r'\b\w+\b', text)
    # Filter by minimum length
    tokens = [t for t in tokens if len(t) >= min_length]
    return tokens


def count_words(text: str, ignore_case: bool = True, min_length: int = 1) -> Counter:
    """Count word frequencies in the given text.
    
    Args:
        text: Input text to analyze.
        ignore_case: If True, treat words case-insensitively.
        min_length: Minimum length of words to count.
    
    Returns:
        Counter object with word frequencies.
    """
    tokens = tokenize(text, ignore_case=ignore_case, min_length=min_length)
    return Counter(tokens)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a counter.
    
    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.
    
    Returns:
        List of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)
