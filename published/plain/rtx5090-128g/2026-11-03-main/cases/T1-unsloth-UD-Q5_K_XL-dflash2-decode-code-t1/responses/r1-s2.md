"""wordfreq: Count word frequencies in a text file and display top N results."""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """Count word occurrences in the given text.

    Words are defined as sequences of alphanumeric characters and underscores,
    extracted using regex \w+. The count is case-insensitive (lowercased).

    Args:
        text: The input text string to analyze.

    Returns:
        A Counter object mapping lowercase words to their occurrence counts.
    """
    words = re.findall(r'\w+', text.lower())
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words from the counter.

    Args:
        counter: A Counter mapping words to their counts.
        n: Number of top results to return.

    Returns:
        A list of tuples (word, count) sorted by count descending, then by
        word ascending for ties.
    """
    # Get all items, sort by count descending, then word ascending
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]


def format_table(top_results: List[Tuple[str, int]]) -> str:
    """Format the top results as a table string.

    Args:
        top_results: A list of tuples (word, count).

    Returns:
        A formatted string with header and rows, aligned in columns.
    """
    if not top_results:
        return "No words found."

    # Calculate column widths
    word_width = max(len(word) for word, _ in top_results)
    word_width = max(word_width, len("Word"))
    count_width = max(len(str(count)) for _, count in top_results)
    count_width = max(count_width, len("Count"))
    rank_width = len(str(len(top_results)))

    # Build header
    header_rank = "Rank".rjust(rank_width)
    header_word = "Word".ljust(word_width)
    header_count = "Count".rjust(count_width)
    header = f"{header_rank} | {header_word
