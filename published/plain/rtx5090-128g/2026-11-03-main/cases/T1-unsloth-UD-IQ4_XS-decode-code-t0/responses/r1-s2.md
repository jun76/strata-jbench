#!/usr/bin/env python3
"""
word_freq.py - A command-line tool to count word frequencies in a text file.

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Extract words from the given text.

    Args:
        text: The input text string.
        case_sensitive: If False, convert all words to lowercase.

    Returns:
        A list of words extracted from the text.
    """
    if not case_sensitive:
        text = text.lower()
    # Match sequences of alphanumeric characters and underscores as words
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word in the list.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) sorted by count descending.
    """
    return counter.most_common(n)
