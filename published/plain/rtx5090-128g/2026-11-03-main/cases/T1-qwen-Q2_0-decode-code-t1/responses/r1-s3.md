#!/usr/bin/env python3
"""Text file word frequency analysis command-line tool."""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_file(path: str) -> str:
    """Read text content from a file.

    Args:
        path: Path to the text file.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        OSError: If there is an error reading the file.
    """
    file_path = Path(path)
    return file_path.read_text(encoding="utf-8")


def tokenize(text: str) -> List[str]:
    """Tokenize text into words.

    Words are sequences of alphanumeric characters (including underscores).
    Case is normalized to lowercase for counting purposes.

    Args:
        text: The text to tokenize.

    Returns:
        A list of lowercase word tokens.
    """
    tokens = []
    current = []
    for ch in text:
        if ch.isalnum() or ch == "_":
            current.append(ch)
        else:
            if current:
                tokens.append("".join(current).lower())
                current = []
    if current:
        tokens.append("".join(current).lower())
    return tokens


def count_words(tokens: List[str]) -> Counter:
    """Count word occurrences.

    Args:
        tokens: List of word tokens.

    Returns:
        A Counter mapping words to their occurrence counts.
    """
    return Counter(tokens)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words.

    Args:
        counter: Word frequency counter.
        n: Number of top entries to retrieve.

    Returns:
        A list of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)
