def read_text_file(filepath: str) -> str:
    """Read a text file and return its contents as a string.
    
    Args:
        filepath: Path to the text file to read.
        
    Returns:
        The contents of the file as a string.
    """
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except IOError as e:
        raise IOError(f"Error reading file: {filepath}: {e}")


def extract_words(text: str) -> List[str]:
    """Extract words from text by splitting on whitespace and punctuation.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of words extracted from the text.
    """
    # Split on whitespace and common punctuation
    import re
    # Normalize: lowercase and extract word-like sequences
    words = re.findall(r"[a-zA-Z]+", text.lower())
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """Count the frequency of each word.
    
    Args:
        words: List of words to count.
        
    Returns:
        A Counter object mapping words to their frequency.
    """
    return Counter(words)


def get_top_n_words(counter: Counter, n: int) -> List[tuple]:
    """Get the top N most frequent words.
    
    Args:
        counter: A Counter object with word frequencies.
        n: Number of top words to return.
        
    Returns:
        A list of tuples (word, count) sorted by frequency descending, then by word ascending.
    """
    # Sort by count descending, then by word ascending for tie-breaking
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
