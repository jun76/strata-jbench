#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位 N 件の単語とその頻度を表形式で出力するコマンドラインツール。

Usage:
    python word_counter.py <input_file> [-n N] [--case-sensitive] [--min-length LEN]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
        min_length: 抽出する単語の最小文字数。

    Returns:
        単語のリスト。
    """
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    # 例: "don't", "well-known" なども単語として扱う
    pattern = r"[a-zA-Z0-9]+(?:['-][a-zA-Z0-9]+)*"
    words = re.findall(pattern, text)

    if not case_sensitive:
        words = [word.lower() for word in words]

    # 最小文字数でフィルタリング
    if min_length > 1:
        words = [word for word in words if len(word) >= min_length]

    return words


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
        min_length: 抽出する単語の最小文字数。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(words)
