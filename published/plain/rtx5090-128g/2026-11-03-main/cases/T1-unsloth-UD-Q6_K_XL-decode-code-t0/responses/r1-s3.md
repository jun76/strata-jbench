"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位 N 件の単語を表形式で出力するコマンドラインツール。
"""

import argparse

from collections import Counter

def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    英数字の連続を単語とし、大文字は小文字に変換する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語のリスト
    """
    # 英数字の連続を抽出し、小文字に変換
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 N 件の単語を返す。
    頻度が同じ場合は辞書順でソートする。
    
    Args:
        counter: 単語出現頻度の Counter
        n: 取得する件数
        
    Returns:
        (単語, 出現回数) のタプルのリスト
    """
    # 頻度降順、同頻度なら単語の辞書順でソート
    sorted_items = sorted(
        counter.items(),
        key=lambda x: (-x[1], x[0])
    )
    return sorted_items[:n]
