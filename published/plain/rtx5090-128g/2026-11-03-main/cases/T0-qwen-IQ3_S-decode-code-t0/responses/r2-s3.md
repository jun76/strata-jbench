word_freq.py - A command-line tool to count word frequencies in a text file
and display the top N most frequent words in a table format.

def read_file(filepath: str) -> str:
    """
    Read the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If there is an error reading the file.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words, ignoring punctuation and non-alphanumeric characters.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens.
    """
    # Extract words consisting of alphanumeric characters and underscores
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of word tokens.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) for the top N most common words.
    """
    return counter.most_common(n)
