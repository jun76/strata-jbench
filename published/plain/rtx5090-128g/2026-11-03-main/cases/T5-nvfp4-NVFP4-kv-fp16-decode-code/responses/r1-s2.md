"""
Word frequency counter command-line tool.

This module provides a command-line tool that reads a text file,
counts word occurrences, and displays the top N most frequent words
in a formatted table.
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_file(filepath: str) -> str:
    """Read and return the content of a text file.

    Args:
        filepath: Path to the text file to read.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If there is an error reading the file.
    """
    path = Path(filepath)
    if not path.exists():
        raise FileNotFoundError(f"File not found: {filepath}")
    try:
        return path.read_text(encoding="utf-8")
    except IOError as e:
        raise IOError(f"Error reading file: {e}")


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words, stripping punctuation.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase words with punctuation removed.
    """
    # Convert to lowercase and replace common punctuation with spaces
    cleaned = text.lower()
    for char in [".", ",", "!", "?", ";", ":", "'", '"', "(", ")", "[", "]", "{", "}", "-", "/", "\\", "@", "#", "$", "%", "^", "&", "*", "+", "=", "~", "`", "<", ">", "|"]:
        cleaned = cleaned.replace(char, " ")
    words = cleaned.split()
    return words


def count_words(text: str) -> Counter:
    """Count word occurrences in the given text.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object with word frequencies.
    """
    words = tokenize(text)
    return Counter(words)
