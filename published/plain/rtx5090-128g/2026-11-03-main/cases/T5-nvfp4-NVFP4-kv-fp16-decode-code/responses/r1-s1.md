def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか。Falseの場合、すべて小文字に変換する。

    Returns:
        抽出された単語のリスト
    """
    if not case_sensitive:
        text = text.lower()
    
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    # 正規表現: \b で単語の境界を指定し、[\w'-]+ で単語文字列をマッチ

    return words

        単語をキー、出現回数を値とする Counter オブジェクト
    """
    return Counter(words)
