#!/usr/bin/env python3
"""
Command-line tool to count word frequencies in a text file and display top N words.

Usage:
    word_freq.py <input_file> [-n N]

Example:
    word_freq.py sample.txt -n 10
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.
    
    Args:
        text: Input text string.
    
    Returns:
        List of lowercase word tokens.
    """
    # Extract words using regex (sequences of alphanumeric characters)
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.
    
    Args:
        text: Input text string.
    
    Returns:
        Counter object mapping words to their frequencies.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: Counter object with word frequencies.
        n: Number of top items to return.
    
    Returns:
        List of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)


def read_file(filepath: str) -> str:
    """
    Read the contents of a text file.
    
    Args:
        filepath: Path to the text file.
    
    Returns:
        Contents of the file as a string.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def main() -> None:
    """Main entry point for the command-line tool."""
    parser = argparse.ArgumentParser(
        description='Count word frequencies in a text file and display top N words.'
    )
    parser.add_argument(
        'input_file',
        type=str,
        help='Path to the input text file.'
    )
    parser.add_argument(
        '-n',
        '--top',
        type=int,
        default=1
