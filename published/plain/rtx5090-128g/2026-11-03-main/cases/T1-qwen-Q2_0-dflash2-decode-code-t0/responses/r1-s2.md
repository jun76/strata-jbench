def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_sensitive: 大小写区分（False 则统一转小写）
        min_length: 单词最小长度（过滤短词）

    Returns:
        单词列表
    """
    if not case_sensitive:
        text = text.lower()

    # 简单分词：按非字母字符分割，保留纯字母单词
    words = []
    current = []
    for ch in text:
        if ch.isalpha():
            current.append(ch)
        else:
            if current:
                word = ''.join(current)
                if len(word) >= min_length:
                    words.append(word)
                current = []
    if current:
        word = ''.join(current)
        if len(word) >= min_length:
            words.append(word)

    return words


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """
    统计文本中单词出现次数。

    Args:
        text: 输入文本
        case_sensitive: 大小写区分
        min_length: 单词最小长度

    Returns:
        Counter 对象，键为单词，值为出现次数
    """
    words = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(words)
