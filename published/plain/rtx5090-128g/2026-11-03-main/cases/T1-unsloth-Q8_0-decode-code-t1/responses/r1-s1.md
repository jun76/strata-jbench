#!/usr/bin/env python3
"""
wordfreq: A command-line tool to count word frequencies from a text file
and display the top N words in a table format.

Usage:
    python wordfreq.py <input_file> [-n TOP_N] [--sort-order {desc,asc}]

Examples:
    python wordfreq.py text.txt -n 10
    python wordfreq.py text.txt --sort-order asc -n 5
"""

import argparse

import re
from collections import Counter
from typing import List, Tuple, Dict


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into words (lowercase, alphanumeric only).

    Args:
        text: Input text string.

    Returns:
        List of tokenized words (lowercase, alphanumeric characters only).
    """
    # Match sequences of alphanumeric characters (words)
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        Counter object with word counts.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int, sort_order: str = 'desc') -> List[Tuple[str, int]]:
    """
    Get the top N words from the counter, sorted by count.

    Args:
        counter: Counter object with word counts.
        n: Number of top words to return.
        sort_order: 'desc' for descending order (most frequent first),
                    'asc' for ascending order (least frequent first).

    Returns:
        List of (word, count) tuples, sorted by count.
    """
    if sort_order == 'desc':
        # Most frequent first
        items = counter.most_common(n)
    elif sort_order == 'asc':
        # Least frequent first, then by alphabetical order for ties
        sorted_items = sorted(counter.items(), key=lambda x: (x[1], x[0]))
        items = sorted_items[:n]
    else:
        raise ValueError(f"Unknown sort order: {sort_order}")

    return items
