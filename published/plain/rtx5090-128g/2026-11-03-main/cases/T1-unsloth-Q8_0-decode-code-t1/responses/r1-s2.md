#!/usr/bin/env python3
"""
word_freq.py - A command-line tool to count word frequencies in text files.

Reads one or more text files, tokenizes the content into words,
counts the occurrence of each word (case-insensitive), and displays
the top N most frequent words in a formatted table.
"""

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from typing import Optional


@dataclass
class WordCount:
    """Represents a word and its frequency."""
    word: str
    count: int


def tokenize(text: str) -> list[str]:
    """
    Tokenize a string into lowercase alphanumeric words.
    
    Args:
        text: The input text to tokenize.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    # Extract sequences of alphanumeric characters
    return re.findall(r"[a-z0-9]+", text.lower())


def count_words(tokens: list[str]) -> Counter[str]:
    """
    Count the occurrences of each token.
    
    Args:
        tokens: A list of tokens (words).
        
    Returns:
        A Counter mapping each token to its count.
    """
    return Counter(tokens)


def top_n_words(counter: Counter[str], n: int) -> list[WordCount]:
    """
    Get the top N most common words from a counter.
    
    Args:
        counter: A Counter mapping words to their frequencies.
        n: The number of top words to return.
        
    Returns:
        A list of WordCount objects, sorted by frequency (descending).
    """
    most_common = counter.most_common(n)
    return [WordCount(word=word, count=count) for word, count in most_common]


def format_table(items: list[WordCount], header: Optional[list[str]] = None) -> str:
    """
    Format a list of WordCount items as a text table.
    
    Args:
        items: A list of WordCount objects.
        header: Optional custom header column names. Defaults to ["Rank", "Word", "Count"].
        
    Returns:
        A formatted string representing the table.
    """
    if header is None:
        header = ["Rank", "Word", "Count
