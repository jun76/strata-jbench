#!/usr/bin/env python3
"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a tabular format.
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase alphanumeric words.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter object mapping each word to its occurrence count.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top entries to return.
        
    Returns:
        A list of (word, count) tuples sorted by count descending,
        then alphabetically for ties.
    """
    # Sort by count descending, then word ascending for stable ordering
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
