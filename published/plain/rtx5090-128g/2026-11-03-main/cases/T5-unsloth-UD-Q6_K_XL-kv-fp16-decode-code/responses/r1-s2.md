#!/usr/bin/env python3
"""Top N word frequency counter.

A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.
"""

import argparse
import re

from collections import Counter

def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    return re.findall(r'[a-zA-Z]+', text.lower())


def count_words(words: List[str]) -> Counter:
    """Count word frequencies.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        A list of tuples (word, count) sorted by frequency descending.
    """
    return counter.most_common(n)


def format_table(words: List[Tuple[str, int]]) -> str:
    """Format word frequency data as a table.

    Args:
        words: A list of tuples (word, count).

    Returns:
        A formatted table string.
    """
    if not words:
        return "No words found."

    # Determine column widths
    word_width = max(len(word) for word, _ in words)
    count_width = max(len(str(count)) for _, count in words)

    # Header
    header_word = "Word".ljust(word_width)
    header_count = "Count".rjust(count_width)
    separator = "-" * (word_width + count_width + 3)

    lines = [header_word, header_count]
    lines.append(separator)

    for word, count in words:
        line = f"{word.ljust(word_width)}   {str(count).rjust(count_width)}"
        lines.append(line)

    return "\n".join(lines)
