"""
wordfreq - Count word frequencies in a text file and display top N words.
"""

import argparse
import re

from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.
    
    Words are defined as sequences of alphanumeric characters and apostrophes.
    Punctuation surrounding words is stripped.
    
    Args:
        text: Input text string.
        
    Returns:
        List of lowercase word tokens.
    """
    # Match sequences of word characters including internal apostrophes
    words = re.findall(r"[a-zA-Z0-9]+(?:'[a-zA-Z0-9]+)*", text)
    return [word.lower() for word in words]


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.
    
    Args:
        text: Input text string.
        
    Returns:
        Counter object with word frequencies.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: Counter object with word frequencies.
        n: Number of top items to return.
        
    Returns:
        List of tuples (word, count) sorted by count descending, then word ascending.
    """
    # most_common returns items sorted by count descending.
    # For ties, we need secondary sort by word ascending.
    items = counter.most_common()
    # Sort by count descending, then by word ascending
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
