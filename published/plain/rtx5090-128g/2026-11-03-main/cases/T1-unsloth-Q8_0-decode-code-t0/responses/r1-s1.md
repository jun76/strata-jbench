#!/usr/bin/env python3
"""
word_freq.py - A command-line tool to count word frequencies in a text file.

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive] [--min-length MIN]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Extract words from the given text.

    Args:
        text: The input text string.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.
        min_length: Minimum length of a word to be included.

    Returns:
        A list of extracted words.
    """
    if not case_sensitive:
        text = text.lower()
    # Extract sequences of alphanumeric characters (words)
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    # Filter by minimum length
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word in the list.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) sorted by count descending, then alphabetically.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
