#!/usr/bin/env python3
"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度を計算して上位 N 件を表形式で表示するコマンドラインツール。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    英数字のみを含むトークンを抽出し、小文字に変換する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語のリスト
    """
    # 英数字のみを含むトークンを抽出し、小文字に変換
    return [word.lower() for word in re.findall(r'\b[a-zA-Z0-9]+\b', text)]


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 N 件を取得する。
    頻度が同じ場合は辞書順でソートする。
    
    Args:
        counter: 単語頻度の Counter
        n: 取得する件数
        
    Returns:
        (単語, 頻度) のタプルのリスト
    """
    # 頻度降順、同頻度なら単語の辞書順でソート
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
