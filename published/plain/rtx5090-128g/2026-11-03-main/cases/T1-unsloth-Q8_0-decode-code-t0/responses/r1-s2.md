#!/usr/bin/env python3
"""
word_freq.py - A command-line tool to count word frequencies in a text file.

This module provides functionality to read a text file, count the frequency
of each word, and display the top N most frequent words in a table format.

Usage:
    python word_freq.py <input_file> [-n TOP_N] [--case-sensitive]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If True, preserve case. If False, convert to lowercase.

    Returns:
        A list of words (tokens) extracted from the text.
    """
    if not case_sensitive:
        text = text.lower()
    # Extract words: sequences of alphanumeric characters and underscores
    words = re.findall(r'\b\w+\b', text)
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words to count.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) for the top N most common words.
    """
    return counter.most_common(n)

    """
