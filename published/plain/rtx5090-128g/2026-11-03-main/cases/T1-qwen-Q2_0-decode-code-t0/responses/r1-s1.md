"""
word_freq.py - テキストファイルの単語出現頻度集計ツール

このツールは指定されたテキストファイルを読み込み、
単語の出現頻度を集計し、上位 N 件を表形式で表示します。

Usage:
    python word_freq.py <file> [--top N] [--case-sensitive] [--min-length L]
"""

import argparse

def read_file(path: str) -> str:
    """指定されたファイルからテキストを読み込む。

    Args:
        path: 読み込むファイルのパス。

    Returns:
        ファイルの内容 (str)。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み込みエラーの場合。
    """
    try:
        with open(path, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {path}")
    except IOError as e:
        raise IOError(f"Error reading file {path}: {e}")


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """テキストを単語に分割する。

    Args:
        text: 分割対象のテキスト。
        case_sensitive: 大小写を区別するか。False なら小写に統一。
        min_length: 単語の最小長。これ未満の単語は除外。

    Returns:
        単語のリスト。
    """
    # 非文字・数字を空白で分割
    # 単語 = 文字・数字・ハイphen・apostrophe を含む連続部分
    import re
    pattern = r"[a-zA-Z0-9'-]+"
    words = re.findall(pattern, text)

    if not case_sensitive:
        words = [w.lower() for w in words]

    words = [w for w in words if len(w) >= min_length]
    return words
