def read_file(path: str) -> str:
    """Read the contents of a text file.

    Args:
        path: Path to the text file.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """Tokenize text into words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If True, preserve case; otherwise, lowercase all words.
        min_length: Minimum length of a word to include.

    Returns:
        A list of words extracted from the text.
    """
    # Extract words using regex: sequences of alphanumeric characters
    words = re.findall(r"[a-zA-Z0-9]+", text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    words = [w for w in words if len(w) >= min_length]
    return words
