#!/usr/bin/env python3
"""Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.

    Args:
        text: Input text string.

    Returns:
        List of lowercase word tokens.
    """
    return re.findall(r'[a-zA-Z]+', text.lower())


def count_words(text: str) -> Counter:
    """Count word occurrences in text.

    Args:
        text: Input text string.

    Returns:
        Counter object with word frequencies.
    """
    return Counter(tokenize(text))


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        List of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)


def format_table(data: List[Tuple[str, int]]) -> str:
    """Format word-frequency pairs as a table.

    Args:
        data: List of (word, count) tuples.

    Returns:
        Formatted table string.
    """
    if not data:
        return "No words found."

    # Calculate column widths
    word_width = max(len(word) for word, _ in data)
    count_width = max(len(str(count)) for _, count in data)
    rank_width = len(str(len(data)))

    # Create header
    header = f"{'Rank':>{rank_width}} | {'Word':<{word_width}} | {'Count':>{count_width}}"
    separator = f"{'-' * rank_width} | {'-' * word_width} | {'-' * count_width}"

    lines = [header, separator]
    for i, (word, count) in enumerate(data, 1):
        line = f"{i:>{rank_width}} | {word:<{word_width}} | {count:>{count_width}}"
        lines.append(line)
