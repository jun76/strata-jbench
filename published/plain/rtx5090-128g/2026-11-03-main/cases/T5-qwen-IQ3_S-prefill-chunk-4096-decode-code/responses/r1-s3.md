def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    英数字の文字列のみを単語とみなし、大文字小文字は無視する。

    Args:
        text (str): 入力テキスト

    Returns:
        List[str]: 単語のリスト（小文字化済み）
    """
    # 簡易的なトークン化: 英数字のみ抽出
    words = []
    current_word = []
    for char in text:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                words.append(''.join(current_word).lower())
                current_word = []
    if current_word:
        words.append(''.join(current_word).lower())
    return words


def count_words(text: str) -> Counter:
    """
    テキスト内の単語の出現頻度をカウントする。

    Args:
        text (str): 入力テキスト

    Returns:
        Counter: 単語をキー、出現回数を値とするカウント辞書
    """
    words = tokenize(text)
    return Counter(words)
