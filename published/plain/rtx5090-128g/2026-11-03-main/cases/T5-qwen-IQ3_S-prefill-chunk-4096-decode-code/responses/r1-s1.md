#!/usr/bin/env python3
"""
word_freq: A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Words are extracted by splitting on non-alphanumeric characters.
    All words are converted to lowercase for case-insensitive counting.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to retrieve.

    Returns:
        A list of tuples, each containing a word and its count,
        sorted by count descending.
    """
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """
    Format the top N words as a table string.

    Args:
        rows: A list of (word, count) tuples.

    Returns:
        A formatted table string with headers and aligned columns.
    """
    if not rows:
        return "No words found."

    # Calculate column widths
    word_width = max(len(word) for word, _ in rows)
    count_width = max(len(str(count)) for _, count in rows)
    # Ensure headers fit
    word_width = max(word_width, len("Word"))
    count_width = max(count_width, len("Count"))

    # Build table
    header = f"{'Word'.ljust(word_width)} | {'Count'.rjust(count_width)}"
    separator = "-" * word_width + " | " + "-" * count_width

    lines = [header, separator]
    for word, count in rows:
        lines.append(f"{word.ljust(word_width)} | {str(count).rjust(count_width
