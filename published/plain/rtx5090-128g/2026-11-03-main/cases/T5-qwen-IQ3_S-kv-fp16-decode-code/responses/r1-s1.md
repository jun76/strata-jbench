#!/usr/bin/env python3
"""
word_freq.py - Count word frequencies in text files and display top N results.

Usage:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt -n 20 --case-insensitive --sort-desc
"""

import argparse
import sys
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def parse_arguments(args: List[str] | None = None) -> argparse.Namespace:
    """Parse command-line arguments for the word frequency tool.

    Args:
        args: Optional list of argument strings. If None, reads from sys.argv.

    Returns:
        argparse.Namespace containing parsed arguments.
    """
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and display top N results."
    )
    parser.add_argument(
        "input_file",
        type=str,
        help="Path to the input text file.",
    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=10,
        help="Number of top words to display (default: 10).",
    )
    parser.add_argument(
        "--case-insensitive",
        action="store_true",
        help="Treat words as case-insensitive (default: False).",
    )
    parser.add_argument(
        "--sort-asc",
        action="store_true",
        help="Sort results in ascending order of frequency (default: descending).",
    )
    parser.add_argument(
        "--min-length",
        type=int,
        default=1,
        help="Minimum word length to include (default: 1).",
    )
    parser.add_argument(
        "--max-length",
        type=int,
        default=1000,
        help="Maximum word length to include (default: 1000).",
    )

    return parser.parse_args(args)
