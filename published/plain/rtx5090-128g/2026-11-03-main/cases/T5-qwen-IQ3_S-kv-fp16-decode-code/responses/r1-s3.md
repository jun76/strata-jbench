### 1. `wordfreq.py`

```python
#!/usr/bin/env python3
"""
wordfreq.py

テキストファイルを読み込み、単語の出現頻度を計算し、上位 N 件を表形式で出力するコマンドラインツール。
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割し、小文字化して返す。
    非アルファベット文字は区切りとして扱われる。
    """
    import re
    # 英数字とアンダースコア以外の文字で分割し、空文字列を除外
    tokens = re.findall(r'[a-zA-Z0-9_]+', text.lower())
    return tokens


def count_words(text: str) -> Counter:
    """
    テキスト内の単語の出現頻度をカウントする。
    """
    return Counter(tokenize(text))


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度が上位 n 件の (word, count) タプルをリストで返す。
    """
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """
    (word, count) のリストを受け取り、整形されたテキストテーブルを返す。
    """
    if not rows:
        return ""
    
    # 列幅の計算
    word_width = max(len(word) for word, _ in rows)
    count_width = max(len(str(count)) for _, count in rows)
    
    # ヘッダー
    header_word = "Word".ljust(word_width)
    header_count = "Count".rjust(count_width)
    
    # 行のフォーマット
    lines = [
        f"{header_word}  {header_count}",
        f"{'-' * word_width}  {'-' * count_width}"
    ]
    
    for word, count in rows:
        lines.append(f"{word.ljust(word_width)}  {str(count).rjust(count_width
