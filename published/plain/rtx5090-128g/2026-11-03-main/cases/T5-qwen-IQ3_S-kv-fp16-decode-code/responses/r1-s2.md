def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """単語を正規化する（小文字化、Unicode正規化など）。

    Args:
        word: 元の単語文字列。
        case_sensitive: Trueなら大小文字を区別する。Falseなら小文字に変換する。

    Returns:
        正規化された単語文字列。
    """
    # Unicode正規化 (NFKC)
    normalized = unicodedata.normalize("NFKC", word)
    if not case_sensitive:
        normalized = normalized.lower()
    return normalized


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストを単語トークンに分割する。

    単語はアルファベット、数字、および一部のUnicode文字（日本語文字など）を
    含む連続した文字列として抽出される。句読点や空白は区切りとして扱われる。

    Args:
        text: 入力テキスト。
        case_sensitive: Trueなら大小文字を区別する。

    Returns:
        単語トークンのリスト。
    """
    tokens: List[str] = []
    current: List[str] = []

    for ch in text:
        if ch.isalnum() or ord(ch) > 127:
            # Unicode文字（日本語など）も単語の一部として扱う
            current.append(ch)
        else:
            if current:
                tokens.append("".join(current))
                current = []

    if current:
        tokens.append("".join(current))

    if not case_sensitive:
        tokens = [normalize_word(t, case_sensitive=False) for t in tokens]

    return tokens
