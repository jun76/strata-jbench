#!/usr/bin/env python3
"""Command-line tool to count word frequencies from a text file and display top N results as a table."""

import argparse
import collections
import re
import sys
from typing import Optional


def count_words(text: str) -> collections.Counter:
    """Count word occurrences in the given text.

    Words are defined as sequences of alphanumeric characters (letters and digits).
    All words are lowercased before counting.

    Args:
        text: The input text to analyze.

    Returns:
        A collections.Counter mapping each word to its occurrence count.
    """
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return collections.Counter(words)


def format_table(counter: collections.Counter, top_n: int) -> str:
    """Format word frequency counter as a table.

    Args:
        counter: A Counter mapping words to their frequencies.
        top_n: Number of top results to display.

    Returns:
        A formatted table string.
    """
    top_items = counter.most_common(top_n)

    if not top_items:
        return "No words found."

    # Determine column widths
    word_width = max(len(word) for word, _ in top_items)
    word_width = max(word_width, len("word"))
    count_width = max(len(str(count)) for _, count in top_items)
    count_width = max(count_width, len("count"))
    rank_width = max(len(str(len(top_items))), len("rank"))

    # Build header
    header = f"{'rank':>{rank_width}}  {'word':<{word_width}}  {'count':>{count_width}}"
    separator = '-' * len(header)

    lines = [header, separator]

    for i, (word, count) in enumerate(top_items, 1):
        line = f"{i:>{rank_width}}  {word:<{word_width}}  {count:>{count_width}}"
        lines.append(line)

    return '\n'.join(lines)
