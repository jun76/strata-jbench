#!/usr/bin/env python3
"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.

Usage:
    python word_freq.py --file input.txt --top 10
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> List[Tuple[str, int]]:
    """Count word occurrences in text.
    
    Args:
        text: Input text string.
        
    Returns:
        List of (word, count) tuples sorted by count descending, then word ascending.
    """
    words = re.findall(r'\b[a-z]+', text.lower())
    counter = Counter(words)
    return sorted(counter.items(), key=lambda x: (-x[1], x[0]))


def format_table(top_words: List[Tuple[str, int]], top_n: int) -> str:
    """Format top N words as a table string.
    
    Args:
        top_words: List of (word, count) tuples.
        top_n: Number of top words to display.
        
    Returns:
        Formatted table string.
    """
    words = top_words[:top_n]
    
    # Determine column widths
    max_word_len = max((len(word) for word, _ in words), default=0)
    max_count_len = max((len(str(count)) for _, count in words), default=0)
    
    header = f"{'Word':<{max_word_len}} | {'Count':>{max_count_len}}"
    separator = f"{'-' * max_word_len}-|-{'-' * max_count_len}"
    
    lines = [header, separator]
    for word, count in words:
        lines.append(f"{word:<{max_word_len}} | {count:>{max_count_len}}")
    
    return "\n".join(lines)


def main():
    """Main entry point for the command-line tool."""
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and display top N."
    )
    parser.add_argument("--file", required=True, help="Input text file path")
    parser.add_argument("--top", type=int, default=10, help="Number of top words to display")
