#!/usr/bin/env python3
"""
Command-line tool that reads a text file, counts word occurrences,
and displays the top N words in a table format.

Usage:
    python word_freq.py <input_file> [--top N] [--encoding UTF-8]

Example:
    python word_freq.py sample.txt --top 10
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Dict


def count_words(text: str) -> Counter[str]:
    """Count the occurrence of each word in the given text.

    Words are extracted by splitting on whitespace.
    The text is lowercased before counting.

    Args:
        text: The input text as a string.

    Returns:
        A Counter mapping each word to its count.
    """
    words = text.lower().split()
    return Counter(words)


def format_table(counter: Counter[str], top_n: int) -> str:
    """Format the top N most frequent words as a table.

    Args:
        counter: A Counter mapping words to their counts.
        top_n: Number of top entries to display.

    Returns:
        A formatted table string with columns: Rank, Word, Count.
    """
    # Get top N common words
    top_words = counter.most_common(top_n)

    # Determine column widths for nice alignment
    word_width = max((len(word) for word, _ in top_words), default=0)
    count_width = max((len(str(count)) for _, count in top_words), default=0)
    rank_width = max(len(str(top_n)), 4)

    # Build header
    header = f"{'Rank':>{rank_width}}  {'Word':<{word_width}}  {'Count':>{count_width}}"
    separator = "-" * len(header)

    lines = [header, separator]
    for rank, (word, count) in enumerate(top_words, start=1):
        lines.append(f"{rank:>{rank_width}}  {word:<{word_width}}  {count:>{count_width}}")

    return "\n".join(lines)
