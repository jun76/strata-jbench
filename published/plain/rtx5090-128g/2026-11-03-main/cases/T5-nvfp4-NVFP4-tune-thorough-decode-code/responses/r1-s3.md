_WORD_PATTERN = re.compile(r"[a-z0-9'\-]+")


def tokenize(text: str) -> List[str]:
    """
    テキストを単語のリストに分割する。

    処理内容:
      1. 全体を小文字化する
      2. 正規表現で有効な単語トークンを抽出する

    Args:
        text: 入力テキスト

    Returns:
        抽出された単語のリスト（重複含む）
    """
    lower_text = text.lower()
    return _WORD_PATTERN.findall(lower_text)


def count_words(text: str) -> Counter:
    """
    テキスト中の単語出現頻度を集計する。

    Args:
        text: 入力テキスト

    Returns:
        単語 -> 出現回数の Counter
    """
    tokens = tokenize(text)
    return Counter(tokens)
