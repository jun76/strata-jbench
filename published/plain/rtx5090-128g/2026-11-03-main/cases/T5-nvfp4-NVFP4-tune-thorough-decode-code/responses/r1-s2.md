wordfreq.py - テキストファイルから単語の出現頻度を集計し、上位N件を表形式で表示するCLIツール。

def read_file(filepath: str) -> str:
    """テキストファイルを読み込み、内容を文字列として返す。

    Args:
        filepath: 読み込むファイルのパス。

    Returns:
        ファイルの内容を示す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み取りに失敗した場合。
    """
    path = Path(filepath)
    try:
        return path.read_text(encoding="utf-8")
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {filepath}")
    except IOError as e:
        raise IOError(f"ファイルの読み取りに失敗しました: {filepath}") from e


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストを単語に分割する。

    単語はアルファベット、数字、ハイフン、アポストロフィを含む連続文字列として定義される。
    句読点や空白は区切りとして扱われる。

    Args:
        text: 分割対象のテキスト。
        case_sensitive: Trueの場合、大文字小文字を区別する。

    Returns:
        単語のリスト。
    """
    import re

    # 単語パターン: アルファベット、数字、ハイフン、アポストロフィを含む
    # 例: "don't", "well-known", "test123"
    word_pattern = re.compile(r"[A-Za-z0-9'-]+")
    matches = word_pattern.findall(text)

    if not case_sensitive:
        matches = [word.lower() for word in matches]

    return matches
