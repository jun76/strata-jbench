@dataclass

    """1 件の単語とその出現回数を保持するデータクラス。

    Attributes:
        word: 正規化された単語。
        count: 出現回数。
    """

    word: str
    count: int


def normalize_text(text: str) -> list[str]:
    """テキストを単語リストに分割し、正規化する。

    正規化の内容:
        - 大文字を小文字に変換
        - 英数字の文字列のみを単語として抽出（区切り文字・句読点を無視）

    Args:
        text: 入力テキスト。

    Returns:
        正規化された単語のリスト。
    """
    # 英数字の連続する文字列を抽出（Unicode 対応）
    tokens = re.findall(r"\w+", text, flags=re.UNICODE)
    # 小文字化して返す
    return [token.lower() for token in tokens]


def count_words(words: list[str]) -> Counter[str]:
    """単語リストから出現頻度を計算する。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
