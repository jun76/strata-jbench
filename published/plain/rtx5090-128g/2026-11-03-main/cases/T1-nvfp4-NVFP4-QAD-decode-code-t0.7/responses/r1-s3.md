word_freq.py

def tokenize(text: str, *, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 単語の最小長。これ未満の単語は除外。

    Returns:
        単語のリスト（正規化後）。
    """
    if not case_sensitive:
        text = text.lower()

    # 英数字を含む連続文字列を単語とみなす
    words = re.findall(r"[a-zA-Z0-9]+", text)

    # 最小長でフィルタ
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]

    return words

        words: 単語のリスト。

    Returns:
        Counter オブジェクト。
    """
    return Counter(words)
