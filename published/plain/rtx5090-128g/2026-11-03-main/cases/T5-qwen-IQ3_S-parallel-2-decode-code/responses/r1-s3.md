word_freq.py: テキストファイルから単語の出現頻度を計算し、上位 N 件を表示するツール。

Usage:
    python word_freq.py <filename> [-n TOP_N]

Example:
    python word_freq.py sample.txt -n 10

import argparse

import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    英数字とハイフンを単語の一部として扱い、それ以外は区切りとする。
    全て小文字に変換して正規化する。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト
    """
    # 正規表現で単語を抽出: 英数字とハイフンを含む連続文字列
    words = re.findall(r"[a-zA-Z0-9'-]+", text)
    return [word.lower() for word in words]


def get_word_frequency(text: str) -> Counter[str]:
    """
    テキスト中の単語出現頻度を計算する。

    Args:
        text: 入力テキスト

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    words = tokenize(text)
    return Counter(words)
