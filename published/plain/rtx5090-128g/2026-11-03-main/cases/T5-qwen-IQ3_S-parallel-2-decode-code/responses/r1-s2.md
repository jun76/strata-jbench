def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: Trueなら大文字小文字を区別、Falseなら小文字に変換

    Returns:
        単語トークンのリスト
    """
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    # 例: "don't", "well-known", "hello-world"
    pattern = r"[A-Za-z0-9]+(?:['\-][A-Za-z0-9]+)*"
    tokens = re.findall(pattern, text)
    if not case_sensitive:
        tokens = [token.lower() for token in tokens]
    return tokens


def count_frequencies(tokens: List[str]) -> Counter:
    """
    トークンリストから出現頻度をカウントする。

    Args:
        tokens: 単語トークンのリスト

    Returns:
        collections.Counter オブジェクト
    """
    return Counter(tokens)
