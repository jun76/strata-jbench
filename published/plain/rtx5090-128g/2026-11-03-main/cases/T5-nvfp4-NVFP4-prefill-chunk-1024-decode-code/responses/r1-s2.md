"""Word frequency counter CLI tool.

Reads one or more text files, counts word occurrences (case-insensitive,
punctuation stripped), and displays the top N words in a table format.
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words, stripping punctuation.

    Args:
        text: Input text string.

    Returns:
        List of lowercase words with punctuation removed.
    """
    # Convert to lowercase and extract words using regex
    # Matches sequences of alphanumeric characters and apostrophes
    words = re.findall(r"[a-z0-9']+", text.lower())
    return words


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        Counter mapping words to their occurrence counts.
    """
    tokens = tokenize(text)
    return Counter(tokens)

        String content of the file.

        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    return filepath.read_text(encoding="utf-8")
