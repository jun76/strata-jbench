## 1. `word_freq.py` — メインモジュール

```python
"""
word_freq: テキストファイルの単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

Usage:
    python word_freq.py <input_file> [-n TOP_N] [--case-sensitive] [--min-length MIN_LEN]
"""

from __future__ import annotations

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """
    単語を正規化する。

    Args:
        word: 元の単語文字列。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。

    Returns:
        正規化された単語文字列。
    """
    if not case_sensitive:
        word = word.lower()
    # 先頭・末尾の非アルファベット/非数字文字を除去（例: "hello," -> "hello"）
    return word.strip(".,;:!?\"'()[]{}")


def count_words(
    text: str,
    case_sensitive: bool = False,
    min_length: int = 1,
) -> Counter[str]:
    """
    テキストから単語の出現頻度をカウントする。

    Args:
        text: 分析対象のテキスト文字列。
        case_sensitive: True の場合、大文字小文字を区別する。
        min_length: 集計対象とする単語の最小文字数。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    counter: Counter[str] = Counter()
    # 空白・改行・タブなどで分割
    raw_words = text.split()
    for raw_word in raw_words:
        word = normalize_word(raw_word, case_sensitive=case_sensitive)
        if len(word) >= min_length:
            counter[word] += 1
