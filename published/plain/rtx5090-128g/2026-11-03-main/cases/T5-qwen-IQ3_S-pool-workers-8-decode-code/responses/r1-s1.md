Usage:

def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, convert all words to lowercase.
        min_length: Minimum length of a word to be included.

    Returns:
        A list of word tokens that meet the criteria.
    """
    # Extract words using regex (sequences of alphanumeric characters and underscores)
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    return words


def count_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of word tokens.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)
