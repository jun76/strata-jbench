"""
word_freq: テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。

    正規表現 \\b[a-zA-Z]+\\b を用いて英単語のみ抽出し、
    すべて小文字に変換する。

    Args:
        text: 入力テキスト

    Returns:
        単語リスト（すべて小文字）
    """
    return [word.lower() for word in re.findall(r'\b[a-zA-Z]+\b', text)]


def count_words(text: str) -> Counter[str]:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    words = tokenize(text)
    return Counter(words)
