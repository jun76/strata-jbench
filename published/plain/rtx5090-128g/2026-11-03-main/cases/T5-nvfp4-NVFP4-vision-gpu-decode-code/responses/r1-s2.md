"""
word_freq.py - テキストファイルから単語出現頻度を集計し、上位N件を表形式で表示するCLIツール
"""

import argparse
import re
import sys
from collections import Counter

def tokenize(text: str) -> list[str]:
    """
    テキストを小文字化し、アルファベットと数字のみを含む単語にトークン化する。

    Args:
        text: 入力テキスト。

    Returns:
        トークン化された単語のリスト。
    """
    # 英数字の連続を単語として抽出し、小文字に変換
    return [word.lower() for word in re.findall(r'\b[a-zA-Z0-9]+\b', text)]


def count_words(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_words(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """
    出現頻度上位N件の単語とその回数を返す。

    Args:
        counter: 単語頻度のCounter。
        n: 取得する上位件数。

    Returns:
        (単語, 回数) のタプルのリスト。頻度降順、同じ頻度の場合はアルファベット順。
    """
    # most_common() は頻度降順、同じ頻度の場合は挿入順だが、
    # 安定したソートを得るために明示的にソートする
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
