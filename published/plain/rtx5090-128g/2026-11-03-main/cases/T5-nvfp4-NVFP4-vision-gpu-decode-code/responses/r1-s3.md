#!/usr/bin/env python3
"""単語の出現頻度をカウントし、上位N件を表形式で表示するコマンドラインツール。

使い方:
    python word_count.py input.txt
    python word_count.py input.txt -n 10
    python word_count.py input.txt --top 20 --min-length 3
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path


def normalize_word(word: str) -> str:
    """単語を正規化する（小文字化、前後の空白除去）。

    Args:
        word: 正規化する単語。

    Returns:
        小文字化された単語。
    """
    return word.lower().strip()


def extract_words(text: str) -> list[str]:
    """テキストから単語を抽出する。

    単語は英数字とハイフン、アポストロフィを含む文字列として定義される。
    句読点や特殊文字は区切り文字として扱われる。

    Args:
        text: 解析対象のテキスト。

    Returns:
        抽出された単語のリスト。
    """
    # 英数字、ハイフン、アポストロフィを含む連続した文字列を単語として抽出
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return [normalize_word(w) for w in words if w]


def count_words(text: str, min_length: int = 1) -> Counter:
    """テキスト内の単語の出現頻度をカウントする。

    Args:
        text: 解析対象のテキスト。
        min_length: カウントする単語の最小長。これより短い単語は除外される。

    Returns:
        単語から出現回数へのマッピング。
    """
    words = extract_words(text)
    filtered_words = [w for w in words if len(w) >= min_length]
    return Counter(filtered_words)
