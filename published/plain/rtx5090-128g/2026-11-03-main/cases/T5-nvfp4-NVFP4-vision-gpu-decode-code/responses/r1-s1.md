"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位 N 件の単語を表形式で表示するコマンドラインツール。
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from pathlib import Path
from typing import TextIO


@dataclass(frozen=True)
class WordCount:
    """単語とその出現回数を表すデータクラス。"""
    word: str
    count: int


def tokenize(text: str) -> list[str]:
    """
    テキストから単語を抽出する。

    正規表現 [a-zA-Z0-9]+ を使用して英数字からなるトークンを抽出し、
    すべて小文字に変換する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_words(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。

    Returns:
        単語 -> 出現回数 の Counter オブジェクト。
    """
    words = tokenize(text)
    return Counter(words)
