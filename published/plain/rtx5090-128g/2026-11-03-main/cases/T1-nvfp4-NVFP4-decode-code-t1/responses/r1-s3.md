def tokenize(text: str, ignore_case: bool = True) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        ignore_case: 大文字小文字を区別しない場合 True。

    Returns:
        単語のリスト（正規化済み）。
    """
    # 英数字のみを単語として抽出
    pattern = r"[a-zA-Z0-9]+"
    words = re.findall(pattern, text)
    if ignore_case:
        words = [w.lower() for w in words]
    return words


def count_word_frequencies(text: str, ignore_case: bool = True) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:

        ignore_case: 大文字小文字を区別しない場合 True。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    words = tokenize(text, ignore_case=ignore_case)
    return Counter(words)
