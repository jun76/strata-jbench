"""
wordfreq.py - 単語出現頻度集計ツール

テキストファイルを読み込み、単語の出現頻度を数え上げ、
上位 N 件の単語を整形された表形式で表示します。

使用方法:
    python wordfreq.py input.txt

    python wordfreq.py input.txt --case-sensitive
    python wordfreq.py input.txt --min-length 3 --output results.csv

def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> list[str]:
    """テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字・小文字を区別するかどうか。
        min_length: 抽出する単語の最小文字数。

    Returns:
        単語のリスト（重複含む）。
    """
    if not case_sensitive:
        text = text.lower()

    # 英数字とハイフン、アポストロフィを含む文字列を単語として抽出
    words = re.findall(r"[a-z0-9'\-]+", text, flags=re.IGNORECASE if not case_sensitive else 0)

    # 最小長さでフィルタ
    words = [w for w in words if len(w) >= min_length]
    return words


def count_word_frequencies(
    text: str,
    case_sensitive: bool = False,
    min_length: int = 1,
) -> Counter[str]:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字・小文字を区別するかどうか。
        min_length: カウントする単語の最小文字数。

    Returns:
        単語と出現回数の Counter。
    """
    words = extract_words(text, case_sensitive, min_length)
    return Counter(words)
