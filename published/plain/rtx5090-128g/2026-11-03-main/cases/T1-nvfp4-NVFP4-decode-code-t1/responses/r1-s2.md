def tokenize(text: str) -> list[str]:
    """Tokenize text into words, lowercased and stripped of punctuation.

    Args:
        text: Input text string.

    Returns:
        A list of words in lowercase.
    """
    # Extract sequences of word characters (letters, digits, underscores)
    # We use a simple regex to find word-like tokens
    return re.findall(r'\b\w+\b', text.lower())


def count_word_frequencies(words: list[str]) -> Counter[str]:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(frequencies: Counter[str], n: int) -> list[tuple[str, int]]:
    """Get the top N most common words.

    Args:
        frequencies: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending, then word ascending.
    """
    # Counter.most_common() returns elements ordered by count (highest first)
    # For ties, Counter.most_common() does not guarantee order, so we sort manually
    # to ensure consistent output: by count descending, then word ascending
    sorted_items = sorted(frequencies.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
