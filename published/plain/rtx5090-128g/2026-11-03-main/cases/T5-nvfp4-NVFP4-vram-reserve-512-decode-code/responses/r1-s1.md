"""

from __future__ import annotations

def tokenize(text: str, *, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """テキストを単語リストに分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するか。
        min_len: この長さ未満の単語を除外する。

    Returns:
        単語のリスト。
    """
    # 英数字の連続を単語とする（日本語等は別途必要なら正規表現を調整）
    words = re.findall(r"[A-Za-z0-9']+", text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    if min_len > 1:
        words = [w for w in words if len(w) >= min_len]
    return words


def count_frequencies(words: List[str]) -> Counter[str]:
    """単語リストから出現頻度を計算する。

    Args:
        words: 単語のリスト。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    return Counter(words)
