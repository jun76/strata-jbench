"""テキストファイルから単語出現頻度をカウントし、上位 N 件を表形式で表示するコマンドラインツール。

Usage:
    python word_count.py <input_file> [-n N] [--case-sensitive] [--min-len MIN_LEN]

Examples:
    python word_count.py sample.txt
    python word_count.py sample.txt -n 10
    python word_count.py sample.txt --case-sensitive --min-len 3
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。デフォルトは False。
        min_len: 抽出する単語の最小長。デフォルトは 1。

    Returns:
        抽出された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とアンダースコアを含む連続した文字列を単語として抽出
    words = re.findall(r'\w+', text)
    # min_len 以上の長さの単語のみ残す
    if min_len > 1:
        words = [w for w in words if len(w) >= min_len]
    return words


def count_words(text: str, case_sensitive: bool = False, min_len: int = 1) -> Counter:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。デフォルトは False。
        min_len: 抽出する単語の最小長。デフォルトは 1。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    words = tokenize(text, case_sensitive, min_len)
    return Counter(words)
