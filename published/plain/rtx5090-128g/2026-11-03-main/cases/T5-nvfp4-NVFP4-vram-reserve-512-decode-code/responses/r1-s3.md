def tokenize(text: str, *, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストを単語トークンに分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。False の場合、全て小文字に変換する。
        min_length: 単語の最小文字数。これ未満の単語は除外する。

    Returns:
        単語トークンのリスト。
    """
    # 英数字とハイフン・アポストロフィを含む連続文字列を単語とみなす
    tokens = re.findall(r"[a-zA-Z0-9'\-]+", text)

    if not case_sensitive:
        tokens = [t.lower() for t in tokens]

    # 最小文字数フィルタ
    tokens = [t for t in tokens if len(t) >= min_length]

    return tokens
