#!/usr/bin/env python3
"""Text file word frequency counter and top-N display tool.

This module provides a command-line utility that reads a text file,
counts word occurrences (case-insensitive, ignoring punctuation),
and displays the top N most frequent words in a formatted table.
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.

    Words are extracted by splitting on whitespace and stripping punctuation.
    Counting is case-insensitive.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object mapping lowercase words to their occurrence counts.
    """
    words: List[str] = []
    for token in text.split():
        # Strip leading and trailing punctuation characters
        cleaned = token.strip(".,;:!?'\"()[]{}<>/\\@#$%^&*-=+_`~|")
        if cleaned:
            words.append(cleaned.lower())
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top items to return.

    Returns:
        A list of (word, count) tuples sorted by count descending,
        then by word ascending for ties.
    """
    # Counter.most_common already sorts by count descending.
    # For ties, we want alphabetical order, so we sort manually.
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(top_n: List[Tuple[str, int]]) -> str:
    """Format the top N word-frequency pairs as a string table.

    Args:
        top_n: A list of (word, count) tuples.

    Returns:
        A formatted string with columns: Rank, Word, Count.
    """
