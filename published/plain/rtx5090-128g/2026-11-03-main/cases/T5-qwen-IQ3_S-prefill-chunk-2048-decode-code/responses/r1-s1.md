"""
word_freq: A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Words are extracted by splitting on non-alphanumeric characters,
    converted to lowercase, and filtered to exclude empty strings.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object with word frequencies.
    """
    # Extract words: sequences of alphanumeric characters (including underscores)
    words = re.findall(r'[a-zA-Z0-9_]+', text.lower())
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object containing word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count), sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by count descending, then by word ascending for consistent ordering
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
