def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    英数字のみを単語とし、大文字小文字は無視して小文字に変換する。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 単語のリスト。
    """
    # 英数字の連続を単語として抽出
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    return [word.lower() for word in words]


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text (str): 入力テキスト。

    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = tokenize(text)
    return Counter(words)
