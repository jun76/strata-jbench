"""
word_count.py - コマンドラインツール: テキストファイルの単語出現頻度を上位N件表示する。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    単語はアルファベットと数字のシーケンスとして定義し、小文字に変換する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        小文字化された単語のリスト
    """
    # 英数字のみを抽出し、小文字に変換
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語をキー、出現回数を値とする Counter
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位N件の (word, count) ペアを返す。
    
    Args:
        counter: 単語出現頻度の Counter
        n: 取得する上位件数
        
    Returns:
        上位N件の (word, count) のリスト。回数の降順、同順位の場合は辞書順。
    """
    # most_common は回数降順だが、同順位の場合は挿入順になるため、
    # 明示的に (count desc, word asc) でソートする
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
