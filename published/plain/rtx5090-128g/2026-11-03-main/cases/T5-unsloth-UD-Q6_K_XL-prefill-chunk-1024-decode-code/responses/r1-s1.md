#!/usr/bin/env python3
"""
Command-line tool to count word frequencies in a text file and display the top N words.

Usage:
    python word_freq.py <file> [--top N] [--min-length M] [--case-sensitive]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Optional


def tokenize_text(text: str, min_length: int = 1, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text to tokenize.
        min_length: Minimum length of a word to include. Default is 1.
        case_sensitive: If False, words are lowercased before counting. Default is False.

    Returns:
        A list of tokens (words) extracted from the text.
    """
    # Extract words using regex: sequences of alphanumeric characters
    tokens = re.findall(r'[a-zA-Z0-9]+', text)
    
    # Filter by minimum length
    if min_length > 1:
        tokens = [t for t in tokens if len(t) >= min_length]
    
    # Convert to lowercase if not case-sensitive
    if not case_sensitive:
        tokens = [t.lower() for t in tokens]
    
    return tokens


def count_word_frequencies(tokens: List[str]) -> Counter:
    """
    Count the frequency of each word in the token list.

    Args:
        tokens: A list of word tokens.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their counts.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to retrieve.

    Returns:
        A list of tuples (word, count) sorted by count descending, then alphabetically.
    """
    # most_common already sorts by count descending
    # For ties, we want alphabetical order for words
    items = list(counter.items())
    # Sort by count descending, then by word ascending for tie-breaking
    items.sort(key=lambda x: (-x[1], x[0]))
