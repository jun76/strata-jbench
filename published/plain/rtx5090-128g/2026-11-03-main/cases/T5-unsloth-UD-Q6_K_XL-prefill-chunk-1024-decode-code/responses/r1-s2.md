#!/usr/bin/env python3
"""
テキストファイルを読み込んで単語の出現頻度を集計し、
上位 N 件を表形式で表示するコマンドラインツール。

Usage:
    python word_freq.py <file> [-n TOP_N] [--case-sensitive]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    テキスト中の単語の出現頻度をカウントする。

    Args:
        text: 分析対象のテキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。
                        False の場合、すべて小文字に変換してカウントする。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    if not case_sensitive:
        text = text.lower()

    # 英数字とアンダースコアを単語とみなす正規表現
    # 日本語など他の文字を除外したい場合はパターンを調整
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 N 件の (単語, 回数) のリストを取得する。

    Args:
        counter: 単語頻度の Counter。
        n: 取得する上位件数。

    Returns:
        (単語, 出現回数) のタプルのリスト。回数の降順に並んでいる。
    """
    return counter.most_common(n)
