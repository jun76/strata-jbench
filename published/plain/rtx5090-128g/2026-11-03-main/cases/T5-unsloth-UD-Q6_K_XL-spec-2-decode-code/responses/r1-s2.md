import argparse

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。

    Returns:
        単語トークンのリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフンを含む連続文字列を単語として抽出
    # 例: "don't" -> ["don", "t"] (アポストロフィで分割), "co-op" -> ["co-op"]
    tokens = re.findall(r"[a-z0-9]+(?:[-'][a-z0-9]+)*", text, re.IGNORECASE)
    return tokens


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: 大文字小文字を区別するかどうか。

    Returns:
        単語をキー、出現回数を値とする collections.Counter。
    """
    tokens = tokenize(text, case_sensitive)
    return Counter(tokens)
