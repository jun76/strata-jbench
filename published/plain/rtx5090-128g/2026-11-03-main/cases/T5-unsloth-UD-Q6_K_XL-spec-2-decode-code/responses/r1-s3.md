wordfreq.py

テキストファイルを読み込み、単語の出現頻度をカウントし、

def tokenize(text: str, case_insensitive: bool = True, min_len: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text (str): 入力テキスト。
        case_insensitive (bool): True の場合、大文字小文字を区別しない。
        min_len (int): この長さ未満の単語は除外する。

    Returns:
        List[str]: 単語のリスト。
    """
    # 英数字とハイフンを含むトークンを抽出（日本語など非ASCII文字も考慮する場合は正規表現を調整）
    # ここでは一般的な英単語を想定し、\w+ で抽出
    tokens = re.findall(r'\w+', text)
    
    if case_insensitive:
        tokens = [token.lower() for token in tokens]
    
    # 最小長でフィルタリング
    if min_len > 1:
        tokens = [token for token in tokens if len(token) >= min_len]
        
    return tokens


def count_frequencies(tokens: List[str]) -> Counter:
    """
    トークンリストから頻度カウントを生成する。

    Args:
        tokens (List[str]): 単語のリスト。

    Returns:
        Counter: 単語をキー、頻度を値とする Counter。
    """
    return Counter(tokens)
