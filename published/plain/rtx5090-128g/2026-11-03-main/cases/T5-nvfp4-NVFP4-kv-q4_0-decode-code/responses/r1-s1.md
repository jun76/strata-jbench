"""
word_freq.py - A command-line tool to count word frequencies in text files.

This module provides functionality to read text files, count word occurrences,
and display the top N most frequent words in a table format.
"""

import argparse
import re
import sys
from collections import Counter

def tokenize_text(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    # Find all sequences of alphanumeric characters (words)
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_word_frequencies(text: str) -> Counter:
    """
    Count the frequency of each word in the given text.
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter object with words as keys and their counts as values.
    """
    words = tokenize_text(text)
    return Counter(words)


def get_top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from a counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to retrieve.
        
    Returns:
        A list of tuples, each containing a word and its count,
        sorted by count descending, then by word ascending.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
