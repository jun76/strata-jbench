"""テキストファイルの単語出現頻度を上位N件で表示するCLIツール。

Usage:
    python word_freq.py <filepath> [--top N] [--case-sensitive] [--min-length M]
"""

import argparse

def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するか。Falseなら小文字化。
        min_length: 抽出する単語の最小文字数。

    Returns:
        抽出された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()

    # 英数字で構成される単語のみ抽出（正規表現で区切る）
    words = re.findall(r"[a-z0-9]+", text, flags=re.IGNORECASE if case_sensitive else 0)
    # 大文字小文字を区別しない場合は既に小文字化済みだが、
    # 区別する場合は元のテキストから抽出するため、正規表現で再抽出
    if case_sensitive:
        words = re.findall(r"[a-zA-Z0-9]+", text)

    return [w for w in words if len(w) >= min_length]


def count_frequencies(words: List[str]) -> Counter[str]:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語リスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
