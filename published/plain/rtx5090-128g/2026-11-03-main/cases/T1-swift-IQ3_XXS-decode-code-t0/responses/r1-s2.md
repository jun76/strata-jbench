#!/usr/bin/env python3
"""
wordfreq.py - テキストファイルから単語の出現頻度を集計し、上位 N 件を表示するツール。

Usage:
    python wordfreq.py <file> [-n N]
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    大文字小文字を区別せず、アルファベットと数字のみのトークンを抽出する。
    """
    # 簡易的なトークナイザ: 英数字の連続を単語とみなす
    import re
    return [word.lower() for word in re.findall(r'[a-zA-Z0-9]+', text)]


def count_words(text: str) -> Counter:
    """
    テキスト中の単語出現頻度を Counter で返す。
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (word, count) を返す。
    同数値の場合は辞書順で安定ソートする。
    """
    # most_common はカウント降順だが、同カウント時の順序は保証されないため
    # 明示的に (count 降順, word 昇順) でソートする
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
