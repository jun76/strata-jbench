#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位N件表示する。

Usage:
    python word_freq.py input.txt -n 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    テキストから単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    # 単語を抽出: 英数字とハイフンを単語として扱い、他は区切り
    import re
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (word, count) を返す。

    Args:
        counter: 単語頻度の Counter
        n: 上位件数

    Returns:
        (word, count) のリスト（降順）
    """
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """
    (word, count) のリストを表形式の文字列に変換する。

    Args:
        rows: (word, count) のリスト

    Returns:
        表形式の文字列
    """
    if not rows:
        return ""

    # 列幅を計算
    word_width = max(len(word) for word, _ in rows)
    count_width = max(len(str(count)) for _, count in rows)

    # ヘッダー
    header = f"{'Word':<{word_width}}  {'Count':>{count_width}}"
    separator = f"{'-' * word_width}  {'-' * count_width}"

    lines = [header, separator]
    for word, count in rows:
        lines.append(f"{word:<{word_width}}  {count:>{count_width}}")

    return "\n".join(lines)
