import argparse

from collections import Counter
from typing import List, Tuple


def read_file(filepath: str) -> str:
    """指定されたファイルを読み込み、テキスト内容を返す。

    Args:
        filepath: 読み込むファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み取りエラーが発生した場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。

    単語は英数字とアンダースコアで構成される連続文字列として抽出される。
    大文字・小文字は区別せず、すべて小文字に変換する。

    Args:
        text: 入力テキスト。

    Returns:
        単語のリスト（小文字化済み）。
    """
    import re
    # 英数字とアンダースコアの連続を単語として抽出
    words = re.findall(r'[a-zA-Z0-9_]+', text)
    return [w.lower() for w in words]


def count_frequencies(words: List[str]) -> Counter:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (word, count) を返す。

    Args:
        counter: 単語頻度の Counter。
        n: 取得する上位件数。

    Returns:
        (word, count) のタプルのリスト。
    """
    return counter.most_common(n)
