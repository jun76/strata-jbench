コマンドラインツール: テキストファイルを読み込んで単語の出現頻度を集計し、

    python word_freq.py sample.txt --top 10
    python word_freq.py sample.txt --top 5 --case-sensitive
"""

from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語のリストを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。

    Returns:
        単語のリスト（正規表現 [a-zA-Z0-9']+ で抽出）。
    """
    if not case_sensitive:
        text = text.lower()
    # 単語として英数字とアポストロフィを含む文字列を抽出
    words = re.findall(r"[a-zA-Z0-9']+", text)
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        各単語とその出現回数を格納した Counter。
    """
    return Counter(words)
