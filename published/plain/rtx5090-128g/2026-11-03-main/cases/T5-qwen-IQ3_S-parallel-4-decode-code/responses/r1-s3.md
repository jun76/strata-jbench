#!/usr/bin/env python3
"""A command-line tool that reads a text file, counts word frequencies, and displays the top N words in a table format."""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.

    Words are defined as sequences of alphanumeric characters (including underscores)
    separated by non-alphanumeric characters. The counting is case-insensitive.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping each word to its frequency count.
    """
    import re
    # Extract words: sequences of word characters (letters, digits, underscores)
    words = re.findall(r'\w+', text)
    # Convert to lowercase for case-insensitive counting
    words_lower = [w.lower() for w in words]
    return Counter(words_lower)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples, each containing a word and its count, sorted by count descending.
        If there are ties, the order is based on the order they appear in the Counter's internal ordering
        (which is insertion order in Python 3.7+).
    """
    return counter.most_common(n)
