#!/usr/bin/env python3
"""wordfreq — テキストファイルの単語出現頻度を表形式で表示する CLI。

使い方:
  python wordfreq.py FILE [FILE ...] [-n 20] [-w 50]

オプション:
  -n, --top N      表示する上位件数 (デフォルト: 20)
  -w, --width W    テーブル全体の幅 (デフォルト: 50)
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from pathlib import Path
from typing import List, Sequence, Tuple


@dataclass(frozen=True)
class WordCount:
    """単語とその出現回数を保持する不変オブジェクト。"""
    word: str
    count: int


def normalize(text: str) -> List[str]:
    """テキストを単語リストに正規化する。

    - 英字/数字/ハイフン/アポストロフィを単語文字として扱い、それ以外は区切りとする。
    - 大文字は小文字に統一する。
    - 空文字列は除去する。

    Args:
        text: 生テキスト。

    Returns:
        正規化された単語のリスト（出現順、重複含む）。
    """
    # \b を使わず、単語文字を明示的にマッチさせる
    return re.findall(r"[a-zA-Z0-9]+(?:['-][a-zA-Z0-9]+)*", text.lower())


def count_words(tokens: Sequence[str]) -> List[WordCount]:
    """トークン列から出現頻度を計算し、(count desc, word asc) の順でソートして返す。"""
    counter: Counter[str] = Counter(tokens)
    items: List[WordCount] = [WordCount(w, c) for w, c in counter.items()]
    items.sort(key=lambda wc: (-wc.count, wc.word))
    return items
