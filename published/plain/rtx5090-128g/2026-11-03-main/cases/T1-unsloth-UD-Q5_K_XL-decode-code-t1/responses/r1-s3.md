#!/usr/bin/env python3
"""A command-line tool that reads a text file, counts word frequencies, and displays the top N words in a table format."""

import argparse

from collections import Counter
from typing import List, Tuple, Dict


def count_words(text: str) -> Counter:
    """Count the occurrence of each word in the given text.

    Words are extracted by splitting on non-alphabetic/non-numeric characters and
    converted to lowercase.

    Args:
        text: The input text to process.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    # Split text into words: split on anything that is not a letter or digit, then filter out empty strings
    words = []
    current_word = []
    for char in text.lower():
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from the counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) for the top N most common words.
    """
    return counter.most_common(n)


def read_file(filepath: str) -> str:
    """Read the content of a text file.

    Args:
        filepath: The path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If there is an error reading the file.
    """
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except IOError as e:
        raise IOError(f"Error reading file {filepath}: {e}")
