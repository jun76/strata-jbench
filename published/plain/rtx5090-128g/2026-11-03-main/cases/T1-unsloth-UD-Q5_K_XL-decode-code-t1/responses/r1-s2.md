import argparse

from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    単語の定義:
    - 英数字の連続する部分 (\w+)
    - 大文字/小文字は無視 (小文字に統一)
    
    Args:
        text (str): 入力テキスト
        
    Returns:
        List[str]: 単語のリスト (小文字)
    """
    # \w+ は文字・数字・アンダースコアにマッチする
    import re
    return [word.lower() for word in re.findall(r'\w+', text)]


def count_frequencies(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。
    
    Args:
        words (List[str]): 単語のリスト
        
    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト
    """
    return Counter(words)
