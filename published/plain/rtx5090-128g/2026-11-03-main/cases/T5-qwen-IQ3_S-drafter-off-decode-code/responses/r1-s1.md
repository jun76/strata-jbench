#!/usr/bin/env python3
"""
word_freq.py - テキストファイル内の単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    英数字で構成される連続した文字列を単語とみなし、すべて小文字に変換する。

    Args:
        text: 入力テキスト文字列。

    Returns:
        抽出された単語のリスト。
    """
    # 英数字で構成されるトークンを抽出し、小文字化
    words = re.findall(r'\b[a-z0-9]+\b', text.lower())
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする collections.Counter。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 N 件を返す。

    Args:
        counter: 単語と出現回数の Counter。
        n: 取得する上位件数。

    Returns:
        (単語, 出現回数) のタプルを並べたリスト。
    """
    return counter.most_common(n)
