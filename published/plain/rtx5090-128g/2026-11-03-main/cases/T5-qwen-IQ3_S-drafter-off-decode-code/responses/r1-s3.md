def read_text(filepath: str) -> str:
    """指定されたテキストファイルを読み込み、内容を文字列として返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を返す。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み取り中にエラーが発生した場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """テキストから単語の出現頻度をカウントする。

    単語は空白文字で区切られ、句読点や記号は除去される。

    Args:
        text: 分析対象のテキスト。
        case_sensitive: 大文字・小文字を区別するかどうか。False の場合、
                        すべて小文字に変換してからカウントする。

    Returns:
        単語とその出現回数を含む Counter オブジェクト。
    """
    import re

    if not case_sensitive:
        text = text.lower()

    # 単語を抽出: 英数字とハイフンを含む連続文字列
    words = re.findall(r"[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*", text)
    return Counter(words)
