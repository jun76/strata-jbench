word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位N件表示する。

使い方:

例:
    python word_freq.py sample.txt -n 10

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストから単語を抽出する。

    Args:
        text: 入力テキスト

    Returns:
        小文字化された単語のリスト
    """
    # 英数字の連続を単語として抽出
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_words(text: str) -> Counter:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語をキー、出現回数を値とするCounter
    """
    words = tokenize(text)
    return Counter(words)

    """Counterから上位n件の単語と出現回数を取得する。

        (単語, 出現回数)のタプルのリスト。出現回数の降順、同数時は辞書順。
    """
    # most_commonは出現回数の降順で返す。同数の場合は挿入順だが、
    # 辞書順にソートするために、(-count, word)でソートする

    return items[:n]
