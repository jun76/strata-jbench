"""
Top N word frequency tool.

Reads a text file, counts word occurrences, and displays the top N most frequent words
in a tabular format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Optional


def read_text_file(filepath: str) -> str:
    """Read and return the content of a text file.

    Args:
        filepath: Path to the text file to read.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """Extract words from text, lowercased and stripped of punctuation.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Use regex to find sequences of word characters (letters and digits, and underscores)
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text.lower())
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """Count the frequency of each word in the list.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top items to return.

    Returns:
        A list of tuples (word, count), sorted by count descending, then alphabetically.
    """
    # Sort by count descending, then by word ascending for consistent ordering
    return sorted(counter.items(), key=lambda x: (-x[1], x[0]))[:n]
