import argparse

import re
from collections import Counter
from typing import List, Tuple


def read_text(filepath: str) -> str:
    """指定されたファイルからテキストを読み込む。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。

    単語は英数字（Unicode対応）で構成される連続文字列として定義する。
    すべての文字を小文字に変換して正規化する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    # Unicodeの文字・数字・_ を単語として扱う
    words = re.findall(r"[^\W_]+", text, flags=re.UNICODE)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語ごとのカウントを表す Counter オブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """出現頻度上位 N 件の (word, count) のリストを返す。

    同数の場合は辞書順（アルファベット順）で並び替える。

    Args:
        counter: 単語とカウントのCounter。
        n: 上位何件を取得するか。

    Returns:
        (word, count) のタプルのリスト（カウント降順、同数は単語昇順）。
    """
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
