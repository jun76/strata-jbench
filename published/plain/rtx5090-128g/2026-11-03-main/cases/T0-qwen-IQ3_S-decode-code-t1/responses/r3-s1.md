"""
text_freq.py - 単語出現頻度集計ツール

テキストファイルを読み込み、単語の出現頻度を上位 N 件で表形式表示する CLI ツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストから単語に分割する。大文字小文字は無視し、非文字/数字は除外する。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト
    """
    return re.findall(r"[a-zA-Z0-9']+(?=[^\w']|\b)", text.lower())


def count_frequencies(text: str, top_n: int) -> List[Tuple[str, int]]:
    """テキスト中の単語出現頻度を上位 top_n 件で返す。

    Args:
        text: 入力テキスト
        top_n: 上位何件を取り出すか

    Returns:
        (単語, 頻度) のタプルのリスト。頻度の高い順。頻度が同数なら辞書順。
    """
    words = tokenize(text)
    counter = Counter(words)
    # Counter.most_common は頻度降順で返すが、同頻度の並び順は保証しない
    # 辞書順も考慮するため、頻度降順→単語昇順でソート
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:top_n]
