"""A command-line tool that reads a text file, counts word frequencies,
and displays the top N words in a table format."""

import argparse

import re
from collections import Counter
from typing import List, Tuple, Dict


def extract_words(text: str) -> List[str]:
    """Extract lowercase words from text using a simple regex.
    
    A word is defined as one or more alphanumeric characters or underscores.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase words found in the text.
    """
    # \w matches [a-zA-Z0-9_] in Python regex by default for str
    return [word.lower() for word in re.findall(r'\w+', text)]


def count_words(words: List[str]) -> Dict[str, int]:
    """Count the frequency of each word.
    
    Args:
        words: A list of words to count.
        
    Returns:
        A dictionary mapping each word to its frequency count.
    """
    return dict(Counter(words))


def get_top_n(
    word_counts: Dict[str, int],
    n: int
) -> List[Tuple[str, int]]:
    """Get the top N most common words.
    
    Args:
        word_counts: A dictionary mapping words to their frequencies.
        n: The number of top words to return.
        
    Returns:
        A list of tuples (word, count) sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by count descending, then by word ascending for stable ordering
    sorted_items = sorted(
        word_counts.items(),
        key=lambda item: (-item[1], item[0])
    )
    return sorted_items[:n]
