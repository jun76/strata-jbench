#!/usr/bin/env python3
"""A command-line tool that reads a text file, counts word frequencies,
and displays the top N words in a table format.

Usage:
    python wordfreq.py <input_file> [--top N] [--case-insensitive/--case-sensitive]

Example:
    python wordfreq.py sample.txt --top 10 --case-insensitive
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def parse_arguments(argv: List[str] | None = None) -> argparse.Namespace:
    """Parse command-line arguments.

    Args:
        argv: List of command-line arguments. If None, sys.argv is used.

    Returns:
        Namespace containing parsed arguments:
            - file: Path to the input text file.
            - top: Number of top words to display (default: 10).
            - case_insensitive: Whether to treat words case-insensitively (default: False).
    """
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and display the top N words."
    )
    parser.add_argument(
        "file",
        type=str,
        help="Path to the input text file."
    )
    parser.add_argument(
        "--top",
        type=int,
        default=10,
        help="Number of top words to display (default: 10)."
    )
    parser.add_argument(
        "--case-insensitive",
        action="store_true",
        default=False,
        help="Treat words case-insensitively (default: False)."
    )
    parser.add_argument(
        "--case-sensitive",
        action="store_false",
        dest="case_insensitive",
        help="Treat words case-sensitively (explicitly)."
    )
    return parser.parse_args(argv)
