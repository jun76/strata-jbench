#!/usr/bin/env python3
"""A command-line tool that reads a text file, counts word frequencies,
and displays the top N words in a table format.

Usage:
    python wordfreq.py <input_file> [--top N] [--case-insensitive/--case-sensitive]

Example:
    python wordfreq.py sample.txt --top 10 --case-insensitive
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def parse_arguments(argv: List[str] | None = None) -> argparse.Namespace:
    """Parse command line arguments.

    Args:
        argv: List of command line arguments. If None, sys.argv is used.

    Returns:
        argparse.Namespace with parsed arguments.
    """
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and display top N words."
    )
    parser.add_argument(
        "input_file",
        help="Path to the input text file"
    )
    parser.add_argument(
        "--top",
        type=int,
        default=10,
        help="Number of top words to display (default: 10)"
    )
    parser.add_argument(
        "--case-insensitive",
        action="store_true",
        default=True,
        help="Case-insensitive word counting (default: True)"
    )
    parser.add_argument(
        "--case-sensitive",
        action="store_false",
        dest="case_insensitive",
        help="Case-sensitive word counting"
    )
    return parser.parse_args(argv)


def extract_words(text: str, case_insensitive: bool = True) -> List[str]:
    """Extract words from text using regex.

    Words are defined as sequences of alphanumeric characters and underscores.

    Args:
        text: The input text string.
        case_insensitive: If True, convert all words to lowercase.

    Returns:
        List of extracted words.
    """
    if case_insensitive:
        text = text.lower()
    # Match sequences of alphanumeric characters and underscores as words
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return words
