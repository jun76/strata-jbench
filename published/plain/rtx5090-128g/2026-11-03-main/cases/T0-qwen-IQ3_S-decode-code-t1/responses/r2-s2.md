import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def read_text(filepath: str) -> str:
    """指定されたファイルからテキストを読み込む。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。

    単語は英数字のみを含む連続文字列として定義される。
    全て小文字に変換される。

    Args:
        text: 入力テキスト。

    Returns:
        抽出された単語のリスト。
    """
    # 英数字のみの連続文字列を抽出し、小文字に変換
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_word_frequency(words: List[str]) -> Counter:
    """単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語とその出現回数をキー・値とするCounter。
    """
    return Counter(words)


def get_top_n_words(freq: Counter, n: int) -> List[Tuple[str, int]]:
    """出現頻度の高い上位n件の単語と回数を得る。

    Args:
        freq: 単語頻度のCounter。
        n: 取得する上位件数。

    Returns:
        (word, count) のタプルのリスト。頻度降順。同数の場合は辞書順。
    """
    # common most common returns items sorted by count descending, then by key order (alphabetical for ties in Python's Counter)
    # Actually, Counter.most_common(n) returns items with highest counts first, and ties are broken arbitrarily (by insertion order in Python 3.7+).
    # For consistent alphabetical breaking on ties, we need custom sorting.
    items = list(freq.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
