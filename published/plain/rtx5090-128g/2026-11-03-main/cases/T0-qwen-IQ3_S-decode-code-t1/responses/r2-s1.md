"""
Top N Word Frequency Command-Line Tool

Reads a text file, counts word frequencies, and displays the top N most frequent words
in a tabular format.
"""

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from pathlib import Path
from typing import List, Tuple


@dataclass
class WordFrequency:
    """Represents a word and its frequency count."""
    word: str
    count: int


def tokenize_text(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Convert to lowercase and extract words (sequences of alphanumeric characters)
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n_frequencies(counter: Counter, n: int) -> List[WordFrequency]:
    """
    Get the top N most frequent words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top entries to retrieve.

    Returns:
        A list of WordFrequency objects for the top N words.
    """
    if n <= 0:
        return []
    # most_common returns a list of tuples (word, count)
    top_items = counter.most_common(n)
    return [WordFrequency(word=word, count=count) for word, count in top_items]
