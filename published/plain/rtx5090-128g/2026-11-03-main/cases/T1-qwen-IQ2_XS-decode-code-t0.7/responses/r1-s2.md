    """テキストを単語リストに分割する。

        text: 任意のテキスト。

    """
    return WORD_PATTERN.findall(text)

    """複数のテキストから単語出現頻度を Counter として返す。

    Returns:

    """
    counter: Counter[str] = Counter()
    for text in texts:
        counter.update(tokenize(text))
