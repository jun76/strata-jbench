#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file and display top N words."""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def read_words_from_file(filepath: str) -> List[str]:
    """Read a text file and return a list of words.

    Args:
        filepath: Path to the text file.

    Returns:
        A list of words, lowercased and stripped of punctuation.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        content = f.read()

    # Normalize: lowercase, split by whitespace, strip punctuation from each word
    words: List[str] = []
    for token in content.split():
        # Strip leading/trailing punctuation
        word = token.strip('.,!?;:\'"()[]{}-')
        if word:
            words.append(word.lower())

    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter mapping words to their frequencies.
    """
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top words to return.

    Returns:
        A list of tuples (word, count) sorted by frequency descending.
    """
    return counter.most_common(n)
