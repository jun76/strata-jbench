#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file and display top N results."""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def parse_args(argv: List[str] | None = None) -> argparse.Namespace:
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and display top N results."
    )
    parser.add_argument(
        "file",
        type=Path,
        help="Path to the text file to analyze",
    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=10,
        help="Number of top words to display (default: 10)",
    )
    return parser.parse_args(argv)


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in the given text.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object mapping words to their frequency.
    """
    # Split on whitespace and strip punctuation from each token
    words: List[str] = []
    for token in text.split():
        # Strip common punctuation from the start and end of each word
        cleaned = token.strip(".,;:!?\"'()[]{}")
        if cleaned:
            words.append(cleaned.lower())
    return Counter(words)


def top_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top N words by frequency.

    Args:
        counter: Counter mapping words to frequencies.
        n: Number of top words to return.

    Returns:
        List of (word, frequency) tuples in descending order of frequency.
    """
    return counter.most_common(n)
