#!/usr/bin/env python3
"""
テキストファイルを読み込んで単語の出現頻度を集計し、
上位 N 件を表形式で表示するコマンドラインツール。

Usage:

Examples:
    python word_freq.py sample.txt
    python word_freq.py sample.txt --top 10
    python word_freq.py sample.txt --top 5 --case-insensitive --sort-by freq
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_text_file(path: str) -> str:
    """
    テキストファイルを読み込んで、その内容を文字列として返す。

    Args:
        path: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        PermissionError: ファイルの読み込み権限がない場合。
        UnicodeDecodeError: 文字コードエラーの場合。
    """
    p = Path(path)
    if not p.exists():
        raise FileNotFoundError(f"File not found: {path}")
    try:
        with open(p, 'r', encoding='utf-8') as f:
            return f.read()
    except UnicodeDecodeError:
        raise
    except PermissionError:
        raise

    """
    テキストを単語に分割する。

    Args:
        text: 分割対象のテキスト。
        case_insensitive: True の場合、単語を小文字化して比較する。

    Returns:
        単語のリスト。
    """

    words = []
    current_word = []
    for char in text:

            current_word.append(char)

            if current_word:

                    word = word.lower()
