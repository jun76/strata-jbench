#!/usr/bin/env python3
"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.

Usage:
    python wordfreq.py <input_file> [--top N] [--output FILE]

Example:
    python wordfreq.py sample.txt --top 10
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def read_file(filepath: str) -> str:
    """Read the contents of a text file.

    Args:
        filepath: Path to the input text file.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        PermissionError: If the file cannot be read.
    """
    path = Path(filepath)
    return path.read_text(encoding="utf-8")


def tokenize(text: str) -> List[str]:
    """Tokenize text into words, converting to lowercase.

    Words are defined as sequences of alphanumeric characters.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens.
    """
    tokens: List[str] = []
    for word in text.split():
        # Strip non-alphanumeric characters from the ends
        word = word.strip(".,;:!?\"'()[]{}")
        if word:
            tokens.append(word.lower())
    return tokens


def count_words(tokens: List[str]) -> Dict[str, int]:
    """Count the frequency of each word in the token list.

    Args:
        tokens: A list of word tokens.

    Returns:
        A dictionary mapping words to their occurrence counts.
    """
    return Counter(tokens)


def top_n_words(counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words.

    Args:
        counts: Dictionary of word -> count.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples sorted by count descending.
    """
    sorted_items = sorted(counts.items(), key=lambda x: x[1], reverse=True)
    return sorted_items[:n
