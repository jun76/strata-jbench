"""Command-line tool to count word frequencies in a text file and display top N results."""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """Extract words from text, lowercased, removing punctuation."""
    # Split on non-alphanumeric characters, convert to lowercase
    words = re.findall(r'[a-z]+', text.lower())
    return words


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text."""
    words = extract_words(text)
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top n most common words as (word, count) tuples."""
    return counter.most_common(n)


def format_table(top_items: List[Tuple[str, int]]) -> str:
    """Format top items as a table string."""
    if not top_items:
        return "No words found."
    
    # Calculate max word length for alignment
    max_word_len = max(len(word) for word, _ in top_items)
    max_count_len = max(len(str(count)) for _, count in top_items)
    
    # Header
    header_word = "Word"
    header_count = "Count"
    
    # Build table lines
    lines = []
    lines.append(f"{'Word'.ljust(max_word_len)} | {'Count'.rjust(max_count_len)}")
    lines.append("-" * max_word_len + " | " + "-" * max_count_len)
    
    for word, count in top_items:
        lines.append(f"{word.ljust(max_word_len)} | {str(count).rjust(max_count_len)}")
    
    return "\n".join(lines)
