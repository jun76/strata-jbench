def tokenize(text: str, case_insensitive: bool = True) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト文字列。
        case_insensitive: True の場合、単語を小文字に正規化する。

    Returns:
        抽出された単語のリスト。

    Note:
        単語は英数字とハイフン、アポストロフィを含む連続文字列として定義される。
        例: "don't" -> "don't", "state-of-the-art" -> "state-of-the-art"
    """
    # 英数字、ハイフン、アポストロフィを含む単語パターン
    pattern = re.compile(r"[a-zA-Z0-9]+(?:['-][a-zA-Z0-9]+)*")
    words = pattern.findall(text)
    if case_insensitive:
        words = [w.lower() for w in words]
    return words
