"""Core logic for counting word frequencies in text."""

from collections import Counter
from typing import List, Tuple


def count_words(text: str, ignore_case: bool = True, min_length: int = 1) -> List[Tuple[str, int]]:
    """Count word frequencies in the given text.

    Words are tokenized by splitting on whitespace and stripping punctuation
    from the start and end of each token.

    Args:
        text: The input text to analyze.
        ignore_case: If True, words are lowercased before counting.
        min_length: Minimum word length to include in the count.

    Returns:
        A list of (word, count) tuples sorted by count descending,
        then by word ascending for ties.
    """
    import string

    raw_tokens = text.split()
    cleaned_tokens: List[str] = []
    for token in raw_tokens:
        # Strip leading and trailing punctuation (and whitespace)
        stripped = token.strip(string.punctuation + string.whitespace)
        if not stripped:
            continue
        if ignore_case:
            stripped = stripped.lower()
        if len(stripped) >= min_length:
            cleaned_tokens.append(stripped)

    counter = Counter(cleaned_tokens)
    # Sort by count descending, then by word ascending for stable ordering
    sorted_items: List[Tuple[str, int]] = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items
