wordfreq.py - 単語出現頻度を上位 N 件表形式で表示するコマンドラインツール。

使い方:
    python wordfreq.py FILE [-n N] [--min-length LEN] [--case-sensitive]

from __future__ import annotations

import argparse
import re

from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        抽出された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン・アポストロフィを含む単語を抽出
    words: List[str] = re.findall(r"[a-z0-9]+(?:[-''][a-z0-9]+)*", text, re.IGNORECASE)
    return words


def count_words(words: List[str], min_length: int = 1) -> Counter[str]:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語リスト。
        min_length: 最小単語長（この長さ未満の単語は除外）。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    filtered = [w for w in words if len(w) >= min_length]
    return Counter(filtered)
