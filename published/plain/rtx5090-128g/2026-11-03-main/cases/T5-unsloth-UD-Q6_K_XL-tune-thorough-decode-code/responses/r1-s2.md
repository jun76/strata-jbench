#!/usr/bin/env python3
"""
word_count.py - A command-line tool to count word frequencies in a text file.

Reads a text file, counts the occurrence of each word, and displays the top N
words in a tabular format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize the input text into a list of lowercase words.

    Words are defined as sequences of alphanumeric characters and underscores.
    Non-word characters are treated as separators.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase word tokens.
    """
    # Use regex to find all word-like tokens (sequences of word characters)
    tokens = re.findall(r'\b\w+\b', text.lower())
    return tokens


def count_words(tokens: List[str]) -> Counter:
    """
    Count the frequency of each word in the token list.

    Args:
        tokens: A list of word tokens.

    Returns:
        A Counter object mapping each word to its count.
    """
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count), sorted by count descending,
        then by word ascending for ties.
    """
    # most_common returns by count descending, then insertion order.
    # To ensure consistent ordering for ties (alphabetical), we sort explicitly.
    items = counter.most_common()
    # Sort by count descending, then by word ascending
    sorted_items = sorted(items, key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
