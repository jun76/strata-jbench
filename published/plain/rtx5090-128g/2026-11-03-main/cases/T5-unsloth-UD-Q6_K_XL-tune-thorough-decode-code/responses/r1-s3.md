### 1. 実装: `word_freq.py`

```python
#!/usr/bin/env python3
"""
単語の出現頻度をカウントし、上位 N 件を表形式で表示するコマンドラインツール。

このツールはテキストファイルを読み込み、単語単位で頻度を数えます。
単語の判定は、英数字のみを単語として扱い、それ以外は区切り文字として扱います。
大文字小文字は無視してカウントします。

Usage:
    python word_freq.py <file> [-n TOP_N]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストに分割する。

    英数字の連続を単語とみなし、それ以外（空白、句読点など）は区切りとして扱う。
    大文字は小文字に変換する。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 単語のリスト。
    """
    # [a-zA-Z0-9]+ にマッチするすべての単語を抽出
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_frequencies(words: List[str]) -> Counter[str]:
    """
    単語リストから頻度をカウントする。

    Args:
        words (List[str]): 単語のリスト。

    Returns:
        Counter[str]: 単語とその出現回数をまとめた Counter オブジェクト。
    """
    return Counter(words)


def get_top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    最も出現頻度が高い上位 n 個の単語とその回数を取得する。

    Args:
        counter (Counter[str]): 頻度をまとめた Counter オブジェクト。
        n (int): 取得する上位の件数。

    Returns:
        List[Tuple[str, int]]: (単語, 出現回数) のタプルのリスト。
    """
    return counter.most_common
