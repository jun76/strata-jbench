#!/usr/bin/env python3
"""
単語出現頻度集計ツール

テキストファイルを読み込み、単語の出現頻度を計算して上位 N 件を表形式で表示します。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize_text(text: str) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。非英数字文字は区切りとして扱われる。
    """
    # 英数字のみを抽出し、小文字に変換
    words: List[str] = re.findall(r'\b[a-zA-Z0-9]+\b', text.lower())
    return words


def count_word_frequencies(text: str) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = tokenize_text(text)
    return Counter(words)


def get_top_n(frequencies: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度が高い順に上位 n 件の単語と回数を返す。

    Args:
        frequencies: 単語頻度のカウント結果。
        n: 取得する件数。

    Returns:
        (単語, 出現回数) のタプルのリスト。降順。
    """
    return frequencies.most_common(n)
