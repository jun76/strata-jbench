#!/usr/bin/env python3
"""
単語出現頻度集計ツール

テキストファイルを読み込んで単語の出現頻度を集計し、
上位 N 件を表形式で表示するコマンドラインツール。

使用例:
    python word_freq.py input.txt -n 10
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import Optional


def tokenize_text(text: str) -> list[str]:
    """
    テキストから単語を抽出する。

    単語は英数字およびハイフンで構成される文字列として抽出され、
    すべて小文字に正規化される。

    Args:
        text: 解析対象のテキスト。

    Returns:
        単語のリスト。
    """
    import re

    # 英数字とハイフンを許容するトークン
    tokens = re.findall(r"[a-z0-9]+(?:-[a-z0-9]+)*", text.lower())
    return tokens


def count_word_frequencies(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 解析対象のテキスト。

    Returns:
        単語から出現回数をマッピングした Counter オブジェクト。
    """
    tokens = tokenize_text(text)
    return Counter(tokens)


def get_top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """
    Counter から上位 n 件の (単語, 回数) を取得する。

    Args:
        counter: 単語出現頻度の Counter。
        n: 取得する件数。

    Returns:
        (単語, 出現回数) のタプルのリスト。頻度が高い順、同じ頻度なら辞書順。
    """
    # most_common は頻度降順だが、同じ頻度の場合は辞書順ではない可能性がある
    # 明示的にソートする
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
