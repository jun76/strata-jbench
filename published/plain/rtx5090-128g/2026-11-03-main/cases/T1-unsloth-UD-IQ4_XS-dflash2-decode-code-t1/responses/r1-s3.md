コマンドラインインターフェース: テキストファイルを読み込み、単語出現頻度を
集計して上位 N 件を表形式で出力する。

Usage:

"""

from __future__ import annotations

import argparse

def tokenize(
    text: str,
    case_sensitive: bool = False,
    keep_punctuation: bool = False,
) -> list[str]:
    """
    テキストを単語トークンに分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字と小文字を区別する。
        keep_punctuation: True の場合、句読点を保留する（デフォルト: 除去）。

    Returns:
        単語トークンのリスト。
    """
    if not case_sensitive:
        text = text.lower()

    if keep_punctuation:
        # 空白区切りで十分（句読点付きのトークンをそのまま残す）
        tokens = text.split()
    else:
        # アルファベットと数字のみを含む単語に分割
        tokens = re.findall(r"[a-z0-9]+", text)

    return tokens
