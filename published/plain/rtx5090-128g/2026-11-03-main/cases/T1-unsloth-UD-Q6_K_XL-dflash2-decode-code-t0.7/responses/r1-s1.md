## wordfreq.py

```python
"""
wordfreq.py - テキストファイルの単語出現頻度を上位 N 件で表示する CLI ツール。

使い方:
    python wordfreq.py input.txt [-n 10] [--case-sensitive] [--min-len 1]
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from pathlib import Path
from typing import TextIO


# 単語を抽出するための正規表現パターン
# Unicode 文字、数字、ハイフン、アポストロフィを含む連続した文字列
WORD_PATTERN = re.compile(r"[\w'-]+", re.UNICODE)


@dataclass(frozen=True)
class WordCount:
    """単語とその出現回数を保持する不変データクラス。

    Attributes:
        word: 集計対象の単語（正規化後）。
        count: 出現回数。
    """

    word: str
    count: int

    def __lt__(self, other: WordCount) -> bool:
        """降順ソート用比較: 回数の多い順、同数なら辞書順。"""
        if self.count != other.count:
            return self.count > other.count
        return self.word < other.word


def extract_words(text: str, *, case_sensitive: bool = False) -> list[str]:
    """テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
                        False の場合、小文字に正規化して比較する。

    Returns:
        抽出された単語のリスト（出現順）。
    """
    if not case_sensitive:
        text = text.lower()
    return WORD_PATTERN.findall(text)
