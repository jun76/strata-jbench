"""

テキストファイルを読み込み、単語の出現頻度を計算し、
上位 N 件の単語とその頻度を表形式で出力するコマンドラインツール。
"""

import argparse

def read_file(filepath: str) -> str:
    """指定されたパスのテキストファイルを読み込み、内容（文字列）を返す。

    Args:
        filepath (str): 読み込むテキストファイルのパス。

    Returns:
        str: ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み取り中にエラーが発生した場合。
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。

    単語は英数字、ハイフン、アポストロフィを含む連続文字列として定義され、
    すべて小文字に変換される。

    Args:
        text (str): 解析対象のテキスト。

    Returns:
        List[str]: 抽出された単語のリスト（小文字）。
    """
    # 英数字、ハイフン、アポストロフィを含む単語を抽出
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter:
    """単語リストから出現頻度をカウントする。

    Args:
        words (List[str]): 単語のリスト。

    Returns:
        Counter: 単語と出現回数のカウンター。
    """
    return Counter(words)
