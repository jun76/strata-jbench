"""
word_freq_counter.py

A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    Count the frequency of each word in the given text.

    Words are split by non-alphanumeric characters and lowercased.
    Empty strings are ignored.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object mapping each word to its occurrence count.
    """
    # Normalize to lowercase and split by non-alphanumeric characters
    words = re.findall(r'[a-z0-9]+', text.lower())
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top items to return.

    Returns:
        A list of tuples (word, count) for the top N most common words.
    """
    return counter.most_common(n)


def format_table(data: List[Tuple[str, int]]) -> str:
    """
    Format the word frequency data as a table.

    Args:
        data: A list of tuples (word, count).

    Returns:
        A formatted string representing the table.
    """
    if not data:
        return "No words found."

    # Determine column widths
    rank_width = len(str(len(data)))
    word_width = max(len(word) for word, _ in data)
    count_width = max(len(str(count)) for _, count in data)

    # Build header
    header = f"{'Rank':>{rank_width}} | {'Word':<{word_width}} | {'Count':>{count_width}}"
    separator = f"{'-' * rank_width}-+-{'-' * word_width}-+-{'-' * count_width}"
