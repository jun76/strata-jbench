def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Split text into words (whitespace-separated), optionally lower-cased,
    and filter out tokens shorter than `min_length`.

    Args:
        text: The input text string.
        case_sensitive: If False, convert tokens to lowercase.
        min_length: Minimum token length to include.

    Returns:
        A list of token strings.
    """
    tokens = text.split()
    if not case_sensitive:
        tokens = [t.lower() for t in tokens]
    return [t for t in tokens if len(t) >= min_length]


def count_words(tokens: List[str]) -> Counter[str]:
    """
    Count occurrences of each token.

    Args:
        tokens: List of word strings.

    Returns:
        A Counter mapping each token to its count.
    """
    return Counter(tokens)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Return the top N most common words, sorted by count descending,
    then by word ascending for tie-breaking.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples.
    """
    # most_common returns ties in arbitrary order; sort by (-count, word)
    items = counter.most_common()
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
