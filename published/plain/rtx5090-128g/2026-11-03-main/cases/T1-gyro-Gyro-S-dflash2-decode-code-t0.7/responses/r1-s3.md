"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を集計し上位N件を表示する。

使い方:
    python word_freq.py --file input.txt --top 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def read_text(filepath: str) -> str:
    """テキストファイルを読み込み、内容を返す。

    Args:
        filepath: テキストファイルのパス。

    Returns:
        ファイルの全文。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        OSError: 読み込みエラーが発生した場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def count_words(text: str) -> Counter:
    """テキスト中の単語出現頻度を集計する。

    単語は空白区切りで分割し、小文字に正規化される。

    Args:
        text: 集計対象のテキスト。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    words = text.lower().split()
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (word, count) を取得する。

    Args:
        counter: 単語と出現回数の Counter。
        n: 取得する件数。

    Returns:
        上位 n 件の (word, count) のリスト。
    """
    return counter.most_common(n)
