#!/usr/bin/env python3
"""
word_freq.py — コマラインツールでテキストファイルの単語出現頻度を集計し、
上位 N 件を表形式で表示する。

使い方:
    python word_freq.py INPUT_FILE [--top N] [--ignore-case] [--min-len LEN]

例:
    python word_freq.py sample.txt --top 10 --ignore-case --min-len 3

依存:
    標準ライブラリのみ (argparse, collections, sys, os, re)
"""

from __future__ import annotations

import argparse
import sys
from collections import Counter
from pathlib import Path
import re


def tokenize(text: str, *, ignore_case: bool = True, min_len: int = 1) -> list[str]:
    """テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        ignore_case: True の場合、単語を小文字化して集計する。
        min_len: 単語の最小文字数。これ未満の単語は除外する。

    Returns:
        単語のリスト（既に正規化済み）。
    """
    # 単語を抽出する正規表現: 英数字とハイフンを含むトークン
    # \b で境界を指定し、英数字およびハイフンを許可する。
    pattern = re.compile(r"[A-Za-z0-9]+(?:-[A-Za-z0-9]+)*")
    tokens = pattern.findall(text)
    if ignore_case:
        tokens = [t.lower() for t in tokens]
    tokens = [t for t in tokens if len(t) >= min_len]
    return tokens


def count_words(text: str, *, ignore_case: bool = True, min_len: int = 1) -> Counter[str]:
    """テキストから単語の出現頻度を Counter で返す。

    Args:
        text: 入力テキスト。
        ignore_case: True の場合、単語を小文字化して集計する。
        min_len: 単語の最小文字数。これ未満の単語は除外する。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    tokens = tokenize(text, ignore_case=ignore_case, min_len=min_len)
    return Counter(tokens)
