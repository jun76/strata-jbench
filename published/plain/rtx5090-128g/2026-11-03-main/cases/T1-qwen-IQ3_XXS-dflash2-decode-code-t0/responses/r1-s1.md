def read_file(file_path: Path) -> str:
    """
    指定されたファイルを読み込み、テキスト内容を返す。

    Args:
        file_path: 読み込むファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        PermissionError: 読み取り権限がない場合。
    """
    try:
        return file_path.read_text(encoding="utf-8")
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {file_path}")
    except PermissionError:
        raise PermissionError(f"ファイルの読み取り権限がありません: {file_path}")


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 分割対象のテキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 単語の最小長（これ未満の単語は除外）。

    Returns:
        単語のリスト。
    """
    # 非英数字を空白で置き換えて分割
    import re
    # 英数字とハイフン、アポストロフィを保持し、それ以外は空白で分割
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)

    if not case_sensitive:
        words = [w.lower() for w in words]

    # 最小長でフィルタ
    words = [w for w in words if len(w) >= min_length]

    return words
