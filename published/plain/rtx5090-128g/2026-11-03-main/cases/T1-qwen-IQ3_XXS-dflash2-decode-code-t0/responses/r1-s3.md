#!/usr/bin/env python3
"""
word_counter.py - テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

Usage:
    python word_counter.py <file> [-n N] [--case-sensitive] [--min-length L]
"""

import argparse

from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 単語の最小文字数（これ未満の単語は除外）。

    Returns:
        抽出された単語のリスト。
    """
    # 英数字とハイフンを含む単語を抽出（例: "don't" は "don" と "t" に分割される）
    # より厳密にしたい場合は正規表現を調整可能
    pattern = r"[a-zA-Z0-9]+"
    words = re.findall(pattern, text)

    if not case_sensitive:
        words = [w.lower() for w in words]

    # 最小長でフィルタ
    words = [w for w in words if len(w) >= min_length]

    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)
