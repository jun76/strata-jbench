word_counter.py - テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 単語の最小長（これ未満の単語は除外）。

    Returns:
        抽出された単語のリスト。
    """
    # 英数字とハイフンを含む単語を抽出（例: "don't" は "don" と "t" に分割される）
    # より厳密にしたい場合は正規表現を調整可能
    words = re.findall(r"[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*", text)

    if not case_sensitive:
        words = [w.lower() for w in words]

    words = [w for w in words if len(w) >= min_length]
    return words


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。

        min_length: 単語の最小長。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = extract_words(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(words)
