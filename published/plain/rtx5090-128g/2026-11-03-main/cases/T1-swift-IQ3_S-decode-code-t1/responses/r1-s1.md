"""
Top N Word Frequency Command-Line Tool

Reads a text file, counts word frequencies, and displays the top N most frequent words
in a formatted table.
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.
    
    Words are split by whitespace and punctuation is stripped.
    Case-insensitive counting.
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter object with word frequencies.
    """
    # Normalize: lowercase, remove punctuation from each word
    words = []
    for token in text.split():
        # Strip common punctuation from start and end of token
        word = token.strip(".,;:!?\"'()[]{}<>")
        if word:  # Skip empty strings
            words.append(word.lower())
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top entries to return.
        
    Returns:
        A list of tuples (word, count) sorted by count descending,
        then alphabetically for ties.
    """
    # most_common() already sorts by count descending.
    # For ties, Python's Counter.most_common uses insertion order,
    # so we sort explicitly for consistent alphabetical tie-breaking.
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
