"""
wordfreq.py - Text file word frequency counter CLI tool.

Reads a text file, counts word frequencies, and displays the top N words
in a table format.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    Extract words from text, converting to lowercase and removing punctuation.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase words.
    """
    # Find all sequences of alphanumeric characters (words)
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word.
    
    Args:
        words: A list of words.
        
    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.
        
    Returns:
        A list of (word, count) tuples for the top N words.
    """
    return counter.most_common(n)
