#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を集計し、
上位 N 件を表形式で表示する。

Usage:
    python word_freq.py <input_file> [-n TOP_N] [--case-sensitive]
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def load_words(file_path: str, case_sensitive: bool = False) -> List[str]:
    """
    指定されたテキストファイルを読み込み、単語のリストを返す。

    Args:
        file_path: 読み込むテキストファイルのパス。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、
                        すべて小文字に変換する。

    Returns:
        単語のリスト。非英数字は単語の区切りとして扱われ、単語自体から
        非英数字は取り除かれる。
    """
    with open(file_path, 'r', encoding='utf-8') as f:
        text = f.read()

    if not case_sensitive:
        text = text.lower()

    # 非英数字をスペースで区切る
    cleaned_chars: List[str] = []
    for char in text:
        if char.isalnum():
            cleaned_chars.append(char)
        else:
            cleaned_chars.append(' ')

    cleaned_text = ''.join(cleaned_chars)
    words = cleaned_text.split()
    # 空文字列を除外
    words = [w for w in words if w]
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語の出現頻度を集計する。

    Args:
        words: 単語のリスト。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    return Counter(words)
