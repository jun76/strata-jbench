def read_file(filepath: str) -> str:
    """指定されたパスのテキストファイルを読み込み、内容を文字列として返す。

    Args:
        filepath: 読み込み対象のファイルパス。

    Returns:
        ファイルの内容を示す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み取りに失敗した場合。
    """
    path = Path(filepath)
    if not path.exists():
        raise FileNotFoundError(f"File not found: {filepath}")
    return path.read_text(encoding="utf-8")


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。単語はアルファベット、数字、ハイフン、アポストロフィを含む。
    大文字小文字は無視して小文字に変換する。

    Args:
        text: 入力テキスト。

    Returns:
        単語のリスト（小文字化済み）。
    """
    words: List[str] = []
    current_word: List[str] = []
    for char in text:
        if char.isalnum() or char == '-' or char == "'":
            current_word.append(char)
        else:
            if current_word:
                words.append(''.join(current_word).lower())
                current_word = []
    if current_word:
        words.append(''.join(current_word).lower())
    return words


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        各単語と出現回数を格納した Counter オブジェクト。
    """
    return Counter(words)
