```python
#!/usr/bin/env python3
"""
word_freq: コマンドラインツールでテキストファイルを読み込み、単語出現頻度の上位 N 件を表形式で表示する。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    単語とは、Unicode のアルファベット・数字・ハイフンを含む連続文字列とする。
    すべて小文字に正規化する。
    """
    # Unicode 対応の単語抽出: 英数字・ハイフン・アポストロフィを含む
    pattern = re.compile(r"[a-zA-Z0-9'\-]+")
    return [word.lower() for word in pattern.findall(text)]


def compute_frequencies(text: str) -> Counter:
    """
    テキスト中の単語出現頻度を計算する。
    """
    tokens = tokenize(text)
    return Counter(tokens)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度が高い上位 n 件の (word, count) のリストを返す。
    頻度が同じ場合は辞書順（アルファベット順）で並べる。
    """
    # most_common は頻度降順で、同じ頻度の場合は挿入順だが、
    # ここでは辞書順にソートし直す
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(rows: List[Tuple[str, int]]) -> str:
    """
    (word, count) のリストを ASCII テーブルとしてフォーマットする。
    """
    if not rows:
        return ""

    # 列の幅を計算
    word_width = max(len(word) for word, _ in rows)
    count_width = max(len(str(count)) for _, count in rows)

    # ヘッダー
    lines = []
    header = f"{'Word':<{word_width}} | {'Count':>{count_width}}"
    separator = f"{'-' * word_width}-+-{'-' * count_width}"

    lines.append(header)
    lines.append(separator)
