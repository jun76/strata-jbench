"""
word_frequency.py

テキストファイルを読み込み、単語の出現頻度を計算して上位 N 件を表形式で表示するコマンドラインツール。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Dict, Optional


def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストに分割する。
    英数字の連続を単語とし、それ以外（句読点、空白など）は区切りとして扱う。
    単語は小文字に変換される。

    Args:
        text (str): 入力テキスト

    Returns:
        List[str]: 単語のリスト
    """
    # 英数字の連続を抽出し、小文字化
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_frequencies(words: List[str]) -> Dict[str, int]:
    """
    単語リストから頻度を計算する。

    Args:
        words (List[str]): 単語のリスト

    Returns:
        Dict[str, int]: 単語とその出現回数の辞書
    """
    return dict(Counter(words))
