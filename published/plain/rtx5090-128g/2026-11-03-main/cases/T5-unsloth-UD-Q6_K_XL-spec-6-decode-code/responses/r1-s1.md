```python
#!/usr/bin/env python3
"""
word_counter.py — テキストファイルから単語出現頻度を上位N件表示するCLIツール。

Usage:
    python word_counter.py input.txt [-n 10] [-v] [--case-sensitive] [--min-length 1]
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: Trueの場合、大文字小文字を区別する。Falseの場合、小文字に変換。
        min_length: 抽出する単語の最小文字数。この長さ未満の単語は除外。

    Returns:
        抽出された単語のリスト（出現順、重複含む）。
    """
    if not case_sensitive:
        text = text.lower()

    # 単語の定義: 英数字・ハイフン・アポストロフィを含む連続文字列
    # 正規表現ではなく、簡単なパースで対応
    words: List[str] = []
    current_word: List[str] = []
    for char in text:
        if char.isalnum() or char in ("-", "'"):
            current_word.append(char)
        else:
            if current_word:
                word = "".join(current_word)
                # 先頭・末尾のハイフンやアポストロフィを除去
                word = word.strip("-'")
                if word and len(word) >= min_length:
                    words.append(word)
                current_word = []
    if current_word:
        word = "".join(current_word)
        word = word.strip("-'")
        if word and len(word) >= min_length:
            words.append(word)

    return words
