"""
word_counter.py

テキストファイルの単語出現頻度を上位 N 件まで表形式で表示するコマンドラインツール。
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path


def normalize_words(text: str) -> list[str]:
    """
    テキストから単語を抽出し、小文字化してリストとして返す。

    Args:
        text: 入力テキスト

    Returns:
        正規化された単語のリスト
    """
    # 単語のみを抽出（英字、数字、ハイフン、アポストロフィを含む）
    words = re.findall(r"[a-zA-Z0-9'\-]+", text.lower())
    return words


def count_word_frequencies(words: list[str]) -> Counter[str]:
    """
    単語リストの出現頻度を Counter で集計する。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    return Counter(words)


def get_top_n_words(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """
    出現頻度の高い上位 n 件の (単語, 回数) のリストを返す。

    Args:
        counter: 単語の出現頻度を表す Counter
        n: 取得する上位件数

    Returns:
        [(単語, 出現回数), ...] のリスト（頻度降順）
    """
    return counter.most_common(n)
