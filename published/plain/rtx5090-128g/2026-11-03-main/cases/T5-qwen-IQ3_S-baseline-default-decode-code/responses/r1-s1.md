#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントして、
上位 N 件の単語とその頻度を表形式で表示するコマンドラインツール。
"""

import argparse
from collections import Counter
import re
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    入力テキストから単語を抽出します。
    
    単語は、英数字とハイフン、アポストロフィを含む文字列として定義されます。
    大文字小文字は無視して小文字に変換されます。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 抽出された単語のリスト。
    """
    # 英数字、ハイフン、アポストロフィを含む文字列を単語として抽出
    words = re.findall(r"[a-z0-9'-]+", text.lower())
    return words


def count_words(words: List[str]) -> Counter[str]:
    """
    単語リストの出現頻度をカウントします。

    Args:
        words (List[str]): 単語のリスト。

    Returns:
        Counter[str]: 単語とその出現回数を格納した Counter オブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counter オブジェクトから最も頻度の高い上位 n 件の単語を返します。

    Args:
        counter (Counter[str]): 単語の頻度を表す Counter オブジェクト。
        n (int): 返す上位単語の数。

    Returns:
        List[Tuple[str, int]]: (単語, 頻度) のタプルのリスト。
    """
    return counter.most_common(n)
