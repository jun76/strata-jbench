Command-line tool to count word frequencies in a text file and display the top N words.

Usage:
    python word_counter.py <file> [-n N] [-c] [-s]

Options:
    -n, --top N       Number of top words to display (default: 10)
    -c, --case-sensitive  Case-sensitive counting (default: False, case-insensitive)
    -s, --sort-by-count   Sort by count descending (default: True)
"""

import argparse

from collections import Counter

def read_file(filepath: str) -> str:
    """Read the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """Tokenize text into words.

    Words are defined as sequences of alphanumeric characters and underscores.
    If case_sensitive is False, all words are converted to lowercase.

    Args:
        text: The input text to tokenize.
        case_sensitive: Whether to preserve case. Defaults to False.

    Returns:
        A list of word tokens.
    """
    # Extract words using regex: sequences of word characters (letters, digits, underscores)
    words = re.findall(r"\b\w+\b", text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_words(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: A list of word tokens.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    return Counter(words)
