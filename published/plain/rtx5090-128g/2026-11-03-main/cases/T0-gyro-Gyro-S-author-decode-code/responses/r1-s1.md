"""
CLI tool that reads a text file, counts word frequencies, and displays the top N words.
"""
import argparse
import sys
from collections import Counter
from typing import Dict, List, Tuple


def count_words(text: str) -> Dict[str, int]:
    """
    Count word frequencies in the given text.

    Words are defined as sequences of alphanumeric characters (letters and digits).
    Case-insensitive counting is performed.

    Args:
        text: The input text to analyze.

    Returns:
        A dictionary mapping each word (lowercased) to its occurrence count.
    """
    # Extract words: sequences of alphanumeric characters
    import re
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return dict(Counter(words))


def top_n_words(
    word_counts: Dict[str, int], n: int
) -> List[Tuple[str, int]]:
    """
    Return the top N words by frequency, with ties broken by alphabetical order.

    Args:
        word_counts: Dictionary mapping words to their counts.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count), sorted by count descending, then by word ascending.
    """
    items = sorted(word_counts.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
