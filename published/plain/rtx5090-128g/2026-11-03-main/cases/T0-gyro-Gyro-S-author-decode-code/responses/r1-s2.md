def read_words(filepath: str) -> List[str]:
    """
    Read a text file and extract words.

    Words are split by whitespace and lowercased.

    Args:
        filepath: Path to the text file.

    Returns:
        A list of lowercase words extracted from the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        content = f.read()
    words = content.split()
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object mapping each word to its count.
    """
    return Counter(words)


def get_top_n_words(
    word_counts: Counter, n: int
) -> List[Tuple[str, int]]:
    """
    Get the top N words by frequency.

    Args:
        word_counts: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count), sorted by descending count,
        then by word alphabetically for ties.
    """
    # Sort by count descending, then by word ascending for consistent tie-breaking
    sorted_items = sorted(
        word_counts.items(), key=lambda item: (-item[1], item[0])
    )
    return sorted_items[:n]
