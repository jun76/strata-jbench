#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルから単語の出現頻度を集計し、
上位 N 件を表形式で表示する。

使い方:
    python word_freq.py <input_file> [--top N] [--sort {freq,alpha}]

例:
    python word_freq.py sample.txt --top 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def parse_args(argv: List[str] = None) -> argparse.Namespace:
    """コマンドライン引数を解析する。

    Args:
        argv: 引数リスト。None の場合 sys.argv を使用。

    Returns:
        解析された引数オブジェクト。
    """
    parser = argparse.ArgumentParser(
        description="テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示します。",
    )
    parser.add_argument(
        "input_file",
        help="入力テキストファイルのパス",
    )
    parser.add_argument(
        "--top",
        type=int,
        default=10,
        help="表示する上位単語数（デフォルト: 10）",
    )
    parser.add_argument(
        "--sort",
        choices=["freq", "alpha"],
        default="freq",
        help="ソート順: freq（頻度順）または alpha（アルファベット順）",
    )
    return parser.parse_args(argv)


def read_words_from_file(filepath: str) -> List[str]:
    """ファイルから単語リストを読み込む。

    Args:
        filepath: ファイルパス。

    Returns:
        単語のリスト。空白・改行で分割し、小文字化して返す。
    """
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            content = f.read()
    except FileNotFoundError as e:
        raise FileNotFoundError(f"ファイルが見つかりません: {filepath}") from e
    except IOError as e:
        raise IOError(f"ファイルの読み込みエラー: {filepath}") from e

    # 空白・改行で分割し、小文字化
    words = content.lower().split()
    return words
