def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    単語は英数字とハイフンで構成される文字列とし、
    大文字小文字は無視して小文字に変換する。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 抽出された単語のリスト。
    """
    # 正規表現で単語を抽出（英数字、ハイフン、アポストロフィを含む）

    words = re.findall(r"[a-zA-Z0-9'-]+", text)
    return [word.lower() for word in words]
