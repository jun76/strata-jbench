def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    Args:
        text: 入力テキスト文字列。
        
    Returns:
        小文字化された単語のリスト。
    """
    # 英数字とハイフンを含む単語を抽出（日本語対応は簡易的に空白区切りも考慮）
    # ここでは一般的な英語/コード用テキストを想定し、英数字の連続を単語とする
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    return [word.lower() for word in words]

    """
