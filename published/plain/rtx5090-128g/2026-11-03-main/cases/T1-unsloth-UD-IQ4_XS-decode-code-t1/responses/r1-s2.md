# デフォルトの単語分割ルール:
# 英字・数字・アンダースコア・日本語などの文字を単語とし、
# それ以外を区切りとする。
DEFAULT_WORD_REGEX = r"[\w\u3040-\u30ff\u4e00-\u9fff]+"


def tokenize(
    text: str,
    *,
    case_sensitive: bool = False,
    pattern: str | None = None,
) -> List[str]:
    """テキストから単語トークンのリストを取り出す。

    Args:
        text: 入力テキスト。
        case_sensitive: True なら大文字小文字を区別する。
        pattern: 単語マッチに使う正規表現。None なら DEFAULT_WORD_REGEX。

    Returns:
        単語トークンのリスト（出現順）。
    """
    if pattern is None:
        pattern = DEFAULT_WORD_REGEX

    regex = re.compile(pattern, re.UNICODE)
    tokens = [m.group(0) for m in regex.finditer(text)]

    if not case_sensitive:
        tokens = [t.lower() for t in tokens]

    return tokens
