#!/usr/bin/env python3
"""
wordfreq — 単語出現頻度集計ツール

テキストファイルを読み込み、単語ごとの出現頻度を計算し、
上位 N 件を表形式で出力するコマンドラインツール。

Usage:
    python wordfreq.py input.txt -n 10
    python wordfreq.py input.txt --top 20 --case-sensitive
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def parse_arguments(argv: List[str] | None = None) -> argparse.Namespace:
    """コマンドライン引数を解析する。

    Args:
        argv: 引数リスト。None の場合、sys.argv[1:] を使用する。

    Returns:
        解析された引数を含む Namespace オブジェクト。
    """
    parser = argparse.ArgumentParser(
        description="テキストファイルから単語の出現頻度をカウントし、上位 N 件を表形式で表示する。",
        formatter_class=argparse.RawTextHelpFormatter,
    )
    parser.add_argument(
        "input_file",
        help="読み込むテキストファイルのパス",
    )
    parser.add_argument(
        "-n", "--top",
        type=int,
        default=10,
        help="表示する上位単語の数（デフォルト: 10）",
    )
    parser.add_argument(
        "--case-sensitive",
        action="store_true",
        default=False,
        help="大文字・小文字を区別する（デフォルト: 区別しない）",
    )
    parser.add_argument(
        "--min-length",
        type=int,
        default=1,
        help="カウント対象にする単語の最小文字数（デフォルト: 1）",
    )
    parser.add_argument(
        "--output-format",
        choices=["table", "csv", "json"],
        default="table",
        help="出力形式: table（表形式）, csv, json（デフォルト: table）",
