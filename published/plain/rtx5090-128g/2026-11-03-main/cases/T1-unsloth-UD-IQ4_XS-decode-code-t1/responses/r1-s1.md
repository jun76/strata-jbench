word_freq.py - Text file word frequency analysis tool.

This module provides a command-line tool that reads a text file, counts word

import argparse
import re

from collections import Counter
from typing import Dict, List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into words, preserving case by default (caller handles case folding).

    This function splits text into tokens based on whitespace and non-alphanumeric
    characters. Words are defined as sequences of alphanumeric characters and apostrophes.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of word tokens extracted from the text.
    """
    # Match sequences of word characters (including apostrophes for contractions)
    # This treats apostrophes within words as part of the word (e.g., "don't" -> "don't")
    pattern = re.compile(r"[a-zA-Z0-9']+|[0-9]+")
    return pattern.findall(text)


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    Count word occurrences in the given text.

    Args:
        text: The input text to analyze.
        case_sensitive: If True, preserve case when counting words.
                       If False, convert all words to lowercase before counting.

    Returns:
        A Counter object mapping each word to its occurrence count.
    """
    if case_sensitive:
        words = tokenize(text)
    else:
        words = [w.lower() for w in tokenize(text)]
    return Counter(words)
