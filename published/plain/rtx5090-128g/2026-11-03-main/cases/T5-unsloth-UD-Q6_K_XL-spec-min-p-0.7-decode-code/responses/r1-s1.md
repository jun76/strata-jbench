"""
wordfreq.py - テキストファイルから単語の出現頻度を算出し、上位N件を表形式で出力するCLIツール。

使い方:
    python wordfreq.py input.txt -n 10
    python wordfreq.py input.txt -n 5 --case-sensitive
    python wordfreq.py input.txt -n 10 -o output.csv
"""

from __future__ import annotations

import argparse

import re

from collections import Counter
from pathlib import Path

def tokenize(text: str, case_sensitive: bool = False) -> list[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: Trueの場合、大文字・小文字を区別して分割する。

    Returns:
        単語のリスト。空白や句読点で区切られたトークンを返す。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含むトークンを抽出
    tokens = re.findall(r"[a-z0-9][a-z0-9'-]*|[0-9]+", text, re.IGNORECASE)
    return tokens


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: Trueの場合、大文字・小文字を区別する。

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    tokens = tokenize(text, case_sensitive=case_sensitive)
    return Counter(tokens)
