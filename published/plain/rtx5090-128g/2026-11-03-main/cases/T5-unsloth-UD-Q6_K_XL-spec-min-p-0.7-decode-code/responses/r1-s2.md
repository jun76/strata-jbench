#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルから単語の出現頻度を集計し、上位 N 件を表示します。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出します。
    英数字の連続を単語とみなし、小文字に変換します。
    
    Args:
        text: 入力テキスト
        
    Returns:
        小文字化された単語のリスト
    """
    # 正規表現で英数字の連続を抽出
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントします。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語と出現回数の Counter オブジェクト
    """
    words = extract_words(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現回数の多い順に上位 N 件の単語と回数を取り出します。
    
    Args:
        counter: 単語と出現回数の Counter オブジェクト
        n: 取得する上位件数
        
    Returns:
        (単語, 出現回数) のタプルのリスト
    """
    return counter.most_common(n)
