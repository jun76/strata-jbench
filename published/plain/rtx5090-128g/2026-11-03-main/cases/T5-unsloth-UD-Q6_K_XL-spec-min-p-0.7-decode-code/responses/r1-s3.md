コマンドラインツール: テキストファイルを読み込んで単語の出現頻度を集計し、
上位 N 件を表形式で表示する。

使い方:
    python word_freq.py <input_file> [--top N] [--case-sensitive] [--min-length MIN_LEN]

例:
    python word_freq.py sample.txt --top 10
    python word_freq.py data.txt --case-sensitive --min-length 3
"""

def parse_arguments(args: List[str] = None) -> argparse.Namespace:
    """

    Returns:

    """
    parser = argparse.ArgumentParser(
        description="テキストファイルの単語出現頻度を上位 N 件表示するツール。"
    )
    parser.add_argument(
        "input_file",
        type=str,
        help="読み込むテキストファイルのパス",
    )
    parser.add_argument(
        "--top",
        "-n",
        type=int,
        default=10,
        help="表示する上位件数 (デフォルト: 10)",
    )
    parser.add_argument(
        "--case-sensitive",
        action="store_true",
        help="大文字小文字を区別する (デフォルト: 区別しない)",
    )
    parser.add_argument(
        "--min-length",
        type=int,
        default=1,
        help="単語の最小長 (デフォルト: 1)",
    )
    return parser.parse_args(args)
