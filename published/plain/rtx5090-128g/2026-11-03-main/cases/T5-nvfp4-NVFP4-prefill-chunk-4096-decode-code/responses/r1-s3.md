def tokenize(text: str, case_insensitive: bool = True) -> List[str]:
    """
    テキストを単語トークンに分割する。

    Args:
        text: 入力テキスト文字列。
        case_insensitive: True の場合、単語を小文字に変換してから分割する。

    Returns:
        単語トークンのリスト（句読点は除去される）。
    """
    if case_insensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含むトークンを抽出
    # 例: "don't", "well-known" を単語として扱う

    return tokens

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。

    return Counter(tokens)
