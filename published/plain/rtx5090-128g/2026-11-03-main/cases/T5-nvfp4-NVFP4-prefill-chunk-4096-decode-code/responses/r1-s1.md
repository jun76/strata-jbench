#!/usr/bin/env python3
"""
word_freq.py — テキストファイルの単語出現頻度を上位 N 件まで集計・表示する CLI ツール。

使い方:
    python word_freq.py input.txt --top 10
    python word_freq.py input.txt --top 5 --case-sensitive
    cat input.txt | python word_freq.py - --top 20

オプション:
    file            : 読み込むテキストファイルパス（'-' で標準入力）
    --top N         : 表示する上位単語数（デフォルト: 10）
    --case-sensitive: 大文字小文字を区別する（デフォルト: 無視）
    --min-length N  : この長さ未満の単語を除外（デフォルト: 1）
    --output FILE   : 結果を CSV で保存（省略時は標準出力）
"""

from __future__ import annotations

import argparse
import csv
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


# ---------------------------------------------------------------------------
# コアロジック
# ---------------------------------------------------------------------------

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語リストに分割する。

    正規表現 \\b\\w+\\b を使用し、英数字・アンダースコアで構成される
    連続文字列を単語として抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別して保持する。
                        False の場合、すべて小文字に変換する。

    Returns:
        単語のリスト（順序はテキスト中の出現順）。

    Examples:
        >>> tokenize("Hello, World! hello.", case_sensitive=False)
        ['hello', 'world', 'hello']
        >>> tokenize("Hello, World! hello.", case_sensitive=True)
        ['Hello', 'World', 'hello
