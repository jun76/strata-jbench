def tokenize(text: str, *, min_length: int = 1, case_sensitive: bool = False) -> list[str]:
    """Tokenize text into words.

    Words are defined as sequences of alphanumeric characters and underscores.
    Non-word characters are used as separators.

    Args:
        text: The input text to tokenize.
        min_length: Minimum word length to include in results.
        case_sensitive: If False, words are converted to lowercase before counting.

    Returns:
        A list of tokens (words) extracted from the text.
    """
    if not case_sensitive:
        text = text.lower()
    # Extract words: sequences of word characters (letters, digits, underscores)
    tokens = re.findall(r"\w+", text)
    if min_length > 1:
        tokens = [tok for tok in tokens if len(tok) >= min_length]
    return tokens
