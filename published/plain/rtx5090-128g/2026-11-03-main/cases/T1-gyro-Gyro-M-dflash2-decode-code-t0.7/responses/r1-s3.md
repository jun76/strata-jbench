#!/usr/bin/env python3
"""Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.

Usage:
    python wordfreq.py <file> [--top N] [--case-sensitive] [--output-format {table,csv,json}]
"""

import argparse
import json
import re
import sys
from collections import Counter
from typing import List, Tuple


def load_text(filepath: str) -> str:
    """Load text from a file.

    Args:
        filepath: Path to the text file.

    Returns:
        The full text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """Tokenize text into words.

    Words are sequences of alphanumeric characters and underscores.
    If case_sensitive is False, all words are lowercased.

    Args:
        text: The input text string.
        case_sensitive: Whether to preserve case. Defaults to False.

    Returns:
        A list of word tokens.
    """
    # Extract words: sequences of word characters (letters, digits, underscores)
    tokens = re.findall(r"\b\w+\b", text)
    if not case_sensitive:
        tokens = [t.lower() for t in tokens]
    return tokens


def count_words(tokens: List[str]) -> Counter:
    """Count occurrences of each token.

    Args:
        tokens: A list of word tokens.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(tokens)
