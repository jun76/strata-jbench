"""
text_word_counter.py

A command-line tool that reads a text file, counts word frequencies,
and displays the top N words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import Dict, List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Extracts words consisting of letters and digits, separated by non-alphanumeric
    characters. All words are converted to lowercase for case-insensitive counting.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase tokens (words).
    """
    # Extract words consisting of alphanumeric characters
    tokens = re.findall(r'[a-zA-Z0-9]+', text)
    return [token.lower() for token in tokens]


def count_words(text: str) -> Counter:
    """
    Count the frequency of each word in the given text.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object with word frequencies.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top entries to return.

    Returns:
        A list of tuples (word, count) sorted by frequency descending,
        then alphabetically for ties.
    """
    # Sort by count descending, then by word ascending for tie-breaking
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
