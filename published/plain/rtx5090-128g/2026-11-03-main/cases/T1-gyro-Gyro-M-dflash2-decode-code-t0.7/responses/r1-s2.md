"""

import argparse

from collections import Counter
from typing import Dict, List, Tuple


def read_text_file(path: str) -> str:
    """指定されたパスのテキストファイルを読み込み、その内容の文字列を返す。

    Args:
        path: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を示す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み込み中にエラーが発生した場合。
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストを単語に分割する。

    Args:
        text: 分割対象のテキスト。
        case_sensitive: Trueの場合、大文字・小文字を区別する。Falseの場合、すべて小文字に変換してから分割する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 基本的なトークン化：空白区切り、句読点を除去
    # 簡易的な実装：非英数字を区切りとして扱う
    import re
    tokens = re.findall(r"[a-zA-Z]+", text)
    return tokens


def count_word_frequency(text: str, case_sensitive: bool = False) -> Dict[str, int]:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 分析対象のテキスト。
        case_sensitive: Trueの場合、大文字・小文字を区別する。

    Returns:
        単語をキー、出現回数を値とする辞書。
    """
    tokens = tokenize(text, case_sensitive)
    return Counter(tokens)
