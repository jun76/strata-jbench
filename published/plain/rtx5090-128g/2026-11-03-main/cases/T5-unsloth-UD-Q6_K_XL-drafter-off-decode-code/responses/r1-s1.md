def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。False の場合、すべて小文字に変換する。

    Returns:
        単語のリスト。単語は英数字とハイフンで構成される文字列とする。
    """
    if not case_sensitive:
        text = text.lower()

    tokens = re.findall(r"[a-z0-9]+(?:-[a-z0-9]+)*", text)
    return tokens
