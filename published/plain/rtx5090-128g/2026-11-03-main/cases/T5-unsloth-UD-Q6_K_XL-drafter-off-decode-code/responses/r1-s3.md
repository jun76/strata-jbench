#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語の出現頻度をカウントし、上位N件を表形式で表示するCLIツール。
"""

import argparse

def read_file(filepath: str) -> str:
    """
    指定されたパスのテキストファイルを読み込み、内容を返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み取りに失敗した場合。
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    単語は英数字およびハイフンを含む連続した文字列とし、大文字は小文字に変換する。

    Args:
        text: 分割対象のテキスト。

    Returns:
        単語のリスト。
    """
    # 英数字とハイフンを単語とみなす。大文字を小文字に変換。
    words = re.findall(r'[a-zA-Z0-9-]+', text.lower())
    return words


def count_words(words: List[str]) -> Dict[str, int]:
    """
    単語リストの出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする辞書。
    """
    return dict(Counter(words))
