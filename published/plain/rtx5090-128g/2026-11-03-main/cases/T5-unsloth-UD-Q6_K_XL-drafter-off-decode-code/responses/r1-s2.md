def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出します。
    
    Args:
        text (str): 入力テキスト。
        
    Returns:
        List[str]: 小文字化された単語のリスト。
    """
    # 英数字とハイフンを含む単語を抽出（例: "don't", "state-of-the-art" などに対応）
    # 日本語など非ASCII文字はここでは英単語抽出のみの仕様とします。
    # より汎用的なマルチリンガル対応が必要な場合は、正規表現やMeCab等の依存が必要になります。
    return re.findall(r"[a-zA-Z]+(?:'[a-zA-Z]+)*", text.lower())


def count_words(text: str) -> Dict[str, int]:
    """
    テキスト内の単語出現頻度をカウントします。
    
    Args:
        text (str): 入力テキスト。
        
    Returns:
        Dict[str, int]: 単語とその出現回数の辞書。
    """
    words = extract_words(text)
    return dict(Counter(words))


def get_top_n_words(word_counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    出現頻度上位 N 件の単語を取得します。
    
    Args:
        word_counts (Dict[str, int]): 単語と出現回数の辞書。
        n (int): 取得する件数。
        
    Returns:
        List[Tuple[str, int]]: 頻度順にソートされた (単語, 頻度) のリスト。
    """
    # 頻度の降順、同じ頻度ならアルファベット順でソート
    sorted_items = sorted(word_counts.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
