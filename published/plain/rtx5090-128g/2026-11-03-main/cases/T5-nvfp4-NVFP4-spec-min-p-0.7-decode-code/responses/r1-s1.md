#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を上位 N 件表示する。

使い方:
    python word_freq.py input.txt --top 10

出力例:
    Rank  Word       Count
    ----  ----       -----
    1     the        150
    2     and        120
    ...
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。

    単語は英数字の文字列として定義され、大文字小文字は無視する。

    Args:
        text: 入力テキスト

    Returns:
        抽出された単語のリスト（正規化済み）
    """
    # 英数字の連続を単語として抽出
    words = re.findall(r'[a-zA-Z0-9]+', text)
    # 小文字に変換して正規化
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語と出現回数の Counter
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度カウンターから上位 N 件の単語を取得する。

    同率の場合は辞書順（アルファベット順）でソートする。

    Args:
        counter: 単語と出現回数の Counter
        n: 取得する件数

    Returns:
        (単語, 出現回数) のタプルのリスト（上位順）
    """
    # 頻度の降順、同率の場合は辞書順でソート
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
