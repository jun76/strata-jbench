"""
word_freq.py — テキストファイルから単語の出現頻度を数え、上位 N 件を表形式で表示する CLI ツール。

使い方:
    python word_freq.py <ファイルパス> [-n N] [--min-len M] [--ignore-case] [--output-format {table,csv,json}]

例:
    python word_freq.py sample.txt -n 10

"""

from __future__ import annotations

@dataclass(frozen=True)
class WordCount:
    """単語とその出現回数を保持する不変データクラス。"""

    word: str
    count: int


def tokenize(text: str, *, min_len: int = 1, ignore_case: bool = False) -> list[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト文字列。
        min_len: 抽出する最小単語長（デフォルト 1）。
        ignore_case: True の場合、単語を小文字に正規化する。

    Returns:
        抽出された単語のリスト（順序は出現順）。
    """
    if ignore_case:
        text = text.lower()

    # 英数字とアンダースコアを含む連続文字列を単語として抽出
    tokens = re.findall(r"[a-zA-Z0-9_]+", text)

    # 最小長でフィルタ
    if min_len > 1:
        tokens = [t for t in tokens if len(t) >= min_len]

    return tokens
