"""

import argparse
import re

from collections import Counter

def tokenize(text: str) -> List[str]:
    """
    与えられたテキストから単語を抽出する.

    単語は英数字とハイフンで構成される文字列とし、小文字に正規化する.

    Args:
        text: 入力テキスト.

    Returns:
        小文字化された単語のリスト.
    """
    # 英数字とハイフンを含むトークンを抽出し、小文字に変換
    words = re.findall(r"[a-zA-Z0-9\-]+", text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする.

    Args:
        words: 単語のリスト.

    Returns:
        単語をキー、出現回数を値とする Counter.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位N件の単語と頻度を取得する.

    同頻度の単語は辞書順でソートされる.

    Args:
        counter: 単語頻度の Counter.
        n: 取得する上位件数.

    Returns:
        (単語, 頻度) のタプルのリスト.上位順.
    """
    # 頻度降順、同頻度なら辞書順でソート
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
