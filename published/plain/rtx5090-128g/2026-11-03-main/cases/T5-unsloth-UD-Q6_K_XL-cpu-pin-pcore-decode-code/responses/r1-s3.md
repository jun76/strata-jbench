"""テキストファイルから単語の出現頻度をカウントし、上位 N 件を表形式で表示するツール。"""

import argparse

def tokenize(text: str) -> List[str]:
    """テキストを単語リストに分割する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    # 英数字とハイフンを含む単語を抽出（一般的な英語テキスト向け）
    return re.findall(r'[a-z0-9]+(?:-[a-z0-9]+)*', text.lower())


def count_words(words: List[str]) -> Counter[str]:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)


def get_top_n(counts: Counter[str], n: int) -> List[Tuple[str, int]]:
    """最も出現頻度の高い上位 N 件の単語とその回数を得る。

    Args:
        counts: 単語の出現頻度 Counter。
        n: 取得する上位件数。

    Returns:
        (単語, 出現回数) のタプルのリスト。頻度が高い順。
    """
    return counts.most_common(n)
