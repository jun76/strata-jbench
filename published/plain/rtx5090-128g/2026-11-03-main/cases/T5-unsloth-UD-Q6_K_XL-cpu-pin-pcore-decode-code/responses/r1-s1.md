_WORD_PATTERN = re.compile(r"[a-zA-Z0-9']+")


def extract_words(text: str, ignore_case: bool = True, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text (str): 入力テキスト。
        ignore_case (bool, optional): 大文字小文字を区別しない場合は True。デフォルトは True。
        min_length (int, optional): 抽出する単語の最小文字数。デフォルトは 1。

    Returns:
        List[str]: 抽出された単語のリスト。
    """
    if min_length < 1:
        raise ValueError("min_length は 1 以上でなければなりません。")

    words = _WORD_PATTERN.findall(text)
    if ignore_case:
        words = [word.lower() for word in words]
    # 最小文字数でフィルタ
    words = [word for word in words if len(word) >= min_length]
    return words
