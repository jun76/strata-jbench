"""単語出現頻度の集計ロジックを提供するモジュール."""

from collections import Counter
from typing import List, Tuple


def normalize_word(word: str) -> str:
    """単語を正規化する。

    句読点を除去し、小文字に変換する。

    Args:
        word: 元の単語文字列。

    Returns:
        正規化された単語文字列。
    """
    # 句読点と空白を除去
    cleaned = ""
    for char in word:
        if char.isalnum() or char == "'":  # アポストロフィは保持（例: don't）
            cleaned += char
    return cleaned.lower().strip()


def count_words(text: str) -> Counter[str]:
    """テキストから単語の出現頻度をカウントする。

    Args:
        text: 分析対象のテキスト文字列。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = text.split()
    normalized_words = [normalize_word(w) for w in words]
    # 空文字列を除外
    filtered_words = [w for w in normalized_words if w]
    return Counter(filtered_words)
