def read_file(filepath: str) -> str:
    """Read the content of a text file.

    Args:
        filepath: Path to the text file to read.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """Tokenize text into words.

    Words are defined as sequences of alphanumeric characters and underscores.
    By default, words are converted to lowercase for case-insensitive counting.

    Args:
        text: The input text to tokenize.
        case_sensitive: If True, preserve the case of words. If False, convert to lowercase.

    Returns:
        A list of words (tokens) extracted from the text.
    """
    # Match words: sequences of word characters (letters, digits, underscores)
    tokens = re.findall(r'\w+', text)
    if not case_sensitive:
        tokens = [token.lower() for token in tokens]
    return tokens


def count_words(tokens: List[str]) -> Counter:
    """Count the frequency of each word in a list of tokens.

    Args:
        tokens: A list of words to count.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(tokens)
