#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語の出現頻度を集計し、
上位 N 件を表形式で表示するコマンドラインツール。

使用例:
    python word_freq.py sample.txt -n 10
    python word_freq.py sample.txt -n 5 --ignore-case --sort-by count
"""

@dataclass(frozen=True)
class WordCount:
    """単語とその出現回数を保持する不変データクラス。

    Attributes:
        word: 集計対象の単語（正規化済み）。
        count: 出現回数。
    """

    word: str
    count: int


def tokenize(text: str, *, ignore_case: bool = True) -> List[str]:
    """テキストを単語トークンに分割する。

    単語は英数字およびアンダースコアで構成される連続文字列として定義される。
    ignore_case が True の場合、単語は小文字に変換される。

    Args:
        text: 分割対象のテキスト。
        ignore_case: 大文字小文字を区別しない場合 True。

    Returns:
        単語トークンのリスト。
    """
    # \w+ は英数字・アンダースコアの連続文字列に一致
    tokens = re.findall(r"\w+", text, flags=re.UNICODE)
    if ignore_case:
        tokens = [tok.lower() for tok in tokens]
    return tokens


def count_words(text: str, *, ignore_case: bool = True) -> Counter[str]:
    """テキスト内の単語出現頻度を Counter として返す。

    Args:
        text: 集計対象のテキスト。
        ignore_case: 大文字小文字を区別しない場合 True。

    Returns:
        単語をキー、出現回数を値とする collections.Counter。
    """
    tokens = tokenize(text, ignore_case=ignore_case)
    return Counter(tokens)
