#!/usr/bin/env python3
"""
word_freq - Count word frequencies in a text file and display the top N results.

Usage:
    python word_freq.py <input_file> [-n TOP_N] [--case-sensitive]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Extract words from text, ignoring punctuation and non-alphanumeric characters.

    Args:
        text: The input text string.
        case_sensitive: If True, preserve case. If False, convert to lowercase.

    Returns:
        A list of extracted words.
    """
    if not case_sensitive:
        text = text.lower()
    # Match sequences of alphanumeric characters and underscores (common word definition)
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word in the list.

    Args:
        words: A list of words.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n_words(freq_counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words.

    Args:
        freq_counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count), sorted by frequency descending.
        Ties are broken lexicographically (alphabetically) for consistency.
    """
    # Common counter.most_common returns by frequency, but for ties it preserves insertion order.
    # To ensure deterministic output, sort by (-count, word)
    items = list(freq_counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
