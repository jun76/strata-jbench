"""
Word Frequency CLI Tool

Reads a text file, counts word occurrences, and displays the top N words.
"""

from __future__ import annotations

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    Count word occurrences in the given text.
    
    Words are split by whitespace and punctuation is stripped.
    Words are case-insensitive.
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    # Normalize: lowercase, strip punctuation, split by whitespace
    words = text.lower().split()
    # Strip common punctuation from each word
    cleaned_words = [word.strip(".,!?;:'\"()") for word in words]
    # Filter out empty strings
    filtered_words = [word for word in cleaned_words if word]
    return Counter(filtered_words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from a Counter.
    
    Args:
        counter: A Counter object containing word counts.
        n: The number of top words to return.
        
    Returns:
        A list of tuples (word, count) sorted by frequency descending.
    """
    return counter.most_common(n)
