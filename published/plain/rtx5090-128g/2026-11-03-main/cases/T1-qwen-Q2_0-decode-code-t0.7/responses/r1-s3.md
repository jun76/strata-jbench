"""
Command-line tool to count word frequencies in a text file and display the top N.

- Uses argparse for CLI arguments.

- Handles punctuation and case-insensitive counting.
- Outputs results in a formatted table.
"""

    """

    Returns:

    """
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and display the top N words."
    )
    parser.add_argument(
        "input_file",
        type=str,
        help="Path to the text file to analyze.",
    )
    parser.add_argument(
        "--top",
        type=int,
        default=10,

    )
    parser.add_argument(
        "--case-sensitive",
        action="store_true",
        help="Preserve case when counting words (default: case-insensitive).",
    )
    return parser.parse_args()
