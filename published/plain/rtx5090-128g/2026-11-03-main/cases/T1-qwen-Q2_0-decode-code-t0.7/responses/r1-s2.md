## 1. 実装（`freq_cli.py`）

```python
#!/usr/bin/env python3
"""
freq_cli.py

コマンドラインツール: 指定したテキストファイルから単語出現頻度を集計し,
上位 N 件を表形式で表示する.

例:
  python freq_cli.py input.txt --top 10
"""

from __future__ import annotations

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを空白・句読点・非アルphanumeric 文字で分割して単語リストを返す.

    Args:
        text: 入力テキスト.

    Returns:
        小字化された単語リスト.

    Notes:
        例: "Hello, world! hello." -> ["hello", "world", "hello"]
    """
    tokens: List[str] = []
    current: List[str] = []
    for ch in text:
        if ch.isalnum():
            current.append(ch)
        else:
            if current:
                tokens.append("".join(current))
                current = []
    if current:
        tokens.append("".join(current))

    return [t.lower() for t in tokens]


def count_words(text: str) -> Counter[str]:
    """
    単語出現頻度 Counter を返す.

    Args:
        text: 入力テキスト.

    Returns:
        Counter[word] = 出現回数.
    """
    return Counter(tokenize(text))


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件 (word, count) を取得.

    Args:
        counter: Counter[word] = 出現回数.
        n: 上位件数. n <= 0 なら [].

    Returns:
        [(word, count), ...] (高->低).
    """
    if n <= 0:
        return []
    return counter.most_common(n)
