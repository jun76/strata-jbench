#!/usr/bin/env python3
"""
テキストファイルを読み込んで単語の出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツール。

Usage:
    python word_frequency.py <file_path> [--top N] [--case-sensitive]

Example:
    python word_frequency.py sample.txt --top 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    words = re.findall(r"[a-z0-9]+(?:[-'][a-z0-9]+)*", text, re.IGNORECASE if case_sensitive else re.IGNORECASE)
    # 実際には case_sensitive が False のときは既に lower されているので IGNORECASE は不要だが、
    # 一貫性のため、case_sensitive が True のときはそのまま、False のときは lower 済みなので再マッチで問題ない
    # より正確には:
    if not case_sensitive:
        # text は既に lower されている
        words = re.findall(r"[a-z0-9]+(?:[-'][a-z0-9]+)*", text)
    else:
        words = re.findall(r"[a-zA-Z0-9]+(?:[-'][a-zA-Z0-9]+)*", text)
    return words


def count_word_frequencies(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)
