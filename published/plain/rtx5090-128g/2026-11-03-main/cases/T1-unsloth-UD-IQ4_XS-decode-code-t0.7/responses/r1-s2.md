word_count.py - A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.

Usage:
    python word_count.py input.txt -n 10
    python word_count.py input.txt --top 5 --case-sensitive
"""

import argparse

import re
from collections import Counter
from typing import List, Tuple, Dict


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize the given text into words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, convert all words to lowercase before tokenizing.

    Returns:
        A list of words extracted from the text.
    """
    if not case_sensitive:
        text = text.lower()
    # Use regex to find sequences of word characters (letters, digits, underscores)
    words = re.findall(r'\b\w+\b', text)
    return words


def count_words(text: str, case_sensitive: bool = False) -> Dict[str, int]:
    """
    Count the frequency of each word in the given text.

    Args:
        text: The input text to analyze.
        case_sensitive: If False, treat words case-insensitively.

    Returns:
        A dictionary mapping each word to its frequency count.
    """
    words = tokenize(text, case_sensitive=case_sensitive)
    return dict(Counter(words))


def get_top_n(counter: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from a word frequency counter.

    Args:
        counter: A dictionary mapping words to their frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples, each containing a word and its count, sorted by frequency
        descending, then alphabetically ascending for ties.
    """
    # Sort by count descending, then by word ascending for stable ordering
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
