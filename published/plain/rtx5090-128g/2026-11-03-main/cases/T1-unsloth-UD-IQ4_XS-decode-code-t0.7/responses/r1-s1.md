"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位N件表示する。

使い方:
    python word_freq.py input.txt --top 10
    python word_freq.py input.txt -n 5 --case-sensitive
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Optional


def extract_words(text: str, case_sensitive: bool = False) -> list[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字・小文字を区別するかどうか。False の場合、単語を小文字に変換する。

    Returns:
        単語のリスト。英数字のみからなるトークンを抽出する。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字のみからなるトークンを抽出
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return words


def count_words(words: list[str]) -> Counter[str]:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """
    Counter から上位 N 件の (単語, 出現回数) を取得する。

    Args:
        counter: 単語の出現頻度を格納した Counter。
        n: 取得する上位の件数。

    Returns:
        上位 N 件の (単語, 出現回数) のリスト。出現回数が多い順に並ぶ。
        同数の場合は辞書順で安定してソートされる。
    """
    # Counter.most_common は出現回数順だが、同数の場合の順序は不定なため、
    # 明示的にソートする。
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
