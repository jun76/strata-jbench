def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    正規表現を使用して、アルファベット・数字・ハイフンを含む連続文字列を単語とみなす。
    大文字小文字は無視して小文字化する。

    Args:
        text (str): 入力テキスト

    Returns:
        List[str]: 抽出された単語のリスト（小文字化済み）

    # 単語パターン: 英数字、ハイフン、アポストロフィを含む

    pattern = r"[a-zA-Z0-9'-]+"
    words = re.findall(pattern, text.lower())
    return words
