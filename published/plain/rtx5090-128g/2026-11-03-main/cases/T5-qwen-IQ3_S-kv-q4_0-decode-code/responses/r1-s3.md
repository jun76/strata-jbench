def parse_args(args: List[str] | None = None) -> argparse.Namespace:

        description="Count word frequencies in a text file and display the top N entries."
    )
    parser.add_argument(
        "input_file",
        type=str,
        help="Path to the input text file.",
    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=10,
        help="Number of top words to display (default: 10).",
    )
    parser.add_argument(
        "--case-sensitive",
        action="store_true",
        help="Perform case-sensitive counting (default: case-insensitive).",
    )
    return parser.parse_args(args)


def read_file(filepath: str) -> str:
    """Read the content of a text file.

    Args:
        filepath: Path to the input file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If there is an error reading the file.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """Tokenize text into words.

    Words are defined as sequences of alphanumeric characters.

    Args:
        text: The input text.
        case_sensitive: If False, convert all words to lowercase.

    Returns:
        A list of word tokens.
    """
    pattern = r"[a-zA-Z0-9]+"
    words = re.findall(pattern, text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words
