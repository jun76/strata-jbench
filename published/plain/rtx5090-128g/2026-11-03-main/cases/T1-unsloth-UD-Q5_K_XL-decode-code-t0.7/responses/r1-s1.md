def tokenize(text: str) -> List[str]:
    """
    テキストを小文字化し、英数字のみのトークンに分割する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化されたトークンのリスト。
    """
    # 英数字のみを抽出（正規表現不要の簡易実装）
    tokens: List[str] = []
    current: List[str] = []
    for char in text:
        if char.isalnum():
            current.append(char)
        else:
            if current:
                tokens.append("".join(current).lower())
                current = []
    if current:
        tokens.append("".join(current).lower())
    return tokens


def count_words(text: str) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(tokenize(text))
