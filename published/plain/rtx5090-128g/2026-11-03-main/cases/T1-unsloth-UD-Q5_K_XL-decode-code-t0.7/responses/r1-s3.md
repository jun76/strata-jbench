テキストファイルを読み込み、単語の出現頻度を計算して、

def normalize_word(word: str) -> str:
    """
    単語を正規化する。
    - 小文字に変換
    - 前後の空白と句読点を除去
    - 英数字のみを含む単語のみを返す（それ以外は空文字を返す）
    
    Args:
        word: 正規化対象の文字列
        
    Returns:
        正規化された単語（有効でない場合は空文字列）
    """
    # 小文字に変換
    word = word.lower()
    # 英数字のみを残す
    word = re.sub(r'[^a-z0-9]', '', word)
    return word


def count_words(text: str) -> Counter:
    """
    テキストから単語の出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    # 空白で分割
    words = text.split()
    # 正規化して有効な単語のみを抽出
    normalized_words = [normalize_word(w) for w in words]
    # 空文字列を除去
    valid_words = [w for w in normalized_words if w]
    return Counter(valid_words)
