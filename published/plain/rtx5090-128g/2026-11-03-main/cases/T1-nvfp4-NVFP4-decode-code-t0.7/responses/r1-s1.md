def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """
    Normalize a word for counting.
    
    Args:
        word: The raw word token.
        case_sensitive: If False, convert to lowercase.
    
    Returns:
        The normalized word string.
    """
    if not case_sensitive:
        word = word.lower()
    return word


def extract_words(text: str, min_length: int = 1, max_length: Optional[int] = None) -> list[str]:
    """
    Extract words from text using regex.
    
    Args:
        text: The input text.
        min_length: Minimum word length to include.
        max_length: Maximum word length to include (None means no limit).
    
    Returns:
        A list of extracted word tokens.
    """
    # Extract sequences of alphanumeric characters and underscores as "words"
    words = re.findall(r'\b\w+\b', text)
    result = []
    for w in words:
        if len(w) >= min_length:
            if max_length is not None and len(w) > max_length:
                continue
            result.append(w)
    return result
