使い方:
    python word_freq.py input.txt

from __future__ import annotations

# 単語を抽出するための正規表現（英数字およびアンダースコアを含む連続文字列）
_WORD_PATTERN = re.compile(r"\w+")


def tokenize(text: str, *, ignore_case: bool = True, min_length: int = 1) -> list[str]:
    """テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト。
        ignore_case: True の場合、小文字化して比較する。
        min_length: この長さ未満の単語は除外する。

    Returns:
        抽出された単語のリスト（出現順）。
    """
    if min_length < 1:
        raise ValueError("min_length must be at least 1")

    raw_words = _WORD_PATTERN.findall(text)
    if ignore_case:
        raw_words = [w.lower() for w in raw_words]
    return [w for w in raw_words if len(w) >= min_length]


def count_words(text: str, *, ignore_case: bool = True, min_length: int = 1) -> Counter[str]:
    """テキスト中の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト。
        ignore_case: True の場合、小文字化して比較する。
        min_length: この長さ未満の単語は除外する。

    Returns:
        単語 → 出現回数の Counter。
    """
    tokens = tokenize(text, ignore_case=ignore_case, min_length=min_length)
    return Counter(tokens)
