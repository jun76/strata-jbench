# 単語出現頻度集計ツール

## `word_freq.py`

"""
word_freq.py — テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

使い方:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt --top 20 --case-insensitive
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_insensitive: bool = True) -> List[str]:
    """テキストから単語のリストを抽出する。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、単語を小文字に変換して大文字小文字を区別しない。

    Returns:
        単語のリスト。
    """
    if case_insensitive:
        text = text.lower()

    words: List[str] = []
    current_word: List[str] = []

    for char in text:
        if char.isalnum() or char == "_":
            current_word.append(char)
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))

    return words


def count_words(text: str, case_insensitive: bool = True) -> Counter:
    """テキスト内の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、単語を小文字に変換して大文字小文字を区別しない。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    words = tokenize(text, case_insensitive=case_insensitive)
    return Counter(words)
