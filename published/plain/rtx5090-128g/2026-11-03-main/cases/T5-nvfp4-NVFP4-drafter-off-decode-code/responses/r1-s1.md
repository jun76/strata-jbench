word_counter.py — テキストファイル内の単語出現頻度をカウントし、
上位 N 件を表形式で表示するコマンドラインツール。

Usage:
    python word_counter.py input.txt -n 10
    python word_counter.py input.txt --top 20 --case-insensitive
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str, case_insensitive: bool = True) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト文字列。
        case_insensitive: True の場合、単語を小文字に変換する。

    Returns:
        抽出された単語のリスト。
    """
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    # 例: "don't" -> "don't", "well-known" -> "well-known"
    pattern = r"[a-zA-Z0-9]+(?:[-'][a-zA-Z0-9]+)*"
    words = re.findall(pattern, text)
    if case_insensitive:
        words = [w.lower() for w in words]
    return words


def count_words(text: str, case_insensitive: bool = True) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト文字列。
        case_insensitive: True の場合、単語を小文字に変換してからカウントする。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = extract_words(text, case_insensitive)
    return Counter(words)
