def tokenize(text: str) -> List[str]:
    """テキストから単語リストを抽出する。

    単語は英数字、ハイフン、アポストロフィを含む連続文字列として定義される。
    大文字小文字は無視し、すべて小文字に変換する。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト
    """
    return re.findall(r"[a-z0-9]+(?:['-][a-z0-9]+)*", text.lower())


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    words = tokenize(text)
    return Counter(words)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """頻度上位 N 件の単語を (単語, 回数) のタプルリストとして返す。

    回数が同じ場合は単語の辞書順でソートする。

    Args:
        counter: 単語頻度の Counter
        n: 上位件数

    Returns:
        (単語, 回数) のタプルのリスト
    """
    # 回数の降順、同率なら単語の昇順でソート
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
