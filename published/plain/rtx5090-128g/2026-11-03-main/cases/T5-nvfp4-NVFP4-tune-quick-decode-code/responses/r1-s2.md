#!/usr/bin/env python3
"""
wordfreq: テキストファイルから単語頻度を集計し、上位N件を表形式で表示するCLIツール。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    # 英数字とハイフン、アポストロフィを単語として扱い、それ以外を区切りとする
    # 日本語などマルチバイト文字も考慮し、\w+ でUnicode単語を抽出
    words = re.findall(r"\w+", text, flags=re.UNICODE)
    return [word.lower() for word in words]


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とするCounterオブジェクト。
    """
    tokens = tokenize(text)
    return Counter(tokens)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counterから上位n件の(単語, 頻度)ペアを返す。

    Args:
        counter: 単語頻度Counter。
        n: 取得する上位件数。

    Returns:
        頻度順にソートされた(単語, 頻度)のリスト。
    """
    return counter.most_common(n)
