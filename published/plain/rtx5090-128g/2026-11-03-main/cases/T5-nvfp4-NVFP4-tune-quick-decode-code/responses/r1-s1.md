def tokenize(text: str, ignore_case: bool = False) -> List[str]:
    """
    Tokenize text into words.

    A word is defined as a sequence of alphanumeric characters, underscores,
    and apostrophes. This allows contractions like "don't" to be treated as
    a single word.

    Args:
        text: The input text to tokenize.
        ignore_case: If True, convert all words to lowercase before counting.

    Returns:
        A list of words found in the text.
    """
    # Match sequences of alphanumeric chars, underscores, and apostrophes
    pattern = r"[a-zA-Z0-9_']+"
    words = re.findall(pattern, text)
    if ignore_case:
        words = [word.lower() for word in words]
    return words
