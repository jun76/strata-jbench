def tokenize(text: str, case_insensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_insensitive: 大文字小文字を区別するか否か

    Returns:
        単語のリスト
    """
    # 句読点や記号を除去して単語のみ抽出
    import re
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    if case_insensitive:
        words = [w.lower() for w in words]
    return words


def count_words(text: str, case_insensitive: bool = False) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_insensitive: 大文字小文字を区別するか否か

    Returns:
        単語 -> 出現回数の Counter オブジェクト
    """
    words = tokenize(text, case_insensitive)
    return Counter(words)
