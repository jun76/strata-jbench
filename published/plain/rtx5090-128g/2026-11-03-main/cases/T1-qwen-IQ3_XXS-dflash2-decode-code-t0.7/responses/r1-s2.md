#!/usr/bin/env python3
"""
単語出現頻度集計ツール

テキストファイルを読み込み、単語の出現頻度を数え、
上位 N 件の単語を頻度順に表形式で表示します。

Usage:
    python word_freq.py input.txt [-n 10]
"""

from __future__ import annotations

import argparse
import sys
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    単語は英数字とアンダースコア、ハイフンで構成される連続文字列として定義される。
    大文字小文字は無視し、すべて小文字に変換する。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト（小文字）
    """
    # 単語パターン: 英数字、ハイフン、アンダースコアを含む連続文字列
    tokens = re.findall(r"[a-zA-Z0-9_-]+", text)
    return [token.lower() for token in tokens]


def count_words(text: str) -> Counter[str]:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語 -> 出現回数の Counter
    """
    tokens = tokenize(text)
    return Counter(tokens)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    上位 n 件の単語を (word, count) のタプルリストで返す。
    頻度が高い順に並び、同数であればアルファベット順に並ぶ。

    Args:
        counter: 単語出現頻度の Counter
        n: 取得する上位件数

    Returns:
        [(word, count), ...] のリスト（頻度降順、同数ならアルファベット順）
    """
    # Counter.most_common は同数時の順序が保証されないため、
    # 明示的にソートする
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
