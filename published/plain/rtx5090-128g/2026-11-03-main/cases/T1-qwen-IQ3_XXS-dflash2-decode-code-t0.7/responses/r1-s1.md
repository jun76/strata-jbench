def read_text_file(filepath: str) -> str:
    """テキストファイルを読み込み、内容を文字列として返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容（UTF-8 でデコードされた文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        UnicodeDecodeError: ファイルが UTF-8 でデコードできない場合。
    """
    with open(filepath, mode="r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストを単語に分割する。

    単語は正規表現 \\b[a-zA-Z0-9_]+\\b に一致する連続文字列として抽出される。
    大文字小文字はデフォルトでは無視される（case_sensitive=False）。

    Args:
        text: 分割対象のテキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、すべて小文字に変換する。

    Returns:
        単語のリスト。
    """
    pattern = r"\b[a-zA-Z0-9_]+\b"
    words = re.findall(pattern, text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words
