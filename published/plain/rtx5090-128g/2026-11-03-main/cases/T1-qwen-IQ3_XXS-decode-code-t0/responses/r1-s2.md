def read_file(filepath: str) -> str:
    """指定されたファイルを読み込み、その内容を文字列として返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み取り中にエラーが発生した場合。
    """
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except IOError as e:
        raise IOError(f"Error reading file {filepath}: {e}")


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストを単語に分割する。

    単語は、英数字とアンダースコアで構成される連続した文字列として定義される。
    大文字小文字の区別は case_sensitive パラメータで制御する。

    Args:
        text: 分割対象のテキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、すべて小文字に変換する。

    Returns:
        単語のリスト。
    """
    import re
    # 英数字とアンダースコアを含む単語を抽出
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    if not case_sensitive:
        words = [word.lower() for word in words]
    return words
