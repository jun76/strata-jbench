#!/usr/bin/env python3
"""
word_freq: テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

Usage:
    python word_freq.py input.txt --top 10
    python word_freq.py input.txt --top 20 --case-sensitive
    python word_freq.py input.txt --top 5 --output json
"""

from __future__ import annotations

import argparse
import json
import re
import sys
from collections import Counter
from typing import Any


def tokenize(text: str, case_sensitive: bool = False) -> list[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        単語トークンのリスト（空文字列は除外）。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字・ハイフン・アポストロフィを含むトークンを抽出
    tokens = re.findall(r"[a-z0-9]+(?:[-'][a-z0-9]+)*", text)
    return [token for token in tokens if token]


def count_words(
    text: str,
    case_sensitive: bool = False,
) -> Counter[str]:
    """
    テキスト中の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: 大文字小文字の区別フラグ。

    Returns:
        単語 → 出現回数の Counter オブジェクト。
    """
    tokens = tokenize(text, case_sensitive=case_sensitive)
    return Counter(tokens)
