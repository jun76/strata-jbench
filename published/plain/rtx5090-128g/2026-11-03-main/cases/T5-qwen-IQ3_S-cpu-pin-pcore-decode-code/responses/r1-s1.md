def read_file(filepath: str) -> str:
    """
    Read the content of a text file.

    Args:
        filepath: Path to the input text file.

    Returns:
        The entire content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Extract words from text, optionally case-sensitive and with a minimum length.

    Args:
        text: Input text.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.
        min_length: Minimum word length to include.

    Returns:
        A list of words extracted from the text.
    """
    # Extract sequences of alphabetic characters (words)
    words = re.findall(r"[a-zA-Z]+", text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    return words


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)
