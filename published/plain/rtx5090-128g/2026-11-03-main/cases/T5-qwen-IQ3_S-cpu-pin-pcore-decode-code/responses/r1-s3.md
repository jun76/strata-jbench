Command-line tool to read a text file, count word frequencies,
and display the top N most frequent words in a table format.

Usage:
    python word_freq.py <file> [-n TOP_N] [--case-sensitive] [--min-len MIN_LEN]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse

from collections import Counter

def extract_words(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    Extract words from text based on given criteria.
    
    Args:
        text: The input text string.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.
        min_len: Minimum length of a word to be counted.
        
    Returns:
        A list of words extracted from the text.
    """
    if not case_sensitive:
        text = text.lower()
    
    # Extract words: sequences of alphanumeric characters and underscores
    words = re.findall(r'\w+', text)
    
    # Filter by minimum length
    if min_len > 1:
        words = [w for w in words if len(w) >= min_len]
    
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.
    
    Args:
        words: A list of words.
        
    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their frequencies.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top entries to return.
        
    Returns:
        A list of tuples (word, count) sorted by frequency (descending),
        then alphabetically for ties.
    """
    # Sort by frequency descending, then by word ascending for ties
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
