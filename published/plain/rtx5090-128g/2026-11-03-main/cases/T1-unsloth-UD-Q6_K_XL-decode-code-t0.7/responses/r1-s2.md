#!/usr/bin/env python3
"""
word_count.py - A command-line tool to count word frequencies in a text file.

Usage:
    python word_count.py <input_file> [--top N] [--min-length LENGTH] [--case-insensitive]

Example:
    python word_count.py sample.txt --top 10 --min-length 3 --case-insensitive
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def parse_args(argv: List[str] | None = None) -> argparse.Namespace:
    """Parse command-line arguments.

    Args:
        argv: Optional list of command-line arguments. If None, sys.argv[1:] is used.

    Returns:
        An argparse.Namespace containing the parsed arguments.
    """
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and display the top N words.",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
Examples:
  %(prog)s sample.txt
  %(prog)s sample.txt --top 20
  %(prog)s sample.txt --min-length 4 --case-insensitive
        """,
    )
    parser.add_argument(
        "input_file",
        type=str,
        help="Path to the input text file.",
    )
    parser.add_argument(
        "--top",
        type=int,
        default=10,
        help="Number of top words to display (default: 10).",
    )
    parser.add_argument(
        "--min-length",
        type=int,
        default=1,
        help="Minimum word length to consider (default: 1).",
    )
    parser.add_argument(
        "--case-insensitive",
        action="store_true",
        help="Perform case-insensitive counting.",
    )
    return parser.parse_args(argv)
