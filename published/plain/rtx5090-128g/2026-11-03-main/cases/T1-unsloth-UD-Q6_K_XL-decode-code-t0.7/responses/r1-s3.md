### 1. `word_freq.py`

```python
#!/usr/bin/env python3
"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度をカウントして
上位 N 件を表形式で出力するコマンドラインツール。

用法:
    python word_freq.py FILE [-n N] [--case-sensitive] [--top-only]
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語のリストを抽出する。

    単語は、英数字とハイフン、アポストロフィを含む連続した文字列として定義される。
    それ以外の文字（句読点、空白など）は区切りとして扱われる。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。
                        False の場合、すべて小文字に変換する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    
    # 英数字、ハイフン、アポストロフィ以外の文字で分割
    # 正規表現を使わず、文字単位で処理して単語を抽出
    words = []
    current_word = []
    for char in text:
        if char.isalnum() or char in ("-", "'"):
            current_word.append(char)
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))
    
    return words


def count_word_frequencies(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    テキスト中の単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字・小文字を区別するかどうか。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = extract_words(text, case_sensitive)
    return Counter(words)
