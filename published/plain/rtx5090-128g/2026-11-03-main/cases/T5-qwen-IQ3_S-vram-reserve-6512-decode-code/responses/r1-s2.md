def read_text(filepath: str) -> str:
    """ファイルを読み込んでテキストを返す。

    Args:
        filepath: 読み込むファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイル読み込みエラーの場合。
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 対象テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    # 英数字のみを含む単語を抽出する
    import re
    words = re.findall(r'\w+', text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (word, count) タプルリストを返す。

    Args:
        counter: 単語頻度の Counter。
        n: 取得する上位件数。

    Returns:
        上位 n 件の (word, count) リスト。
    """
    return counter.most_common(n)
