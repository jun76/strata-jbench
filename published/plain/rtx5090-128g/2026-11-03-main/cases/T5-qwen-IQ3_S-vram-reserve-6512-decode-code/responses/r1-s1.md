def read_file(path: str) -> str:
    """指定されたパスのテキストファイルを読み込み、内容を文字列として返す。

    Args:
        path: 読み込み対象のファイルパス。

    Returns:
        ファイルの内容を示す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み取りに失敗した場合。
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語トークンリストを取得する。

    単語はアルファベット、数字、ハイフン、アポストロフィを含む連続文字列として認識される。
    大文字小文字の区別は case_sensitive パラメータで制御する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別して単語をカウントする。

    Returns:
        単語トークンのリスト。
    """
    import re

    # 単語パターン: 英数字、ハイフン、アポストロフィを含む
    pattern = r"[A-Za-z0-9'-]+"
    tokens = re.findall(pattern, text)

    if not case_sensitive:
        tokens = [token.lower() for token in tokens]

    return tokens


def count_frequencies(tokens: List[str]) -> Counter:
    """トークンリストから単語出現頻度を計算する。

    Args:
        tokens: 単語トークンのリスト。

    Returns:
        単語から出現回数をマッピングした Counter オブジェクト。
    """
    return Counter(tokens)
