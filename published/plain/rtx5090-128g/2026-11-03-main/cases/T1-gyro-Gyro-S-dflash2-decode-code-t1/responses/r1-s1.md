"""
word_freq CLI: Count word frequencies in a text file and display top N.

Usage:
    python word_freq.py [--top N] [--case-sensitive] [--min-length M] <file>

Features:
    - Read text from file
    - Tokenize by whitespace/punctuation (default) or configurable
    - Count occurrences
    - Display top N words in a tabular format
"""

import argparse
import sys
import re
from collections import Counter
from pathlib import Path
from typing import Optional


def tokenize(text: str, case_sensitive: bool = False) -> list[str]:
    """
    Tokenize text into words by splitting on whitespace and stripping punctuation.

    Args:
        text: The input text string.
        case_sensitive: If False, convert all words to lowercase.

    Returns:
        A list of tokens (words).
    """
    # Lowercase if not case-sensitive
    if not case_sensitive:
        text = text.lower()

    # Split by whitespace and remove punctuation from each token
    tokens = []
    for token in text.split():
        # Remove leading and trailing punctuation
        cleaned = re.sub(r'^[^\w]+|[^\w]+$', '', token)
        # Remove internal punctuation? Usually we want to strip only outer.
        # For simplicity, just keep word characters.
        # Use regex to extract word-like sequences
        matches = re.findall(r'[a-zA-Z0-9]+', cleaned)
        tokens.extend(matches)

    return tokens


def count_words(tokens: list[str]) -> Counter[str]:
    """
    Count the frequency of each token.

    Args:
        tokens: A list of token strings.

    Returns:
        A Counter mapping each token to its frequency.
    """
    return Counter(tokens)


def get_top_words(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """
    Get the top N most frequent words from a Counter.

    Args:
        counter: A Counter of word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, frequency), sorted by frequency descending.
    """
    return counter.most_common(n)
