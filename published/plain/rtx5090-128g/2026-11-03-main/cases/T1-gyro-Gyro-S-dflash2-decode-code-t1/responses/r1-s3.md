import argparse

from collections import Counter

def tokenize(text: str, case_insensitive: bool = True) -> List[str]:
    """
    入力テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、単語は小文字化される。

    Returns:
        単語のリスト（正規表現 \\w+ にマッチするトークン）。
    """
    tokens = re.findall(r'\b\w+\b', text)
    if case_insensitive:
        tokens = [token.lower() for token in tokens]
    return tokens


def count_words(text: str, case_insensitive: bool = True) -> Counter:
    """
    入力テキスト内の単語出現回数を集計する。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、単語は小文字化してから集計する。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    tokens = tokenize(text, case_insensitive)
    return Counter(tokens)


def top_n_words(
    counter: Counter,
    n: int,
) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (単語, 回数) のリストを返す。
    回数が同じ場合は、単語の辞書順で安定してソートする。

    Args:
        counter: 単語と出現回数の Counter。
        n: 取得する件数。

    Returns:
        (単語, 回数) のタプルのリスト。
    """
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
