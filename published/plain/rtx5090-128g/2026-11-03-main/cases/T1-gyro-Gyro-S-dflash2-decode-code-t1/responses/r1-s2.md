#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルから単語の出現頻度を集計し、上位 N 件を出力します。

使い方:
    python word_freq.py --top 10 input.txt

import argparse
import sys
from collections import Counter
from typing import Dict, List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを空白区切りでトークン化し、小文字に正規化する。

    Args:
        text: 入力テキスト文字列。

    Returns:
        小文字化した単語のリスト。
    """
    return text.lower().split()


def count_words(text: str) -> Counter:
    """テキスト内の単語出現頻度を集計する。

    Args:
        text: 入力テキスト文字列。

    Returns:
        単語 -> 出現回数の Counter。
    """
    tokens = tokenize(text)
    return Counter(tokens)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """出現頻度上位 N 件の (単語, 回数) を取得する。

    Args:
        counter: 単語出現頻度の Counter。
        n: 上位件数。

    Returns:
        頻度降順の (単語, 回数) リスト。頻度同率の場合は辞書順で安定化。
    """
    if n <= 0:
        return []
    # most_common は頻度降順。同率の場合は挿入順だが、テストの安定性のため
    # 辞書順での二次キーはここでは simple に most_common を使う。
    # より deterministic な順序が必要なら sorted を使う。
    return [(word, count) for word, count in counter.most_common(n)]
