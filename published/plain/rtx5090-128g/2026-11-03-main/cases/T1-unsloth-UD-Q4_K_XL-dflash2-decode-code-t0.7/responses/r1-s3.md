"""
wordcount.py

テキストファイルを読み込み、単語の出現頻度を計算し、
上位 N 件を表形式で出力するコマンドラインツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。

    アルファベット、数字、ハイフンを含む文字列を単語とみなす。
    すべて小文字に変換して返す。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト
    """
    # 単語の定義: 英数字、ハイフン、アポストロフィを含む連続文字列
    # 例: "don't", "state-of-the-art", "hello123"
    words = re.findall(r"[a-zA-Z0-9'\-]+", text.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 N 件を返す。

    頻度が同じ場合は単語の辞書順でソートする。

    Args:
        counter: 単語と出現回数の Counter
        n: 取得する件数

    Returns:
        [(word, count), ...] のリスト（降順）
    """
    # most_common は頻度順だが、同頻度のときの順序は保証されない
    # 明示的にソートする
    items = sorted(
        counter.items(),
        key=lambda x: (-x[1], x[0])
    )
    return items[:n]
