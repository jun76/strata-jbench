#!/usr/bin/env python3
"""
word_freq.py - テキストファイルの単語出現頻度を上位N件で表示するCLIツール.

このモジュールは、指定されたテキストファイルを読み込み、
単語の出現頻度を集計し、上位N件を整形された表形式で出力します。

Usage:
    python word_freq.py <file> [-n N] [--min-len LEN] [--case-sensitive]
"""

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from typing import List, Tuple


@dataclass
class WordFrequencyResult:
    """単語頻度集計の結果を表すデータクラス.

    Attributes:
        word: 単語そのもの.
        count: 出現回数.
    """

    word: str
    count: int

    def __lt__(self, other: "WordFrequencyResult") -> bool:
        """ソート用: 頻度降順、同じ頻度なら辞書順.

        Args:
            other: 比較対象のWordFrequencyResult.

        Returns:
            True: selfがotherより「先」に来る場合（頻度高 or 同じ頻度で辞書順前）.
        """
        if self.count != other.count:
            return self.count > other.count  # 頻度が高いほど「小さい」=先
        return self.word < other.word
