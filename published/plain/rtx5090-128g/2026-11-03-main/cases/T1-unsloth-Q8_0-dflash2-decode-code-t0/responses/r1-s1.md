#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツール: テキストファイルから単語出現頻度を上位N件表示

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive] [--min-length LEN]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか
        min_length: 単語の最小長（これ未満の単語は除外）

    Returns:
        抽出された単語のリスト
    """
    # 英数字とハイフン、アポストロフィを含むトークンを抽出
    # 例: "don't" -> "don't", "well-known" -> "well-known"
    tokens = re.findall(r"[a-zA-Z0-9]+(?:['-][a-zA-Z0-9]+)*", text)
    
    if not case_sensitive:
        tokens = [token.lower() for token in tokens]
    
    # 最小長でフィルタ
    if min_length > 1:
        tokens = [token for token in tokens if len(token) >= min_length]
    
    return tokens


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とするCounter
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度カウンターから上位N件の単語と回数を取り出す。

    Args:
        counter: 単語頻度のCounter
        n: 取得する件数

    Returns:
        (単語, 回数) のタプルのリスト（回数の降順）
    """
    return counter.most_common(n)
