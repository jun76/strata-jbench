word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位 N 件の単語とその頻度を表形式で出力するコマンドラインツール。
"""

import argparse

def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    英数字のみを単語とし、大文字は小文字に変換する。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 単語のリスト。
    """
    # 英数字のみを抽出し、小文字に変換
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_words(text: str) -> Counter:
    """
    テキスト内の単語の出現頻度をカウントする。

    Args:
        text (str): 入力テキスト。

    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = tokenize(text)
    return Counter(words)
