#!/usr/bin/env python3
"""Top N word frequency counter.

A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.
"""

import argparse

from collections import Counter

def extract_words(text: str) -> List[str]:
    """Extract words from text, converting to lowercase and splitting on non-word characters.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    return re.findall(r"\b[a-zA-Z]+\b", text.lower())


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        A Counter object with word counts.
    """
    words = extract_words(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a counter.

    Args:
        counter: A Counter object with word counts.
        n: Number of top words to retrieve.

    Returns:
        A list of (word, count) tuples, sorted by count descending, then word ascending.
    """
    # Sort by count descending, then by word ascending for tie-breaking
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
