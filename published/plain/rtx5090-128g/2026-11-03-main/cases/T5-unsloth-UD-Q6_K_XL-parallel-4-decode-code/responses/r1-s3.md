def read_file(filepath: str) -> str:
    """
    テキストファイルを読み込み、内容を文字列として返す。

    Args:
        filepath (str): 読み込むファイルのパス。

    Returns:
        str: ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        PermissionError: 読み取り権限がない場合。
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    アルファベット、数字、ハイフン、アポストロフィを単語文字として扱い、
    その他の文字を区切りとして分割する。
    すべて小文字に変換する。

    Args:
        text (str): 分割対象のテキスト。

    Returns:
        List[str]: 単語のリスト。
    """
    # 正規表現を使わずに、文字ごとに判定して分割
    # より正確なトークン化のため、単語文字のみ抽出
    words: List[str] = []
    current_word: List[str] = []
    
    for char in text:
        # 単語を構成する文字: アルファベット、数字、ハイフン、アポストロフィ
        if char.isalnum() or char in ('-', "'"):
            current_word.append(char)
        else:
            if current_word:
                words.append(''.join(current_word).lower())
                current_word = []
    if current_word:
        words.append(''.join(current_word).lower())
        
    return words
