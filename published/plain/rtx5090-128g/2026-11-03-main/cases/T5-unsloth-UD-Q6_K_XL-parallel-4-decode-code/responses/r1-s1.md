#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を上位 N 件まで表形式で表示する。

    python word_freq.py input.txt --top 10
    python word_freq.py input.txt -n 20 --case-sensitive
"""

import argparse

def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        抽出された単語のリスト。
    """
    # 単語は英数字とハイフン、アポストロフィを含む連続文字列として抽出
    # 例: don't, well-known, 123
    pattern = r"[a-zA-Z0-9]+(?:[-'][a-zA-Z0-9]+)*"
    words = re.findall(pattern, text, flags=re.IGNORECASE if not case_sensitive else 0)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (word, count) を取得する。

    Args:
        counter: 単語頻度の Counter。
        n: 取得する件数。

    Returns:
        (word, count) のタプルのリスト。頻度降順、同頻度の場合はアルファベット順。
    """
    return counter.most_common(n)
