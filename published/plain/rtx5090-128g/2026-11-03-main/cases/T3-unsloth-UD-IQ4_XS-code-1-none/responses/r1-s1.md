def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    if not intervals:
        return []
    
    # Sort by start time, then by end time for consistent ordering
    sorted_intervals = sorted(intervals, key=lambda x: (x[0], x[1]))
    
    merged = [sorted_intervals[0][:]]  # Create a copy to avoid modifying input
    
    for i in range(1, len(sorted_intervals)):
        current_start, current_end = sorted_intervals[i]
        last_start, last_end = merged[-1]
        
        # Check for overlap or touching: if current_start <= last_end, they overlap or touch
        if current_start <= last_end:
            merged[-1][1] = max(last_end, current_end)
        else:
            merged.append([current_start, current_end])
    
    return merged
