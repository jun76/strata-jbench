def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    """
    Merge all overlapping or touching closed intervals and return them sorted by start.
    
    Args:
        intervals: A list of intervals, each represented as [start, end].
                   The input may be unsorted and may be empty.
    
    Returns:
        A new list of merged intervals, sorted by start time.
    """
    if not intervals:
        return []
    
    # Sort intervals by start time, then by end time
    sorted_intervals = sorted(intervals, key=lambda x: (x[0], x[1]))
    
    merged = []
    for current in sorted_intervals:
        if not merged:
            merged.append(current)
        else:
            last = merged[-1]
            # If the current interval overlaps or touches the last merged interval
            if current[0] <= last[1]:
                # Merge them by updating the end of the last interval
                merged[-1] = [last[0], max(last[1], current[1])]
            else:
                merged.append(current)
    
    return merged
