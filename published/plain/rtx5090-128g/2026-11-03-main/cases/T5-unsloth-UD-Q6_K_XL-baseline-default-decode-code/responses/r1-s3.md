#!/usr/bin/env python3
"""
wordfreq - Count word frequencies in text files and display top N results.

This command-line tool reads one or more text files, tokenizes the content
into words (lowercased, stripped of punctuation), counts occurrences, and
displays the top N most frequent words in a formatted table.
"""

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from typing import List, Dict


@dataclass(frozen=True)
class WordFrequency:
    """Represents a word and its frequency count."""
    word: str
    count: int


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.
    
    Words are defined as sequences of alphanumeric characters (including
    underscores). Consecutive non-word characters are treated as separators.
    All tokens are converted to lowercase.
    
    Args:
        text: The input text string to tokenize.
        
    Returns:
        A list of lowercase word tokens.
    """
    return re.findall(r'\w+', text.lower())


def count_words(text: str) -> Counter:
    """
    Count the frequency of each word in the given text.
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter object mapping each word to its frequency.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[WordFrequency]:
    """
    Get the top N most frequent words from a Counter.
    
    In case of ties, words are sorted alphabetically.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.
        
    Returns:
        A list of WordFrequency objects, sorted by count (descending)
        then by word (ascending) for ties.
    """
    # Get all items and sort by (-count, word) to handle ties alphabetically
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return [WordFrequency(word=word, count=count) for word, count in items[:n]]
