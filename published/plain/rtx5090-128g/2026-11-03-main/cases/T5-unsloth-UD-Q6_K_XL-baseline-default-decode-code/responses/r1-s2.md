#!/usr/bin/env python3
"""
word_freq - Count word frequencies in text files and display top N entries.

Usage:
    word_freq [options] <file>

Options:
    -n, --top N       Show top N words (default: 10)
    -c, --case-sensitive  Be case-sensitive (default: case-insensitive)
    --json            Output in JSON format
"""

import argparse
import json
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, words are lowercased.

    Returns:
        A list of word tokens.
    """
    if not case_sensitive:
        text = text.lower()
    # Extract words consisting of alphanumeric characters and underscores
    words = re.findall(r'\w+', text)
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text.
        case_sensitive: If False, words are lowercased before counting.

    Returns:
        A Counter object with word frequencies.
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)
