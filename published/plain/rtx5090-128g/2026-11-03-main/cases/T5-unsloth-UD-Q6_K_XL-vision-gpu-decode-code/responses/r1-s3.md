#!/usr/bin/env python3
"""
Text frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a tabular format.
"""

import argparse

import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    Extract words from text.

    Words are defined as sequences of alphanumeric characters (including
    underscores). The text is lowercased before extraction.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words found in the text.
    """
    return re.findall(r'\b\w+\b', text.lower())


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples sorted by count descending,
        then by word alphabetically for ties.
    """
    # Sort by count descending, then by word ascending for deterministic output
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
