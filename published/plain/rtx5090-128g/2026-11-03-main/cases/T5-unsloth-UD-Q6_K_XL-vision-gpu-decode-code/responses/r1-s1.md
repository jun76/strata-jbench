"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位N件表示する。

使い方:
    python word_freq.py input.txt --top 10
    python word_freq.py input.txt -n 5 --case-sensitive
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def load_text(filepath: str) -> str:
    """テキストファイルを読み込んで内容を文字列として返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        OSError: ファイルの読み取りに失敗した場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語トークンを抽出する。

    単語は英数字とハイフン、アポストロフィを含む連続文字列として定義される。
    句読点や空白は区切りとして扱われる。

    Args:
        text: トークン化するテキスト。
        case_sensitive: Trueの場合、大文字と小文字を区別する。Falseの場合、すべて小文字に変換する。

    Returns:
        単語トークンのリスト。
    """
    # 英数字、ハイフン、アポストロフィを含む連続文字列を単語として抽出
    pattern = r"[a-zA-Z0-9'\-]+"
    tokens = re.findall(pattern, text)
    if not case_sensitive:
        tokens = [token.lower() for token in tokens]
    return tokens


def count_words(tokens: List[str]) -> Counter:
    """トークンリストから単語の出現頻度をカウントする。

    Args:
        tokens: 単語トークンのリスト。

    Returns:
        単語をキー、出現回数を値とするCounterオブジェクト。
    """
    return Counter(tokens)
