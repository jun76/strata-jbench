#!/usr/bin/env python3
"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度を集計して、
上位 N 件の単語とその出現回数を表形式で出力するコマンドラインツールです。

単語の定義:
- 英数字で構成される連続した文字列（正規表現 \w+）
- すべて小文字に変換して集計します。

使用例:
    python word_freq.py input.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple

# 単語を抽出するための正規表現パターン
WORD_PATTERN = re.compile(r'\w+', re.IGNORECASE)


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出し、すべて小文字に変換したリストを返します。

    Args:
        text: 入力テキスト文字列

    Returns:
        小文字化された単語のリスト
    """
    return [word.lower() for word in WORD_PATTERN.findall(text)]


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントします。

    Args:
        text: 入力テキスト文字列

    Returns:
        単語をキー、出現回数を値とする collections.Counter オブジェクト
    """
    words = extract_words(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter オブジェクトから上位 n 件の (単語, 回数) を返します。

    Args:
        counter: 単語出現頻度の Counter
        n: 取得する上位件数

    Returns:
        (単語, 出現回数) のタプルのリスト（回数が多い順）
    """
    return counter.most_common(n)
