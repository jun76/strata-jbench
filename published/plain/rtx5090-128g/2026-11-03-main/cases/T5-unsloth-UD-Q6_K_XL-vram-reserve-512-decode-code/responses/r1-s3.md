#!/usr/bin/env python3
"""
テキストファイルから単語出現頻度を統計し、上位 N 件を表形式で表示する CLI ツール。

Usage:
    python word_frequency.py <input_file> [--top N] [--case-sensitive]

Example:
    python word_frequency.py sample.txt --top 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    単語は英数字およびハイフン・アポストロフィを含まない連続した英字列として扱う。
    大文字小文字の区別はフラグで制御する。

    Args:
        text: 入力テキスト文字列
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、すべて小文字に変換する。

    Returns:
        抽出された単語のリスト（出現順）
    """
    if not case_sensitive:
        text = text.lower()
    # 英字のみを単語として抽出（数字は除外。必要に応じて [a-zA-Z0-9]+ などに変更可能）
    words = re.findall(r"[a-zA-Z]+", text)
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 N 件の (単語, 出現回数) を降順で返す。

    Args:
        counter: 出現頻度を格納した Counter
        n: 取得する上位件数

    Returns:
        (単語, 出現回数) のタプルのリスト。出現回数の降順。
        同率の場合は辞書順で安定ソートされる（Counter.most_common の仕様に基づく）。
    """
    return counter.most_common(n)
