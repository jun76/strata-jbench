#!/usr/bin/env python3
"""
wordcount.py — テキストファイルから単語の出現頻度を算出し、上位 N 件を表形式で出力する CLI ツール。

使用例:
    python wordcount.py input.txt -n 10
    python wordcount.py input.txt -n 5 --case-sensitive
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。False の場合、すべて小文字に変換する。

    Returns:
        単語トークンのリスト。正規表現で [a-zA-Z']+ にマッチする連続文字列を抽出する。
        アポストロフィを含む単語 (例: don't, it's) も単語として扱う。
    """
    if not case_sensitive:
        text = text.lower()
    # アポストロフィを含む英単語を抽出 (例: don't, it's, can't)
    pattern = r"[a-zA-Z']+"
    tokens = re.findall(pattern, text)
    # 先頭・末尾のアポストロフィを除去し、空文字列を除外
    cleaned = [t.strip("'") for t in tokens if t.strip("'")]
    return cleaned


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト内の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字・小文字を区別するかどうか。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    tokens = tokenize(text, case_sensitive)
    return Counter(tokens)
