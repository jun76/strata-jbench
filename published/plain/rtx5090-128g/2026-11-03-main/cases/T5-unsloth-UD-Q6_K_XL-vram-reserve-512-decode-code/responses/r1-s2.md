word_freq.py - 単語出現頻度集計ツール

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位N件の単語をフォーマットされた表形式で表示するコマンドラインツール。

Usage:
    python word_freq.py input.txt -n 10

    cat input.txt | python word_freq.py -n 5
"""

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    単語は英数字の連続と見なし、それ以外の文字を区切りとして扱う。
    大文字小文字を区別しない場合、全て小文字に変換する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        単語のリスト。

    Examples:
        >>> tokenize("Hello, World! hello")
        ['hello', 'world', 'hello']
        >>> tokenize("Hello, World!", case_sensitive=True)
        ['Hello', 'World']
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字（およびアンダースコアを含む単語）を抽出
    words = re.findall(r'[a-zA-Z0-9]+(?:_[a-zA-Z0-9]+)*', text)
    return words
