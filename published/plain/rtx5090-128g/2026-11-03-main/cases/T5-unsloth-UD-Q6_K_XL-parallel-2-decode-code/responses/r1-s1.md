def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If True, preserve case. If False, convert to lowercase.
        min_length: Minimum length of a word to include.

    Returns:
        A list of tokenized words.
    """
    # Extract words using regex (alphanumeric sequences)
    words = re.findall(r'[a-zA-Z0-9]+', text)
    
    if not case_sensitive:
        words = [w.lower() for w in words]
    
    # Filter by minimum length
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    
    return words


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter[str]:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text.

        min_length: Minimum length of a word to include.

    Returns:
        A Counter object with word frequencies.
    """
    tokens = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(tokens)
