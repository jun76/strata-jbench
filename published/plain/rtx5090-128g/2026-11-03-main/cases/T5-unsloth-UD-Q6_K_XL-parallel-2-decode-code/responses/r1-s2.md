#!/usr/bin/env python3
"""A command-line tool to count word frequencies in a text file and display the top N results in a table.

Usage:
    python word_count.py <input_file> [-n TOP_N] [-s]

Options:
    -n, --top N        Number of top results to display (default: 10)
    -s, --sort-asc     Sort by ascending frequency instead of descending (default: False)
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """Extract words from text, converting to lowercase and ignoring non-alphabetic characters.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Convert to lowercase and extract words consisting of letters only
    words = re.findall(r'[a-zA-Z]+', text.lower())
    return words


def count_word_frequencies(text: str) -> Counter:
    """Count the frequency of each word in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter object with word frequencies.
    """
    words = extract_words(text)
    return Counter(words)


def get_top_words(counter: Counter, n: int, ascending: bool = False) -> List[Tuple[str, int]]:
    """Get the top N most frequent words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top results to return.
        ascending: If True, sort by ascending frequency; otherwise descending.

    Returns:
        A list of tuples (word, count) sorted by frequency.
    """
    if ascending:
        sorted_items = sorted(counter.items(), key=lambda x: x[1])
    else:
        sorted_items = counter.most_common(n)
        # If ascending, we still take the last n items
        if ascending:
            sorted_items = sorted(counter.items(), key=lambda x: x[1])[:n]
        else:
            sorted_items = counter.most_common(n)
    return sorted_items[:n]
