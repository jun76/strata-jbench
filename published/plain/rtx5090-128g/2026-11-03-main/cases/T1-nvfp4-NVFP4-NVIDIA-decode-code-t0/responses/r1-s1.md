## 1. 本体: `word_freq.py`

```python
#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

使い方:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt -n 5 --case-sensitive
    cat input.txt | python word_freq.py -n 3
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む連続文字列を単語とみなす
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 N 件を取得する。

    Args:
        counter: 単語頻度の Counter。
        n: 取得する件数。

    Returns:
        (単語, 出現回数) のタプルのリスト。頻度順、同頻度の場合は辞書順。
    """
    # most_common は頻度順に返すが、同頻度時の順序は保証されないため、
    # 明示的にソートする
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
