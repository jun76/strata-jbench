#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語出現頻度を集計し、上位N件を表形式で表示するCLIツール。

使い方:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt -n 5 --case-sensitive
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: Trueの場合、大文字小文字を区別する。Falseの場合、すべて小文字に変換する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    # 例: "don't" -> "don't", "state-of-the-art" -> "state-of-the-art"
    words = re.findall(r"[a-zA-Z0-9]+(?:['-][a-zA-Z0-9]+)*", text)
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。

    Returns:
        単語をキー、出現回数を値とするCounterオブジェクト。
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)
