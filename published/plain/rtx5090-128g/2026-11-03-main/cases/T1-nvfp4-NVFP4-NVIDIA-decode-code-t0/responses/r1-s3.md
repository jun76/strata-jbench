word_freq.py - テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

    cat input.txt | python word_freq.py -n 3

from __future__ import annotations

import argparse

from collections import Counter

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む連続文字列を単語とみなす
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return words

    """

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    words = tokenize(text, case_sensitive=case_sensitive)
    return Counter(words)
