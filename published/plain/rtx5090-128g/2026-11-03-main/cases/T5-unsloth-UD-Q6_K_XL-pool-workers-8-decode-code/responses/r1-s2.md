#!/usr/bin/env python3
"""
text_freq.py - A command-line tool to count word frequencies in a text file
and display the top N most frequent words in a table format.

Usage:
    python text_freq.py <file> [-n N] [--case-sensitive] [--min-length LEN]
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize_text(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, words are lowercased before counting.
        min_length: Minimum length of a word to be included in the results.

    Returns:
        A list of extracted words.
    """
    if not case_sensitive:
        text = text.lower()

    # Extract sequences of letters (and optionally digits/underscores?)
    # We'll use a simple regex to extract words consisting of letters
    words = re.findall(r'[a-zA-Z]+', text)

    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]

    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words to count.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their counts.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) for the top N words.
    """
    return counter.most_common(n)
