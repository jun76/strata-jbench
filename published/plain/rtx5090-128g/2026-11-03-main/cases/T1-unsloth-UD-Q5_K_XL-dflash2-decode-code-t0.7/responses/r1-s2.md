import argparse

from collections import Counter

def load_text(filepath: str) -> str:
    """
    テキストファイルを読み込み、内容を文字列として返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み取りに失敗した場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    単語は英数字とハイフン、アポストロフィを含む連続した文字列として定義される。
    case_sensitive が False の場合、単語は小文字に変換される。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字・小文字を区別する場合 True。

    Returns:
        抽出された単語のリスト。
    """
    # 英数字、ハイフン、アポストロフィを含む単語を抽出
    pattern = r"[a-zA-Z0-9'-]+"
    words = re.findall(pattern, text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)
