word_counter.py

A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.

Usage:
    python word_counter.py <file> [--top N] [--case-sensitive]
"""

import argparse

from collections import Counter
from typing import List, Tuple


def read_file(filepath: str) -> str:
    """Read and return the content of a text file.

    Args:
        filepath: Path to the text file to read.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If an I/O error occurs while reading the file.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """Tokenize text into words.

    Words are defined as sequences of alphanumeric characters, possibly
    including underscores. Punctuation and other non-word characters are
    treated as separators.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, all words are converted to lowercase.

    Returns:
        A list of word tokens.
    """
    if not case_sensitive:
        text = text.lower()
    # Match sequences of word characters (letters, digits, underscores)
    tokens = re.findall(r'\b\w+\b', text)
    return tokens


def count_words(tokens: List[str]) -> Counter:
    """Count the frequency of each word in the token list.

    Args:
        tokens: A list of word tokens.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(tokens)
