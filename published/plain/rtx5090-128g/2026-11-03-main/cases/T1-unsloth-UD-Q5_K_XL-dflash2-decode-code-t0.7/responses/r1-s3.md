#!/usr/bin/env python3
"""
単語出現頻度集計ツール

テキストファイルを読み込み、単語の出現頻度を上位N件まで表形式で表示します。
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def read_text_file(file_path: str) -> str:
    """
    テキストファイルを読み込み、内容を文字列として返す。

    Args:
        file_path: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み込みに失敗した場合。
    """
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {file_path}")
    except IOError as e:
        raise IOError(f"ファイルの読み込みに失敗しました: {file_path}. 理由: {e}")


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。

    単語は英数字およびハイフン・アポストロフィを含む文字列として認識されます。
    すべて小文字に変換されます。

    Args:
        text: 処理対象のテキスト。

    Returns:
        単語のリスト。
    """
    # 正規表現で単語を抽出: 英数字、ハイフン、アポストロフィを含む連続した文字列
    # \b は単語境界。ここではよりシンプルなパターンを使用
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    # すべて小文字に変換
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とするCounterオブジェクト。
    """
    return Counter(words)
