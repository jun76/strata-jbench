"""
word_freq.py - コマンドラインツールでテキストファイルから単語出現頻度を集計し、
上位N件を表形式で表示する。

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def normalize_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出し、正規化する。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。

    Returns:
        単語のリスト
    """
    # 単語を抽出: 英数字とアンダースコアの連続
    words = re.findall(r'[a-zA-Z0-9_]+', text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_word_frequencies(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキストから単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか

    Returns:
        単語 -> 出現回数 の Counter
    """
    words = normalize_words(text, case_sensitive)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度上位 N 件の単語を返す。同頻度の場合は辞書順でソート。

    Args:
        counter: 単語 -> 出現回数の Counter
        n: 取得する件数

    Returns:
        [(word, count), ...] のリスト
    """
    # 頻度で降順、頻度が同じ場合は単語で昇順
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
