def tokenize(text: str) -> List[str]:
    """テキストを単語リストに分割する。

    英数字の連続を単語とし、非英数字を区切りとして扱う。
    大文字/小文字は区別しない（すべて小文字に変換）。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    words: List[str] = []
    current: List[str] = []
    for char in text:
        if char.isalnum():
            current.append(char.lower())
        else:
            if current:
                words.append("".join(current))
                current = []
    if current:
        words.append("".join(current))
    return words


def count_words(text: str) -> Counter:
    """テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    words = tokenize(text)
    return Counter(words)
