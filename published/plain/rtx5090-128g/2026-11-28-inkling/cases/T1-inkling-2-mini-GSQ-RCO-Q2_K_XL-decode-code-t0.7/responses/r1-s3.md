import argparse
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def count_words(text: str) -> Counter:
    """テキストを読み込み、単語の出現頻度を集計する。"""
    return Counter(text.split())


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """集計結果から上位 N 件を返す。"""
    return counter.most_common(n)


def render_table(items: List[Tuple[str, int]]) -> str:
    """上位 N 件を表形式で整形する。"""
    if not items:
        return "No words found."
    lines = ["Word\tCount"]
    for word, count in items:
        lines.append(f"{word}\t{count}")
    return "\n".join(lines)


def main() -> None:
    parser = argparse.ArgumentParser(description="テキストファイルから単語頻度を集計し、上位 N 件を表示する。")
    parser.add_argument("file", type=Path, help="読み込むテキストファイル")
    parser.add_argument("-n", type=int, default=10, help="表示する件数（デフォルト: 10）")
    args = parser.parse_args()

    text = args.file.read_text(encoding="utf-8")
    counter = count_words(text)
    print(render_table(top_n(counter, args.n)))


if __name__ == "__main__":
    main()
