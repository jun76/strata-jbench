import argparse
from collections import Counter
from pathlib import Path
from typing import Iterable


def count_words(text: str) -> Counter:
    """テキストから単語を抽出し、出現頻度を集計する。"""
    return Counter(word.lower() for word in text.split())


def top_words(counter: Counter, n: int) -> list[tuple[str, int]]:
    """上位 n 件の (単語, 頻度) を返す。"""
    return counter.most_common(n)


def format_table(rows: list[tuple[str, int]]) -> str:
    """表形式の文字列を生成する。"""
    if not rows:
        return "(結果なし)"
    width = max(len(word) for word, _ in rows)
    lines = [f"{'単語'.ljust(width)}  頻度", f"{'-' * width}  -----"]
    for word, freq in rows:
        lines.append(f"{word.ljust(width)}  {freq}")
    return "\n".join(lines)


def main() -> None:
    parser = argparse.ArgumentParser(description="テキストファイルの単語頻度を集計し上位 N 件を表示します。")
    parser.add_argument("file", type=Path, help="入力テキストファイル")
    parser.add_argument("-n", type=int, default=10, help="表示件数（既定: 10）")
    args = parser.parse_args()

    text = args.file.read_text(encoding="utf-8")
    rows = top_words(count_words(text), args.n)
    print(format_table(rows))


if __name__ == "__main__":
    main()
