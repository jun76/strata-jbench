#!/usr/bin/env python3
"""テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示するCLI。"""

from __future__ import annotations

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import Iterable


def tokenize(text: str) -> list[str]:
    """テキストを小文字化して単語に分割する。"""
    return re.findall(r"[a-zA-Z0-9_]+", text.lower())


def count_words(path: Path) -> Counter[str]:
    """ファイルを読み込み、単語頻度を集計する。"""
    text = path.read_text(encoding="utf-8")
    return Counter(tokenize(text))


def format_table(counts: Counter[str], top_n: int) -> str:
    """上位 N 件を表形式で返す。"""
    if top_n <= 0:
        return "表示件数は 0 です。"
    rows = counts.most_common(top_n)
    if not rows:
        return "単語が見つかりませんでした。"
    max_word_len = max(len(word) for word, _ in rows)
    max_count_len = max(len(str(count)) for _, count in rows)
    header = f"{'Word'.ljust(max_word_len)} | {'Count'.rjust(max_count_len)}"
    separator = f"{'-' * max_word_len}-+-{'-' * max_count_len}"
    lines = [header, separator]
    for word, count in rows:
        lines.append(f"{word.ljust(max_word_len)} | {str(count).rjust(max_count_len)}")
    return "\n".join(lines)
