import argparse
from collections import Counter
from pathlib import Path
from typing import Dict, List


def count_words(text: str) -> Dict[str, int]:
    """単語の出現頻度を集計する。"""
    words = text.lower().split()
    return Counter(words)


def top_n(counts: Dict[str, int], n: int) -> List[tuple[str, int]]:
    """上位 N 件を降順で返す。"""
    return sorted(counts.items(), key=lambda item: (-item[1], item[0]))[:n]


def format_table(rows: List[tuple[str, int]]) -> str:
    """表形式で文字列を返す。"""
    if not rows:
        return "なし"
    lines = ["単語\t出現数"]
    for word, count in rows:
        lines.append(f"{word}\t{count}")
    return "\n".join(lines)


def load_text(path: Path) -> str:
    """テキストファイルを読み込む。"""
    return path.read_text(encoding="utf-8")


def main() -> None:
    parser = argparse.ArgumentParser(
        description="テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示します。"
    )
    parser.add_argument("file", type=Path, help="入力テキストファイル")
    parser.add_argument("-n", type=int, default=10, help="表示する上位件数（既定: 10）")
    args = parser.parse_args()

    text = load_text(args.file)
    counts = count_words(text)
    rows = top_n(counts, args.n)
    print(format_table(rows))
