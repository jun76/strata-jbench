```python
import argparse
from collections import Counter
from pathlib import Path


def count_words(text: str) -> dict[str, int]:
    """トークン化して単語の出現頻度を集計する。"""
    return dict(Counter(text.lower().split()))


def top_n(text: str, n: int) -> list[tuple[str, int]]:
    """上位 n 件の (単語, 出現回数) を返す。"""
    return Counter(text.lower().split()).most_common(n)


def render_table(items: list[tuple[str, int]]) -> str:
    """表形式の文字列を返す。"""
    if not items:
        return "(結果なし)"
    rows = [f"{word:<20} {count:>6}" for word, count in items]
    header = f"{'word':<20} {'count':>6}"
    return "\n".join([header, "-" * len(header), *rows])


def main() -> None:
    parser = argparse.ArgumentParser(description="テキストファイルの単語出現頻度を集計して上位 N 件を表示する")
    parser.add_argument("path", type=Path, help="読み込むテキストファイルのパス")
    parser.add_argument("-n", type=int, default=10, help="上位 N 件を表示する（既定: 10）")
    args = parser.parse_args()

    text = args.path.read_text(encoding="utf-8")
    print(render_table(top_n(text, args.n)))


if __name__ == "__main__":
    main()
```
