```python
"""テキストファイルから単語の出現頻度を集計して上位N件を表示するCLIツール。"""
import argparse
import re
from collections import Counter
from pathlib import Path

def read_text(path: str) -> str:
    return Path(path).read_text(encoding="utf-8")

def count_words(text: str) -> Counter:
    words = re.findall(r"\w+", text.lower(), flags=re.UNICODE)
    return Counter(words)

def top_n(counter: Counter, n: int) -> list[tuple[str, int]]:
    return counter.most_common(n)

def format_table(rows: list[tuple[str, int]]) -> str:
    lines = ["word\tcount"]
    for word, count in rows:
        lines.append(f"{word}\t{count}")
    return "\n".join(lines)

def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="テキストの単語頻度を集計し上位N件を表示")
    parser.add_argument("file", help="入力テキストファイル")
    parser.add_argument("-n", type=int, default=10, help="表示する上位件数")
    args = parser.parse_args(argv)
    counter = count_words(read_text(args.file))
    print(format_table(top_n(counter, args.n)))
    return 0

if __name__ == "__main__":
    raise SystemExit(main())
```
