import argparse
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count word occurrences in the text.

    Args:
        text: Input text.

    Returns:
        Counter mapping words to their occurrence counts.
    """
    return Counter(text.split())


def top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N words by frequency.

    Args:
        counter: Word frequency counter.
        n: Number of top words to return.

    Returns:
        List of (word, count) tuples.
    """
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """Format rows as a simple table.

    Args:
        rows: List of (word, count) tuples.

    Returns:
        Formatted table string.
    """
    lines = ["Word\tCount"]
    for word, count in rows:
        lines.append(f"{word}\t{count}")
    return "\n".join(lines)


def main(args: argparse.Namespace) -> None:
    """Entry point for the command-line tool.

    Args:
        args: Parsed command-line arguments.
    """
    with open(args.path, "r", encoding="utf-8") as f:
        text = f.read()
    counts = count_words(text)
    rows = top_words(counts, args.n)
    print(format_table(rows))
