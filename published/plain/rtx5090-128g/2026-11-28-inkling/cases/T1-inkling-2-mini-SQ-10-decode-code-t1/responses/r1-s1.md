以下は、テキストファイルを読み込み、単語の出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツールです。

```python
"""
word_frequency.py

テキストファイルを読み込み、単語の出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。
"""

from __future__ import annotations

import argparse
import re
from collections import Counter
from pathlib import Path


def tokenize_text(text: str) -> list[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト

    Returns:
        小文字化された単語のリスト
    """
    return re.findall(r"[a-zA-Z0-9]+", text.lower())


def count_word_frequency(text: str) -> Counter:
    """
    テキスト中の単語の出現頻度を集計する。

    Args:
        text: 入力テキスト

    Returns:
        Counter オブジェクト
    """
    return Counter(tokenize_text(text))


def top_words(counter: Counter, n: int) -> list[tuple[str, int]]:
    """
    出現頻度が高い上位 N 件の単語を返す。

    Args:
        counter: 単語出現頻度の Counter
        n: 取得する件数

    Returns:
        (単語, 頻度) のリスト（頻度降順）
    """
    return counter.most_common(n)
