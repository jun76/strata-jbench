**freq_counter.py**

```python
from __future__ import annotations
import argparse
from collections import Counter
from pathlib import Path

def count_words(text: str) -> Counter[str]:
    """
    テキストから単語の出現頻度を集計する。
    """
    words = text.lower().split()
    return Counter(words)

def read_text(path: Path) -> str:
    """テキストファイルを読み込み、UTF-8 で解釈して返す。"""
    return path.read_text(encoding="utf-8")

def top_n_words(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """出現頻度の高い上位 N 件を返す。"""
    return counter.most_common(n)

def main() -> None:
    parser = argparse.ArgumentParser(description="テキストファイルの単語頻度を集計し上位 N 件を表示する")
    parser.add_argument("path", type=Path, help="入力テキストファイル")
    parser.add_argument("--n", type=int, default=10, help="表示する上位 N 件（デフォルト: 10）")
    args = parser.parse_args()

    if args.n < 1:
        raise ValueError("--n は 1 以上である必要があります")

    text = read_text(args.path)
    counter = count_words(text)
    top = top_n_words(counter, args.n)

    print(f"上位 {args.n} 件:")
    for word, count in top:
        print(f"{word}\t{count}")

if __name__ == "__main__":
    main()
```
