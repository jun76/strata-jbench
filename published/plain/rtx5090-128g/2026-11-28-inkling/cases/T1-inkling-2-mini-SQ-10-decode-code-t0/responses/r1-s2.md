```python
# word_freq.py
from __future__ import annotations

import argparse
from collections import Counter
from pathlib import Path
from typing import Iterable


def count_words(text: str) -> Counter:
    """テキストから単語を抽出し、出現頻度を集計する。"""
    words = text.lower().split()
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> list[tuple[str, int]]:
    """上位 N 件の (単語, 頻度) を返す。"""
    return counter.most_common(n)


def read_text(path: Path) -> str:
    """テキストファイルを読み込む。"""
    return path.read_text(encoding="utf-8")


def main() -> None:
    parser = argparse.ArgumentParser(description="テキストファイルの単語出現頻度を集計する")
    parser.add_argument("file", type=Path, help="入力テキストファイル")
    parser.add_argument("-n", type=int, default=10, help="表示する上位件数")
    args = parser.parse_args()

    text = read_text(args.file)
    counter = count_words(text)
    top = top_n_words(counter, args.n)

    print(f"{'word':<20} {'count':>6}")
    print("-" * 28)
    for word, count in top:
        print(f"{word:<20} {count:>6}")


if __name__ == "__main__":
    main()
```
