def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    if not intervals:
        return []

    merged = []
    for start, end in sorted(intervals, key=lambda interval: (interval[0], interval[1])):
        if not merged:
            merged.append([start, end])
            continue

        last = merged[-1]
        if start <= last[1]:
            if end > last[1]:
                last[1] = end
        else:
            merged.append([start, end])

    return merged
