以下は、テキストファイルを読み込み、単語の出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツールです。

**main.py**

```python
import argparse
from collections import Counter
from pathlib import Path
from typing import List


def load_words(file_path: Path) -> List[str]:
    """テキストファイルを読み込み、単語のリストを返す。"""
    text = file_path.read_text(encoding="utf-8")
    return text.split()


def count_words(words: List[str]) -> Counter:
    """単語リストから出現頻度を集計する。"""
    return Counter(words)


def top_words(counter: Counter, n: int) -> List[tuple]:
    """出現頻度の上位 N 件を返す。"""
    return counter.most_common(n)


def print_table(top: List[tuple]) -> None:
    """上位 N 件を表形式で表示する。"""
    print(f"{'単語':<30} {'出現数':>10}")
    print("-" * 40)
    for word, count in top:
        print(f"{word:<30} {count:>10}")


def main() -> None:
    parser = argparse.ArgumentParser(
        description="テキストファイルの単語出現頻度を集計し、上位 N 件を表形式で表示します。"
    )
    parser.add_argument("file", type=Path, help="対象テキストファイルのパス")
    parser.add_argument(
        "-n",
        type=int,
        default=10,
        help="表示する上位 N 件の数（デフォルト: 10）",
    )
    args = parser.parse_args()

    words = load_words(args.file)
    counter = count_words(words)
