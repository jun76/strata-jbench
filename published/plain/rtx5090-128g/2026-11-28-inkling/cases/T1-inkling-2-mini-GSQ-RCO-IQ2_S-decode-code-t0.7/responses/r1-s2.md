"""テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で表示する。"""

from __future__ import annotations

import argparse
import re
from collections import Counter
from pathlib import Path


def count_words(text: str) -> Counter:
    """テキストから単語を抽出して出現頻度を返す。

    Args:
        text: 対象テキスト。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    words = re.findall(r"[A-Za-z0-9_]+", text.lower())
    return Counter(words)


def top_n(counter: Counter, n: int) -> list[tuple[str, int]]:
    """Counter の上位 n 件を (単語, 回数) のリストで返す。"""
    return counter.most_common(n)


def format_table(rows: list[tuple[str, int]]) -> str:
    """(単語, 回数) のリストを表形式の文字列にする。"""
    if not rows:
        return "（結果なし）"
    word_w = max(len(w) for w, _ in rows)
    lines = [f"{'単語':<{word_w}}  回数", f"{'-' * word_w}  ---"]
    for word, count in rows:
        lines.append(f"{word:<{word_w}}  {count}")
    return "\n".join(lines)
