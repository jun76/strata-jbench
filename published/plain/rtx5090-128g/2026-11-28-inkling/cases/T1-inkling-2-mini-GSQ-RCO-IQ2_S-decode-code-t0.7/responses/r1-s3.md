import argparse
from collections import Counter
from pathlib import Path


def count_words(text: str) -> Counter:
    """Return a Counter of lowercase word frequencies."""
    words = text.lower().split()
    return Counter(words)


def top_words(counter: Counter, n: int) -> list[tuple[str, int]]:
    """Return the top n words as (word, count) tuples."""
    return counter.most_common(n)


def render_table(rows: list[tuple[str, int]]) -> str:
    """Render rows as a simple aligned table."""
    if not rows:
        return "No words found."
    max_word = max(len(word) for word, _ in rows)
    lines = [f"{'Word'.ljust(max_word)}  Count"]
    lines.append("-" * (max_word + 8))
    for word, count in rows:
        lines.append(f"{word.ljust(max_word)}  {count}")
    return "\n".join(lines)


def main() -> None:
    parser = argparse.ArgumentParser(description="Read a text file and show the top N words.")
    parser.add_argument("file", type=Path, help="Path to the text file")
    parser.add_argument("-n", "--top", type=int, default=10, help="Number of top words to show")
    args = parser.parse_args()

    text = args.file.read_text(encoding="utf-8")
    counter = count_words(text)
    print(render_table(top_words(counter, args.top)))


if __name__ == "__main__":
    main()
