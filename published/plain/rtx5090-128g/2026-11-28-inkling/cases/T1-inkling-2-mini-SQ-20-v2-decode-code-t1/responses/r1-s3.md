import argparse
from collections import Counter
from pathlib import Path


def count_words(text: str) -> dict[str, int]:
    """Count word frequencies in text (case-insensitive, alphabetic words)."""
    words = []
    for raw in text.lower().split():
        # Strip non-alphabetic punctuation from edges of each token
        token = raw.strip(".,;:!?\"'()[]{}<>")
        if token.isalpha():
            words.append(token)
    return dict(Counter(words))


def format_table(counter: dict[str, int], top_n: int) -> str:
    """Format top N words as a simple aligned table."""
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))[:top_n]
    if not items:
        return "(no words found)"
    width = max(len(word) for word, _ in items)
    lines = [f"{'Word'.ljust(width)}  Count"]
    lines.append("-" * (width + 8))
    for word, count in items:
        lines.append(f"{word.ljust(width)}  {count}")
    return "\n".join(lines)


def main() -> None:
    parser = argparse.ArgumentParser(
        description="Read a text file and display the top-N most frequent words."
    )
    parser.add_argument("filepath", type=Path, help="Path to the text file")
    parser.add_argument(
        "-n", "--top", type=int, default=10, help="Number of top words to show"
    )
    args = parser.parse_args()

    text = args.filepath.read_text(encoding="utf-8")
    counter = count_words(text)
    print(format_table(counter, args.top))


if __name__ == "__main__":
    main()
