#!/usr/bin/env python3
"""テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する CLI。"""
from __future__ import annotations

import argparse
import re
from collections import Counter
from pathlib import Path


def count_words(text: str) -> Counter[str]:
    """テキストを小文字化し、英数字の連続（単語）を抽出して頻度を集計する。"""
    tokens = re.findall(r"[A-Za-z0-9]+", text.lower())
    return Counter(tokens)


def read_text(path: Path) -> str:
    """指定パスのテキストファイルをUTF-8で読み込む。"""
    return path.read_text(encoding="utf-8")


def top_n_words(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """頻度が高い順に上位 n 件の (word, count) を返す。"""
    return counter.most_common(n)


def format_table(rows: list[tuple[str, int]]) -> str:
    """行リストを簡易表形式の文字列に整形する。"""
    if not rows:
        return "(該当する単語はありません)"
    word_width = max(len(word) for word, _ in rows)
    count_width = max(len(str(count)) for _, count in rows)
    header = f"{'Word'.ljust(word_width)}  {'Count'.rjust(count_width)}"
    separator = "-" * len(header)
    lines = [header, separator]
    for word, count in rows:
        lines.append(f"{word.ljust(word_width)}  {str(count).rjust(count_width)}")
    return "\n".join(lines)
