#!/usr/bin/env python3
"""
テキストファイルの単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

例:
    python word_freq.py sample.txt --top 10
"""
from __future__ import annotations

import argparse
from collections import Counter
from pathlib import Path
from typing import Iterable


def count_words(text: str) -> Counter:
    """テキストから単語を抽出し、出現頻度を返す。

    単語は英数字を含む連続文字列として扱い、小文字化して集計する。
    """
    words = []
    current = []
    for ch in text:
        if ch.isalnum():
            current.append(ch.lower())
        else:
            if current:
                words.append("".join(current))
                current = []
    if current:
        words.append("".join(current))
    return Counter(words)


def read_text(path: Path) -> str:
    """指定パスのテキストファイルを読み込む。"""
    return path.read_text(encoding="utf-8")


def format_table(counter: Counter, top_n: int) -> str:
    """上位 N 件を表形式で整形する。"""
    rows = counter.most_common(top_n)
    if not rows:
        return "単語が見つかりませんでした。"

    header = f"{'word':<30} {'count':>10}"
    separator = "-" * len(header)
    lines = [header, separator]
    for word, count in rows:
        lines.append(f"{word:<30} {count:>10}")
    return "\n".join(lines)
