#!/usr/bin/env python3
"""テキストファイルから単語出現頻度を集計し上位 N 件を表形式で表示するCLI。"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """テキストから単語を抽出し、出現回数を集計する。"""
    words = re.findall(r"[A-Za-z0-9]+", text.lower())
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """上位 N 件の (word, count) を返す。同数は辞書順で安定化。"""
    return sorted(counter.items(), key=lambda item: (-item[1], item[0]))[:n]
