以下は、テキストファイルを読み込んで単語の出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツールです。

```python
#!/usr/bin/env python3
"""
テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示するCLIツール。
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを単語トークンに分割する。"""
    return re.findall(r"\w+", text.lower())


def count_words(text: str) -> Counter:
    """テキストから単語の出現頻度を集計する。"""
    return Counter(tokenize(text))


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """集計結果から上位 N 件を返す。"""
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """上位 N 件を表形式の文字列に整形する。"""
    if not items:
        return "(no words found)"
    word_width = max(len(word) for word, _ in items)
    lines = [
        f"{'Word':<{word_width}}  {'Count':>6}",
        f"{'-' * word_width}  {'-' * 6}",
    ]
    for word, count in items:
        lines.append(f"{word:<{word_width}}  {count:>6}")
    return "\n".join(lines)
