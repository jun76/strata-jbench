以下は、テキストファイルを読み込んで単語の出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツールです。

```python
#!/usr/bin/env python3
"""テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する。"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """テキストから単語を抽出して出現頻度を集計する。"""
    words = re.findall(r"\b\w+\b", text.lower())
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """出現頻度の高い上位 N 件を返す。"""
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """上位 N 件を表形式の文字列に整形する。"""
    if not items:
        return "(データなし)"
    max_word_len = max(len(word) for word, _ in items)
    max_count_len = max(len(str(count)) for _, count in items)
    header = f"{'単語'.ljust(max_word_len)} | {'頻度'.rjust(max_count_len)}"
    separator = f"{'-' * max_word_len}-+-{'-' * max_count_len}"
    rows = [f"{word.ljust(max_word_len)} | {str(count).rjust(max_count_len)}" for word, count in items]
    return "\n".join([header, separator] + rows)
