以下は、テキストファイルを読み込んで単語の出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツールです。

```python
#!/usr/bin/env python3
"""テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する。"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """テキストから単語を抽出して出現回数を集計する。"""
    words = re.findall(r"\b\w+\b", text.lower())
    return Counter(words)


def read_file(path: Path) -> str:
    """指定パスのテキストファイルを読み込む。"""
    return path.read_text(encoding="utf-8")


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """出現頻度の高い上位 N 件を返す。"""
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """上位 N 件を表形式の文字列に整形する。"""
    if not items:
        return "(データなし)"
    max_word_len = max(len(word) for word, _ in items)
    lines = [f"{'単語':<{max_word_len}}  出現回数"]
    lines.append("-" * (max_word_len + 12))
    for word, count in items:
        lines.append(f"{word:<{max_word_len}}  {count}")
    return "\n".join(lines)


def main() -> None:
    parser = argparse.ArgumentParser(description="テキストファイルの単語出現頻度を集計して上位 N 件を表示します。")
