#!/usr/bin/env python3
"""テキストファイルを読み込んで単語の出現頻度を集計し、上位 N 件を表示する CLI ツール。"""

from __future__ import annotations

import argparse
from collections import Counter
from pathlib import Path


def count_words(text: str) -> Counter[str]:
    """テキストを単純に小文字化し、空白で分割して単語頻度を数える。"""
    words = text.lower().split()
    return Counter(words)


def top_n_words(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """頻度上位 N 件を、頻度降順・同率は単語昇順で返す。"""
    items = sorted(counter.items(), key=lambda item: (-item[1], item[0]))
    return items[:n]


def format_table(items: list[tuple[str, int]]) -> str:
    """上位 N 件を簡易表形式の文字列にする。"""
    if not items:
        return "(単語が見つかりません)"

    name_width = max(len(word) for word, _ in items)
    lines = [f"{'単語':<{name_width}} | 出現数", f"{'-' * name_width}-+--------"]
    for word, count in items:
        lines.append(f"{word:<{name_width}} | {count}")
    return "\n".join(lines)
