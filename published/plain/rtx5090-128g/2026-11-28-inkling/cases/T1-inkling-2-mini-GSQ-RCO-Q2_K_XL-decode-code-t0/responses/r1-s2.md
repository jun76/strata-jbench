import argparse
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """テキストから単語を抽出して出現頻度を集計する。"""
    words = text.lower().split()
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """出現頻度が高い上位 N 件を返す。"""
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """表形式の文字列を生成する。"""
    if not rows:
        return "(データなし)"
    width = max(len(word) for word, _ in rows)
    lines = [f"{'単語':<{width}}  {'頻度':>5}"]
    lines.append("-" * (width + 7))
    for word, freq in rows:
        lines.append(f"{word:<{width}}  {freq:>5}")
    return "\n".join(lines)


def main() -> None:
    parser = argparse.ArgumentParser(description="テキストファイルの単語出現頻度を集計し上位 N 件を表示します。")
    parser.add_argument("file", type=Path, help="読み込むテキストファイル")
    parser.add_argument("-n", type=int, default=10, help="表示する上位件数（デフォルト: 10）")
    args = parser.parse_args()

    text = args.file.read_text(encoding="utf-8")
    counter = count_words(text)
    rows = top_n_words(counter, args.n)
    print(format_table(rows))


if __name__ == "__main__":
    main()
