以下は、テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で表示するコマンドラインツールです。

```python
#!/usr/bin/env python3
"""テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する。"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """テキストを単語に分割して出現頻度を集計する。"""
    words = re.findall(r"[A-Za-z0-9_]+", text.lower())
    return Counter(words)


def top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """出現頻度が高い順に上位 N 件を返す。"""
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """上位 N 件を表形式の文字列として返す。"""
    if not rows:
        return "(単語が見つかりませんでした)"

    word_width = max(len(word) for word, _ in rows)
    count_width = max(len(str(count)) for _, count in rows)

    header = f"{'単語':<{word_width}}  {'頻度':>{count_width}}"
    separator = "-" * len(header)

    lines = [header, separator]
    for word, count in rows:
        lines.append(f"{word:<{word_width}}  {count:>{count_width}}")
    return "\n".join(lines)


def main() -> None:
    parser = argparse.ArgumentParser(
        description="テキストファイルから単語の出現頻度を集計し、上位 N 件を表示します。"
    )
