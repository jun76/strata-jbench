import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """テキストから単語を抽出し、出現頻度を集計する。"""
    words = re.findall(r"\b\w+\b", text.lower())
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """集計結果から上位 N 件を返す。"""
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """上位 N 件を表形式で整形する。"""
    if not items:
        return "No words found."

    word_width = max(len(word) for word, _ in items)
    count_width = max(len(str(count)) for _, count in items)

    header = f"{'Word'.ljust(word_width)}  {'Count'.rjust(count_width)}"
    separator = "-" * (word_width + 2 + count_width)

    rows = [
        f"{word.ljust(word_width)}  {str(count).rjust(count_width)}"
        for word, count in items
    ]

    return "\n".join([header, separator] + rows)
