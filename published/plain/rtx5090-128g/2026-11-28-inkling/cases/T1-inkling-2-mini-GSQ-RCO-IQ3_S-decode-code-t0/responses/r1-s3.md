import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """テキストから単語を抽出して頻度を集計する。"""
    words = re.findall(r"[A-Za-z0-9_]+", text.lower())
    return Counter(words)


def top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """頻度上位 n 件を返す。同数の場合は単語順で安定化する。"""
    return sorted(counter.items(), key=lambda item: (-item[1], item[0]))[:n]


def format_table(rows: List[Tuple[str, int]]) -> str:
    """表形式の文字列を生成する。"""
    if not rows:
        return "No words found."

    width = max(len(word) for word, _ in rows)
    lines = [f"{'Word':<{width}}  Count", f"{'-' * width}  -----"]
    for word, count in rows:
        lines.append(f"{word:<{width}}  {count}")
    return "\n".join(lines)
