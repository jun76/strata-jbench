以下は、テキストファイルを読み込んで単語の出現頻度を集計し、上位 N 件を表形式で表示する Python CLI ツールです。

**main.py**
```python
"""テキストファイルから単語出現頻度を集計して上位N件を表示するCLIツール。"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """テキストから単語を抽出して頻度を集計する。"""
    words = re.findall(r"\b\w+\b", text.lower())
    return Counter(words)


def read_text(path: Path) -> str:
    """指定パスのテキストファイルを読み込む。"""
    return path.read_text(encoding="utf-8")


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """頻度の高い順に上位n件を返す。同数はアルファベット順。"""
    return sorted(counter.items(), key=lambda x: (-x[1], x[0]))[:n]


def format_table(rows: List[Tuple[str, int]]) -> str:
    """上位N件を表形式に整形して返す。"""
    if not rows:
        return "(該当する単語はありません)"
    width = max(len(word) for word, _ in rows)
    lines = [f"{'Word':<{width}}  {'Count':>5}"]
    lines.append(f"{'-' * width}  -----")
    for word, count in rows:
        lines.append(f"{word:<{width}}  {count:>5}")
    return "\n".join(lines)
