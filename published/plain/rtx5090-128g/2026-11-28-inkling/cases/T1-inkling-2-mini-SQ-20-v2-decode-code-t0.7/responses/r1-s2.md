#!/usr/bin/env python3
"""テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path


def tokenize(text: str) -> list[str]:
    """テキストを小文字の単語に分割する。"""
    return re.findall(r"[a-zA-Z0-9]+", text.lower())


def count_words(text: str) -> Counter[str]:
    """テキストから単語の出現頻度を数える。"""
    return Counter(tokenize(text))


def top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """頻度順（降順）で上位 N 件を返す。同数はアルファベット順で安定化。"""
    items = sorted(counter.items(), key=lambda kv: (-kv[1], kv[0]))
    return items[:n]


def format_table(items: list[tuple[str, int]]) -> str:
    """表形式の文字列を生成する。"""
    if not items:
        return "（単語が見つかりませんでした）"
    word_w = max(len(w) for w, _ in items)
    lines = [f"{'単語'.ljust(word_w)}  {'回数':>6}", "-" * (word_w + 8)]
    for word, count in items:
        lines.append(f"{word.ljust(word_w)}  {count:>6}")
    return "\n".join(lines)
