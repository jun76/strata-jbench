#!/usr/bin/env python3
"""テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で表示するCLIツール。"""

from __future__ import annotations

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def count_words(text: str) -> Counter:
    """テキストを小文字化して単語トークンに分割し、頻度を集計する。"""
    tokens = re.findall(r"[a-zA-Z0-9_]+", text.lower())
    return Counter(tokens)


def load_text(path: Path) -> str:
    """指定パスのテキストファイルを読み込む。"""
    return path.read_text(encoding="utf-8")


def top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """頻度上位 n 件を (word, count) のリストで返す。"""
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """上位単語を表形式に整形して返す。"""
    if not rows:
        return "No words found."

    width = max(len(word) for word, _ in rows)
    lines = [f"{'Word':<{width}}  Count", f"{'-' * width}  -----"]
    for word, count in rows:
        lines.append(f"{word:<{width}}  {count}")
    return "\n".join(lines)
