### コード (`word_freq.py`)

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def count_words(text: str) -> Counter:
    """テキストから単語を抽出して出現頻度を集計する。"""
    words = re.findall(r"[\w]+", text.lower(), flags=re.UNICODE)
    return Counter(words)


def load_text(file_path: Path) -> str:
    """指定パスのテキストファイルを読み込む。"""
    with file_path.open("r", encoding="utf-8") as f:
        return f.read()


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """集計結果から上位 N 件を返す。"""
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """単語と頻度を表形式の文字列として返す。"""
    if not items:
        return "No words found."
    max_word_len = max(len(word) for word, _ in items)
    max_count_len = max(len(str(count)) for _, count in items)
    header = f"{'Word':<{max_word_len}}  {'Count':>{max_count_len}}"
    separator = "-" * (max_word_len + 2 + max_count_len)
    rows = [f"{word:<{max_word_len}}  {count:>{max_count_len}}" for word, count in items]
    return "\n".join([header, separator] + rows)
