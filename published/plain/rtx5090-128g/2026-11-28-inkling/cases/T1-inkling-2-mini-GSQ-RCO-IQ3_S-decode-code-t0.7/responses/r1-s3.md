import argparse
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def count_words(text: str) -> Counter:
    """
    テキストから単語を抽出し、出現頻度を集計する。

    :param text: 読み込んだテキスト
    :return: 単語と出現回数のカウンター
    """
    words = text.lower().split()
    return Counter(words)


def top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度上位 N 件を返す。

    :param counter: 単語出現頻度カウンター
    :param n: 表示件数
    :return: (単語, 出現回数) のリスト
    """
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """
    上位単語を表形式の文字列として返す。

    :param rows: (単語, 出現回数) のリスト
    :return: 表形式の文字列
    """
    if not rows:
        return "No words found."

    word_width = max(len(word) for word, _ in rows)
    lines = [
        f"{'Word':<{word_width}}  Count",
        f"{'-' * word_width}  -----",
    ]
    for word, count in rows:
        lines.append(f"{word:<{word_width}}  {count}")
    return "\n".join(lines)


def read_text_file(path: Path) -> str:
    """
    テキストファイルを読み込む。

    :param path: ファイルパス
    :return: ファイル内容
    """
    return path.read_text(encoding="utf-8")
