```python
import argparse
import sys
from collections import Counter
from pathlib import Path


def count_words(text: str) -> Counter:
    """テキストを空白と句読点などで分割し、小文字に変換して単語頻度を集計する"""
    words = [w.strip(".,;:!?\"'()[]{}").lower() for w in text.split()]
    words = [w for w in words if w]
    return Counter(words)


def read_file(path: Path) -> str:
    """テキストファイルを読み込む"""
    return path.read_text(encoding="utf-8")


def top_words(counter: Counter, n: int) -> list[tuple[str, int]]:
    """上位 N 件を頻度降順、同率なら辞書順で返す"""
    return sorted(counter.items(), key=lambda x: (-x[1], x[0]))[:n]


def main() -> int:
    parser = argparse.ArgumentParser(description="テキストファイルから単語頻度を集計して上位 N 件を表示する")
    parser.add_argument("file", type=Path, help="入力テキストファイル")
    parser.add_argument("-n", type=int, default=10, help="表示件数 (default: 10)")
    args = parser.parse_args()

    text = read_file(args.file)
    counter = count_words(text)
    results = top_words(counter, args.n)

    print(f"{'word':<30} {'count':>8}")
    print("-" * 38)
    for word, count in results:
        print(f"{word:<30} {count:>8}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
