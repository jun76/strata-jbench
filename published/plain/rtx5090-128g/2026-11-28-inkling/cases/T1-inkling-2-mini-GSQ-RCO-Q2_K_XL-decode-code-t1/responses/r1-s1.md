import argparse
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def count_words(text: str) -> Dict[str, int]:
    """テキストを単語に分割して出現頻度を返す。"""
    words = text.split()
    return dict(Counter(words))


def top_n_words(counter: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """出現頻度が高い上位 N 件を返す。"""
    return sorted(counter.items(), key=lambda item: (-item[1], item[0]))[:n]


def render_table(rows: List[Tuple[str, int]]) -> str:
    """表形式の文字列を生成する。"""
    lines = ["word\tcount"]
    lines.append("----\t-----")
    for word, count in rows:
        lines.append(f"{word}\t{count}")
    return "\n".join(lines)


def main() -> None:
    parser = argparse.ArgumentParser(description="テキストファイルの単語頻度を集計し上位 N 件を表示する。")
    parser.add_argument("path", type=Path, help="入力テキストファイル")
    parser.add_argument("-n", type=int, default=10, help="表示する上位件数")
    args = parser.parse_args()

    text = args.path.read_text(encoding="utf-8")
    counter = count_words(text)
    rows = top_n_words(counter, args.n)
    print(render_table(rows))


if __name__ == "__main__":
    main()
