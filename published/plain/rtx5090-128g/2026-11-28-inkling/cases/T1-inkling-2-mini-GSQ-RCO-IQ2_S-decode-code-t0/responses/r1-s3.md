以下は、テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で表示する Python コマンドラインツールです。

#!/usr/bin/env python3
"""テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する。"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """テキストを小文字化し、英数字の単語として抽出して頻度を集計する。"""
    words = re.findall(r"[a-zA-Z0-9]+", text.lower())
    return Counter(words)


def top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """頻度の高い順に上位 n 件を返す。"""
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """上位 N 件を表形式の文字列として返す。"""
    if not rows:
        return "単語が見つかりませんでした。"
    max_word_len = max(len(word) for word, _ in rows)
    lines = [f"{'単語':<{max_word_len}}  {'頻度':>5}"]
    lines.append("-" * (max_word_len + 7))
    for word, count in rows:
        lines.append(f"{word:<{max_word_len}}  {count:>5}")
    return "\n".join(lines)
