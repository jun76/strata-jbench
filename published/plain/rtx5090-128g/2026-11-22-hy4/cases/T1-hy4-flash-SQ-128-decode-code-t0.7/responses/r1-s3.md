"""単語出現頻度集計 CLI ツール.

テキストファイルを読み込み、単語の出現回数を集計して上位 N 件を表形式で表示します。
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Sequence


TOKEN_RE = re.compile(r"[A-Za-z0-9_]+|[ぁ-んァ-ン一-龯]+")


def tokenize(text: str) -> list[str]:
    """テキストを単語トークンに分割する.

    Args:
        text: 入力テキスト.

    Returns:
        小文字化されたトークンのリスト.

    Examples:
        >>> tokenize("Hello hello 世界 世界!")
        ['hello', 'hello', '世界', '世界']
    """
    return [m.group(0).lower() for m in TOKEN_RE.finditer(text)]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を集計する.

    Args:
        text: 入力テキスト.

    Returns:
        単語をキー、出現回数を値とする Counter.
    """
    return Counter(tokenize(text))


def read_text(path: Path, encoding: str = "utf-8") -> str:
    """テキストファイルを読み込む.

    Args:
        path: 読み込むファイルパス.
        encoding: 文字エンコーディング.

    Returns:
        ファイル全体の文字列.

    Raises:
        FileNotFoundError: ファイルが存在しない場合.
        UnicodeDecodeError: エンコーディングが不正な場合.
    """
    return path.read_text(encoding=encoding)
