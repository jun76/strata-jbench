"""テキストファイル内の単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。"""

_WORD_RE = re.compile(r"[A-Za-z0-9]+(?:'[A-Za-z0-9]+)?")


def extract_words(text: str) -> list[str]:
    """テキストから単語を抽出する。

    Args:
        text: 対象テキスト。

    Returns:
        小文字に正規化された単語のリスト。

    Examples:
        >>> extract_words("Hello, hello world!")
        ['hello', 'hello', 'world']
    """
    return [match.group(0).lower() for match in _WORD_RE.finditer(text)]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を集計する。

    Args:
        text: 対象テキスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(extract_words(text))
