#!/usr/bin/env python3
"""単語出現頻度集計 CLI ツール.

テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で表示する。
"""

from __future__ import annotations

import re

_WORD_RE = re.compile(r"[A-Za-z0-9_]+|[\u3040-\u30ff\u4e00-\u9fff]+")


def tokenize(text: str) -> list[str]:
    """テキストを単語トークンに分割する（英数字・日本語に対応）.

    Args:
        text: 分割対象のテキスト.

    Returns:
        小文字化された単語リスト.

    Examples:
        >>> tokenize("Hello hello 世界")
        ['hello', 'hello', '世界']
    """
    return [m.group(0).lower() for m in _WORD_RE.finditer(text)]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を集計する.

    Args:
        text: 集計対象のテキスト.

    Returns:
        単語をキー、出現回数を値とする Counter.
    """
    return Counter(tokenize(text))
