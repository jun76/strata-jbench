#!/usr/bin/env python3
"""単語出現頻度集計 CLI ツール.

テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で表示する。

Usage:
    python wordfreq.py sample.txt -n 5
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Sequence

WORD_RE = re.compile(r"[A-Za-z]+(?:'[A-Za-z]+)?")


def extract_words(text: str) -> list[str]:
    """テキストから単語のみを抽出して小文字化する.

    Args:
        text: 入力テキスト.

    Returns:
        抽出された単語のリスト（小文字）。
    """
    return [m.group(0).lower() for m in WORD_RE.finditer(text)]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を集計する.

    Args:
        text: 入力テキスト.

    Returns:
        単語 -> 出現回数 の Counter.
    """
    return Counter(extract_words(text))


def read_file(path: Path, encoding: str = "utf-8") -> str:
    """ファイルを読み込んで文字列を返す.

    Args:
        path: 読み込むファイルパス.
        encoding: 文字エンコーディング（デフォルト: utf-8）.

    Returns:
        ファイル内容.

    Raises:
        FileNotFoundError: ファイルが存在しない場合.
        UnicodeDecodeError: エンコーディングが不正な場合.
    """
    return path.read_text(encoding=encoding)
