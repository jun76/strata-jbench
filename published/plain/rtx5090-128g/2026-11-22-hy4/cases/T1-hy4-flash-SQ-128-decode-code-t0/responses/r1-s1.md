#!/usr/bin/env python3
"""テキストファイル内の単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Sequence


WORD_RE = re.compile(r"[A-Za-z]+(?:'[A-Za-z]+)?")


def extract_words(text: str) -> list[str]:
    """テキストから単語のみを抽出し、小文字に変換して返す。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    return [m.group(0).lower() for m in WORD_RE.finditer(text)]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を集計する。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(extract_words(text))


def format_table(counter: Counter[str], top_n: int) -> str:
    """集計結果を上位 N 件の表形式文字列に変換する。

    Args:
        counter: 単語頻度の Counter。
        top_n: 表示する上位件数。

    Returns:
        表形式の文字列（ヘッダー付き）。
    """
    rows = counter.most_common(top_n)
    word_width = max((len(w) for w, _ in rows), default=5)
    word_width = max(word_width, len("Word"))

    lines = [
        f"{'Word':<{word_width}} | Count",
        f"{'-' * word_width}-+------",
    ]
    lines.extend(f"{word:<{word_width}} | {count:>5}" for word, count in rows)
    return "\n".join(lines)
