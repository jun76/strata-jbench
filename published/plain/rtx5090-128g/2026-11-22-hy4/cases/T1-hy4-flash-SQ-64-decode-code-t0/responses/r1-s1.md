#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file.

Reads a UTF-8 text file, tokenizes words case-insensitively,
counts occurrences, and prints the top-N results as a table.
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Sequence


WORD_RE = re.compile(r"[A-Za-z0-9]+(?:'[A-Za-z0-9]+)?")


def tokenize(text: str) -> list[str]:
    """Extract lowercase word tokens from text.

    Args:
        text: Raw input text.

    Returns:
        List of lowercase word tokens (apostrophes preserved).
    """
    return [m.group(0).lower() for m in WORD_RE.finditer(text)]


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in the given text.

    Args:
        text: Raw input text.

    Returns:
        Counter mapping word -> frequency.
    """
    return Counter(tokenize(text))


def read_file(path: Path) -> str:
    """Read a UTF-8 text file.

    Args:
        path: Path to the input file.

    Returns:
        File contents as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        UnicodeDecodeError: If the file is not valid UTF-8.
    """
    return path.read_text(encoding="utf-8")
