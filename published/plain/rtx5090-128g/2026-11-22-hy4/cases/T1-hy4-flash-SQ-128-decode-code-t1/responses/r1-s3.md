#!/usr/bin/env python3
"""テキストファイル内の単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Sequence


WORD_RE = re.compile(r"\b[a-zA-Z]+(?:'[a-zA-Z]+)?\b")


def extract_words(text: str) -> list[str]:
    """テキストから単語を抽出し、小文字に正規化して返す。

    Args:
        text: 入力テキスト。

    Returns:
        正規化された単語のリスト。
    """
    return [match.group(0).lower() for match in WORD_RE.finditer(text)]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を集計する。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(extract_words(text))


def format_table(counter: Counter[str], top_n: int) -> str:
    """集計結果を上位 N 件の表形式文字列に変換する。

    Args:
        counter: 単語頻度の Counter。
        top_n: 表示する上位件数。

    Returns:
        表形式の文字列（ヘッダ付き）。
    """
    rows = counter.most_common(top_n)
    header = f"{'Rank':>4}  {'Word':<20} {'Count':>6}"

    body = "\n".join(
        f"{i:>4}  {word:<20} {count:>6}"
        for i, (word, count) in enumerate(rows, start=1)
