#!/usr/bin/env python3
"""テキストファイル内の単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Sequence


WORD_RE = re.compile(r"[A-Za-z]+(?:'[A-Za-z]+)?")


def extract_words(text: str) -> list[str]:
    """テキストから単語を抽出し、小文字に正規化して返す。

    Args:
        text: 入力テキスト。

    Returns:
        抽出された単語のリスト（小文字）。
    """
    return [m.group(0).lower() for m in WORD_RE.finditer(text)]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を集計する。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(extract_words(text))


def format_table(items: Sequence[tuple[str, int]], limit: int) -> str:
    """上位 N 件を表形式文字列に整形する。

    Args:
        items: (単語, 出現回数) のシーケンス（頻度降順）。
        limit: 表示件数。

    Returns:
        整形済みの表文字列。
    """
    header = f"{'単語':<20} {'回数':>6}"
    sep = "-" * len(header)
    lines = [header, sep]
    for word, count in items[:limit]:
        lines.append(f"{word:<20} {count:>6}")
    return "\n".join(lines)
