"""テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

Usage:
    python wordfreq.py sample.txt --top 10
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Sequence


WORD_RE = re.compile(r"[A-Za-z]+(?:['’-][A-Za-z]+)?")


def tokenize(text: str) -> list[str]:
    """テキストを小文字化し、単語トークンに分割する。

    Args:
        text: 元のテキスト文字列。

    Returns:
        小文字化された単語のリスト（記号・数字は除外）。

    Examples:
        >>> tokenize("Hello, hello! World's OK.")
        ['hello', 'hello', 'world's', 'ok']
    """
    return [m.group(0).lower() for m in WORD_RE.finditer(text)]


def count_words(tokens: Sequence[str]) -> Counter[str]:
    """トークン列から出現頻度を集計する。

    Args:
        tokens: 単語トークンのシーケンス。

    Returns:
        Counter[str]: 単語 -> 出現回数 のマッピング。
    """
    return Counter(tokens)
