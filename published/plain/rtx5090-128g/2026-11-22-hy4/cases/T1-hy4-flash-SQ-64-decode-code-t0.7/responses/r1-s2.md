#!/usr/bin/env python3
"""テキストファイルの単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

Usage:
    python wordfreq.py input.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


WORD_PATTERN = re.compile(r"[A-Za-z0-9]+(?:'[A-Za-z]+)?|[\u3040-\u30ff\u4e00-\u9fff]+")


def tokenize(text: str) -> List[str]:
    """テキストから単語トークンを抽出し、小文字化して返す。

    Args:
        text: 生テキスト文字列。

    Returns:
        トークンのリスト（英数字・日本語・ハイフン付き英語を含む）。
    """
    return [m.group(0).lower() for m in WORD_PATTERN.finditer(text)]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現回数を Counter で返す。

    Args:
        text: 生テキスト文字列。

    Returns:
        Counter[str]: 単語 -> 出現回数のマッピング。
    """
    return Counter(tokenize(text))


def read_file(path: Path) -> str:
    """指定されたパスのテキストファイルを UTF-8 で読み込む。

    Args:
        path: 読み込むファイルの Path。

    Returns:
        ファイル内容の文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        UnicodeDecodeError: UTF-8 としてデコードできない場合。
    """
    return path.read_text(encoding="utf-8")
