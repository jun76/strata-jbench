```python
"""Command-line tool to count word frequencies in a text file.

Usage example:
    python wordfreq.py sample.txt --top 10
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


WORD_PATTERN = re.compile(r"[A-Za-z]+(?:'[A-Za-z]+)?")


def tokenize(text: str) -> List[str]:
    """Extract words from text, lowercased.

    Args:
        text: Raw text content.

    Returns:
        List of lowercase word tokens.

    >>> tokenize("Hello, hello! World's world.")
    ['hello', 'hello', "world's", 'world']
    """
    return [m.group(0).lower() for m in WORD_PATTERN.finditer(text)]


def count_words(text: str) -> Counter[str]:
    """Count word occurrences in text.

    Args:
        text: Raw text content.

    Returns:
        Counter mapping word -> frequency.
    """
    return Counter(tokenize(text))


def read_file(path: Path) -> str:
    """Read text from a file using UTF-8 encoding.

    Args:
        path: Path to the text file.

    Returns:
        File contents as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
    """
    return path.read_text(encoding="utf-8")


def top_words(counter: Counter[str], top_n: int) -> List[Tuple[str, int]]:
    """Return the top-N most common words.

    Args:
        counter: Word frequency counter.
        top_n: Number of entries to return.

    Returns:
        List of (word, count) tuples sorted by descending count,
        then alphabetically for ties.
    """
    return sorted(
        counter.items(),
        key=lambda item: (-item[1], item[0]),
    )[:top_n]
