#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file and show top-N results."""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Sequence


WORD_RE = re.compile(r"[A-Za-z]+(?:'[A-Za-z]+)?")


def tokenize(text: str) -> list[str]:
    """Extract lowercase alphabetic tokens (including apostrophes) from text.

    Args:
        text: Raw input text.

    Returns:
        List of lowercase words.

    Example:
        >>> tokenize("Hello, hello world!")
        ['hello', 'hello', 'world']
    """
    return [m.group(0).lower() for m in WORD_RE.finditer(text)]


def count_words(tokens: Sequence[str]) -> Counter[str]:
    """Count word occurrences.

    Args:
        tokens: Sequence of words.

    Returns:
        Counter mapping word -> frequency.
    """
    return Counter(tokens)


def format_table(rows: list[tuple[str, int]]) -> str:
    """Format (word, count) pairs as a simple aligned text table.

    Args:
        rows: Pairs of word and frequency.

    Returns:
        Multi-line string with header and aligned columns.
    """
    header = "WORD        COUNT"
    lines = [header, "-" * len(header)]
    for word, count in rows:
        lines.append(f"{word:<10}  {count:>5}")
    return "\n".join(lines)


def read_and_count(path: Path) -> Counter[str]:
    """Read a UTF-8 text file and count its words.

    Args:
        path: Path to the text file.

    Returns:
        Counter of word frequencies.

    Raises:
        FileNotFoundError: If the file does not exist.
        UnicodeDecodeError: If the file is not valid UTF-8.
    """
    text = path.read_text(encoding="utf-8")
    return count_words(tokenize(text))
