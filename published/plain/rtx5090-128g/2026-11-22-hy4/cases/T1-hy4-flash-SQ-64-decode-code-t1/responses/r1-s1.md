#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file.

Reads a UTF-8 encoded text file, tokenizes words case-insensitively,
counts their occurrences, and prints the top-N results as a table.
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Sequence

WORD_RE = re.compile(r"\b[a-zA-Z]+(?:'[a-zA-Z]+)?\b")


def tokenize(text: str) -> list[str]:
    """Extract lowercase word tokens from text.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase words (including apostrophes like "don't").
    """
    return [m.group(0).lower() for m in WORD_RE.finditer(text)]


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        A ``Counter`` mapping word -> frequency.
    """
    return Counter(tokenize(text))
