#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file.

Reads a UTF-8 text file, tokenizes words case-insensitively,
counts occurrences, and displays the top-N results as a table.

Usage:
    python wordfreq.py input.txt --top 10
"""

import argparse
import re
from collections import Counter
from pathlib import Path


WORD_PATTERN = re.compile(r"[A-Za-z]+(?:'[A-Za-z]+)?")


def tokenize(text: str) -> list[str]:
    """Extract lowercase alphabetic words from text.

    Args:
        text: Raw input text.

    Returns:
        List of lowercase word tokens.

    Example:
        >>> tokenize("Hello, hello! Don't stop.")
        ['hello', 'hello', 'don't', 'stop']
    """
    return [m.group(0).lower() for m in WORD_PATTERN.finditer(text)]


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in the given text.

    Args:
        text: Raw input text.

    Returns:
        Counter mapping word -> frequency.
    """
    return Counter(tokenize(text))


def read_file(path: Path) -> str:
    """Read a UTF-8 encoded text file.

    Args:
        path: Path to the input file.

    Returns:
        File contents as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        UnicodeDecodeError: If the file is not valid UTF-8.
    """
    return path.read_text(encoding="utf-8")
