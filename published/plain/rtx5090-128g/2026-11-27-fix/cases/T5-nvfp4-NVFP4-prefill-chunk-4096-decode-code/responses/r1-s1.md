#!/usr/bin/env python3
"""
wordfreq.py — テキストファイル内の単語出現頻度を計算し、上位 N 件を表形式で表示する CLI ツール。

使い方:
    python wordfreq.py <input_file> [-n N] [-i] [--min-length MIN]

オプション:
    -n, --top N          上位 N 件を表示 (デフォルト: 10)
    -i, --ignore-case    大文字小文字を区別しない (デフォルト: 区別する)
    --min-length MIN     指定文字数以上の単語のみ集計 (デフォルト: 1)
    --sort-by {freq,alpha}  ソート順 (デフォルト: freq)

例:
    python wordfreq.py sample.txt -n 5 -i
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from pathlib import Path
from typing import List, Tuple


@dataclass
class WordCount:
    """単語とその出現回数を保持するデータクラス。"""
    word: str
    count: int


def tokenize(text: str, *, ignore_case: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト。
        ignore_case: True の場合、単語を小文字に変換する。
        min_length: 抽出する単語の最小文字数。

    Returns:
        単語のリスト。
    """
    # 英数字の連続を単語とみなす（日本語対応は別途必要）
    pattern = re.compile(r"[a-zA-Z0-9]+")
    tokens = pattern.findall(text)
    if ignore_case:
        tokens = [t.lower() for t in tokens]
    tokens = [t for t in tokens if len(t) >= min_length]
    return tokens
