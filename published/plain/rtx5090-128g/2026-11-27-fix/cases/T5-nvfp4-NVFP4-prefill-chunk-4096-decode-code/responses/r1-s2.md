Word frequency counter command-line tool.

This tool reads a text file, counts word occurrences, and displays the top N
most frequent words in a table format.

Usage:
    python word_freq.py <input_file> [-n TOP_N]

Example:
    python word_freq.py sample.txt -n 10

import argparse
import re
from collections import Counter

def tokenize(text: str) -> list[str]:
    """
    Tokenize text into lowercase words.

    Extracts words consisting of alphanumeric characters and underscores,
    converts them to lowercase, and returns them as a list.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase word tokens.
    """
    return re.findall(r'\b\w+\b', text.lower())


def count_words(text: str) -> Counter[str]:
    """
    Count word occurrences in the given text.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object mapping each word to its occurrence count.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n_words(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """
    Get the top N most common words from a word counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples sorted by count descending, then by
        word ascending for ties.
    """
    # Counter.most_common() already sorts by count descending.
    # To break ties by word ascending, we sort explicitly.
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
