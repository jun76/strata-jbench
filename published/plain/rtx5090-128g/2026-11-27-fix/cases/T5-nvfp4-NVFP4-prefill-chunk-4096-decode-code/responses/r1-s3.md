"""
Text word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words
in a tabular format.
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into words.

    Words are sequences of alphanumeric characters, lowercased.

    Args:
        text: Input text string.

    Returns:
        List of lowercase words.
    """
    return re.findall(r'[a-z0-9]+', text.lower())


def count_words(text: str) -> Counter:
    """
    Count word occurrences in the given text.

    Args:
        text: Input text string.

    Returns:
        Counter object with word frequencies.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to retrieve.

    Returns:
        List of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)
