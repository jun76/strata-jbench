def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。

    単語は英数字およびハイフンで構成される連続文字列として定義される。
    大文字小文字の区別はここで保持される（呼び出し側で小文字化する場合もある）。

    Args:
        text: 入力テキスト。

    Returns:
        単語のリスト。
    """
    # 英数字とハイフンを許可するパターン
    return re.findall(r"[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*", text)


def count_words(text: str, case_insensitive: bool = False) -> Counter[str]:
    """テキスト中の単語出現頻度をCounterで返す。

    Args:
        text: 入力テキスト。
        case_insensitive: Trueの場合、単語を小文字に変換してから集計する。

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    words = tokenize(text)
    if case_insensitive:
        words = [w.lower() for w in words]
    return Counter(words)
