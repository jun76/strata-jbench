"""wordfreq.py

テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で出力するコマンドラインツール。

Usage:
    python wordfreq.py <file> [-n TOP_N] [--case-sensitive] [--sort {count,alpha}]
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Iterable


def tokenize(text: str, *, case_sensitive: bool = False) -> list[str]:
    """テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字と小文字を区別する。
                        False の場合、すべて小文字に変換する。

    Returns:
        単語トークンのリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む連続文字列を単語とみなす
    tokens: list[str] = re.findall(r"[a-z0-9'-]+", text) if not case_sensitive else re.findall(r"[A-Za-z0-9'-]+", text)
    return tokens


def count_frequencies(text: str, *, case_sensitive: bool = False) -> Counter[str]:
    """テキスト中の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字の区別を行うか。

    Returns:
        単語 -> 出現回数の Counter。
    """
    tokens = tokenize(text, case_sensitive=case_sensitive)
    return Counter(tokens)
