def read_text(source: str) -> str:
    """
    ファイルパスまたは '-' (標準入力) からテキストを読み込む。

    Args:
        source: 読み込むファイルパス。'-' の場合は標準入力から読み込む。

    Returns:
        読み込んだテキストの文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み込み中にエラーが発生した場合。
    """
    if source == "-":
        return sys.stdin.read()
    else:
        with open(source, "r", encoding="utf-8") as f:
            return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> list[str]:
    """

    Args:
        text: 分割対象のテキスト。
        case_sensitive: True の場合、大文字と小文字を区別する。False の場合、すべて小文字に変換する。

    Returns:
        単語のリスト。
    """

        text = text.lower()

    # 簡易的なトークナイザー: 英数字のみを単語の一部として扱い、それ以外を区切りとする

    import re
    words = re.findall(r"[a-z0-9]+", text)
    return words
