#!/usr/bin/env python3
"""
単語出現頻度集計ツール

テキストファイルを読み込み、単語の出現頻度を上位N件まで表形式で表示する。
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト文字列。

    Returns:
        小文字化された単語のリスト。
    """
    # 英数字・ハイフン・アポストロフィを含むトークンを抽出
    return re.findall(r"[a-zA-Z0-9'\-]+", text.lower())


def count_words(text: str) -> Counter[str]:
    """
    テキスト中の単語出現回数をカウントする。

    Args:
        text: 入力テキスト文字列。

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    出現回数上位N件の単語と回数を取得する。

    同点の場合は辞書順（アルファベット順）でソートする。

    Args:
        counter: 単語と出現回数のCounter。
        n: 取得する件数。

    Returns:
        (単語, 出現回数) のタプルのリスト。上位N件。
    """
    # まず回数で降順、同点なら単語で昇順
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
