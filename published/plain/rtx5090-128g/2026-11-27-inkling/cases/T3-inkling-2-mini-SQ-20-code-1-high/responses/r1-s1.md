def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    if not intervals:
        return []
    iv = sorted(((s, e) for s, e in intervals), key=lambda x: (x[0], x[1]))
    merged = []
    cs, ce = iv[0]
    for s, e in iv[1:]:
        if s <= ce:
            if e > ce:
                ce = e
        else:
            merged.append([cs, ce])
            cs, ce = s, e
    merged.append([cs, ce])
    return merged
