def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    """
    Merge all overlapping or touching closed intervals and return them sorted by start.
    
    Args:
        intervals: A list of closed intervals [start, end]. The input may be unsorted
                   and may be empty. It will not be modified.
    
    Returns:
        A list of merged closed intervals [start, end] sorted by start.
    """
    if not intervals:
        return []

    # Sort a copy of the intervals by start time
    sorted_intervals = sorted(intervals, key=lambda x: x[0])

    merged: list[list[int]] = []
    for interval in sorted_intervals:
        if not merged:
            merged.append(list(interval))
        else:
            last = merged[-1]
            if interval[0] <= last[1]:
                # Overlapping or touching: merge them
                last[1] = max(last[1], interval[1])
            else:
                merged.append(list(interval))

    return merged
