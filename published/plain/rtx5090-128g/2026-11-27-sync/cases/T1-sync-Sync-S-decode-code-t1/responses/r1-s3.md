def load_text(filepath: str, encoding: str = "utf-8") -> str:
    """
    テキストファイルを読み込んでその内容を文字列として返す。

    Args:
        filepath: 読み込むテキストファイルのパス。
        encoding: ファイルのエンコーディング（デフォルト: utf-8）。

    Returns:
        ファイルの内容を示す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        UnicodeDecodeError: 指定したエンコーディングでデコードできない場合。
    """
    return Path(filepath).read_text(encoding=encoding)
