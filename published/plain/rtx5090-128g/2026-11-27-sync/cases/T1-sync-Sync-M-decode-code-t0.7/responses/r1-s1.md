and displays the top N words in a tabular format.

Usage:

Example:

import argparse

from collections import Counter

from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text string.
        case_sensitive: If True, words are case-sensitive; otherwise, lowercased.
        min_len: Minimum length of a word to be included.

    Returns:
        A list of tokens (words) extracted from the text.
    """
    if not case_sensitive:
        text = text.lower()

    # Split by whitespace and strip punctuation from edges
    words: List[str] = []
    for token in text.split():
        # Strip common punctuation from start and end
        cleaned = token.strip(".,;:!?\"'()[]{}-_/\\@#$%^&*~`<>")
        if len(cleaned) >= min_len:
            words.append(cleaned)

    return words


def count_words(text: str, case_sensitive: bool = False, min_len: int = 1) -> Counter[str]:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text string.
        case_sensitive: If True, words are case-sensitive; otherwise, lowercased.
        min_len: Minimum length of a word to be included.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    tokens = tokenize(text, case_sensitive=case_sensitive, min_len=min_len)
    return Counter(tokens)
