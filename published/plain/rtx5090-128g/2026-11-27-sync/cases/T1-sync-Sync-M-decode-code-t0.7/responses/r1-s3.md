def read_text_file(path: str) -> str:
    """Read the entire contents of a text file.

    Args:
        path: Path to the text file.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words, stripping punctuation.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words with punctuation removed.
    """
    import re

    # Split on whitespace, then strip common punctuation from each token
    tokens = text.split()
    words: List[str] = []
    for token in tokens:
        # Strip leading and trailing punctuation, convert to lowercase
        word = token.strip(".,!?;:'\"()[]{}<>@#$%^&*~`|/\\_+-=").lower()
        if word:
            words.append(word)
    return words


def count_words(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a counter.

    Args:
        counter: A Counter object mapping words to their frequencies.
        n: The number of top entries to return.

    Returns:
        A list of (word, count) tuples for the top N most common words.
    """
    return counter.most_common(n)
