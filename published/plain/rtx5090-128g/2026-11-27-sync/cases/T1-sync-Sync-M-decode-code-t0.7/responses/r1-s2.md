"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words in a table format.
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    Count word frequencies in the given text.

    Words are defined as sequences of alphanumeric characters and underscores.
    Matching is case-insensitive.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter mapping each lowercase word to its occurrence count.
    """
    words = re.findall(r"\b\w+\b", text.lower())
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.

    Args:
        counter: A Counter object containing word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count), sorted by count descending, then alphabetically.
    """
    # Sort by count descending, then by word ascending for ties
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
