#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を上位 N 件を表形式で表示する。

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive] [--min-length MIN_LEN]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか
        min_length: 単語の最小長

    Returns:
        単語のリスト
    """
    # 英数字とハイフンを含む単語を抽出
    words = re.findall(r'[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*', text)
    
    if not case_sensitive:
        words = [w.lower() for w in words]
    
    # 最小長でフィルタ
    words = [w for w in words if len(w) >= min_length]
    
    return words


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか
        min_length: 単語の最小長

    Returns:
        単語と出現回数の Counter
    """
    words = tokenize(text, case_sensitive, min_length)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 N 件の (word, count) を取得する。

    Args:
        counter: 単語出現頻度の Counter
        n: 上位件数

    Returns:
        (word, count) のリスト（頻度順）
    """
    return counter.most_common(n)
