コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を上位 N 件を表形式で表示する。

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive] [--min-length L]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか
        min_length: 単語の最小長

    Returns:
        単語のリスト
    """
    # 英数字とハイフンを含む単語を抽出
    words = re.findall(r'[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*', text)
    
    if not case_sensitive:
        words = [w.lower() for w in words]
    
    # 最小長でフィルタ
    words = [w for w in words if len(w) >= min_length]
    
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語と出現回数の Counter
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の単語と出現回数を取得する。

    Args:
        counter: 単語と出現回数の Counter
        n: 上位何件を取得するか

    Returns:
        (単語, 出現回数) のリスト
    """
    return counter.most_common(n)
