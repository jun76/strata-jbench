```python
"""Top word frequency command-line tool.

Reads a text file, counts word occurrences (case-insensitive,
punctuation stripped), and prints the top N words in a table.

Usage:
    python word_freq.py <file> [-n N] [--by-frequency]
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> list[str]:
    """Tokenize text into lowercase words by removing punctuation.

    Args:
        text: Input string to tokenize.

    Returns:
        List of lowercase words.
    """
    # Remove all non-word characters (keep letters, digits, underscores)
    # Convert to lowercase
    cleaned = re.sub(r'[^\w\s]', '', text).lower()
    return cleaned.split()


def count_words(text: str) -> Counter:
    """Count word frequencies in text.

    Args:
        text: Input string.

    Returns:
        Counter object with word frequencies.
    """
    words = tokenize(text)
    return Counter(words)


def top_words(counts: Counter, n: int) -> list[tuple[str, int]]:
    """Get top N most common words.

    Args:
        counts: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        List of (word, count) tuples, sorted by frequency descending,
        then alphabetically.
    """
    # Counter.most_common already sorts by count descending
    # For ties, sort alphabetically
    items = list(counts.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
