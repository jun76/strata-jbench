#!/usr/bin/env python3
"""Top word frequency counter CLI tool.

Reads text files, counts word frequencies, and displays the top N words in a table.
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words, stripping non-alphanumeric characters.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens.
    """
    import re
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_words(words: List[str]) -> Counter:
    """Count word frequencies.

    Args:
        words: A list of word tokens.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N words by frequency.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples sorted by frequency descending.
    """
    return counter.most_common(n)


def read_files(paths: List[str]) -> str:
    """Read and concatenate contents of multiple files.

    Args:
        paths: A list of file path strings.

    Returns:
        The concatenated text from all files.

    Raises:
        FileNotFoundError: If a file does not exist.
        PermissionError: If a file cannot be read.
        UnicodeDecodeError: If a file cannot be decoded.
    """
    contents = []
    for path_str in paths:
        path = Path(path_str)
        try:
            contents.append(path.read_text(encoding='utf-8'))
        except FileNotFoundError:
            raise FileNotFoundError(f"File not found: {path_str}")
        except PermissionError:
            raise PermissionError(f"Permission denied: {path_str}")
    return "\n".join(contents)
