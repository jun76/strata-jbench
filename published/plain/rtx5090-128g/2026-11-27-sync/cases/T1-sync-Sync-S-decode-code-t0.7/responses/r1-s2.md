使い方:
    python wordfreq.py <file> [-n N] [-w WIDTH] [--case-sensitive]

例:
    python wordfreq.py sample.txt -n 10

from __future__ import annotations

@dataclass(frozen=True)
class WordFrequency:
    """単語とその出現回数を保持する不変データクラス。

    Attributes:
        word: 単語（正規化済み）
        count: 出現回数
    """

    word: str
    count: int


def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """単語を正規化する。

    Args:
        word: 元の単語
        case_sensitive: True の場合大文字小文字を区別する

    Returns:
        正規化された単語（句読点除去、必要に応じて小文字化）
    """
    # 先頭・末尾の句読点を除去
    stripped = word.strip(".,;:!?\"'()[]{}<>-")
    if not case_sensitive:
        stripped = stripped.lower()
    return stripped


def tokenize(text: str, case_sensitive: bool = False) -> list[str]:
    """テキストを単語トークンに分割する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか

    Returns:
        正規化された単語のリスト（空文字列は除外）
    """
    tokens = text.split()
    result = []
    for token in tokens:
        normalized = normalize_word(token, case_sensitive=case_sensitive)
        if normalized:
            result.append(normalized)
    return result
