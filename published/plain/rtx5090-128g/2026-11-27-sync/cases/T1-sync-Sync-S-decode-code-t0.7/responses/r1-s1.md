from __future__ import annotations

# 単語抽出用正規表現: 英数字およびハイフン/アポストロフィを含む連続文字列
_WORD_PATTERN = re.compile(r"[A-Za-z0-9][A-Za-z0-9'\-]*")


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語のリストを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別して抽出する。
            False の場合、小文字化してから抽出する。

    Returns:
        出現順の単語リスト (空リストもあり得る)。
    """
    if not case_sensitive:
        text = text.lower()
    return _WORD_PATTERN.findall(text)


def count_words(words: List[str]) -> collections.Counter:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        collections.Counter インスタンス。
    """
    return collections.Counter(words)
