def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    if not intervals:
        return []
    
    # Create a sorted copy to avoid modifying the original input
    sorted_intervals = sorted(intervals, key=lambda x: x[0])
    
    merged = [sorted_intervals[0][:]]  # Copy of the first interval
    
    for current in sorted_intervals[1:]:
        last = merged[-1]
        # Check if current interval overlaps or touches with the last merged interval
        # For closed intervals, [1,3] and [4,6] are touching (3+1 == 4? No, but "touching" means they share a point or are adjacent with no gap)
        # Actually, for closed intervals, "touching" usually means they are adjacent like [1,2] and [3,4] don't touch, but [1,3] and [3,5] overlap at 3.
        # The problem says "overlapping or touching". Touching could mean adjacent like [1,2] and [2,3] (share endpoint) or [1,2] and [3,4] (gap of 1?). 
        # Typically, merging touching intervals means if last_end >= current_start - 1? No, standard merging is if last_end >= current_start (overlap/touch at point).
        # Let's stick to the standard definition: if last[1] >= current[0], they overlap or touch at a point.
        if last[1] >= current[0]:
            last[1] = max(last[1], current[1])
        else:
            merged.append(current[:])
    
    return merged
