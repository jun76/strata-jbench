def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    if not intervals:
        return []
    
    # Sort intervals by start value, without modifying the original input
    sorted_intervals = sorted(intervals, key=lambda x: x[0])
    
    merged = [sorted_intervals[0][:]]  # Start with the first interval
    
    for current in sorted_intervals[1:]:
        last = merged[-1]
        # If current interval overlaps or touches the last merged interval
        if current[0] <= last[1]:
            # Merge them by updating the end of the last merged interval
            last[1] = max(last[1], current[1])
        else:
            # No overlap, add the current interval as a new merged interval
            merged.append(current[:])
    
    return merged
