def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。単語は英数字+[']で構成される（例: don't は don と t に分かれるわけではないが、
    正規表現 \w+ で単語単位を抽出し、小文字化する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        小文字化された単語のリスト
    """
    # \w+ は [a-zA-Z0-9_] にマッチ。単語として扱う。
    words = re.findall(r'\w+', text)
    return [w.lower() for w in words]


def count_words(words: List[str]) -> Counter:
    """単語リストから出現頻度をカウントする。
    
    Args:
        words: 単語のリスト
        
    Returns:
        単語 -> 出現回数 の Counter
    """
    return Counter(words)

        (単語, 出現回数) のタプルのリスト、出現回数の降順

    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
