#!/usr/bin/env python3
"""
word_freq.py - A command-line tool to count word frequencies in text files.

Reads one or more text files, counts the occurrences of each word (case-insensitive),
and displays the top N most frequent words in a formatted table.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize the input text into lowercase words.
    
    Words are defined as sequences of alphanumeric characters and apostrophes
    (to handle contractions like "don't").
    
    Args:
        text: Input text string.
        
    Returns:
        A list of lowercase word tokens.
    """
    # Match words that may contain apostrophes (e.g., "don't", "it's")
    words = re.findall(r"[a-zA-Z0-9]+(?:'[a-zA-Z0-9]+)*", text.lower())
    return words


def count_words(text: str) -> Counter[str]:
    """
    Count the frequency of each word in the given text.
    
    Args:
        text: Input text string.
        
    Returns:
        A Counter object mapping each word to its frequency.
    """
    words = tokenize(text)
    return Counter(words)


def read_file(filepath: str) -> str:
    """
    Read the contents of a text file.
    
    Args:
        filepath: Path to the text file.
        
    Returns:
        The contents of the file as a string.
        
    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If an error occurs while reading the file.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()
