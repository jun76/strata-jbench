def extract_words(text: str) -> List[str]:
    """
    Extract words from text, treating them as case-insensitive sequences of
    letters and digits, ignoring punctuation and other non-word characters.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase words found in the text.
    """
    # Match sequences of word characters (letters, digits, underscores)
    # Convert to lowercase for case-insensitive counting
    return re.findall(r'\b[a-zA-Z0-9_]+\b', text.lower())


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word.
    
    Args:
        words: A list of words to count.
        
    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their frequencies.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.
        
    Returns:
        A list of tuples (word, count) sorted by frequency descending,
        then alphabetically for ties.
    """
    # Most common first; for ties, sort alphabetically
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
