def read_file(filepath: str) -> str:
    """Read and return the content of a text file.

    Args:
        filepath: Path to the input text file.

    Returns:
        The full content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If an error occurs while reading the file.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """Extract words from text, converting to lowercase and stripping punctuation.

    Words are defined as sequences of alphanumeric characters.
    All words are converted to lowercase for case-insensitive counting.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    words: List[str] = []
    current_word: List[str] = []
    for char in text:
        if char.isalnum():
            current_word.append(char.lower())
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    if current_word:
        words.append(''.join(current_word))
    return words


def count_words(words: List[str]) -> Counter:
    """Count the frequency of each word in the list.

    Args:
        words: A list of words (as strings).

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)
