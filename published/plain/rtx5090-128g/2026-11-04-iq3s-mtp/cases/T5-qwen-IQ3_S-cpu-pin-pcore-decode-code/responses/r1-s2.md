word_freq: テキストファイルの単語出現頻度を上位N件で表示するCLIツール。

使い方:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt -n 5 --case-sensitive
    python word_freq.py input.txt -n 10 --output json
"""

import argparse

import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出してリストとして返す。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: Trueの場合、大文字小文字を区別する。

    Returns:
        単語文字列のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字で構成される単語を抽出（ハイフン付き単語も許可）
    words = re.findall(r"[a-z0-9]+(?:-[a-z0-9]+)*", text, re.IGNORECASE if case_sensitive else re.IGNORECASE)
    return words


def count_words(words: List[str]) -> Counter[str]:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語文字列のリスト。

    Returns:
        単語をキー、出現回数を値とするCounterオブジェクト。
    """
    return Counter(words)
