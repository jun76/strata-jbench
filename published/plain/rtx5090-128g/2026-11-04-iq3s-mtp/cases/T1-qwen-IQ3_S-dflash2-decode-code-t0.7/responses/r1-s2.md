#!/usr/bin/env python3
"""
word_freq_counter.py

A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def count_word_frequencies(text: str) -> Counter:
    """
    Count word frequencies in the given text.
    
    Words are defined as sequences of alphanumeric characters.
    The counting is case-insensitive.
    
    Args:
        text: The input text to analyze.
        
    Returns:
        A Counter object mapping words to their frequencies.
    """
    # Find all words (sequences of alphanumeric characters)
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    # Convert to lowercase for case-insensitive counting
    words = [word.lower() for word in words]
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.
        
    Returns:
        A list of tuples (word, frequency) for the top N words.
    """
    return counter.most_common(n)
