#!/usr/bin/env python3
"""
wordfreq - Count word frequencies in text files and display top-N results.

This command-line tool reads one or more text files, tokenizes the content
into words (alphanumeric sequences), counts their frequencies, and displays
the top-N most frequent words in a formatted table.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Dict


def extract_words(text: str) -> List[str]:
    """
    Extract words from a given text string.
    
    Words are defined as sequences of alphanumeric characters (including
    underscores and apostrophes within words). The extraction is case-insensitive.
    
    Args:
        text: Input text string.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    # Match sequences of word characters: letters, digits, underscores, and apostrophes
    # This handles contractions like "don't" and words with numbers like "abc123"
    words = re.findall(r"[a-zA-Z0-9_']+", text)
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word in a list of words.
    
    Args:
        words: A list of lowercase words.
        
    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: Number of top entries to return.
        
    Returns:
        A list of (word, count) tuples, sorted by frequency descending.
        Ties are broken alphabetically.
    """
    # Counter.most_common() sorts by count descending, then by insertion order.
    # To ensure deterministic alphabetical tie-breaking:
    items = list(counter.items())
    # Sort by (-count, word) to get descending count, ascending word for ties
    sorted_items = sorted(items, key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
