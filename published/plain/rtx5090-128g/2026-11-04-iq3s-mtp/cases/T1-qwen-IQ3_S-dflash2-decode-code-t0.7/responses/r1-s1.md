#!/usr/bin/env python3
"""
テキストファイルを読み込み、単語の出現頻度を上位 N 件まで表示するツール。
"""

import argparse

def count_words(text: str) -> Counter[str]:
    """
    与えられたテキストから単語を抽出し、出現頻度をカウントする。
    
    Args:
        text (str): 解析対象のテキスト。
        
    Returns:
        Counter[str]: 単語をキー、出現回数を値とする Counter オブジェクト。
    """
    # 簡易的な単語分割: 空白、改行、タブなどで区切る
    # 必要に応じて正規表現でより厳密な定義を行うことも可能
    words = text.split()
    # 単語を小文字に統一してカウント (ケースインセンシティブなカウント)
    normalized_words = [word.lower() for word in words]
    return Counter(normalized_words)


def get_top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counter オブジェクトから最も出現頻度の高い上位 n 件の単語と回数を取り出す。
    
    Args:
        counter (Counter[str]): 単語の頻度カウンター。
        n (int): 取得したい上位件数。
        
    Returns:
        List[Tuple[str, int]]: (単語, 回数) のタプルのリスト。
    """
    return counter.most_common(n)
