def count_words(
    text: str,
    case_sensitive: bool = False,
    min_length: int = 1
) -> Counter[str]:
    """
    テキスト中の単語の出現頻度をカウントする。

    Args:
        text: 解析対象のテキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 最低限の単語長（1未満なら1とみなす）。

    Returns:
        単語をキー、出現回数を値とするCounterオブジェクト。
    """
    if min_length < 1:
        min_length = 1

    if not case_sensitive:
        text = text.lower()

    # 単語を抽出: 英数字とハイフンを含む文字列
    import re
    words: List[str] = re.findall(r'\b[a-z0-9][a-z0-9\-]*[a-z0-9]\b|\b[a-z0-9]\b', text, re.IGNORECASE)

    # 最低長でフィルタ
    filtered_words: List[str] = [w for w in words if len(w) >= min_length]

    counter: Counter[str] = Counter(filtered_words)
    return counter
