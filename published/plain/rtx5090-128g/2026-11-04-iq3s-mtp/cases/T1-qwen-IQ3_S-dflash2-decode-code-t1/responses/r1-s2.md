#!/usr/bin/env python3
"""テキストファイルを読み込み、単語出現頻度の上位N件を表形式で表示するCLIツール."""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path


def tokenize(text: str) -> list[str]:
    """テキストから単語のリストを返す.

    単語はアルファベット・数字・アポストロフィを含む連続文字列として抽出される。
    すべて小文字に正規化される。

    Args:
        text: 入力テキスト.

    Returns:
        小文字化された単語のリスト.

    Examples:
        >>> tokenize("Hello, world! HELLO.")
        ['hello', 'world', 'hello']
    """
    return re.findall(r"[a-zA-Z0-9']+", text.lower())


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を Counter として返す.

    Args:
        text: 入力テキスト.

    Returns:
        単語をキー、出現回数を値とする Counter.

    Examples:
        >>> c = count_words("apple banana apple")
        >>> c["apple"]
        2
    """
    return Counter(tokenize(text))
