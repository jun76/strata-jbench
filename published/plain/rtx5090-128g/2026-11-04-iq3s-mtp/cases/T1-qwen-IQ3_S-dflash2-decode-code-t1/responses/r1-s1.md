def read_text(filepath: str | Path) -> str:
    """
    Read the entire contents of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The file content as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    path = Path(filepath)
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, ignore_case: bool = False, strip_punctuation: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text string.
        ignore_case: If True, convert all words to lowercase.
        strip_punctuation: If True, remove non-alphanumeric characters from word boundaries.

    Returns:
        A list of tokens (words).
    """
    words = text.split()
    if strip_punctuation:
        # Remove leading/trailing punctuation and keep internal hyphens/apostrophes as needed
        words = [re.sub(r"^\W+|\W+$", "", w) for w in words]
        words = [w for w in words if w]  # Remove empty strings
    if ignore_case:
        words = [w.lower() for w in words]
    return words


def count_words(words: List[str]) -> Counter[str]:
    """
    Count the frequency of each word.

    Args:
        words: A list of word tokens.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)
