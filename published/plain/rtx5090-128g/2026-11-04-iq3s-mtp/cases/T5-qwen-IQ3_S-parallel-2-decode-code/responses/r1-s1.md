word_freq.py - Count word frequencies in text files and display top N results.

Usage:
    python word_freq.py <file> [--top N] [--case-sensitive]

def read_text_file(filepath: str) -> str:
    """Read and return the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        PermissionError: If the file cannot be read due to permissions.
        UnicodeDecodeError: If the file cannot be decoded as UTF-8.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """Tokenize text into words.

    Words are defined as sequences of alphanumeric characters.
    By default, case is folded to lowercase for counting.

    Args:
        text: The input text to tokenize.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.

    Returns:
        A list of word tokens.
    """
    if not case_sensitive:
        text = text.lower()
    words: List[str] = []
    current_word: List[str] = []
    for char in text:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))
    return words


def count_words(tokens: List[str]) -> Counter:
    """Count the frequency of each word in the token list.

    Args:
        tokens: A list of word tokens.

    Returns:
        A Counter object mapping words to their counts.
    """
    return Counter(tokens)
