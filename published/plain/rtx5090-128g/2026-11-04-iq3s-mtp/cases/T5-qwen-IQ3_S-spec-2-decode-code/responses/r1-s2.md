def read_text_file(filepath: str) -> str:
    """
    Read and return the contents of a text file.

    Args:
        filepath: Path to the input text file.

    Returns:
        The full text content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize_words(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    Tokenize text into words, stripping punctuation and optionally preserving case.

    Words are defined as sequences of alphabetic characters. Punctuation and
    digits are treated as separators.

    Args:
        text: The input text string.
        case_sensitive: If False, words are lowercased for counting.
        min_len: Minimum length of a word to include (default 1).

    Returns:
        A list of tokenized words.
    """
    # Extract sequences of alphabetic characters
    words = re.findall(r'[A-Za-z]+', text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return [w for w in words if len(w) >= min_len]

    Args:
        words: List of words to count.

    Returns:
        A Counter object with word counts.
    """
    return Counter(words)
