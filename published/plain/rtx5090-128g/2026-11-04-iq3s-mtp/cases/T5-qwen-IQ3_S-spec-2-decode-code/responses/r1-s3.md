#!/usr/bin/env python3
"""
word_freq: コマンドラインツールでテキストファイルを読み込み、単語出現頻度の上位 N 件を表形式で表示する。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    単語とは、Unicode のアルファベット・数字・アンダースコアからなる連続文字列と定義する。
    小文字に正規化する。
    """
    return re.findall(r'\w+', text.lower())


def count_words(words: List[str]) -> Counter[str]:
    """
    単語リストから出現頻度をカウントする。
    """
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 n 件を返す。
    頻度が同じ場合は辞書順で安定ソートする。
    """
    # Counter.most_common は頻度降順、同じ頻度の場合は挿入順（不定）なので、
    # 明示的に (頻度, 単語) でソートして安定性を保証する。
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(data: List[Tuple[str, int]]) -> str:
    """
    (単語, 頻度) のリストを表形式の文字列として整形する。
    """
    if not data:
        return "No words found."

    # 列幅を計算
    word_width = max(len(word) for word, _ in data)
    freq_width = max(len(str(freq)) for _, freq in data)

    # ヘッダー
    header = f"{'Word':<{word_width}} | {'Count':> {freq_width}}"
    separator = f"{'-' * word_width}-+-{'-' * freq_width}"

    lines = [header, separator]
    for word, freq in data:
        line = f"{word:<{word_width}} | {freq:>{freq_width}}"
        lines.append(line)

    return "\n".join(lines)
