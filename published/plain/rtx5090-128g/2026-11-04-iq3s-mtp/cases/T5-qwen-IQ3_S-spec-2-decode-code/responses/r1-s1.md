word_count.py

テキストファイルを読み込み、単語の出現頻度を計算して上位N件を表形式で出力するツール。

def read_text_file(file_path: str) -> str:
    """指定されたファイルパスからテキストを読み込み、内容を返す。

    Args:
        file_path: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        UnicodeDecodeError: ファイルがUTF-8でデコードできない場合。
    """
    with open(file_path, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。
    単語は英字のみで構成される連続文字列として定義され、大文字小文字は無視される。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    # 英字のみを含む単語を抽出し、小文字に変換
    words = re.findall(r"[a-zA-Z]+", text)
    return [word.lower() for word in words]

    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語とそのカウントを含む collections.Counter オブジェクト。
    """
    return Counter(words)
