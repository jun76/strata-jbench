#!/usr/bin/env python3
"""
word_freq: A command-line tool to count word frequencies in text files
and display the top N most frequent words in a table format.
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    Extract words from text, treating sequences of alphanumeric characters as words.
    
    Args:
        text: Input text string.
        
    Returns:
        List of lowercase words.
    """
    # Use regex to find sequences of alphanumeric characters (words)
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.
    
    Args:
        words: List of words.
        
    Returns:
        Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their frequencies.
    
    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to retrieve.
        
    Returns:
        List of tuples (word, frequency) sorted by frequency descending,
        then by word ascending for ties.
    """
    # Counter.most_common returns elements sorted by count descending
    # For ties, we need to sort alphabetically as secondary key
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
