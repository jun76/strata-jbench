#!/usr/bin/env python3
"""Text word frequency counter command-line tool.

Reads a text file, counts word occurrences (case-insensitive, alphanumeric-only words),
and displays the top N most frequent words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """Extract lowercase alphanumeric words from text.

    Args:
        text: Input text string.

    Returns:
        List of lowercase words containing only alphanumeric characters.
    """
    return re.findall(r'[a-z0-9]+', text.lower())


def count_words(words: List[str]) -> Counter:
    """Count occurrences of each word.

    Args:
        words: List of words.

    Returns:
        Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top n most common words.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples, sorted by count descending, then word ascending.
    """
    items = list(counter.items())
    # Sort by count descending, then by word ascending for ties
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(entries: List[Tuple[str, int]]) -> str:
    """Format entries as a table string.

    Args:
        entries: List of (word, count) tuples.

    Returns:
        Formatted table string with column headers.
    """
    if not entries:
        return "No words found."

    # Calculate column widths
    word_width = max(len("Word"), max(len(word) for word, _ in entries))
    count_width = max(len("Count"), max(len(str(count)) for _, count in entries))

    header = f"{'Word':<{word_width}}  {'Count':>{count_width}}"
    separator = f"{'-' * word_width}  {'-' * count_width}"

    lines = [header, separator]
    for word, count in entries:
        lines
