#!/usr/bin/env python3
"""
word_freq: テキストファイルを読み込み、単語の出現頻度を集計して
上位 N 件を表形式で表示するコマンドラインツール。

Usage:
    python word_freq.py -f input.txt -n 10
    python word_freq.py --file input.txt --top 20 --case-insensitive
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path


def read_text_file(file_path: Path) -> str:
    """
    テキストファイルを読み込み、その内容を文字列として返す。

    Args:
        file_path: 読み込むファイルのパス。

    Returns:
        ファイルのテキスト内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        PermissionError: ファイルの読み取り権限がない場合。
    """
    return file_path.read_text(encoding="utf-8")


def tokenize(text: str, case_insensitive: bool = False) -> list[str]:
    """
    テキストから単語を抽出する。

    単語は英数字とアンダースコアで構成される連続した文字列として
    正規表現 `\w+` で抽出される。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、すべての単語を小文字に変換して
                          大文字小文字を区別せずに集計する。

    Returns:
        単語のリスト。
    """
    tokens = re.findall(r"\w+", text)
    if case_insensitive:
        tokens = [t.lower() for t in tokens]
    return tokens
