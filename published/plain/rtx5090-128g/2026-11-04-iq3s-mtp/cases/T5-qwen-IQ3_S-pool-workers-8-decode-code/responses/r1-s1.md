#!/usr/bin/env python3
"""
word_freq.py - Count word frequencies from a text file and display top N results.

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive] [--min-length M]
"""

import argparse

import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: Input text string.
        case_sensitive: If False, convert all words to lowercase.
        min_length: Minimum length of a word to include.

    Returns:
        A list of tokens (words) extracted from the text.
    """
    # Extract words: sequences of alphanumeric characters (including underscores)
    words = re.findall(r'\b\w+\b', text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    words = [w for w in words if len(w) >= min_length]
    return words


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: Input text string.
        case_sensitive: If False, treat words case-insensitively.
        min_length: Minimum word length to include.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    tokens = tokenize(text, case_sensitive, min_length)
    return Counter(tokens)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a counter.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
        Ties are broken alphabetically.
    """
    # Sort by count descending, then by word ascending for stable output
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
