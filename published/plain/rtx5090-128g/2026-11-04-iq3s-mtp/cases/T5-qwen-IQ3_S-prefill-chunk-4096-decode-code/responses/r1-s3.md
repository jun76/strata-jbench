"""
wordfreq - Count word frequencies in a text file and display top N words.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words (alphanumeric sequences)."""
    return re.findall(r'[a-z0-9]+', text.lower())


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text."""
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words as (word, count) tuples."""
    return counter.most_common(n)


def format_table(top_words: List[Tuple[str, int]]) -> str:
    """Format top words as a table string."""
    if not top_words:
        return ""
    
    word_width = max(len(word) for word, _ in top_words)
    count_width = max(len(str(count)) for _, count in top_words)
    
    header_word = "Word".ljust(word_width)
    header_count = "Count".rjust(count_width)
    separator = "-" * (word_width + 1 + count_width)
    
    lines = [
        f"{header_word} {header_count}",
        separator,
    ]
    for word, count in top_words:
        lines.append(f"{word.ljust(word_width)} {str(count).rjust(count_width)}")
    
    return "\n".join(lines)


def read_file(filepath: str) -> str:
    """Read the content of a text file."""
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()
