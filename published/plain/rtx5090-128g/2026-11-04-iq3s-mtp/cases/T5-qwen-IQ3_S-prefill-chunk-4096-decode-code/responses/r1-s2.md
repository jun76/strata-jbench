#!/usr/bin/env python3
"""Text word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """Extract words from text, lowercased.

    Args:
        text: Input text string.

    Returns:
        List of lowercase words.
    """
    # Match sequences of alphanumeric characters (words)
    words = re.findall(r'[a-zA-Z0-9]+(?:\'[a-zA-Z]+)?', text.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """Count occurrences of each word.

    Args:
        words: List of words.

    Returns:
        Counter mapping words to their counts.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        List of (word, count) tuples, sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by count descending, then by word ascending
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(rows: List[Tuple[str, int]]) -> str:
    """Format rows as a table string.

    Args:
        rows: List of (word, count) tuples.

    Returns:
        Formatted table string.
    """
    if not rows:
        return ""

    # Calculate column widths
    max_word_len = max(len(word) for word, _ in rows)
    max_count_len = max(len(str(count)) for _, count in rows)

    word_col_width = max(max_word_len, len("Word"))
    count_col_width = max(max_count_len, len("Count"))
