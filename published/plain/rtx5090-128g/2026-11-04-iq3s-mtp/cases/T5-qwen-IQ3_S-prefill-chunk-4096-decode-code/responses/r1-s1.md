#!/usr/bin/env python3
"""
word_freq: A command-line tool to count word frequencies in text files
and display the top N most frequent words in a table format.
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    Extract words from text, ignoring case and punctuation.
    
    Args:
        text: Input text string.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    # Convert to lowercase and extract words using regex (alphanumeric sequences)
    return re.findall(r'\b[a-zA-Z0-9_]+\b', text.lower())


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word.
    
    Args:
        words: A list of words.
        
    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their counts.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top items to return.
        
    Returns:
        A list of tuples (word, count) sorted by count descending,
        then alphabetically for ties.
    """
    # Sort by count descending, then by word ascending for ties
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(rows: List[Tuple[str, int]]) -> str:
    """
    Format rows as a simple text table.
    
    Args:
        rows: A list of tuples (word, count).
        
    Returns:
        A formatted string representing the table.
    """
    if not rows:
        return "No words found."
    
    # Calculate column widths
    word_width = max(len(word) for word, _ in rows)
    count_width = max(len(str(count)) for _, count in rows)
    
    # Header
    header_word = "Word".ljust(word_width)
    header_count = "Count".rjust(count_width)
