def count_words(text: str) -> Counter:
    """
    テキストから単語を抽出し、出現頻度をカウントする。

    Args:
        text: カウント対象のテキスト文字列。

    Returns:
        Counter オブジェクト。キーは単語（小文字化、句読点除去後）、値は出現回数。
    """
    # 単語の正規化:
    # 1. 小文字に変換
    # 2. 英数字のみに分割（句読点や記号を除去）
    # 簡易的な正規表現の代わりに str.translate を使用して高速化
    import re
    # 単語境界を考慮した分割: 英数字のみを単語として抽出
    words = re.findall(r'\b[a-z0-9]+\b', text.lower())
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
