def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    英数字の文字列のみを単語とみなし、大文字は小文字に正規化する。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 小文字化された単語のリスト。
    """
    # 正規表現で英数字の連続部分列を抽出
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words (List[str]): 単語のリスト。

    Returns:
        Counter: 単語とその出現回数を記録した Counter オブジェクト。
    """
    return Counter(words)
