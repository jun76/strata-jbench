def tokenize(text: str, ignore_case: bool = False) -> List[str]:
    """Tokenize text into words.
    
    Args:
        text: Input text string.
        ignore_case: If True, convert all words to lowercase.
        
    Returns:
        List of word tokens (strings).
    """
    if ignore_case:
        text = text.lower()
    # Extract words: sequences of alphanumeric characters (including underscores)
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return words


def count_words(text: str, ignore_case: bool = False) -> Counter:
    """Count word frequencies in the given text.
    
    Args:
        text: Input text string.
        ignore_case: If True, treat words case-insensitively.
        
    Returns:
        Counter object mapping words to their counts.
    """
    words = tokenize(text, ignore_case=ignore_case)
    return Counter(words)

    Raises:
        FileNotFoundError: If the file does not exist.
        PermissionError: If the file cannot be read.
    """
    return filepath.read_text(encoding='utf-8', errors='replace')
