"""
Top Words Counter - A command-line tool to count word frequencies in a text file.

This tool reads a text file, counts the occurrence of each word, and displays
the top N most frequent words in a tabular format.

Usage:
    python top_words.py <input_file> [--top N] [--case-sensitive]
"""

@dataclass

    """Represents a word and its frequency count."""
    word: str

def read_file(filepath: str) -> str:
    """
    Read the content of a text file.

    Args:
        filepath: Path to the input text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Extract words from the given text.

    Words are defined as sequences of alphanumeric characters and underscores.
    Punctuation and other non-word characters are used as delimiters.

    Args:
        text: The input text string.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.

    Returns:
        A list of extracted words.
    """
    # Extract words using regex: sequences of word characters
    words = re.findall(r'\b\w+\b', text)

    if not case_sensitive:
        words = [word.lower() for word in words]

    return words

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)
