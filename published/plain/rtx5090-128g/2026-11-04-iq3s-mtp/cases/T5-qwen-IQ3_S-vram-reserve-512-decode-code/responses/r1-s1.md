import argparse

from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.
    
    Words are defined as sequences of alphanumeric characters.
    Case is normalized to lowercase.
    
    Args:
        text: Input text string.
        
    Returns:
        Counter object mapping words to their frequencies.
    """
    # Split on non-alphanumeric characters to extract words
    import re
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return Counter(word.lower() for word in words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.
    
    Args:
        counter: Counter object with word frequencies.
        n: Number of top items to return.
        
    Returns:
        List of tuples (word, count) sorted by count descending, then word ascending.
    """
    # Sort by count descending, then by word ascending for tie-breaking
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
