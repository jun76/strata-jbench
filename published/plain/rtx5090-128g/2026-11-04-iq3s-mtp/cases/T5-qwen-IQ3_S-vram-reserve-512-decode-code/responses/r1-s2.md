def read_text_file(file_path: Path) -> str:
    """
    指定されたファイルパスからテキストをUTF-8エンコーディングで読み込む。

    Args:
        file_path (Path): 読み込むテキストファイルのパス。

    Returns:
        str: ファイル内のテキスト内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        UnicodeDecodeError: ファイルがUTF-8として解釈できない場合。
    """
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {file_path}")
    except UnicodeDecodeError as e:
        raise UnicodeDecodeError(
            e.encoding, e.object, e.start, e.stop, e.reason
        )


def count_words(text: str) -> Counter:
    """
    テキストから単語を抽出し、出現頻度をカウントする。
    単語は英数字とハイフン、アポストロフィのみを含むものとし、
    大文字小文字は区別しない（すべて小文字に変換）。
    句読点や特殊文字は除去する。

    Args:
        text (str): 解析対象のテキスト。

    Returns:
        Counter: 単語をキー、出現回数を値とするCounterオブジェクト。
    """
    # 簡易的なトークナイザー: 英数字、ハイフン、アポストロフィの連続を単語とみなす
    # 例: "don't", "co-op", "test123"
    import re
    words = re.findall(r"[a-z0-9'\-]+", text.lower())
    return Counter(words)
