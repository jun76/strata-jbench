#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file and display top N results."""

import argparse

def read_file(filepath: str) -> str:
    """Read the contents of a text file.

    Args:
        filepath: Path to the text file to read.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If there is an error reading the file.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.

    Extracts sequences of alphanumeric characters (including underscores)
    and converts them to lowercase.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase word tokens.
    """
    return re.findall(r'[a-zA-Z0-9_]+', text.lower())


def count_words(text: str) -> Counter:
    """Count the frequency of each word in the given text.

    Args:
        text: The text to analyze.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples, each containing a word and its count,
        sorted by descending frequency.
    """
    return counter.most_common(n)
