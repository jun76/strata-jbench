"""
word_freq.py - Text file word frequency analyzer.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.

import argparse

def read_file(filepath: str) -> str:
    """Read and return the contents of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        PermissionError: If the file cannot be read.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """Extract words from text, lowercased and stripped of punctuation.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase words.
    """
    words: List[str] = []
    current: str = ""
    for char in text:
        if char.isalnum():
            current += char.lower()
        else:
            if current:
                words.append(current)
                current = ""
    if current:
        words.append(current)
    return words


def count_words(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)
