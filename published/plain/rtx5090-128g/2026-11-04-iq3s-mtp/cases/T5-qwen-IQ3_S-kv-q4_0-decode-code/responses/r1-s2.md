#!/usr/bin/env python3
"""
word_freq.py: テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で出力する CLI ツール。
"""

import argparse

def read_file(filepath: str) -> str:
    """
    指定されたファイルパスからテキストを読み込み、UTF-8 として返す。

    Args:
        filepath (str): 読み込むテキストファイルのパス。

    Returns:
        str: ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        UnicodeDecodeError: UTF-8 でデコードできない場合。
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    - 小文字化
    - 非英数文字を区切りとして分割
    - 空文字列を除外

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 単語のリスト。
    """
    # 英数字のみを抽出（日本語等非 ASCII はこの正規表現では除外されるため、必要に応じて拡張可能）
    # ここでは一般的な英単語ベースの解析を想定
    words = re.findall(r'[a-z0-9]+', text.lower())
    return [w for w in words if w]


def count_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words (List[str]): 単語のリスト。

    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)
