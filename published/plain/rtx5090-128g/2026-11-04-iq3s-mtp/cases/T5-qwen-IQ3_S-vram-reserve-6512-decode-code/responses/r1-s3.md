word_freq.py

テキストファイルを読み込み、単語の出現頻度を集計して
上位 N 件の単語とその回数を表形式で出力するコマンドラインツール。
"""

import argparse

from collections import Counter

def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出します。
    単語は英数字（およびアンダースコア）の連続と見なし、
    すべて小文字に変換します。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語のリスト（重複含む）
    """
    # \w+ は [a-zA-Z0-9_] に一致する
    # 日本語などの文字を除外したい場合は正規表現を調整する必要があります。
    # ここでは一般的な英単語抽出を想定しています。
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> List[Tuple[str, int]]:
    """
    単語リストから出現頻度を集計し、頻度が高い順にソートして返します。
    頻度が同じ場合は、単語の辞書順で昇順に並び替えます。
    
    Args:
        words: 単語のリスト
        
    Returns:
        (単語, 出現回数) のタプルのリスト
    """
    counter = Counter(words)
    # 頻度降順、同頻度なら単語の辞書順昇順
    sorted_items = sorted(
        counter.items(),
        key=lambda x: (-x[1], x[0])
    )
    return sorted_items
