Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N most frequent words
in a table format.
"""

import argparse

from collections import Counter

def count_words(text: str) -> Counter:
    """
    Count the frequency of each word in the given text.
    
    Words are extracted using regex pattern that matches sequences of alphanumeric
    characters and underscores. All words are converted to lowercase for case-insensitive
    counting.
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter object mapping words to their frequencies.
    """
    # Extract words: sequences of alphanumeric characters and underscores
    words = re.findall(r'\b\w+\b', text.lower())
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: Number of top items to return.
        
    Returns:
        A list of tuples (word, frequency) for the top N items.
    """
    return counter.most_common(n)
