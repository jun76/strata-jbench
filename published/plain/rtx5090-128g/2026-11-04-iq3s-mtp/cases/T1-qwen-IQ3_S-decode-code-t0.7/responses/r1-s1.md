import argparse

def normalize_words(text: str) -> List[str]:
    """
    テキストを正規化し、単語のリストを返す。
    
    Args:
        text: 入力テキスト文字列。
        
    Returns:
        正規化された単語のリスト（小文字化、非アルファベット文字の除去）。
    """
    # 英数字のみを含んだトークンを抽出し、小文字に変換
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text.lower())
    return words


def count_frequencies(words: List[str]) -> Counter[str]:
    """
    単語リストから頻度カウンターを作成する。
    
    Args:
        words: 単語のリスト。
        
    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    頻度カウンターから上位 n 件の (word, count) タプルを返す。
    
    Args:
        counter: 単語頻度の Counter。
        n: 取得する上位件数。
        
    Returns:
        (word, count) タプルのリスト。
    """
    return counter.most_common(n)
