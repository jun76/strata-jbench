"""
word_freq.py - コマンドラインツールでテキストファイルから単語出現頻度を集計し、
上位N件を表形式で表示する。

Usage:
    python word_freq.py <file> [-n TOP_N] [--case-sensitive]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するか否か

    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    # 日本語などの非ASCII文字も単語として扱う場合は [^\W\d_]+ に変更
    words = re.findall(r"[a-zA-Z0-9']+", text)
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数の Counter
    """
    return Counter(words)


def get_top_n_frequencies(
    freq_counter: Counter, n: int
) -> List[Tuple[str, int]]:
    """
    上位N件の単語と頻度を取得する。

    Args:
        freq_counter: 単語頻度のCounter
        n: 上位何件を取得するか

    Returns:
        (word, count) のタプルのリスト、頻度が高い順
    """
    return freq_counter.most_common(n)
