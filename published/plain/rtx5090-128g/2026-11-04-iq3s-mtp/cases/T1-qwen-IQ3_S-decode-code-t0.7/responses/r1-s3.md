コマンドラインツール: テキストファイルを読み込んで単語の出現頻度を

使い方:
    python wordfreq.py <file> [-n N] [--case-sensitive]

def count_words(text: str, case_sensitive: bool = False) -> List[Tuple[str, int]]:
    """
    テキストから単語を抽出し、出現頻度をカウントする。

    Args:
        text: 解析対象のテキスト文字列。
        case_sensitive: True の場合、大文字・小文字を区別する。

    Returns:
        (word, count) のタプルのリスト。頻度の高い順にソートされている。
        頻度が同じ場合はアルファベット順（辞書順）にソートされる。
    """
    if not case_sensitive:
        text = text.lower()

    # 単語を抽出: アルファベット、数字、ハイフン、アポストロフィを含む連続文字列
    # より一般的には、空白や句読点で分割するアプローチを使う
    import re
    # 英数字、ハイフン、アポストロフィを単語の一部として扱う
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    if not case_sensitive:
        # already lowercased above, but ensure consistency
        words = [w.lower() for w in words]
    else:
        words = [w for w in words if w]

    counter = Counter(words)
    # 頻度降順、同じ頻度なら語の辞書順昇順
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items
