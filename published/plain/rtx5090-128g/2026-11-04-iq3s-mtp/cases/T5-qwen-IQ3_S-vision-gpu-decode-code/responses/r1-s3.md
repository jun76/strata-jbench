def count_words(text: str) -> Counter[str]:
    """
    テキストから単語を抽出し、出現頻度をカウントします。
    単語は英数字と下線からなる文字列として扱われ、すべて小文字に変換されます。
    """
    # 正規表現で単語（英数字+下線）を抽出
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text.lower())
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counter オブジェクトから上位 N 件の (word, count) タプルリストを返します。
    頻度が同じ場合は単語の辞書順でソートされます。
    """
    # most_common は頻度順だが、同頻度時の順序は不定なため、
    # 明示的に (-count, word) でソートして安定性を確保
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
