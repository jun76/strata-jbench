def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    単語は英数字とハイフンで構成されるものとし、大文字小文字は無視する。

    Args:
        text: 入力テキスト

    Returns:
        小文字化された単語のリスト
    """
    # 英数字、ハイフン、アポストロフィを含む単語パターン
    # 例: "don't", "well-known", "abc123"
    pattern = r"[a-zA-Z0-9]+(?:[-'][a-zA-Z0-9]+)*"
    return [word.lower() for word in re.findall(pattern, text)]

    """

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    return Counter(words)
