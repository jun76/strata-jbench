"""
word_freq_counter.py

A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    Count the frequency of each word in the given text.

    Words are split by non-alphanumeric characters and lowercased.
    Empty strings are ignored.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    # Split text into words using regex: sequences of alphanumeric characters
    words = re.findall(r'[a-zA-Z0-9]+', text)
    # Lowercase all words for case-insensitive counting
    words = [word.lower() for word in words if word]
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their frequencies.

    Args:
        counter: A Counter object containing word frequencies.
        n: The number of top items to return.

    Returns:
        A list of tuples, each containing (word, count).
    """
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """
    Format the list of (word, count) tuples as a table string.

    Args:
        items: A list of tuples, each containing (word, count).

    Returns:
        A string representation of the table.
    """
    if not items:
        return ""

    # Determine column widths
    word_width = max(len(word) for word, _ in items)
    count_width = max(len(str(count)) for _, count in items)

    # Header
    header = f"{'Word':<{word_width}} | {'Count':>{count_width}}"
    separator = f"{'-' * word_width} | {'-' * count_width}"

    # Rows
    rows = [f"{word:<{word_width}} | {count:>{count_width}}" for word, count in items]

    return "\n".join([header, separator]
