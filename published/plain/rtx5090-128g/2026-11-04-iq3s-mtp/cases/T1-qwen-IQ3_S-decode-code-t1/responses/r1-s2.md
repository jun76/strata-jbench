import argparse

import re
from collections import Counter
from typing import List, Tuple


def read_text(filepath: str) -> str:
    """指定されたファイルからテキストを読み込む。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。

    単語は英数字（[a-zA-Z0-9]）の連続で定義され、全て小文字に変換される。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    words = re.findall(r"[a-zA-Z0-9]+", text)
    return [word.lower() for word in words]


def count_frequencies(words: List[str]) -> Counter:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語とその出現回数をキーと値とする Counter オブジェクト。
    """
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """最もよく出現する上位 n 件の単語とその頻度を返す。

    Args:

        n: 取得する上位件数。

    Returns:
        (単語, 頻度) のタプルのリスト。頻度が高い順に並んでいる。
    """
    return counter.most_common(n)
