"""
text_freq.py - 単語出現頻度を集計し、上位N件を表形式で表示するコマンドラインツール。

使用方法:
    python text_freq.py <input_file> [-n N] [--case-sensitive]

import argparse

from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語トークンに分割する。

    単語はアルファベットと数字で構成される連続した文字列として抽出される。
    大文字・小文字の区別は case_sensitive フラグで制御する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字・小文字を区別する場合 True

    Returns:
        単語トークンのリスト
    """
    import re
    # アルファベットと数字で構成される単語を抽出
    pattern = r'[a-zA-Z0-9]+'
    words = re.findall(pattern, text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_frequencies(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語トークンのリスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    return Counter(words)


def top_n(frequency_counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度が高い上位 n 件の (word, count) タプルを返す。

    Args:
        frequency_counter: 単語と出現回数の Counter
        n: 返す上位件数

    Returns:
        上位 n 件の (word, count) のリスト
    """
    return frequency_counter.most_common(n)
