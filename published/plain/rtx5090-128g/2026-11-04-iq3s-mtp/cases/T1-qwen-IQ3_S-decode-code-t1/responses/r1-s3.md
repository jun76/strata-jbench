#!/usr/bin/env python3
"""コマンドラインツール：テキストファイルを読み込んで単語の出現頻度を上位 N 件をテーブル表示する."""

from __future__ import annotations

import argparse
import re

from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """指定したテキストから単語トークンを抽出する.

    単語は英数字 (a-zA-Z0-9) に加えてアンダースコアで構成されるものとし、
    すべて小文字に変換する。それ以外の文字は区切りとして扱う。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト
    """
    tokens = re.findall(r'[a-zA-Z0-9_]+', text)
    return [t.lower() for t in tokens]


def count_words(text: str) -> Counter[str]:
    """テキスト中の単語出現頻度を Counter で返す.

    Args:
        text: 入力テキスト

    Returns:
        単語 -> 出現回数の Counter
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Counter から頻度上位 n 件の (単語, 頻度) のリストを返す.

    同頻度の場合は辞書順（アルファベット順）で並び替える。

    Args:
        counter: 単語 -> 出現回数の Counter
        n: 取得する上位件数

    Returns:
        (単語, 頻度) のタプルのリスト
    """
    # 頻度降順、同頻度ならアルファベット昇順
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
