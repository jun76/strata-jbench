def tokenize(text: str) -> List[str]:
    """
    テキストを単語（アルファベット、数字、一部の記号を含む連続文字列）に分割する。
    大文字小文字は区別せず、小文字に変換する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        小文字化された単語のリスト

    # 正規表現で単語境界を定義: 英数字とハイフン、アポストロフィを含む

    words = re.findall(r"[a-z0-9]+(?:['-][a-z0-9]+)*", text.lower())
    return words
