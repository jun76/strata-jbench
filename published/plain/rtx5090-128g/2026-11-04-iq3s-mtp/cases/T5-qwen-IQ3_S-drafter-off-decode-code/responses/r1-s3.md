#!/usr/bin/env python3
"""
word_count.py - Count word frequencies in a text file and display top N.

This command-line tool reads a text file, counts the occurrences of each word
(case-insensitive), and prints the top N most frequent words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def read_file(filepath: str) -> str:
    """
    Read the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Words are extracted using a regex that matches sequences of alphanumeric
    characters and underscores. Words are converted to lowercase for case-insensitive
    counting.

    Args:
        text: The input text string.

    Returns:
        A Counter object with words as keys and their counts as values.
    """
    # Match words: sequences of word characters (letters, digits, underscore)
    words = re.findall(r'\b\w+\b', text)
    # Convert to lowercase for case-insensitive counting
    words_lower = [word.lower() for word in words]
    return Counter(words_lower)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their counts.

    Args:
        counter: A Counter object with word counts.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) sorted by count descending, then by word ascending.
    """
    # most_common returns a list of (element, count) tuples sorted by count descending
    # For ties, we want alphabetical order, so we sort manually
    items = list(counter.items())
    # Sort by count descending, then by word ascending
    items.sort(key=lambda x: (-x[1], x[0]))
