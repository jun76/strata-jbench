"""
word_freq_counter.py

A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    Count the frequency of each word in the given text.

    Words are converted to lowercase and split using a regex that matches
    sequences of alphanumeric characters.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    # Extract words using regex: sequences of alphanumeric characters
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text.lower())
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top items to return.

    Returns:
        A list of tuples, each containing a word and its count,
        sorted by frequency descending, then by word ascending for ties.
    """
    # Get most common N words; for ties, Counter.most_common uses insertion order,
    # but we want deterministic tie-breaking by word alphabetically.
    # most_common already returns items sorted by count desc, then by insertion order.
    # To ensure deterministic tie-breaking by word, we sort explicitly.
    items = list(counter.items())
    # Sort by count descending, then by word ascending
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
