#!/usr/bin/env python3
"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度を計算して
上位 N 件を表形式で出力するコマンドラインツール。

Usage:
    python word_freq.py <input_file> [--top N] [--case-insensitive]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_insensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text (str): 入力テキスト。
        case_insensitive (bool): 大文字小文字を区別しない場合は True。

    Returns:
        List[str]: 抽出された単語のリスト。
    """
    if case_insensitive:
        text = text.lower()
    # 英数字とハイフン・アポストロフィを含む単語を抽出
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words (List[str]): 単語のリスト。

    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の単語と出現回数を取り出す。

    Args:
        counter (Counter): 単語頻度カウンター。
        n (int): 上位何件を取り出すか。

    Returns:
        List[Tuple[str, int]]: (単語, 出現回数) のリスト。
    """
    return counter.most_common(n)
