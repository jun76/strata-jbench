#!/usr/bin/env python3
"""
word_counter.py - Count word frequencies in a text file and display top N entries.

Usage:
    python word_counter.py <input_file> [--top N] [--case-sensitive]

Example:
    python word_counter.py sample.txt --top 10
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Extract words from the given text.
    
    Args:
        text: Input text string.
        case_sensitive: If False, words are converted to lowercase.
        
    Returns:
        A list of extracted words.
    """
    # Use regex to find word-like sequences (letters, digits, underscores)
    words = re.findall(r'\b\w+\b', text)
    if not case_sensitive:
        words = [word.lower() for word in words]
    return words


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word.
    
    Args:
        words: List of words to count.
        
    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words with their counts.
    
    Args:
        counter: Counter object with word frequencies.
        n: Number of top entries to return.
        
    Returns:
        A list of tuples (word, count), sorted by count descending, then alphabetically.
    """
    # most_common returns elements in descending order by count.
    # For ties, we want alphabetical order. Counter.most_common does not guarantee
    # alphabetical order for ties, so we sort explicitly.
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
