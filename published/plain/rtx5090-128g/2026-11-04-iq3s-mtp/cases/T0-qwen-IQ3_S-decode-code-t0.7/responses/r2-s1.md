以下は、テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で出力するコマンドラインツールです。

```python
#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive]
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字・小文字を区別するかどうか。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフンを含む単語を抽出（日本語は未対応）
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    return words


def count_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)


def get_top_n(frequencies: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の単語と頻度を取得する。

    Args:
        frequencies: 単語と頻度の Counter。
        n: 取得する件数。

    Returns:
        (単語, 頻度) のタプルのリスト（頻度降順）。
    """
    return frequencies.most_common(n)

    Args:
        entries: (単語, 頻度) のタプルのリスト。

    Returns:
        フォーマット化された表文字列。
    """

        return "単語が見つかりません。"
