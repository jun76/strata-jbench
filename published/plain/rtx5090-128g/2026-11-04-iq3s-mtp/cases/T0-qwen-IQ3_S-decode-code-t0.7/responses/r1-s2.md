word_freq.py - コマンドラインツールでテキストファイルから単語出現頻度を集計し、

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキストから単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、
                        すべて小文字に変換してからカウントする。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    if not case_sensitive:
        text = text.lower()
    # 単語の定義: アルファベット、数字、ハイフン、アポストロフィを含む連続文字列
    words = re.findall(r"[a-zA-Z0-9'-]+", text)
    return Counter(words)
