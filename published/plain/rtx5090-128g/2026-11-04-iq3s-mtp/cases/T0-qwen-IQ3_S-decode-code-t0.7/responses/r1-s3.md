import argparse

from collections import Counter
from typing import List, Tuple, TextIO, Optional


def tokenize(text: str, ignore_case: bool = True) -> List[str]:
    """
    テキストから単語を抽出する。
    
    Args:
        text (str): 入力テキスト。
        ignore_case (bool): 大文字小文字を区別しない場合は True。
        
    Returns:
        List[str]: 抽出された単語のリスト。
    """
    # 英数字とハイフン、アポストロフィを含む文字列を単語として抽出
    # 例: "don't", "co-op", "hello"
    pattern = r"[a-zA-Z0-9'’-]+"
    words = re.findall(pattern, text)
    
    if ignore_case:
        words = [word.lower() for word in words]
        
    return words


def count_words(text: str, ignore_case: bool = True) -> Counter:
    """
    テキスト内の単語の出現頻度をカウントする。
    
    Args:
        text (str): 入力テキスト。
        ignore_case (bool): 大文字小文字を区別しない場合は True。
        
    Returns:
        Counter: 単語とその出現回数を格納した Counter オブジェクト。
    """
    words = tokenize(text, ignore_case)
    return Counter(words)
