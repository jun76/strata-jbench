def load_text(filepath: str) -> str:
    """テキストファイルを読み込んで文字列を返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルを読み込めない場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語（アルファベット・数字・ハイフンを含むトークン）を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に正規化する。

    Returns:
        単語のリスト。
    """
    # 単語としてアルファベット、数字、ハイフンを抽出（例: don't は don と t に分割される）
    # より一般的な単語抽出: 英数字とハイフンを含む連続文字列
    pattern = r"[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*"
    tokens = re.findall(pattern, text)
    if not case_sensitive:
        tokens = [t.lower() for t in tokens]
    return tokens


def count_words(tokens: List[str]) -> Counter:
    """トークンリストから単語出現頻度をカウントする。

    Args:
        tokens: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(tokens)
