#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントして上位 N 件を表形式で表示するコマンドラインツール。

def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    Args:
        text (str): 入力テキスト
        
    Returns:
        List[str]: 正規化された単語リスト（小文字、句読点除去）
    """
    # 単語を抽出: 英数字とハイフン、アポストロフィを許可し、他を空白として扱う
    # 簡易的な正規表現の代わりに str.split と手動クリーンアップを使用（依存関係なし）
    words = []
    for word in text.split():
        # 句読点や特殊文字を除去
        cleaned = word.strip('.,!?;:"\'()[]{}<>-—–').lower()
        if cleaned:
            words.append(cleaned)
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから頻度カウンタを作成する。
    
    Args:
        words (List[str]): 単語リスト
        
    Returns:
        Counter: 単語とその出現回数のカウント
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    カウンタから上位 N 件の単語と頻度を取得する。
    
    Args:
        counter (Counter): 単語頻度カウンタ
        n (int): 取得する件数
        
    Returns:
        List[Tuple[str, int]]: [(word, count), ...] のリスト
    """
    return counter.most_common(n)
