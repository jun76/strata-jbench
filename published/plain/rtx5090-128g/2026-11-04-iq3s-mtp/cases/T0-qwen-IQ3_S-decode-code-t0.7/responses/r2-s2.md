def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、すべて小文字に変換する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()

    # 単語を抽出：アルファベット、数字、ハイフン、アポストロフィを含むトークン
    # 簡易的なトークナイザー：空白および句読点で分割し、非英数字を除去
    import re
    # 単語パターン：英数字、ハイフン、アポストロフィを含む連続文字列
    pattern = r"[a-zA-Z0-9'\-]+"
    tokens = re.findall(pattern, text)
    return tokens
