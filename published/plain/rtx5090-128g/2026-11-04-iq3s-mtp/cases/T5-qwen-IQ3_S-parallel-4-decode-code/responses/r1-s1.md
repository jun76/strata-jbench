Options:

import argparse
import re

from pathlib import Path
from typing import List, Tuple


def read_file(filepath: str) -> str:
    """
    Read the contents of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The file contents as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    path = Path(filepath)
    if not path.exists():
        raise FileNotFoundError(f"File not found: {filepath}")
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, ignore_case: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: Input text string.
        ignore_case: If True, convert all words to lowercase.

    Returns:
        A list of word tokens.
    """
    # Extract words using regex (alphanumeric characters and underscores)
    tokens = re.findall(r"\w+", text)
    if ignore_case:
        tokens = [token.lower() for token in tokens]
    return tokens


def count_frequencies(text: str, ignore_case: bool = False) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: Input text string.
        ignore_case: If True, treat words case-insensitively.

    Returns:
        A Counter object with word frequencies.
    """
    tokens = tokenize(text, ignore_case)
    return Counter(tokens)
