#!/usr/bin/env python3
"""コマンドラインツール: テキストファイルの単語出現頻度を集計し、上位 N 件を表形式で表示する。"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストから単語を抽出する。

    単語は英数字文字列 (\w+) として抽出され、すべて小文字に変換される。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト
    """
    return [word.lower() for word in re.findall(r'\w+', text)]


def count_words(text: str) -> Counter:
    """テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    return Counter(tokenize(text))


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """出現頻度上位 N 件の単語とカウントを返す。

    同順位の場合は辞書順で並べる。

    Args:
        counter: 単語頻度の Counter
        n: 取得する上位件数

    Returns:
        (単語, 出現回数) のタプルのリスト
    """
    items = list(counter.items())
    # 出現回数で降順、同順位は単語で昇順
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
