word_freq.py - コマンドラインツール：テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する。

使い方:
    python word_freq.py <input_file> [--top N] [--case-insensitive] [--min-length LEN]

例:
    python word_freq.py sample.txt --top 10
    python word_freq.py sample.txt -t 5 -ci -ml 3
"""

def parse_arguments(argv: Optional[List[str]] = None) -> argparse.Namespace:
    """コマンドライン引数を解析する。

        argv: 引数のリスト。None の場合は sys.argv を使用。

        解析された引数を含む Namespace オブジェクト。
    """
    parser = argparse.ArgumentParser(

        formatter_class=argparse.RawDescriptionHelpFormatter,
    )
    parser.add_argument(
        "input_file",
        type=str,
        help="読み込むテキストファイルのパス",
    )
    parser.add_argument(
        "-t",
        "--top",
        type=int,
        default=10,
        help="表示する上位件数（デフォルト: 10）",
    )
    parser.add_argument(
        "-ci",
        "--case-insensitive",
        action="store_true",
        help="大文字小文字を区別しない（デフォルト: 区別する）",
    )
    parser.add_argument(
        "-ml",
        "--min-length",
        type=int,
        default=1,
        help="集計対象とする単語の最小文字数（デフォルト: 1）",
    )

    return parser.parse_args(argv)
