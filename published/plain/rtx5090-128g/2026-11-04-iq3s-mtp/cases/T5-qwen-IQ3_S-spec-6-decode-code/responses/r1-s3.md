word_freq_counter.py

A command-line tool that reads a text file, counts the frequency of each word,
and displays the top N most frequent words in a table format.

Usage:
    python word_freq_counter.py <input_file> [-n N] [--case-sensitive] [--min-length LENGTH]

Examples:
    python word_freq_counter.py sample.txt -n 10
    python word_freq_counter.py sample.txt --case-sensitive --min-length 3
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words, optionally preserving case and filtering by minimum length.

    Args:
        text: The input text to tokenize.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.
        min_length: Minimum word length to include (default 1).

    Returns:
        A list of word tokens.
    """
    if not case_sensitive:
        text = text.lower()
    # Match sequences of word characters (letters, digits, underscores)
    words = re.findall(r'\w+', text)
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    return words


def count_word_frequencies(tokens: List[str]) -> Counter:
    """
    Count the frequency of each word in the token list.

    Args:
        tokens: A list of word tokens.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(tokens)
