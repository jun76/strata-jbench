"""
word_freq.py - テキストファイルから単語出現頻度を上位N件で表示するツール.
"""

def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する.
    
    - 英数字のみを単語とみなす (正規表現 \\w+ を使用).
    - すべて小文字に変換する.
    
    Args:
        text: 入力テキスト.
        
    Returns:
        抽出された単語のリスト.
    """
    # \w+ は [a-zA-Z0-9_] に一致する. 日本語などのマルチバイト文字は除外される.
    words = re.findall(r'\w+', text.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする.
    
    Args:
        words: 単語のリスト.
        
    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト.
    """
    return Counter(words)
