word_freq - Count word frequencies in a text file and display the top N results.

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Optional


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text to analyze.
        case_sensitive: If False, words are treated case-insensitively (lowercased).

    Returns:
        A Counter object mapping words to their frequencies.
    """
    if not case_sensitive:
        text = text.lower()
    # Extract words: sequences of alphanumeric characters
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return Counter(words)


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object containing word frequencies.
        n: The number of top words to retrieve.

    Returns:
        A list of (word, frequency) tuples, sorted by frequency descending,
        then by word ascending for ties.
    """
    # Most common first, then alphabetical for ties
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
