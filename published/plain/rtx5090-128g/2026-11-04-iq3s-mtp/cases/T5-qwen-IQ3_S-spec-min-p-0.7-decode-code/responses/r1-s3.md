#!/usr/bin/env python3
"""
wordfreq - Count word frequencies in text files and display top-N results.

This command-line tool reads one or more text files, tokenizes the content
into words (alphanumeric sequences), counts their frequencies, and displays
the top-N most frequent words in a formatted table.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Dict


def extract_words(text: str) -> List[str]:
    """
    Extract words from a given text string.
    
    Words are defined as sequences of alphanumeric characters (letters and digits).
    The extraction is case-insensitive: all words are converted to lowercase.
    
    Args:
        text: The input text string to tokenize.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    return [word.lower() for word in re.findall(r'[a-zA-Z0-9]+', text)]


def count_words(words: List[str]) -> Dict[str, int]:
    """
    Count the frequency of each word in a list of words.
    
    Args:
        words: A list of words (assumed to be already lowercased).
        
    Returns:
        A dictionary mapping each word to its frequency count.
    """
    return dict(Counter(words))


def get_top_n(freq_counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    Get the top-N most frequent words sorted by frequency (descending),
    then alphabetically for ties.
    
    Args:
        freq_counts: A dictionary mapping words to their frequency counts.
        n: The number of top words to return.
        
    Returns:
        A list of tuples (word, count), sorted by count descending,
        then by word ascending for ties.
    """
    sorted_items = sorted(
        freq_counts.items(),
        key=lambda item: (-item[1], item[0])
    )
    return sorted_items[:n]
