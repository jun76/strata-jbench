#!/usr/bin/env python3

テキストファイルを読み込み、単語の出現頻度を集計して上位N件を表形式で表示するコマンドラインツール。

import argparse

def read_file(filepath: str) -> str:
    """
    指定されたファイルパスからテキストを読み込む。

    Args:
        filepath: 読み込むファイルのパス

    Returns:
        ファイルの内容（文字列）

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        IOError: ファイルの読み込みに失敗した場合
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。英単語のみ抽出し、小文字に変換する。

    Args:
        text: 分割するテキスト

    Returns:
        単語のリスト（小文字化済み）
    """
    # 英数字の単語のみ抽出（最小長1）
    words = re.findall(r"[a-zA-Z]+", text)
    return [w.lower() for w in words]


def count_words(tokens: List[str]) -> Counter[str]:
    """
    単語の出現頻度をカウントする。

    Args:
        tokens: 単語のリスト

    Returns:
        単語をキー、出現回数を値とするCounter
    """
    return Counter(tokens)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    出現頻度上位N件の単語と回数を返す。同数値の場合は辞書順でソート。

    Args:
        counter: 単語頻度のCounter
        n: 上位何件を取得するか

    Returns:
        (単語, 頻度) のタプルのリスト（頻度降順、同値時は単語昇順）
    """
    # most_commonでは同数値時の順序が不定なので、明示的にソート
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
