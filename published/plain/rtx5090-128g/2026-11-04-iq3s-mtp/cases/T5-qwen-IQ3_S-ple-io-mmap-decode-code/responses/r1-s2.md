#!/usr/bin/env python3
"""
word_freq: テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

Usage:
    python word_freq.py input.txt --top 10
    python word_freq.py input.txt --top 20 --case-sensitive
    python word_freq.py input.txt --top 5 --output json
"""

from __future__ import annotations

import argparse
import json
import re
import sys
from collections import Counter
from typing import Any


def tokenize(text: str, case_sensitive: bool = False) -> list[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        単語トークンのリスト（句読点や空白は除去される）。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字、ハイフン、アポストロフィを含む連続文字列を単語とみなす
    tokens = re.findall(r"[a-z0-9]+(?:['\-][a-z0-9]+)*", text, re.IGNORECASE)
    return [t for t in tokens if t]


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    テキスト内の単語出現頻度を集計する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    tokens = tokenize(text, case_sensitive)
    return Counter(tokens)
