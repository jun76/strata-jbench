#!/usr/bin/env python3
"""
word_freq.py - Text file word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.
    
    Args:
        text: Input text string.
        
    Returns:
        List of lowercase word tokens.
    """
    # Extract words consisting of alphanumeric characters and underscores
    return re.findall(r'\b\w+\b', text.lower())


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.
    
    Args:
        text: Input text string.
        
    Returns:
        Counter object with word frequencies.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.
        
    Returns:
        List of tuples (word, count) for the top N words.
    """
    return counter.most_common(n)


def format_table(top_words: List[Tuple[str, int]]) -> str:
    """
    Format top words as a table string.
    
    Args:
        top_words: List of tuples (word, count).
        
    Returns:
        Formatted table string.
    """
    if not top_words:
        return "No words found."
    
    # Calculate column widths
    max_word_len = max(len(word) for word, _ in top_words)
    max_count_len = max(len(str(count)) for _, count in top_words)
    
    # Ensure minimum widths
    word_width = max(max_word_len, 5)  # "Word" header is 4 chars
    count_width = max(max_count_len, 5)  # "Count" header is 5 chars
    
    # Build header
    header = f"{'Word':<{word_width}} | {'Count':>{count_width}}"
