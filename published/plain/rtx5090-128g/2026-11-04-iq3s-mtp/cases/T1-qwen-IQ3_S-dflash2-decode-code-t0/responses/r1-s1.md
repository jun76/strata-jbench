#!/usr/bin/env python3
"""
word_freq.py - A command-line tool to count word frequencies in a text file.

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive] [--min-length L]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Extract words from text.

    Args:
        text: The input text string.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.
        min_length: Minimum length of a word to be counted.

    Returns:
        A list of words extracted from the text.
    """
    # Use regex to find sequences of alphanumeric characters (including underscores)
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    
    if not case_sensitive:
        words = [word.lower() for word in words]
    
    # Filter by minimum length
    words = [word for word in words if len(word) >= min_length]
    
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) sorted by count descending, then by word ascending.
    """
    # most_common returns items sorted by count descending
    # For ties, we want alphabetical order, so we sort manually
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
