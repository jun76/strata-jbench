#!/usr/bin/env python3
"""
wordfreq.py - テキストファイルから単語出現頻度を解析し、上位N件を表示するツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出してリストで返す。
    英数字とアンダースコアのみを単語とし、他は区切りとする。
    単語は小文字に変換される。
    """
    # [a-zA-Z0-9_]+ に一致する部分を取り出し、小文字化
    return [word.lower() for word in re.findall(r"[a-zA-Z0-9_]+", text)]


def count_words(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度を Counter オブジェクトとして返す。
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counter から出現回数の多い上位 n 件の (word, count) を返す。
    """
    return counter.most_common(n)

    (word, count) のリストを整形されたテーブル文字列として返す。
    """

        return "No words found."

    count_width = max(len(str(count)) for _, count in rows)
