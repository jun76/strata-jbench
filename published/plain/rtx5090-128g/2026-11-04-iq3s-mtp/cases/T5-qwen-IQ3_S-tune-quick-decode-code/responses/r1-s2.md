"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントして、
上位 N 件の単語とその頻度を表形式で表示するコマンドラインツール。

Usage:
    python word_counter.py input.txt --top 10
    python word_counter.py input.txt --top 20 --case-sensitive

def load_text(filepath: str) -> str:
    """指定されたファイルパスからテキストを読み込む。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み取りに失敗した場合。
    """
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {filepath}")
    except IOError as e:
        raise IOError(f"ファイルの読み取りに失敗しました: {e}")
