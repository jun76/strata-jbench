"""
text_word_counter.py
コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を上位N件を表形式で表示する。
"""

import argparse
import re

from collections import Counter

def extract_words(text: str) -> List[str]:
    """
    テキストから単語（アルファベットおよび数字の連続）を抽出する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        抽出された単語のリスト（小文字化済み）
    """
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_words(words: List[str]) -> Counter:
    """
    単語リストから単語頻度のCounterを生成する。
    
    Args:
        words: 単語のリスト
        
    Returns:
        単語をキー、出現回数を値とするCounter
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    最も出現頻度の高い上位n件の単語とその回数を取得する。
    
    Args:
        counter: 単語頻度Counter
        n: 上位件数
        
    Returns:
        (word, count)タプルのリスト、降順ソート済み
    """
    return counter.most_common(n)
