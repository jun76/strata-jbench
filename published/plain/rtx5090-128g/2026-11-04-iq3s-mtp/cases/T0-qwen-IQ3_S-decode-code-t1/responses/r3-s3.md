    python word_counter.py input.txt --top 5 --case-insensitive

import argparse
import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストから単語リストを抽出する。

    英数字およびアンダースコアを単語文字として扱い、それ以外を区切りとする。

    Args:
        text: 入力テキスト。

    Returns:
        抽出された単語リスト（小文字化しない）。
    """
    return re.findall(r'\w+', text)


def count_words(text: str, case_insensitive: bool = False) -> Counter[str]:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、単語を小文字に変換してカウントする。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    words = tokenize(text)
    if case_insensitive:
        words = [word.lower() for word in words]
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Counter から最も出現頻度の高い上位 n 件を返す。

    頻度が同じ場合は辞書順（アルファベット順）で並べ替える。

    Args:
        counter: 単語と出現回数の Counter。
        n: 取得する上位件数。

    Returns:
        (単語, 出現回数) のタプルリスト。
    """
    # most_common は頻度順だが、同じ頻度の場合は挿入順になるため、
    # 辞書順で安定ソートするためにキー関数で (freq, word) を使う
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
