import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def read_text(filepath: str) -> str:
    """指定されたファイルからテキストを読み込む。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。

    単語は英数字のみを含む連続文字列として定義される。
    全て小文字に変換される。

    Args:
        text: 入力テキスト。

    Returns:
        抽出された単語のリスト。
    """
    # 英数字のみの連続文字列を抽出し、小文字に変換
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_word_frequency(words: List[str]) -> Counter:
    """単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語とその出現回数をキーと値とするCounter。
    """
    return Counter(words)


def get_top_n_words(freq_counter: Counter, n: int) -> List[Tuple[str, int]]:
    """上位N件（出現回数が多い順）の単語と回数を取得する。

    同数の場合は辞書順でソートされる。

    Args:
        freq_counter: 単語頻度のCounter。
        n: 取得する上位件数。

    Returns:
        (単語, 出現回数)のタプルが上位順に並んだリスト。
    """
    if n <= 0:
        return []
    
    # 出現回数で降順、同数なら単語の辞書順（昇順）
    sorted_items = sorted(
        freq_counter.items(),
        key=lambda x: (-x[1], x[0])
    )
    return sorted_items[:n]
