#!/usr/bin/env python3
"""A command-line tool that reads a text file, counts word frequencies,
and displays the top N words in a table format."""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple, Dict


def extract_words(text: str) -> List[str]:
    """Extract lowercase words from text, splitting on non-alphanumeric characters.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens.
    """
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return [w.lower() for w in words]


def count_word_frequencies(words: List[str]) -> Dict[str, int]:
    """Count the frequency of each word in the list.

    Args:
        words: A list of word tokens.

    Returns:
        A dictionary mapping each word to its count.
    """
    return dict(Counter(words))


def get_top_n(frequencies: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words, sorted by count descending, then by word ascending.

    Args:
        frequencies: A dictionary mapping words to their counts.
        n: The number of top entries to return.

    Returns:
        A list of tuples, each containing (word, count).
    """
    sorted_items = sorted(frequencies.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
