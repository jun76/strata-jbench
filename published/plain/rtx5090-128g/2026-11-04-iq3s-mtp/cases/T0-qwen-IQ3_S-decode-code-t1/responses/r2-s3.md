#!/usr/bin/env python3
"""A command-line tool that reads a text file, counts word frequencies,
and displays the top N words in a table format.

Usage:
    python wordfreq.py <input_file> [--top N] [--case-insensitive] [--min-length LEN]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """Tokenize text into words, filtering by minimum length and optionally normalizing case.

    Args:
        text: Input text to tokenize.
        case_sensitive: If True, preserve original case. If False, convert to lowercase.
        min_length: Minimum word length to include.

    Returns:
        A list of tokens (words) extracted from the text.
    """
    if not case_sensitive:
        text = text.lower()
    # Match sequences of word characters (letters, digits, underscores)
    tokens = re.findall(r'\b\w+\b', text)
    # Filter by minimum length
    tokens = [token for token in tokens if len(token) >= min_length]
    return tokens


def count_word_frequencies(tokens: List[str]) -> Counter[str]:
    """Count the frequency of each token in the list.

    Args:
        tokens: A list of tokens (words).

    Returns:
        A Counter mapping each token to its frequency.
    """
    return Counter(tokens)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Get the top N most common tokens from a Counter.

    Args:
        counter: A Counter object mapping tokens to frequencies.
        n: Number of top items to return.

    Returns:
        A list of (token, frequency) tuples sorted by frequency descending,
        then by token ascending for ties.
    """
    # most_common already returns tuples sorted by count desc, then by insertion order
    # For deterministic tie-breaking, sort by (-count, token)
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
