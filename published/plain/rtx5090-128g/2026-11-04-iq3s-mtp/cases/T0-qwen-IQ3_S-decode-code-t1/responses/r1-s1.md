"""

import argparse

from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    入力テキストを単語トークンに分割する。

    単語はアルファベットと数字で構成される連続した文字列として抽出される。
    大文字・小文字の区別は case_sensitive フラグで制御する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。False の場合、すべて小文字に変換する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # アルファベットと数字の連続文字列を抽出
    import re
    tokens = re.findall(r'[a-zA-Z0-9]+', text)
    return tokens


def count_word_frequencies(tokens: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        tokens: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(tokens)


def top_n_words(frequencies: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度カウンターから上位 N 件の単語と回数のタプルリストを返す。

    同率の場合は辞書順（単語のアルファベット順）で並べる。

    Args:
        frequencies: 単語と出現回数をマッピングする Counter。
        n: 取得する上位件数。

    Returns:
        (word, count) のタプルリスト。上位 N 件。
    """
    # most_common は回数の降順だが、同率時は不定なため、手動でソートする
    items = sorted(frequencies.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
