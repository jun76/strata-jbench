import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def read_text(filepath: str) -> str:
    """指定されたファイルからテキストを読み込む。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。

    単語は英数字（Unicode対応）の連続として定義され、すべて小文字に変換される。

    Args:
        text: 解析対象のテキスト。

    Returns:
        抽出された単語のリスト。
    """
    # Unicodeの文字や数字を考慮したパターン
    words = re.findall(r"[\w]+", text.lower())
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """単語リストから出現頻度を集計する。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)


def get_top_n(frequencies: Counter, n: int) -> List[Tuple[str, int]]:
    """上位N件の単語と回数を取り出す。

    同順位の場合は辞書順に並び替える。

    Args:
        frequencies: 単語の出現頻度 Counter。
        n: 取り出す件数。

    Returns:
        (単語, 回数) のタプルのリスト。上位順。
    """
    # まず回数で降順、次に単語で昇順
    items = sorted(frequencies.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
