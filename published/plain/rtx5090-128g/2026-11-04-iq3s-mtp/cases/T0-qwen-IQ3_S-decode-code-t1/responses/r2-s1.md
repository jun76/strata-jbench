"""
テキストファイルを読み込んで単語の出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツール。

使い方:
    python word_count.py <input_file> [-n N]

例:
    python word_count.py sample.txt
    python word_count.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    テキストから単語の出現頻度を数える。

    Args:
        text: 分析対象のテキスト。

    Returns:
        Counter: 単語をキー、出現回数を値とする Counter。
                 単語はすべて小文字に変換され、非文字/数字文字で分割される。
    """
    # 英数字とアポストロフィを含む単語を抽出
    words = re.findall(r"[a-zA-Z0-9']+", text.lower())
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から最も頻度の高い n 件の単語を (word, count) のタプルリストとして返す。

    Args:
        counter: 単語の出現頻度を保持する Counter。
        n: 取得する単語の数。

    Returns:
        List[Tuple[str, int]]: (単語, 出現回数) のタプル。頻度順、同率の場合は辞書順。
    """
    # most_common は頻度順、同率の場合は挿入順だが、安定した出力のために
    # 頻度で並べ替え、同率なら辞書順で並べる
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
