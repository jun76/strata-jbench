import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def read_text(filepath: str) -> str:
    """指定されたファイルからテキストを読み込む。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。

    単語は英数字のみを含む文字列として定義される。
    大文字小文字は区別せず、小文字に変換される。

    Args:
        text: 入力テキスト。

    Returns:
        抽出された単語のリスト。
    """
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_words(words: List[str]) -> Counter:
    """単語リストから各単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語とその出現回数をキー・値とする Counter。
    """
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """最も頻繁に出現する上位 n 個の単語とその出現回数を取得する。

    Args:
        counter: 単語頻度を表す Counter。
        n: 取得する上位単語の数。

    Returns:
        (単語, 出現回数) のタプルのリスト。頻度が高い順。
        頻度が同じ場合は辞書順で安定した並びを考慮する。
    """
    # most_common(n) gives the n most common elements
    # For ties, Python's Counter.most_common maintains insertion order for equal counts.
    # To ensure deterministic ordering by frequency desc then lexicographic asc for ties,
    # we sort explicitly.
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
