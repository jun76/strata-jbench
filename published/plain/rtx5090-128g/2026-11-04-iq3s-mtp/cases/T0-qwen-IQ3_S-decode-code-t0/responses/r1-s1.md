#!/usr/bin/env python3
"""
word_freq.py - A command-line tool to count word frequencies in a text file.

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive] [--min-length L]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, convert all words to lowercase.
        min_length: Minimum length of a word to include.

    Returns:
        A list of tokens (words) extracted from the text.
    """
    # Extract words using regex: sequences of alphanumeric characters
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    
    if not case_sensitive:
        words = [w.lower() for w in words]
    
    # Filter by minimum length
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    
    return words


def count_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        A list of tuples (word, count) sorted by count descending, then alphabetically.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
