#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位N件表示する。

使い方:
    python word_freq.py <file> [-n N] [--case-sensitive] [--min-length L]

例:
    python word_freq.py sample.txt -n 10
    python word_freq.py sample.txt --case-sensitive --min-length 3
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか
        min_length: 単語の最小長（この長さ未満の単語は除外）

    Returns:
        抽出された単語のリスト
    """
    # 英数字とハイフンを含む単語を抽出（日本語などの非ASCII文字も考慮）
    # 簡易的に、Unicodeの文字・数字・ハイフンを含むトークンを抽出
    pattern = r'[\w\-]+'
    words = re.findall(pattern, text, flags=re.UNICODE)

    if not case_sensitive:
        words = [w.lower() for w in words]

    # 最小長でフィルタ
    words = [w for w in words if len(w) >= min_length]

    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とするCounter
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位N件の単語と頻度を取得する。

    Args:
        counter: 単語頻度のCounter
        n: 取得する件数

    Returns:
        (単語, 頻度) のタプルのリスト、頻度降順
    """
    return counter.most_common(n)
