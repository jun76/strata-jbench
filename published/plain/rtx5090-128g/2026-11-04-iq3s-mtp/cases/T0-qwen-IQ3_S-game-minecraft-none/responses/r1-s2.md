  function raycast(){
    const o=camera.position,e=new THREE.Vector3();camera.getWorldDirection(e);
    let x=Math.floor(o.x),y=Math.floor(o.y),z=Math.floor(o.z);
    const sx=e.x>0?1:-1,sy=e.y>0?1:-1,sz=e.z>0?1:-1;
    const ddx=e.x!==0?Math.abs(1/e.x):Infinity,ddy=e.y!==0?Math.abs(1/e.y):Infinity,ddz=e.z!==0?Math.abs(1/e.z):Infinity;
    let tMaxX=e.x!==0?((sx>0?x+1-o.x:o.x-x)*ddx):Infinity;
    let tMaxY=e.y!==0?((sy>0?y+1-o.y:o.y-y)*ddy):Infinity;
    let tMaxZ=e.z!==0?((sz>0?z+1-o.z:o.z-z)*ddz):Infinity;
    let nx=0,ny=0,nz=0,t=0;
    for(let i=0;i<200;i++){
      const b=getBlock(x,y,z);
      if(b!==0&&b!==WATER){return{x,y,z,nx,ny,nz,id:b};}
      if(tMaxX<tMaxY&&tMaxX<tMaxZ){x+=sx;t=tMaxX;tMaxX+=ddx;nx=-sx;ny=0;nz=0;}
      else if(tMaxY<tMaxZ){y+=sy;t=tMaxY;tMaxY+=ddy;nx=0;ny=-sy;nz=0;}
      else{z+=sz;t=tMaxZ;tMaxZ+=ddz;nx=0;ny=0;nz=-sz;}
      if(t>REACH)break;
    }
    return null;
  }

    const R=0.3,H=1.8;
    if(selected!==WATER){const ox=Math.max(px,player.x-R),oy=Math.max(py,player.y),oz=Math.max(pz,player.z-R);
      const ax=Math.min(px+1,player.x+R),ay=Math.min(py+1,player.y+H),az=Math.min(pz+1,player.z+R);
      if(ox<ax&&oy<ay&&oz<az)return;}
    setBlock(px,py,pz,selected);sfxPlace(selected);}

  function updateSky(dt){
    timeOfDay=(timeOfDay+dt/DAY_LEN*(timeBoost?6:1))%1;
    const ang=timeOfDay*Math.PI*2,elev=Math.sin(ang);
    const dir=new THREE.Vector3(Math.cos(ang),elev,0.35).normalize();
    sun.position.copy(dir);
    const day=smoothstep(-0.1,0.25,elev);
    tmpC.copy(NIGHT_SKY).lerp(DAY_SKY,day);
    skyMat.color.copy(tmpC);scene.fog.color.copy(tmpC);
    renderer.setClearColor(tmpC,1);
    ambient.intensity=0.16+0.52*day;
    sun.intensity=0.15+0.85*day;
    sun.color.copy(MOON).lerp(SUN,day);
    starsMat.opacity=clamp(1-day*1.4,0,1)*0.9;
    // ui
    const hh=Math.floor(timeOfDay*24),mm=Math.floor((timeOfDay*24-hh)*60);
    const tt=String(hh).padStart(2,"0")+":"+String(mm).padStart(2,"0");
    clockVal.textContent=tt;timeval.textContent=tt;
    sunA.style.transform=`translate(-50%,-50%) rotate(${(timeOfDay*360)}deg)`;
    moonA.style.transform=`translate(-50%,-50%) rotate(${(timeOfDay*360+180)}deg)`;
  }
