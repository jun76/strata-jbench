function collides(px,py,pz){
  const r=.3;
  const x0=Math.floor(px-r),x1=Math.floor(px+r),y0=Math.floor(py),y1=Math.floor(py+1.8),z0=Math.floor(pz-r),z1=Math.floor(pz+r);
  for(let x=x0;x<=x1;x++)for(let y=y0;y<=y1;y++)for(let z=z0;z<=z1;z++)if(isSolid(x,y,z))return true;
  return false;
}
function stepAxis(axis,delta){
  if(axis==='x')player.pos.x+=delta;else if(axis==='y')player.pos.y+=delta;else player.pos.z+=delta;
  if(collides(player.pos.x,player.pos.y,player.pos.z)){
    if(axis==='x'){player.pos.x-=delta;player.vel.x=0}
    else if(axis==='y'){player.pos.y-=delta;player.vel.y=0;player.grounded=delta<0}
    else{player.pos.z-=delta;player.vel.z=0}
  }
}

## What's inside

- **Procedural voxel world** — fBm value-noise terrain with hills, mountains, beaches, oceans, and scattered trees, generated chunk-by-chunk around you as you walk (never pre-baked).
- **Greedy-ish face culling** — only visible block faces are emitted, with per-block atlas textures (grass top/side, wood grain, stone speckle, translucent water).
- **First-person controller** — WASD, jump, sprint, gravity + AABB collision, and a `F` fly-mode toggle.
- **Break / place** — raycast targeting with a highlight box, live crosshair feedback, and a 3D-feel particle burst of block-colored chips.
- **Living atmosphere** — a day/night cycle with moving sun & moon, drifting sky/fog colors, soft directional shadows, sky dome, subtle grain + vignette.
- **HUD** — live coordinates, a compass, a day-cycle bar, and an interactive hotbar you can click, scroll, or key `1`–`7`.

Click **▶ プレイ** to enter, and **新しい世界** (or `R`) for a fresh seed.
