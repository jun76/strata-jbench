#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルの単語出現頻度を上位 N 件表示する。

使い方:
    python word_freq.py input.txt --top 10
    python word_freq.py input.txt -n 5 --case-sensitive
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()

    # 非英数字文字をスペースに置換し、単語分割
    cleaned = "".join(
        ch if ch.isalnum() or ch == "'" else " "
        for ch in text
    )
    return [word for word in cleaned.split() if word]


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト内の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)
