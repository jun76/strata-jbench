wordfreq - テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

使い方:
    python wordfreq.py <file> [-n N] [--case-sensitive] [--sort-value]

例:
    python wordfreq.py sample.txt -n 10
    python wordfreq.py sample.txt -n 5 --case-sensitive --sort-value
"""

from __future__ import annotations

import argparse

import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。
            False の場合、全て小文字に変換する。

    Returns:
        抽出された単語トークンのリスト。
        単語はアルファベット・数字・ハイフン・アポストロフィで構成される文字列とする。
    """
    # 単語パターン: 英数字、ハイフン、アポストロフィを含む連続文字列
    pattern = r"[A-Za-z0-9][A-Za-z0-9'\-]*[A-Za-z0-9]|[A-Za-z0-9]"
    if not case_sensitive:
        text = text.lower()
    tokens = re.findall(pattern, text)
    # 空文字列や空白のみを除く
    tokens = [t for t in tokens if t.strip()]
    return tokens
