#!/usr/bin/env python3
"""
テキストファイルを読み込み、単語の出現頻度を上位 N 件まで集計・表示する CLI ツール。

使い方例:
    $ python word_freq.py -n 10 sample.txt
    $ cat sample.txt | python word_freq.py -n 5
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def normalize_word(word: str) -> str:
    """
    単語を小文字に変換し、前後の空白を除いて正規化する。

    Args:
        word: 正規化対象の文字列。

    Returns:
        小文字化・トリムされた文字列。
    """
    return word.strip().lower()


def extract_words(text: str) -> List[str]:
    """
    テキストから単語（アルファベット、数字、ハイフン、アポストロフィを含む）を抽出する。

    Args:
        text: 解析対象の文字列。

    Returns:
        正規化された単語のリスト。
    """
    # 単語として [a-zA-Z0-9'-] を含む連続文字列を抽出
    words = re.findall(r"[a-zA-Z0-9'-]+", text)
    return [normalize_word(w) for w in words if normalize_word(w)]


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
