#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語の出現頻度を集計し、上位N件を表形式で表示するCLIツール。

Usage:
    python word_freq.py <file> [-n N] [--case-sensitive] [--min-length LEN]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: Trueの場合、大文字小文字を区別する。Falseの場合、小文字に変換する。
        min_length: この長さ未満の単語は除外する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    # 最小長でフィルタ
    words = [w for w in words if len(w) >= min_length]
    return words


def count_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    return Counter(words)
