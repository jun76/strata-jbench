word_freq_counter.py

A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a table format.
"""

import argparse
import re

from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    Count word occurrences in the given text.
    
    Words are defined as sequences of alphanumeric characters.
    Counting is case-insensitive (all words are lowercased).
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter object mapping each word to its frequency.
    """
    # Extract words using regex: sequences of word characters
    words = re.findall(r'\b[a-z0-9]+\b', text.lower())
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: Number of top entries to return.
        
    Returns:
        A list of (word, count) tuples, sorted by frequency descending,
        then alphabetically for ties.
    """
    # Most common returns up to n items, sorted by count desc, then key asc
    return counter.most_common(n)
