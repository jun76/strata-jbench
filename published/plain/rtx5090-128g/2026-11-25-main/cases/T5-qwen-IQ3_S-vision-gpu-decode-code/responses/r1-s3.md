#!/usr/bin/env python3
"""
word_count.py - Text file word frequency counter.

Reads a text file, counts word occurrences, and displays the top N words
in a tabular format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    Extract words from text, normalizing to lowercase.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Match sequences of alphanumeric characters and underscores (word characters)
    words = re.findall(r'\b\w+\b', text, re.IGNORECASE)
    return [word.lower() for word in words]


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    words = extract_words(text)
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples, each containing (word, count), sorted by count descending.
    """
    return counter.most_common(n)
