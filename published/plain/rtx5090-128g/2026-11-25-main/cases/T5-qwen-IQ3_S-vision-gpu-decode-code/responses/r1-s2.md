#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で表示するコマンドラインツール。
"""

import argparse

from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    テキストから単語を抽出し、出現頻度をカウントする。

    Args:
        text (str): 入力テキスト。

    Returns:
        Counter[str]: 単語をキー、出現回数値を値とする Counter オブジェクト。
        単語は小文字に変換され、句読点や記号は除去される。
    """
    # 句読点や記号を空白に置き換えるなどの簡易的なトークン化
    # 正規表現を使うとより正確だが、ここでは str.translate を使用して高速化
    import string
    translator = str.maketrans(string.punctuation, ' ' * len(string.punctuation))
    clean_text = text.translate(translator)
    
    words = clean_text.lower().split()
    return Counter(words)


def get_top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counter オブジェクトから上位 N 件の単語と頻度を取得する。

    Args:
        counter (Counter[str]): 単語頻度カウンター。
        n (int): 取得する件数。

    Returns:
        List[Tuple[str, int]]: (単語, 頻度) のタプルリスト。
    """
    return counter.most_common(n)
