def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words.
    
    Args:
        text: Input text string.
        case_sensitive: If False, convert to lowercase for counting.
        min_length: Minimum length of a word to be included.
    
    Returns:
        List of tokens (words).
    """
    if not case_sensitive:
        text = text.lower()
    
    # Extract words (sequences of alphanumeric characters)
    words = []
    current_word = []
    for char in text:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                word = ''.join(current_word)
                if len(word) >= min_length:
                    words.append(word)
                current_word = []
    if current_word:
        word = ''.join(current_word)
        if len(word) >= min_length:
            words.append(word)
    
    return words


def count_word_frequencies(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """
    Count the frequency of each word in the given text.
    
    Args:
        text: Input text string.
        case_sensitive: If False, treat words case-insensitively.
        min_length: Minimum length of a word to be included.
    
    Returns:
        Counter object with word frequencies.
    """
    tokens = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(tokens)
