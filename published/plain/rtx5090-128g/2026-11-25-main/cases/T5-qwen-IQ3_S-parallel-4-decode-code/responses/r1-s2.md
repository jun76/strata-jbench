def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別しない。False (デフォルト) の場合、
                        すべての単語を小文字に変換して区別しない。

    Returns:
        単語のリスト。
    """
    # 英数字とハイフンを含む文字列を単語として抽出
    # 例: "don't" -> ["don", "t"] または ["don't"] にするか?
    # ここでは、アルファベットと数字、および内側のハイフン/アポストロフィを含むトークンを抽出する。
    # より一般的なアプローチ: 英数字のみで単語を区切る。
    # \b[a-zA-Z0-9]+(?:[-'][a-zA-Z0-9]+)*\b でハイフンやアポストロフィを含む単語も拾える。
    # シンプルに [a-zA-Z0-9']+ で拾い、先頭末尾の句読点は除外される。
    # 実際には、正規表現 [A-Za-z0-9]+ で十分。
    pattern = r"[A-Za-z0-9']+"
    tokens = re.findall(pattern, text)
    if not case_sensitive:
        tokens = [t.lower() for t in tokens]
    return tokens
