テキストファイルを読み込み、単語の出現頻度を計算し、上位 N 件を表形式で出力するコマンドラインツール。

def count_words(text: str) -> Counter[str]:
    """
    テキストから単語の出現頻度をカウントする。
    単語は小文字に変換され、アルファベットと数字のみを含む文字列として扱われる。

    Args:
        text (str): 入力テキスト

    Returns:
        Counter[str]: 単語をキー、出現回数を値とするカウンタオブジェクト
    """
    # 正規表現を使わず、単純な文字分類で単語抽出
    # より正確な単語抽出には re を使うこともできるが、ここでは標準ライブラリのみで実装
    import re
    # 単語は英数字の連続文字列とする
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    # 小文字に変換
    return Counter(word.lower() for word in words)


def get_top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    カウンタから上位 n 件の単語と頻度を取得する。
    頻度が同じ場合は、辞書順でソートする。

    Args:
        counter (Counter[str]): 単語頻度カウンタ
        n (int): 取得する件数

    Returns:
        List[Tuple[str, int]]: (単語, 頻度) のタプルのリスト。頻度降順、同頻度時は辞書順。
    """
    # 頻度降順、次に単語の辞書順でソート
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
