#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルから単語出現頻度を集計し上位N件を表示する。

使い例:
    python word_freq.py input.txt -n 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple, Optional


def parse_args(argv: Optional[List[str]] = None) -> argparse.Namespace:
    """コマンドライン引数を解析する。

    Args:
        argv: 引数リスト（Noneなら sys.argv を使用）

    Returns:
        argparse.Namespace: パースされた引数
    """
    parser = argparse.ArgumentParser(
        description="テキストファイルから単語出現頻度を集計し上位N件を表示するツール"
    )
    parser.add_argument(
        "file",
        help="対象のテキストファイルパス",
    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=10,
        help="表示する上位単語の件数 (デフォルト: 10)",
    )
    parser.add_argument(
        "--case-insensitive",
        action="store_true",
        help="大文字小文字を区別せず集計する",
    )
    return parser.parse_args(argv)


def tokenize(text: str, case_insensitive: bool = False) -> List[str]:
    """テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_insensitive: Trueなら小文字に変換して集計

    Returns:
        単語のリスト
    """
    words = text.split()
    if case_insensitive:
        words = [w.lower() for w in words]
    return words


def count_words(words: List[str]) -> Counter:
    """単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数の Counter
    """
    return Counter(words)
