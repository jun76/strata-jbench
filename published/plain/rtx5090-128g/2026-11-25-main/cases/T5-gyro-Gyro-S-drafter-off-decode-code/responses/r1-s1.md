# 型エイリアス

def read_file(path: str) -> str:
    """テキストファイルを読み込み、その内容を文字列として返す。

    Args:
        path: 読み込むファイルのパス。

    Returns:
        ファイルの内容を含む文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        OSError: ファイル読み込みが失敗した場合。
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, ignore_case: bool = True) -> List[str]:
    """テキストを単語に分割する。

    Args:
        text: 分割対象のテキスト。
        ignore_case: True の場合、単語を小文字に変換する。

    Returns:
        単語のリスト。
    """
    # 簡易的なトークン化: 空白区切りで分割し、句読点を除去する
    tokens = text.split()
    cleaned = []
    for token in tokens:
        # 句読点や特殊文字を除去
        word = token.strip(".,!?;:\"'()-{}[]")
        if word:
            if ignore_case:
                word = word.lower()
            cleaned.append(word)
    return cleaned


def count_frequencies(tokens: List[str]) -> WordFrequency:
    """単語の出現頻度を集計する。

    Args:
        tokens: 単語のリスト。

    Returns:
        単語とその出現回数の辞書。
    """
    return dict(Counter(tokens))
