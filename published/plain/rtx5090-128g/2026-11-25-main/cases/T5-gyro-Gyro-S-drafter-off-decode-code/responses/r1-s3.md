"""
word_count.py - コマンドラインツール：テキストファイルの単語出現頻度を上位N件で表示

使用方法:
    python word_count.py <file> [--top N] [--case-sensitive] [--min-length L]

例:
    python word_count.py sample.txt --top 10
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple, Optional


def load_text(file_path: Path) -> str:
    """
    テキストファイルを読み込み、その内容を文字列として返す。

    Args:
        file_path: 読み込むファイルのパス

    Returns:
        ファイルのテキスト内容

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        OSError: ファイルが読み込めない場合
    """
    return file_path.read_text(encoding="utf-8")


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 分割対象のテキスト
        case_sensitive: Trueなら大文字小文字を区別、Falseなら小文字に変換して集計
        min_length: この長さ未満のトークンは除外

    Returns:
        単語のリスト
    """
    # 英字、数字、ハイフン、アポストロフィーを含む連続文字列を単語として抽出
    pattern = r"[a-zA-Z0-9]+(?:[-'][a-zA-Z0-9]+)*"
    tokens = re.findall(pattern, text)

    if not case_sensitive:
        tokens = [token.lower() for token in tokens]

    # 長さ制限
    tokens = [token for token in tokens if len(token) >= min_length]

    return tokens


def count_words(tokens: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        tokens: 単語のリスト

    Returns:
        単語 -> 出現回数 のカウント辞書
    """
    return Counter(tokens)
