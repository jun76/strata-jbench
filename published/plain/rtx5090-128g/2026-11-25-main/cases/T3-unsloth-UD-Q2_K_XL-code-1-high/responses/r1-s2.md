def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    """
    Merge all overlapping or touching closed intervals.

    Args:
        intervals: A list of closed intervals, where each interval is a
                   list/tuple of two integers [start, end].

    Returns:
        A new sorted list of merged intervals, without modifying the input.
    """
    merged = []

    for start, end in sorted(intervals, key=lambda interval: (interval[0], interval[1])):
        if not merged or start > merged[-1][1]:
            merged.append([start, end])
        elif end > merged[-1][1]:
            merged[-1][1] = end

    return merged
