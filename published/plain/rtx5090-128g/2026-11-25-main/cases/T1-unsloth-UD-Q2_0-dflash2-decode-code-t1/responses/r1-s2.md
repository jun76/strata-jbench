Usage:
    python word_freq.py --top 10 input.txt

def read_text_file(path: str) -> str:
    """Read and return the contents of a text file.

    Args:
        path: Path to the text file.

    Returns:
        The text content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        PermissionError: If the file cannot be read due to permission issues.
    """
    file_path = Path(path)
    if not file_path.exists():
        raise FileNotFoundError(f"File not found: {path}")
    if not file_path.is_file():
        raise FileNotFoundError(f"Path is not a file: {path}")
    try:
        return file_path.read_text(encoding="utf-8")
    except PermissionError:
        raise PermissionError(f"Permission denied: {path}")


def extract_words(text: str) -> List[str]:
    """Extract words from text, lowercased and stripped of punctuation.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words with surrounding punctuation removed.
    """
    words: List[str] = []
    for word in text.lower().split():
        # Strip common punctuation from both ends
        cleaned = word.strip(".,;:!?\"'()[]{}<>\\/`~@#$%^&*_=-|")
        if cleaned:
            words.append(cleaned)
    return words


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter mapping each word to its frequency.
    """
    return Counter(words)
