from typing import Final, List, Tuple


DEFAULT_TOP_N: Final[int] = 10
WORD_PATTERN: Final[re.Pattern[str]] = re.compile(r"\b\w+\b", re.UNICODE)


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 解析対象のテキスト。
        case_sensitive: True の場合、大文字/小文字を区別する。False の場合、
                        すべて小文字に変換して集計する。

    Returns:
        抽出された単語のリスト（元の順序を保持）。
    """
    words: List[str] = WORD_PATTERN.findall(text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_frequency(words: List[str]) -> Counter[str]:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語 -> 出現回数の Counter。
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 n 件を返す。

    Args:
        counter: 単語の出現頻度 Counter。
        n: 取得する件数。

    Returns:
        (単語, 出現回数) のタプルリスト。降順でソート済み。
    """
    return counter.most_common(n)
