def normalize_text(text: str) -> str:
    """テキストを正規化し、単語単位に分割するための小文字化と句読点除去を行う。

    Args:
        text: 入力テキスト。

    Returns:
        正規化された小文字のテキスト。句読点や特殊文字はスペースに置き換えられる。
    """
    # 句読点や記号をスペースに置換
    normalized = unicodedata.normalize("NFKC", text).lower()
    # 英数字以外の文字をスペースに置換（日本語の文字も保持されるようにするため、
    # 実際には日本語の形態素解析が必要だが、ここでは簡易的に単語分割を想定）
    # より正確な日本語処理は別途形態素解析ライブラリが必要だが、
    # このツールは主に英語・スペース区切りの単語を想定している。
    result = []
    for char in normalized:
        if char.isalnum():
            result.append(char)
        else:
            result.append(" ")
    return "".join(result)
