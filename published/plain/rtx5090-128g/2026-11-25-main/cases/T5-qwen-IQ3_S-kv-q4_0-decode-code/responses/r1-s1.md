#!/usr/bin/env python3
"""
word_freq.py - Text file word frequency analysis tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    Extract words from text, ignoring case and punctuation.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Match sequences of alphanumeric characters (words)
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their frequencies.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) for the top N words.
    """
    return counter.most_common(n)


def format_table(data: List[Tuple[str, int]], title: str = "Top Words") -> str:
    """
    Format the top words data as a text table.

    Args:
        data: A list of tuples (word, count).
        title: The title for the table.

    Returns:
        A formatted string representation of the table.
    """
    if not data:
        return f"{title}\n(No words found)"

    # Determine column widths
    word_width = max(len(word) for word, _ in data)
    count_width = max(len(str(count)) for _, count in data)

    # Ensure minimum widths
    word_width = max(word_width, 4)  # "Word" header width
    count_width = max(count_width, 5)  # "Count"
