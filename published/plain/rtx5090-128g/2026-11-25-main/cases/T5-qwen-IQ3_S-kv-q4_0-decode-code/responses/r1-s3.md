#!/usr/bin/env python3
"""
word_freq.py: テキストファイルから単語の出現頻度を計算し、上位 N 件を表形式で出力するツール。
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple
import re


def read_file(filepath: str) -> str:
    """指定されたファイルパスからテキストを読み込む。

    Args:
        filepath: 読み込むファイルのパス。

    Returns:
        ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み取りに失敗した場合。
    """
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except IOError as e:
        raise IOError(f"Error reading file {filepath}: {e}")


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。大文字小文字は無視し、非英数字は削除する。

    Args:
        text: 入力テキスト。

    Returns:
        正規化された単語のリスト。
    """
    # 正規表現で英数字の連続部分列を抽出し、小文字に変換
    words = re.findall(r'\b[a-z0-9]+\b', text.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
