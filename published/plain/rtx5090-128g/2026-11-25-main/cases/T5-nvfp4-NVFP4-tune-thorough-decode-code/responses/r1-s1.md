## 1. CLIツール本体: `wordfreq.py`

```python
#!/usr/bin/env python3
"""
wordfreq.py - テキストファイルの単語出現頻度を集計し、上位N件を表形式で表示するCLIツール。

使い方:
    python wordfreq.py input.txt
    python wordfreq.py input.txt -n 10 --sort freq
    python wordfreq.py input.txt --case-sensitive --min-length 3
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Literal

# 単語の正規表現パターン（英数字とハイフンを含む単語を抽出）
WORD_PATTERN = re.compile(r"[A-Za-z0-9]+(?:[-'][A-Za-z0-9]+)*")


def tokenize(text: str, *, case_sensitive: bool = False, min_length: int = 1) -> list[str]:
    """テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別する場合 True。
        min_length: 抽出する単語の最小文字数。

    Returns:
        単語のリスト。
    """
    words = WORD_PATTERN.findall(text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return [w for w in words if len(w) >= min_length]


def count_words(words: list[str]) -> Counter[str]:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)


def get_top_n(
    counts: Counter[str],
    n: int,
    *,
    sort_by: Literal["freq", "alpha"] = "freq",
) -> list[tuple[str, int]]:
    """上位 N 件の (単語, 頻度) を返す。

    Args:
        counts: 出現頻度の Counter。
        n: 上位何件まで取得するか。n <= 0 の場合は全件を返す。
        sort_by: 並び替え基準。"freq" は頻度
