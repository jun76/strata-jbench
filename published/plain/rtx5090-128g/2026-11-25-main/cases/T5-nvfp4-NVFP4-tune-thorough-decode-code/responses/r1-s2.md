Command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a tabular format.

Usage:
    python word_freq.py <input_file> [-n TOP_N] [--case-sensitive]
"""

import argparse
import re

from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Extract words from the given text.

    Args:
        text: The input text to extract words from.
        case_sensitive: If True, preserve case. Otherwise, convert to lowercase.

    Returns:
        A list of words extracted from the text.
    """
    if not case_sensitive:
        text = text.lower()
    # Extract sequences of alphanumeric characters and underscores as words
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word in the list.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their frequencies.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, frequency) sorted by frequency descending.
    """
    return counter.most_common(n)
