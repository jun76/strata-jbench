# 単語出現頻度集計ツール

## メインスクリプト: `word_freq.py`

#!/usr/bin/env python3
"""
word_freq.py — テキストファイルから単語の出現頻度を集計し、上位N件を表形式で表示するCLIツール。

使い方:
    python word_freq.py input.txt
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt -n 5 --case-insensitive
    python word_freq.py input.txt -n 10 --min-length 3
"""

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from typing import List, Tuple


@dataclass
class WordFreqEntry:
    """単語とその出現回数を保持するデータクラス。"""

    word: str
    count: int


def tokenize(text: str, *, case_insensitive: bool = True, min_length: int = 1) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト。
        case_insensitive: Trueの場合、小文字に変換して集計する。
        min_length: この長さ未満のトークンは除外する。

    Returns:
        トークンのリスト（順序は原文の出現順）。
    """
    # 英数字、アンダースコア、ハイフンを含むトークンに分割
    # 日本語などの連続文字列も1トークンとして扱う
    tokens = re.findall(r"[a-zA-Z0-9_\-]+|[\u3040-\u309F\u30A0-\u30FF\u4E00-\u9FFF]+|[^\s\W]+", text)

    if case_insensitive:
        tokens = [token.lower() for token in tokens]

    tokens = [token for token in tokens if len(token) >= min_length]
    return tokens
