    """

    単語は英数字・ハイフン・アポストロフィを含む文字列として扱われます。
    すべて小文字に変換されます。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 抽出された単語のリスト。
    """
    # 単語の定義: 英数字、ハイフン、アポストロフィを含む連続文字列
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return [word.lower() for word in words]
