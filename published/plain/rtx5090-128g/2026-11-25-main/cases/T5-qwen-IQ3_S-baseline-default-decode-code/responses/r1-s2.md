import argparse

from typing import List, Tuple, Dict


def load_text(filepath: str) -> str:
    """
    テキストファイルを読み込み、その内容を文字列として返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイル内容の文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み取りに失敗した場合。
    """
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {filepath}")
    except IOError as e:
        raise IOError(f"ファイルの読み取りに失敗しました: {e}")


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語リストに分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、全て小文字に変換する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む単語を抽出（一般的な英語の単語）
    words = re.findall(r"[a-z0-9]+(?:['-][a-z0-9]+)*", text)
    return words


def count_word_frequencies(words: List[str]) -> Dict[str, int]:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする辞書。
    """
    return dict(Counter(words))
