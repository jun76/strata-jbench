"""
word_freq.py - コマンドラインツール：テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する。

使い方:
    python word_freq.py <input_file> [--top N] [--case-insensitive] [--min-length M]

例:
    python word_freq.py sample.txt --top 10
    python word_freq.py sample.txt -t 20 -i
"""

import argparse

def tokenize(text: str, case_insensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_insensitive: 大文字小文字を区別しない場合 True
        min_length: 単語の最小文字数（これ未満の単語は除外）

    Returns:
        抽出された単語のリスト
    """
    # 英数字とアンダースコアのみを含む単語を抽出
    # 日本語などのマルチバイト文字も考慮する場合、より柔軟なパターンが必要だが、
    # ここでは一般的な英語単語中心の処理とする。必要に応じて正規表現を変更可能。
    pattern = r"[a-zA-Z0-9_]+"
    tokens = re.findall(pattern, text)
    
    if case_insensitive:
        tokens = [token.lower() for token in tokens]
    
    if min_length > 1:
        tokens = [token for token in tokens if len(token) >= min_length]
    
    return tokens


def count_words(text: str, case_insensitive: bool = False, min_length: int = 1) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_insensitive: 大文字小文字を区別しない場合 True
        min_length: 単語の最小文字数

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    tokens = tokenize(text, case_insensitive=case_insensitive, min_length=min_length)
    return Counter(tokens)
