"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位 N 件表示する。

Usage:
    python word_freq.py <file> [-n TOP_N] [--case-sensitive]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語リストに分割する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字・小文字を区別するかどうか

    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()
    # 単語として英数字とハイフン、アポストロフィを許容
    words = re.findall(r"[a-z0-9]+(?:[-'_][a-z0-9]+)*", text) if not case_sensitive else re.findall(r"[A-Za-z0-9]+(?:[-'_][A-Za-z0-9]+)*", text)
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字・小文字を区別するかどうか

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)

    Returns:
        (単語, 出現回数) のタプルリスト、上位順

        return []
