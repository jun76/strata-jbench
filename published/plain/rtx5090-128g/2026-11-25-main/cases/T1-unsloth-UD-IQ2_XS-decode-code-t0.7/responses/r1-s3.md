"""
Word Frequency Counter Command Line Tool.

This tool reads one or more text files, counts the frequency of each word,
and displays the top N most frequent words in a formatted table.
"""

import argparse
import re
from collections import Counter

from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercased words.

    Words are defined as sequences of alphanumeric characters and underscores.
    Punctuation and other characters are treated as separators.

    Args:
        text: Input text string.

    Returns:
        List of lowercased word tokens.
    """
    # Use regex to find all word characters (letters, digits, underscores)
    return re.findall(r'\w+', text.lower())


def count_words(text: str) -> Counter[str]:
    """
    Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        Counter object mapping each word to its frequency.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their counts.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top results to return.

    Returns:
        List of tuples (word, count) sorted by frequency descending,
        then alphabetically for ties.
    """
    # most_common returns elements ordered by count descending
    # For ties, we want alphabetical order, so we sort manually
    items = list(counter.items())
    # Sort by count descending, then by word ascending
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
