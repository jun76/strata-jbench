"""
word_counter.py

A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a formatted table.

Usage:
    python word_counter.py <input_file> -n <top_n>
    python word_counter.py <input_file> --top-n <top_n>

Example:
    python word_counter.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_file(filepath: str) -> str:
    """Read and return the contents of a text file.

    Args:
        filepath: Path to the input text file.

    Returns:
        The entire content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        OSError: If the file cannot be read.
    """
    path = Path(filepath)
    if not path.exists():
        raise FileNotFoundError(f"File not found: {filepath}")
    return path.read_text(encoding="utf-8")


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.

    Splits text into words using whitespace and non-alphanumeric characters,
    converts them to lowercase, and filters out empty strings.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase word tokens.
    """
    # Split on non-alphanumeric characters (keeping only letters and digits)
    words = re.split(r'[^a-zA-Z0-9]+', text)
    # Filter out empty strings and convert to lowercase
    return [word.lower() for word in words if word]


def count_words(text: str) -> Counter:
    """Count the frequency of each word in the text.

    Args:
        text: Input text string.

    Returns:
        A Counter mapping each word to its frequency.
    """
    tokens = tokenize(text)
    return Counter(tokens)
