"""

from __future__ import annotations

def tokenize(text: str) -> List[str]:
    """
    テキストから単語リストを抽出する。
    英数字とアンダースコアを含む語を単語として扱い、小文字に変換する。

    Args:
        text: 入力テキスト

    Returns:
        小文字化された単語のリスト
    """
    return re.findall(r'[a-zA-Z0-9_]+', text.lower())


def count_words(text: str) -> Counter[str]:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語 -> 出現回数のCounter
    """
    return Counter(tokenize(text))


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    頻度の上位N件の単語と出現回数を返す。
    同頻度の場合はアルファベット順でソートする。

    Args:
        counter: 単語頻度Counter
        n: 取得したい件数

    Returns:
        (単語, 頻度)のタプルのリスト（頻度降順、同頻度ならアルファベット昇順）
    """
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
