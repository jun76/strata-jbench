#!/usr/bin/env python3
"""テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール."""

from __future__ import annotations

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import Iterable


def tokenize(text: str) -> list[str]:
    """テキストを単語リストに分割する.

    大文字小文字を区別せず、非英数字文字を区切りとして扱う。
    例: "Hello, world! Hello." -> ["hello", "world", "hello"]

    Args:
        text: 入力テキスト.

    Returns:
        小文字化された単語のリスト.
    """
    import re
    # 英数字（および一部の言語対応のため unicode 文字）を単語として抽出
    return [word.lower() for word in re.findall(r'\w+', text)]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を Counter で返す.

    Args:
        text: 入力テキスト.

    Returns:
        単語 -> 出現回数の Counter.
    """
    return Counter(tokenize(text))


def top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """Counter から上位 n 件を (単語, 回数) のタプルリストで返す.

    回数が同じ場合は辞書順でソートする。

    Args:
        counter: 単語頻度 Counter.
        n: 取得する上位件数.

    Returns:
        (単語, 回数) のリスト（降順）.
    """
    # 回数降順、同順位は単語昇順でソート
    return sorted(
        counter.items(),
        key=lambda item: (-item[1], item[0]),
    )[:n]
