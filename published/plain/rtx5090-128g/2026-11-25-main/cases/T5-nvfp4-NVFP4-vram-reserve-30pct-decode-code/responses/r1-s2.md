#!/usr/bin/env python3
"""
word_freq.py — テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

使用例:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt -n 5 --min-len 3 --case-sensitive
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, min_length: int = 1, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語リストにトークナイズする。

    Args:
        text: 入力テキスト。
        min_length: 単語の最小長（この長さ未満の単語は除外）。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に正規化。

    Returns:
        トークン化された単語リスト。
    """
    if not case_sensitive:
        text = text.lower()

    # 英数字のみを許容する簡単なトークナイザ
    # 句読点や記号は区切りとして扱う
    import re
    tokens = re.findall(r'[a-zA-Z0-9]+', text)

    # 最小長でフィルタ
    if min_length > 1:
        tokens = [t for t in tokens if len(t) >= min_length]

    return tokens
