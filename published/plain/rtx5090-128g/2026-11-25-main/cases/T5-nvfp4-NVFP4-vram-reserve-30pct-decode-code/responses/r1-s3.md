#!/usr/bin/env python3
"""
Command-line tool to count word frequencies from text files and display the top N words.

Usage:
    python word_freq.py [options] <file> [<file> ...]
"""

import argparse
import sys
from collections import Counter
from typing import List, Dict


def read_file(filepath: str) -> str:
    """Read a text file and return its content.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words, stripping punctuation.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase words with punctuation stripped.
    """
    words = []
    for word in text.lower().split():
        # Strip leading and trailing punctuation
        cleaned = word.strip('.,!?;:\'"()-[]{}')
        if cleaned:
            words.append(cleaned)
    return words


def count_words(words: List[str]) -> Counter:
    """Count word frequencies.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[tuple]:
    """Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
