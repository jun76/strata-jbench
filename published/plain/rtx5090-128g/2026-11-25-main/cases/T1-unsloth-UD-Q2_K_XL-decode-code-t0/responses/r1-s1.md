"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度を計算して上位 N 件を表形式で表示するコマンドラインツール。

from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    英数字の連続を単語とみなし、すべて小文字に変換する。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 抽出された単語のリスト。
    """
    # 英数字の連続を単語として抽出し、小文字に変換
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words (List[str]): 単語のリスト。

    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)
