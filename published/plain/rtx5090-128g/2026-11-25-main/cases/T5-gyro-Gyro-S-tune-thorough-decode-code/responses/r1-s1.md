#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルの単語出現頻度を集計し、上位N件を表形式で表示する。

Usage:
    python word_freq.py <file> [-n N]

Arguments:
    file: 対象のテキストファイルパス
    -n: 表示する上位単語数 (デフォルト: 10)
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。

    Args:
        text: 対象のテキスト文字列

    Returns:
        単語のリスト (小文字化)
    """
    # 英数字の単語のみ抽出し、小文字に正規化
    words = []
    for token in text.split():
        # 句読点や特殊文字を除去
        cleaned = token.strip(".,!?;:\"'()[]{}")
        if cleaned:
            words.append(cleaned.lower())
    return words


def count_frequency(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を集計する。

    Args:
        text: 対象のテキスト文字列

    Returns:
        単語をキー、出現回数値を値とするCounter
    """
    words = tokenize(text)
    return Counter(words)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """上位N件の単語と頻度を取得する。

    Args:
        counter: 単語頻度のCounter
        n: 上位件数

    Returns:
        (単語, 頻度) のタプルリスト
    """
    return counter.most_common(n)
