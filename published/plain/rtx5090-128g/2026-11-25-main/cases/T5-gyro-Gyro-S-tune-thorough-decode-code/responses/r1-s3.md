"""
wordfreq.py - コマンドラインツール：テキストファイルの単語出現頻度を上位 N 件で表形式表示する。

使い方:
    python wordfreq.py [options] <file>

オプション:
    --top N         表示する上位単語数（デフォルト: 10）
    --case          大文字小文字を区別する（デフォルト: False）
    --sort {freq,word}  ソート基準（デフォルト: freq）
"""

import argparse

from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか

    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()
    # 単語はアルファベット・数字・ハイフン・アポストロフィを含む連続文字列
    # 簡易的な分割: 空白と句読点で分割
    words: List[str] = []
    for line in text.splitlines():
        for word in line.split():
            # 句読点を除去（先頭・末尾の . , ; : ! ? 等）
            cleaned = word.strip()
            # 句読点で囲まれた単語を抽出
            # 簡易: 非アルファベット・数字・ハイフン・アポストロフィを除去
            filtered = "".join(ch for ch in cleaned if ch.isalnum() or ch in "-'")
            if filtered:
                words.append(filtered)
    return words


def count_words(words: List[str]) -> Counter[str]:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 頻度の Counter
    """
    return Counter(words)
