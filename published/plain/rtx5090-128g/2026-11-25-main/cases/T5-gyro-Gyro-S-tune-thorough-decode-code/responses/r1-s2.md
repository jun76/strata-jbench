def read_words_from_file(filepath: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    指定されたファイルを読み込み、単語のリストを返す。

    Args:
        filepath: 読み込むテキストファイルのパス。
        case_sensitive: True の場合、大文字と小文字を区別する。
            False の場合、すべて小文字に正規化する。
        min_length: この長さ未満の単語は除外する。

    Returns:
        単語のリスト（抽出された順）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        OSError: 読み込みエラーが発生した場合。
    """
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            content = f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except OSError as e:
        raise OSError(f"Error reading file: {e}")

    # 単語を抽出: 空白・改行・句読点などで分割
    # 簡易的な分割: 空白区切りのトークンを扱い、句読点を除去する
    import re
    # 単語として扱うのは英数字・下線の連続
    tokens = re.findall(r"[a-zA-Z0-9_]+", content)

    if not case_sensitive:
        tokens = [token.lower() for token in tokens]

    # min_length でフィルタ
    tokens = [token for token in tokens if len(token) >= min_length]

    return tokens
