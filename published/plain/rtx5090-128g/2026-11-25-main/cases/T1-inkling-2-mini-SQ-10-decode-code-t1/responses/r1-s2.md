好的。提供一个 Python 命令行工具（读取文本文件，统计词频，输出前 N 个词的表格），使用 argparse、类型注解与 docstring，并附 pytest 单元测试。

```python
#!/usr/bin/env python3
"""CLI tool: word frequency from a text file."""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List


def count_word_frequencies(text: str) -> Counter:
    """Return a Counter of word frequencies for a text."""
    words = re.findall(r"[A-Za-z0-9]+", text.lower())
    return Counter(words)


def render_table(counts: Counter, n: int) -> str:
    """Render top-N counts as a simple text table."""
    top = counts.most_common(n)
    lines = [f"{'word':<20} {'count':>10}"]
    lines.append(f"{'-'*20} {'-'*10}")
    for word, count in top:
        lines.append(f"{word:<20} {count:>10}")
    return "\n".join(lines)


def main(argv: List[str] | None = None) -> int:
    """CLI entry point."""
    parser = argparse.ArgumentParser(
        description="Read a text file and display top N word frequencies."
    )
    parser.add_argument("path", type=Path, help="Path to text file")
    parser.add_argument("-n", type=int, default=10, help="Number of top words")
    args = parser.parse_args(argv)

    text = args.path.read_text(encoding="utf-8")
    counts = count_word_frequencies(text)
    print(render_table(counts, args.n))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
