### 1. 本体コード `wordfreq.py`

```python
#!/usr/bin/env python3
"""
テキストファイルから単語の出現頻度を集計し、上位 N 件を表示する。

入力: テキストファイルのパス、出力件数 N
出力: 上位 N 件の単語と出現回数を表形式で表示
"""

from __future__ import annotations

import argparse
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def count_words(text: str) -> Counter:
    """
    テキストを単語に分割して出現頻度を集計する。

    Args:
        text: 読み込んだテキスト

    Returns:
        単語と出現回数の辞書
    """
    words = text.lower().split()
    return Counter(words)


def read_text(path: Path) -> str:
    """
    テキストファイルを読み込む。

    Args:
        path: ファイルパス

    Returns:
        読み込んだテキスト
    """
    return path.read_text(encoding="utf-8")


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の高い上位 N 件を返す。

    Args:
        counter: 単語出現頻度の辞書
        n: 表示件数

    Returns:
        上位 N 件の (単語, 出現回数) のリスト
    """
    if n <= 0:
        return []
    return counter.most_common(n)
