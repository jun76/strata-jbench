import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple

def count_words(text: str) -> Counter:
    """テキストから単語を抽出し、出現頻度を集計する。"""
    words = re.findall(r"[A-Za-z]+", text.lower())
    return Counter(words)

def top_words(counts: Counter, n: int) -> List[Tuple[str, int]]:
    """集計結果から上位 N 件を返す。"""
    return counts.most_common(n)

def load_text(file_path: Path) -> str:
    """テキストファイルを読み込む。"""
    return file_path.read_text(encoding="utf-8")

def display_table(items: List[Tuple[str, int]]) -> None:
    """表形式で表示する。"""
    print("| 単語 | 頻度 |")
    print("|------|------|")
    for word, freq in items:
        print(f"| {word} | {freq} |")

def main() -> None:
    """コマンドラインツールのエントリポイント。"""
    parser = argparse.ArgumentParser(description="テキストファイルの単語出現頻度を上位 N 件表示する")
    parser.add_argument("file", type=Path, help="読み込むテキストファイル")
    parser.add_argument("--n", type=int, default=10, help="表示する上位件数")
    args = parser.parse_args()

    text = load_text(args.file)
    counts = count_words(text)
    items = top_words(counts, args.n)
    display_table(items)

if __name__ == "__main__":
    main()
