#!/usr/bin/env python3
"""コマンドラインツール: テキストファイルから単語頻度を集計し上位 N 件を表示する。"""

import argparse
import sys
from collections import Counter
from pathlib import Path


def count_words(text: str) -> Counter[str]:
    """テキスト中の単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト。空白区切りの単語を大文字小文字を区別せず集計する。

    Returns:
        Counter[str]: 単語から出現回数へのマッピング。
    """
    words = [w.strip().lower() for w in text.split()]
    return Counter(w for w in words if w)


def top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """出現頻度上位 N 件の (word, count) のリストを返す。

    Args:
        counter: 単語頻度カウント。
        n: 取得する件数。

    Returns:
        list[tuple[str, int]]: 頻度降順・同頻度時は辞書順の上位 n 件。
    """
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(items: list[tuple[str, int]], width: int = 40) -> str:
    """(word, count) のリストを表形式の文字列に整形する。

    Args:
        items: 表示する (word, count) のリスト。
        width: 単語列の最大幅。

    Returns:
        str: 表形式のテキスト。
    """
    if not items:
        return ""
    header = f"{'Rank':>5}  {'Word':<{width}}  {'Count':>5}"
    sep = "-" * (5 + width + 5 + 2)
    rows = [header, sep]
    for rank, (word, count) in enumerate(items, 1):
        truncated = word[:width]
        rows.append(f"{rank:>5}  {truncated:<{width}}  {count:>5}")
    return "\n".join(rows)
