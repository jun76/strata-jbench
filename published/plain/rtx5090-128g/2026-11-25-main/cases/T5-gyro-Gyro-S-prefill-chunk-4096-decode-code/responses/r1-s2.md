def read_file(filepath: str) -> str:
    """テキストファイルを読み込み、内容を返す.

    Args:
        filepath: ファイルパス

    Returns:
        ファイルの内容

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        IOError: 読み込みエラーの場合
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def count_words(text: str) -> Counter[str]:
    """テキストから単語の出現頻度を集計する.

    Args:
        text: 対象テキスト

    Returns:
        単語と出現回数の Counter

    Raises:
        ValueError: text が None の場合
    """
    if text is None:
        raise ValueError("テキストが None です")
    words = text.split()
    return Counter(words)
