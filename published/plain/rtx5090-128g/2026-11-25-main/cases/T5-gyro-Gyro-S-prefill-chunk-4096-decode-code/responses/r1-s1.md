import argparse

from collections import Counter
from typing import Dict, List, Tuple


def normalize_word(word: str, ignore_case: bool = False) -> str:
    """Normalize a word by stripping punctuation and optionally lowercasing."""
    # Strip common punctuation characters from both ends
    cleaned = word.strip(".,;:!?\"'()[]{}-_")
    if ignore_case:
        cleaned = cleaned.lower()
    return cleaned


def tokenize(text: str, ignore_case: bool = False) -> List[str]:
    """Tokenize text into words, normalizing each word."""
    words = []
    for token in text.split():
        normalized = normalize_word(token, ignore_case=ignore_case)
        if normalized:  # Skip empty strings after normalization
            words.append(normalized)
    return words


def count_words(text: str, ignore_case: bool = False) -> Dict[str, int]:
    """Count word frequencies in the given text."""
    words = tokenize(text, ignore_case=ignore_case)
    return dict(Counter(words))
