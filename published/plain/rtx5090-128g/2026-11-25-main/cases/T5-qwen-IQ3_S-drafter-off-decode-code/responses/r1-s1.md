def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語のリストに分割する。
    
    Args:
        text: 入力テキスト
        case_sensitive: Trueの場合、大文字小文字を区別する
        
    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()
    
    # 非英数字を空白に置換して分割（基本的なトークナイザー）
    # 日本語などのマルチバイト文字にも対応するため、より堅牢な正規表現を使うことも可能だが、
    # ここではシンプルな英数字ベースの分割を使用
    import re
    # 英数字（及びアンダースコア）を含むトークンを抽出
    tokens = re.findall(r'\b\w+\b', text)
    return tokens


def count_frequencies(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。
    
    Args:
        words: 単語のリスト
        
    Returns:
        Counterオブジェクト（単語 -> 出現回数）
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度上位N件の単語と回数を取得する。
    
    Args:
        counter: 単語頻度Counter
        n: 取得する件数
        
    Returns:
        (単語, 回数) のタプルのリスト（回数降順）
    """
    return counter.most_common(n)
