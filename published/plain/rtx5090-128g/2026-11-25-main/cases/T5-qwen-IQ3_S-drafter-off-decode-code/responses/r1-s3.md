#!/usr/bin/env python3
"""
word_count.py

テキストファイルを読み込み、単語の出現頻度を算出するツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def normalize_word(word: str) -> str:
    """
    単語を正規化する（小文字化、前後の非アルファベット文字の除去）。

    Args:
        word (str): 単語

    Returns:
        str: 正規化された単語
    """
    return word.lower().strip(".,;:!?\"'()[]{}")


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text (str): 入力テキスト

    Returns:
        List[str]: 正規化された単語のリスト
    """
    # 単語を抽出（アルファベット、数字、ハイフンを含む）
    words = re.findall(r"[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*", text)
    normalized = [normalize_word(w) for w in words if normalize_word(w)]
    return normalized


def count_words(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words (List[str]): 単語のリスト

    Returns:
        Counter: 単語とその出現回数
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 N 件を取得する。

    Args:
        counter (Counter): 単語の出現回数
        n (int): 上位件数

    Returns:
        List[Tuple[str, int]]: (単語, 出現回数) のリスト
    """
    return counter.most_common(n)
