#!/usr/bin/env python3
"""
word_freq.py: テキストファイルから単語出現頻度を集計し上位N件を表形式で表示するCLIツール.

Usage:
  python word_freq.py input.txt --top 10
  python word_freq.py input.txt --top 5 --sort by_freq
"""

import argparse

from collections import Counter
from pathlib import Path

# 単語分割規則: 文字・数字・underscore を 1 単語とみなす
WORD_PATTERN = re.compile(r"[A-Za-z0-9_]+")


def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストへ分割する.

    Args:
        text: 原文テキスト.

    Returns:
        小文字化された単語のリスト.
    """
    return [m.group().lower() for m in WORD_PATTERN.finditer(text)]


def count_words(text: str) -> Counter:
    """
    出現頻度を Counter として集計する.

    Args:
        text: 原文テキスト.

    Returns:
        {word: count} 形式の Counter.
    """
    return Counter(tokenize(text))


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位 n 件を [(word, count), ...] 形式で返す.

    Args:
        counter: Counter 形式の出現頻度.
        n: 上位件数.

    Returns:
        上位 n 件の (word, count) 列表.
    """
    return counter.most_common(n)
