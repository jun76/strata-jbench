"""
コマンドラインツール: テキストファイルから単語の出現頻度を集計し、
上位 N 件を表形式で表示する。

使い方:
    python wordfreq.py input.txt --top 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple

def read_words_from_file(filepath: str) -> List[str]:
    """ファイルを読み込み、空白区切りの単語リストを返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        単語のリスト。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        OSError: ファイル読み込み失敗の場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        content = f.read()
    # 空白で分割し、空文字を除く
    return [word for word in content.split() if word]

def count_words(words: List[str]) -> Counter[str]:
    """単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語とその出現回数の Counter。
    """
    return Counter(words)

def get_top_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """出現頻度の上位 N 件を (単語, 回数) のタプルで返す。

    Args:
        counter: 単語の Counter。
        n: 取得する上位件数。

    Returns:
        (単語, 回数) のタプルのリスト。回数が多い順にソートされる。
        同じ回数の単語は辞書順でソートされる。
    """
    # 回数降順、単語昇順でソート
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
