#!/usr/bin/env python3
"""
Text file word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N most frequent words
in a tabular format.
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def read_file(file_path: str) -> str:
    """Read the contents of a text file.

    Args:
        file_path: Path to the text file.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If there is an error reading the file.
    """
    with open(file_path, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Split text into words, removing punctuation and converting to lowercase.

    Args:
        text: The input text string.

    Returns:
        A list of words (lowercase, stripped of punctuation).
    """
    words = []
    # Split by whitespace and remove punctuation from each token
    for token in text.split():
        # Strip common punctuation characters
        cleaned = token.strip('.,!?;:'"()[]{}')
        if cleaned:
            words.append(cleaned.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count), sorted by frequency descending.
        If there are ties, words are sorted alphabetically.
    """
    # Sort by count descending, then by word ascending for tie-breaking
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
