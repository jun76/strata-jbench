"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を集計し、
上位 N 件を表形式で表示する。

使い方:
    python word_freq.py --file sample.txt --top 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。

    Args:
        text: 入力テキスト

    Returns:
        小書きされた単語のリスト
    """
    words = text.split()
    return [word.lower() for word in words]


def count_words(text: str) -> Counter[str]:
    """単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語 -> 出現回数の Counter
    """
    return Counter(tokenize(text))


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """上位 N 件の (単語, 出現回数) のリストを取得する。

    Args:
        counter: 単語 -> 出現回数の Counter
        n: 取得する件数

    Returns:
        (単語, 出現回数) のリスト（頻度順）
    """
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """アイテムのリストを表形式でフォーマットする。

    Args:
        items: (単語, 出現回数) のリスト

    Returns:
        表形式の文字列
    """
    if not items:
        return "No words found."

    # 列幅を計算
    word_width = max(len(word) for word, _ in items)
    count_width = max(len(str(count)) for _, count in items)

    header = f"{'Word':<{word_width}} {'Count':>{count_width}}"
    separator = f"{'-' * word_width} {'-' * count_width}"

    lines = [header, separator]
    for word, count in items:
        lines.append(f"{word:<{word_width}} {count:>{count_width}}")

    return "\n".join(lines)
