"""
word_freq.py - テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

使用方法:
    python word_freq.py INPUT_FILE [-n N] [--case-sensitive] [--min-length LEN]

例:
    python word_freq.py sample.txt -n 10
    python word_freq.py sample.txt --case-sensitive --min-length 3
"""

@dataclass(frozen=True)

    """単一の単語とその出現回数を保持する不変データクラス。"""
    word: str
    count: int


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> list[str]:
    """
    テキストを単語トークンに分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。False の場合、すべて小文字に変換。
        min_length: この長さ未満の単語は除外する。

    Returns:
        単語トークンのリスト。
    """
    # 英数字とハイフン・アポストロフィを含むトークンを抽出
    tokens = re.findall(r"[a-zA-Z0-9]+(?:['-][a-zA-Z0-9]+)*", text)
    if not case_sensitive:
        tokens = [t.lower() for t in tokens]
    if min_length > 1:
        tokens = [t for t in tokens if len(t) >= min_length]
    return tokens


def count_words(tokens: list[str]) -> Counter[str]:
    """
    単語トークンリストから出現頻度をカウントする。

    Args:
        tokens: 単語トークンのリスト。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    return Counter(tokens)
