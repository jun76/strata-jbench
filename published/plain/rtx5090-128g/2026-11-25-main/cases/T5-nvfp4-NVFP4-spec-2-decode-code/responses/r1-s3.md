def tokenize(text: str, case_insensitive: bool = True) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_insensitive: 大文字小文字を区別しない場合 True

    Returns:
        単語のリスト
    """
    if case_insensitive:
        text = text.lower()
    # 英数字とハイフンを含む単語を抽出（日本語は1文字ずつではなく、ここでは英単語を想定）
    # より汎用的には jieba 等の分割ライブラリを使うべきだが、ここでは簡易的な正規表現を使用
    words = re.findall(r'\b[a-z0-9-]+\b', text)
    return words

    Returns:
        単語と出現回数の Counter オブジェクト
    """
    words = tokenize(text, case_insensitive)
    return Counter(words)
