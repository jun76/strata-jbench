#!/usr/bin/env python3
"""Text frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_text_file(filepath: Path) -> str:
    """Read the contents of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Tokenize text into words.

    Args:
        text: Input text string.

    Returns:
        List of words (lowercased, alphanumeric characters only).
    """
    # Convert to lowercase and extract words
    words = []
    current_word = []
    for char in text:
        if char.isalnum():
            current_word.append(char.lower())
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))
    return words


def count_words(words: List[str]) -> Counter:
    """Count occurrences of each word.

    Args:
        words: List of words.

    Returns:
        Counter object with word counts.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        List of tuples (word, count), sorted by count descending.
    """
    return counter.most_common(n)
