#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度を集計し、
上位 N 件を表形式で表示するコマンドラインツール。

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        小文字化された単語のリスト
    """
    # 単語を抽出: 英数字の連続文字列
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の単語と頻度を取得する。
    
    Args:
        counter: 単語頻度 Counter
        n: 取得する上位件数
        
    Returns:
        (単語, 頻度) のタプルリスト
    """
    return counter.most_common(n)
