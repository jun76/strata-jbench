def read_file(path: str) -> str:
    """指定されたパスのテキストファイルを読み込み、その内容を返す。

    Args:
        path: 読み取るファイルのパス。

    Returns:
        ファイルの内容を表す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        OSError: ファイル読み取り中にエラーが発生した場合。
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def count_words(text: str) -> Counter:
    """テキスト内の単語の出現頻度をカウントする。

    単語は空白文字で区切られ、句読点（,.!?;:'"）は除去される。
    大文字・小文字は無視される（すべて小文字に変換される）。

    Args:
        text: 分析対象のテキスト。

    Returns:
        単語をキー、出現回数を値とする collections.Counter。
    """

    # 英数字と日本語の文字を単語として扱い、それ以外を区切りとする

    cleaned = re.sub(r"[^\w\s\u3040-\u30ff\u4e00-\u9fff]", " ", text)
    words = cleaned.split()
    # 大文字小文字を区別しないため、すべて小文字にする（日本語には影響しない）
    lower_words = [word.lower() for word in words if word]
    return Counter(lower_words)
