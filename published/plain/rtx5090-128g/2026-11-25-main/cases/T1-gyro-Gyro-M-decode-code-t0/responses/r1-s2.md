"""
word_freq.py - コマンドラインツール: テキストファイルから単語出現頻度を上位N件表示
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    英数字の連続を単語とし、それ以外を区切りとする。
    小文字化して正規化する。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト（小文字化済み）
    """
    # 英数字の連続を抽出し、小文字化
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_word_frequencies(text: str) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (word, count) を取得する。
    同頻度の場合は辞書順でソートする。

    Args:
        counter: 単語頻度の Counter
        n: 取得する件数

    Returns:
        上位 n 件の (word, count) のリスト
    """
    # 頻度降順、同頻度なら単語の辞書順
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]


def format_table(items: List[Tuple[str, int]]) -> str:
    """
    (word, count) のリストを表形式の文字列にフォーマットする。

    Args:
        items: (word, count) のリスト

    Returns:
        表形式の文字列
    """
    if not items:
        return ""

    # 列幅を計算
    max_word_len = max(len(word) for word, _ in items)
    max_count_len = max(len(str(count)) for _, count in items
