def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    if not intervals:
        return []

    sorted_intervals = sorted(intervals, key=lambda x: (x[0], x[1]))
    merged = []

    start, end = sorted_intervals[0]

    for s, e in sorted_intervals[1:]:
        if s <= end:
            if e > end:
                end = e
        else:
            merged.append([start, end])
            start, end = s, e

    merged.append([start, end])
    return merged
