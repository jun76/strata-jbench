#!/usr/bin/env python3
"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N most frequent
words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize the input text into lowercase words.

    Words are sequences of alphanumeric characters. All characters are lowercased.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens.
    """
    return [word.lower() for word in re.findall(r"[a-zA-Z0-9]+", text)]


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples for the top N most frequent words.
        If there are ties, the order among tied words is arbitrary.
    """
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """
    Format the word-frequency pairs as a fixed-width table.

    Args:
        rows: A list of (word, count) tuples.

    Returns:
        A formatted string representing the table.
    """
    if not rows:
        return "No words found."

    # Determine column widths
    max_word_len = max(len(word) for word, _ in rows)
    max_count_len = max(len(str(count)) for _, count in rows)

    # Set minimum widths
    word_col_width = max(max_word_len, 5)  # At least width of "Word"
    count_col_width = max(max_count_len, 5)  # At least width of "Count"
