word_freq.py - Text file word frequency counter.

Reads a text file, counts word occurrences, and displays the top N words
in a formatted table. Supports case-insensitive counting and basic

Example:
    $ python word_freq.py sample.txt --top 10
"""

import argparse

from collections import Counter

from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Args:
        text: Input text string.

    Returns:
        List of lowercase words with punctuation stripped.
    """
    # Convert to lowercase and extract words (sequences of alphanumeric characters)
    text = text.lower()
    words = re.findall(r'[a-z0-9]+', text)
    return words


def count_words(text: str) -> Counter:
    """
    Count word occurrences in text.

    Args:
        text: Input text string.

    Returns:
        Counter object mapping words to their occurrence counts.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top items to return.

    Returns:
        List of tuples (word, count) sorted by count descending, then by word ascending.
    """
    # Sort by count descending, then alphabetically for ties
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
