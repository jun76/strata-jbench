#!/usr/bin/env python3
"""コマンドラインツール：テキストファイルの単語出現頻度を上位 N 件で表形式表示。"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """テキストから単語を抽出し、出現頻度を集計する。

    Args:
        text: 入力テキスト。

    Returns:
        単語 -> 出現回数の Counter。
    """
    # 簡易的な単語分割: 空白区切りで、句読点を除去
    words = []
    for raw in text.split():
        # 句読点を除去して小文字化
        cleaned = "".join(ch for ch in raw if ch.isalnum())
        if cleaned:
            words.append(cleaned.lower())
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (word, count) を返す。

    Args:
        counter: 単語 -> 出現回数の Counter。
        n: 取得する件数。

    Returns:
        上位 n 件の (word, count) のリスト。
    """
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """(word, count) のリストを表形式の文字列に変換する。

    Args:
        items: (word, count) のリスト。

    Returns:
        表形式の文字列。
    """
    if not items:
        return "単語が見つかりません。"

    # 列幅を計算
    word_width = max(len(w) for w, _ in items)
    count_width = max(len(str(c)) for _, c in items)

    header_word = "単語".ljust(word_width)
    header_count = "回数".rjust(count_width)
    header_line = f"{header_word}  {header_count}"
    separator = "-" * len(header_line)

    lines = [header_line, separator]
    for word, count in items:
        lines.append(f"{word.ljust(word_width)}  {str(count).rjust(count_width)}")

    return "\n".join(lines
