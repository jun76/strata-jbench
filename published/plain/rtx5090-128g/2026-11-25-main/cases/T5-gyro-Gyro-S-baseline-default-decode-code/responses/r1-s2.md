"""
wordfreq.py - コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を
集計して上位 N 件を表形式で表示する。

使い方:
    python wordfreq.py FILE [--top N] [--case-sensitive]

例:
    python wordfreq.py sample.txt --top 5
"""

from __future__ import annotations

import argparse

import re
from collections import Counter
from pathlib import Path

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字と小文字を区別する。

    Returns:
        単語のリスト（正規表現で単語境界に分割）。
    """
    # 単語は英字・数字・ハイフン・アポストロフィで構成される
    pattern = r"[A-Za-z0-9'\-]+"
    words = re.findall(pattern, text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_words(words: List[str]) -> Counter[str]:
    """単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        Counter[str]: 各単語の出現回数。
    """
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """上位 N 件の (word, count) を返す。

    Args:
        counter: 単語の頻度カウンタ。
        n: 上位件数。

    Returns:
        (word, count) のリスト。頻度降順でソートされ、頻度が同じ場合は
        単語の辞書順でソートされる。
    """
    # 頻度降順、同じ頻度なら辞書順
    items = sorted(counter.items(), key=lambda item: (-item[1], item[0]))
    return items[:n]
