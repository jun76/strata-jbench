A command-line tool to read a text file, count word frequencies, and display
the top N most frequent words in a tabular format.

Usage:
    python word_counter.py <file> [--top N] [--case-sensitive]

Example:
    python word_counter.py sample.txt --top 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text string.
        case_sensitive: If False, words are converted to lowercase.

    Returns:
        A list of word tokens.
    """
    words = text.split()
    if not case_sensitive:
        words = [word.lower() for word in words]
    return words

def count_words(words: List[str]) -> Counter[str]:
    """
    Count the frequency of each word in the list.

    Args:
        words: A list of word tokens.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    return Counter(words)

def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words and their counts.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count), sorted by count descending.
    """
    return counter.most_common(n)
