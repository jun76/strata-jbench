word_freq: コマンドラインツールでテキストファイルを読み込み、単語出現頻度の上位 N 件を表形式で表示する。

使い方:
    python word_freq.py <file> [-n N]

オプション:
    -n, --top N   上位 N 件を表示 (デフォルト: 10)

import argparse
import re
from collections import Counter
from typing import TextIO


def tokenize(text: str) -> list[str]:
    """テキストから英単語のリストを抽出する。

    Args:
        text: 入力テキスト文字列

    Returns:
        小文字化した単語のリスト
    """
    return [word.lower() for word in re.findall(r'[a-zA-Z]+', text)]


def count_words(text: str) -> Counter:
    """テキスト中の単語出現頻度を数える。

    Args:
        text: 入力テキスト文字列

    Returns:
        単語と出現回数の Counter
    """
    return Counter(tokenize(text))


def top_n_words(counter: Counter, n: int) -> list[tuple[str, int]]:
    """Counter から上位 n 件の (word, count) タプルを返す。

    Args:
        counter: 単語出現頻度の Counter
        n: 取得する件数

    Returns:
        (word, count) のリスト (頻度降順)
    """
    return counter.most_common(n)
