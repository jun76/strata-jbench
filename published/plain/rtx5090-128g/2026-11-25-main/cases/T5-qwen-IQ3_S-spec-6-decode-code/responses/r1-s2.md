def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    単語はアルファベット・数字・ハイフンで構成されるとみなし、
    大文字小文字は区別しない（すべて小文字に変換）。
    
    Args:
        text: 入力テキスト
        
    Returns:
        抽出された単語のリスト（小文字化済み）
    """
    # アルファベット、数字、ハイフン、アポストロフィを単語として扱う
    # 例: "it's" -> "it's", "hello-world" -> "hello-world"
    words = re.findall(r"[a-zA-Z0-9'-]+", text)
    return [word.lower() for word in words]

    """

        text: 入力テキスト
        
    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    words = extract_words(text)
    return Counter(words)
