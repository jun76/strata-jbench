word_freq.py - テキストファイルから単語出現頻度を上位N件表示するCLIツール

Usage:
    python word_freq.py <file> [-n N] [--case-sensitive] [--min-len LEN]
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    テキストを単語リストに分割する。

    Args:
        text: 入力テキスト
        case_sensitive: Trueの場合、大文字小文字を区別する
        min_len: この長さ未満の単語を除外する

    Returns:
        単語のリスト
    """
    # 英数字とハイフン、アポストロフィを含むトークンを抽出
    tokens = re.findall(r"[a-zA-Z0-9'\-]+", text)
    
    if not case_sensitive:
        tokens = [t.lower() for t in tokens]
    
    # 最小長でフィルタ
    if min_len > 1:
        tokens = [t for t in tokens if len(t) >= min_len]
    
    return tokens


def count_frequencies(text: str, case_sensitive: bool = False, min_len: int = 1) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_sensitive: Trueの場合、大文字小文字を区別する
        min_len: この長さ未満の単語を除外する

    Returns:
        単語をキー、出現回数を値とするCounter
    """
    tokens = tokenize(text, case_sensitive=case_sensitive, min_len=min_len)
    return Counter(tokens)
