#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive] [--min-length LENGTH]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 単語の最小長（これ未満の単語は除外）。

    Returns:
        単語のリスト。
    """
    # 英数字とハイフン、アポストロフィを含むトークンを抽出
    # 例: "don't", "state-of-the-art" なども単語として扱う
    tokens = re.findall(r"[a-zA-Z0-9]+(?:['\-][a-zA-Z0-9]+)*", text)

    if not case_sensitive:
        tokens = [token.lower() for token in tokens]

    # 最小長でフィルタ
    tokens = [token for token in tokens if len(token) >= min_length]

    return tokens


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 単語の最小長。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    tokens = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(tokens)
