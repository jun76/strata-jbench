def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    if not intervals:
        return []
    
    # Sort intervals by start, then by end
    sorted_intervals = sorted(intervals, key=lambda x: (x[0], x[1]))
    
    merged = []
    current_start, current_end = sorted_intervals[0]
    
    for interval in sorted_intervals[1:]:
        next_start, next_end = interval
        
        # Check if the current interval overlaps or touches the next one
        if next_start <= current_end:
            # Merge: extend the end if the next interval's end is larger
            current_end = max(current_end, next_end)
        else:
            # No overlap: add the current merged interval and start a new one
            merged.append([current_start, current_end])
            current_start, current_end = next_start, next_end
    
    # Add the last merged interval
    merged.append([current_start, current_end])
    
    return merged
