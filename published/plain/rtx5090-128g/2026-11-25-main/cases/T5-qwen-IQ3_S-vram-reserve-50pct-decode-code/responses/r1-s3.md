#!/usr/bin/env python3

wordfreq.py

テキストファイルを読み込み、単語の出現頻度を計算して上位 N 件を表形式で出力するコマンドラインツール。

import argparse

def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    Args:
        text: 入力テキスト文字列
        
    Returns:
        正規化された単語のリスト（小文字、アルファベットのみ）
    """
    # 単語を抽出: アルファベット文字列のみを抽出し、小文字に変換
    words = re.findall(r'[a-zA-Z]+', text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。
    
    Args:
        words: 単語のリスト
        
    Returns:
        単語と頻度の Counter
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の単語と頻度を取得する。
    
    Args:
        counter: 単語頻度の Counter
        n: 取得する件数
        
    Returns:
        (単語, 頻度) のタプルのリスト、頻度降順・単語昇順でソート
    """
    # most_common(n) は頻度順だが、同頻度時の順序は不定なため
    # 明示的に (頻度降順, 単語昇順) でソートする
    sorted_items = sorted(
        counter.items(),
        key=lambda x: (-x[1], x[0])
    )
    return sorted_items[:n]
