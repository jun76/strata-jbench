def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するか
        min_length: 単語の最小長（これ未満の単語は除外）

    Returns:
        単語リスト
    """
    # 英数字とハイフンを単語文字として扱う
    words = re.findall(r'[a-zA-Z0-9]+(?:[-_][a-zA-Z0-9]+)*', text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度を集計する。

    Args:
        words: 単語リスト

    Returns:
        Counter オブジェクト
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位N件の単語と回数を取り出す。同じ回数の場合は辞書順で安定ソート。

    Args:
        counter: 単語と回数の Counter
        n: 取得する件数

    Returns:
        [(word, count), ...] のリスト
    """
    items = list(counter.items())
    # 回数の降順、同じ回数の場合は単語の昇順
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
