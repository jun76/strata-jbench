"""Command-line tool to read a text file and display the top N most frequent words."""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count the frequency of each word in the given text.

    Words are split by whitespace and converted to lowercase.

    Args:
        text: The input text string.

    Returns:
        A Counter object with word frequencies.
    """
    words = text.lower().split()
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from the counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) sorted by frequency descending.
    """
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """Format the word-frequency pairs into a table string.

    Args:
        items: A list of tuples (word, count).

    Returns:
        A formatted table string.
    """
    if not items:
        return "No words found."

    # Calculate column widths
    max_word_len = max(len(word) for word, _ in items)
    max_count_len = max(len(str(count)) for _, count in items)

    # Header
    header = f"{'Word':<{max_word_len}}  {'Count':>{max_count_len}}"
    separator = f"{'-' * max_word_len}  {'-' * max_count_len}"

    lines = [header, separator]
    for word, count in items:
        line = f"{word:<{max_word_len}}  {count:>{max_count_len}}"
        lines.append(line)

    return "\n".join(lines)
