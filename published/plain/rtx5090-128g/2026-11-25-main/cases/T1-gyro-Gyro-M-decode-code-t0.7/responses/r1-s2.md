Command-line tool to count word occurrences in a text file and display
the top N words in a table format.

def read_words_from_file(filepath: str) -> List[str]:
    """Read words from a text file, splitting on whitespace and lowercasing.

    Args:
        filepath: Path to the text file.

    Returns:
        List of lowercase words found in the file.
    """
    words: List[str] = []
    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            # Split on whitespace and strip punctuation-like characters
            for word in line.split():
                # Remove surrounding non-alphanumeric characters
                cleaned = word.strip(".,;:!\"'()[]{}?")
                if cleaned:
                    words.append(cleaned.lower())
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """Count occurrences of each word.

    Args:
        words: List of words to count.

    Returns:
        Counter mapping words to their occurrence counts.
    """
    return Counter(words)


def get_top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words.

    Args:
        counter: Counter of word frequencies.
        n: Number of top words to return.

    Returns:
        List of (word, count) tuples sorted by frequency descending.
    """
    return counter.most_common(n)
