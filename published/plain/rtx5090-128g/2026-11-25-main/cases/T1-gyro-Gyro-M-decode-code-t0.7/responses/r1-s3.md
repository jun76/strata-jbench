#!/usr/bin/env python3
"""
Command-line tool to count word frequencies in a text file and display
the top N most frequent words in a table format.

Usage:
    python word_counter.py <input_file> [-n N] [-t TOP]

Example:
    python word_counter.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize the input text into lowercase words.

    Words are defined as sequences of alphanumeric characters and underscores.
    All tokens are converted to lowercase.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens.
    """
    # Split on non-word characters, keeping only alphanumeric/underscore tokens
    tokens = re.findall(r'\w+', text.lower())
    return tokens


def count_words(text: str) -> Counter[str]:
    """
    Count the frequency of each word in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from a Counter.

    If there are ties in frequency, words are sorted alphabetically
    as a secondary sort key.

    Args:
        counter: A Counter object mapping words to their frequencies.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by descending frequency,
        then ascending word order for ties.
    """
    # Sort by descending count, then ascending word for ties
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
