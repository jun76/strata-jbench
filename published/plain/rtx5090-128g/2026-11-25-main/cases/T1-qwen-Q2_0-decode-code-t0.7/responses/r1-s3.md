"""Command-line tool to count word frequencies in a text file.

This module reads a text file, tokenizes it into words, counts the
frequency of each word, and prints the top N words in a table format.

Usage:
    python word_counter.py --input FILE --top N [--case-sensitive] [--min-len LEN]
"""

def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """Tokenize a string into a list of words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, lowercase all words before counting.
        min_len: Minimum length of a word to be included.

    Returns:
        A list of word tokens extracted from the text.

    Raises:
        ValueError: If min_len is less than 1.
    """
    if min_len < 1:
        raise ValueError("min_len must be at least 1")

    # Extract sequences of letters (including Unicode letters)
    pattern = r"[a-zA-Z\u0080-\uFFFF]+"
    words = re.findall(pattern, text)

    if not case_sensitive:
        words = [w.lower() for w in words]

    words = [w for w in words if len(w) >= min_len]
    return words
