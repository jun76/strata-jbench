def read_file(path: str) -> str:
    """Read the contents of a text file.

    Args:
        path: Path to the text file to read.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(path, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Tokenize text into words.

    Words are extracted by splitting on whitespace and non-alphanumeric
    characters, then lowercased.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Match alphanumeric sequences (words)
    words = re.findall(r'[a-zA-Z]+', text)
    return [word.lower() for word in words]


def count_words(text: str) -> Counter:
    """Count word frequencies in text.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    words = tokenize(text)
    return Counter(words)
