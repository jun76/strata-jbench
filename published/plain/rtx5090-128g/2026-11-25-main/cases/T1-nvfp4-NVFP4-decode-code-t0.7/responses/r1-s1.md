#!/usr/bin/env python3
"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度を上位 N 件まで表形式で表示する CLI ツール。

Usage:
    python word_freq.py <file_path> [-n N] [--case-sensitive]

Example:
    python word_freq.py sample.txt -n 5
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


# 単語を抽出するための正規表現パターン（英数字とアンダースコア）
_WORD_PATTERN = re.compile(r'\b[\w]+\b')


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語のリストを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、全て小文字に変換する。

    Returns:
        抽出された単語のリスト。
    """
    words = _WORD_PATTERN.findall(text)
    if not case_sensitive:
        words = [word.lower() for word in words]
    return words


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """
    単語のリストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    出現頻度が高い上位 N 件の単語と回数を取得する。

    Args:
        counter: 単語の出現頻度を格納した Counter。
        n: 取得する上位件数。

    Returns:
        (単語, 出現回数) のタプルのリスト（頻度が高い順）。
    """
    return counter.most_common(n)
