    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_insensitive: 大文字小文字を区別しない場合 True
        ignore_punctuation: 句読点や記号を無視する場合 True

    Returns:
        単語のリスト
    """
    if case_insensitive:
        text = text.lower()

    if ignore_punctuation:
        # 英数字とハイフン、アポストロフィを除くすべての文字を空白に置換
        text = re.sub(r"[^\w\s\-']", " ", text)
    else:
        # 空白で分割するだけ
        pass

    # 複数空白をまとめて分割
    words = text.split()
    return words

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とするCounter
    """
    return Counter(words)
