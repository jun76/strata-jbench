This module provides a command-line tool that reads a text file,
counts word occurrences, and displays the top N words in a table format.
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from pathlib import Path
from typing import List, Tuple


@dataclass
class WordCount:
    """Represents a word and its count.

    Attributes:
        word: The word string.
        count: The number of occurrences.
    """

    word: str
    count: int


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.

    Words are defined as sequences of alphanumeric characters (including
    underscores). The text is lowercased before tokenization.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase word tokens.

    Examples:
        >>> tokenize("Hello, World! Hello again.")
        ['hello', 'world', 'hello', 'again']
    """
    return re.findall(r"\b[\w]+\b", text.lower())


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in the given text.

    Args:
        text: The input text.

    Returns:
        A Counter mapping each word to its frequency.

    Examples:
        >>> count_words("apple banana apple")
        Counter({'apple': 2, 'banana': 1})
    """
    return Counter(tokenize(text))
