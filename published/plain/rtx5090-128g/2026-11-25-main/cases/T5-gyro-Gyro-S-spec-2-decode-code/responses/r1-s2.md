import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count word frequencies from the given text.

    Words are split by whitespace and normalized to lowercase.
    Punctuation is stripped from each word.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter mapping each word to its occurrence count.
    """
    words = []
    for word in text.split():
        # Strip leading/trailing punctuation from each word
        cleaned = word.strip(".,!?;:\"'()[]{}")
        if cleaned:
            words.append(cleaned.lower())
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most frequent words from the counter.

    Args:
        counter: A Counter mapping words to their counts.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples sorted by count descending,
        then by word ascending for ties.
    """
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(entries: List[Tuple[str, int]]) -> str:
    """Format word count entries as a table.

    Args:
        entries: A list of (word, count) tuples.

    Returns:
        A formatted table string with columns for rank, word, and count.
    """
    if not entries:
        return ""
    lines = ["Rank  Word       Count"]
    lines.append("-  " + "-" * 10 + "  " + "-" * 7)
    for rank, (word, count) in enumerate(entries, start=1):
        lines.append(f"{rank:>4}  {word:<10} {count:>7}")
    return "\n".join(lines)
