"""
Text frequency counter command line tool.

Reads a text file, counts word occurrences (case-insensitive), and displays
the top N most frequent words in a table format.
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_file(path: str) -> str:
    """Read the contents of a text file.

    Args:
        path: Path to the text file.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        OSError: If the file cannot be read.
    """
    return Path(path).read_text(encoding="utf-8")


def tokenize(text: str) -> List[str]:
    """Tokenize text into words, lowercased.

    Words are split on whitespace. Punctuation is stripped from each word.

    Args:
        text: The input text.

    Returns:
        A list of lowercase words.
    """
    words = []
    for token in text.split():
        # Strip common punctuation characters
        cleaned = token.strip(".,!?;:\"'()[]{}")
        if cleaned:
            words.append(cleaned.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter mapping words to their counts.
    """
    return Counter(words)
