#!/usr/bin/env python3
"""Command-line tool to read a text file and display the top N most frequent words.

This module provides a simple CLI tool that:
1. Reads a text file specified via command-line arguments.
2. Tokenizes the text into words (lowercased, stripped of punctuation).
3. Counts the frequency of each word.
4. Displays the top N most frequent words in a table format.

Usage:
    python word_freq.py <input_file> [-n TOP_N] [-v]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words, stripping non-alphanumeric characters.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Extract words using regex: sequences of alphanumeric characters
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter[str]:
    """Count the frequency of each word.

    Args:
        words: A list of words to count.

    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    return Counter(words)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words from a counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to retrieve.

    Returns:
        A list of tuples (word, count), sorted by frequency descending.
        If counts are tied, words are sorted alphabetically for deterministic output.
    """
    # Sort by count descending, then by word ascending for tie-breaking
    items = sorted(counter.items(), key=lambda item: (-item[1], item[0]))
    return items[:n]
