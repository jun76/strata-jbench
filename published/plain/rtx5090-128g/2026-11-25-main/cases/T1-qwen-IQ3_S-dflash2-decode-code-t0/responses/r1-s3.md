#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度を計算し、
上位 N 件の単語とその頻度を表形式で出力するコマンドラインツール。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    単語は英数字とハイフンで構成される文字列とし、
    大文字小文字は無視して小文字に変換する。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 抽出された単語のリスト。
    """
    # 正規表現で単語を抽出: 英数字とハイフンを含む連続文字列
    # \b は単語境界、[\w-]+ は単語文字（英数字、アンダースコア、ハイフン）
    # ただし、アンダースコアは単語の一部として扱い、ハイフンも単語の一部として扱う
    # より厳密に「単語」を定義する場合、アンダースコアを除外することもあるが、
    # ここでは一般的な単語抽出として \w+ を使用しつつ、ハイフン付き単語も考慮する。
    # 簡潔さのため、\w+ を使用し、必要に応じて調整可能。
    # ここでは、英数字とハイフンを含む単語を抽出する。
    words = re.findall(r"[a-zA-Z0-9-]+", text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度を計算する。

    Args:
        words (List[str]): 単語のリスト。

    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)
