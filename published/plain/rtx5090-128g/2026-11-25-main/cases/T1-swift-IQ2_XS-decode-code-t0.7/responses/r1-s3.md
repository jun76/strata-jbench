#!/usr/bin/env python3
"""wordcount: Count word frequencies in a text file and display top N entries as a table."""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def read_file(path: str) -> str:
    """Read the contents of a text file.

    Args:
        path: Path to the text file.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def count_words(text: str) -> Counter[str]:
    """Count the frequency of each word in the text.

    Words are defined as sequences of alphanumeric characters.
    Case-insensitive counting.

    Args:
        text: The input text.

    Returns:
        A Counter mapping lowercase words to their frequency.
    """
    import re
    words = re.findall(r"[a-zA-Z0-9]+", text)
    return Counter(w.lower() for w in words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]]) -> str:

    Returns:
        A formatted string with aligned columns.
    """

        return ""
