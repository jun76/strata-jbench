#!/usr/bin/env python3
"""
wordfreq.py — 単語出現頻度を上位 N 件、表形式で表示する CLI ツール。

Usage:
    python wordfreq.py --file input.txt --top 10
    python wordfreq.py -f input.txt -n 5 --case-sensitive
"""

from __future__ import annotations

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語リストにトークン化する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
                        False の場合、すべて小文字化して区別しない。

    Returns:
        単語のリスト。句読点などの非英数字は単語の境界として扱う。
    """
    # 句読点・記号を空白に置換して単語分割
    import re
    # 英数字とアンダースコアを単語として保持、それ以外を区切り
    pattern = r'[A-Za-z0-9]+'
    tokens = re.findall(pattern, text)
    if not case_sensitive:
        tokens = [t.lower() for t in tokens]
    return tokens


def count_frequencies(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    テキスト中の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。

    Returns:
        単語 -> 出現回数の Counter。
    """
    tokens = tokenize(text, case_sensitive)
    return Counter(tokens)
