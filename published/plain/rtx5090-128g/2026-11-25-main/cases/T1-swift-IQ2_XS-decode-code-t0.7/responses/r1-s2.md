import argparse
import sys
from collections import Counter
from typing import List, Tuple


def read_text_file(filepath: str) -> str:
    """
    指定されたファイルパスからテキストを読み込み、小文字化して単語リストに変換する。

    Args:
        filepath (str): テキストファイルのパス。

    Returns:
        str: ファイルの内容全体。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み取りに失敗した場合。
    """
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except IOError as e:
        raise IOError(f"Error reading file {filepath}: {e}")


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    単語はアルファベット・数字・アンダースコアで構成される連続した文字列とする。
    大文字小文字は無視して小文字に統一する。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 単語のリスト。
    """
    import re
    # 単語パターン: 英数字とアンダースコアの連続
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度を集計する。

    Args:
        words (List[str]): 単語のリスト。

    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)
