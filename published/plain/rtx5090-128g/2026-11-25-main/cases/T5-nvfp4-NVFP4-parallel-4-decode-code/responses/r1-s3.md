_WORD_PATTERN = re.compile(r"[a-zA-Z0-9_'\-]+")


def tokenize(text: str, *, case_insensitive: bool = True) -> list[str]:
    """テキストから単語トークンのリストを抽出する.

    Returns:
        抽出された単語のリスト (順序は出現順).
    """
    if case_insensitive:
        text = text.lower()
    return _WORD_PATTERN.findall(text)
