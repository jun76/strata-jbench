word_freq.py

テキストファイルを読み込み、単語の出現頻度をカウントし、

使い方:

from collections import Counter
from typing import Dict, List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語のリストを抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合、大文字小文字を区別する

    Returns:
        単語のリスト（正規表現で英数字の連続を抽出）
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字の連続を単語として抽出（日本語等の文字は含まれない）
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return words


def count_words(words: List[str]) -> Dict[str, int]:
    """
    単語リストから各単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とする辞書
    """
    return dict(Counter(words))
