def read_text(file_path: Path) -> str:
    """
    テキストファイルを読み込み、内容を返す。

    Args:
        file_path: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        OSError: ファイルの読み取りに失敗した場合。
    """
    return file_path.read_text(encoding="utf-8")


def tokenize(text: str, case_insensitive: bool = True) -> List[str]:
    """
    テキストから単語を抽出する。

    単語は英数字（およびハイフン、アポストロフィを含む一部文字）の
    連続として定義される。日本語などの非ASCII文字は個別の文字として
    扱うのではなく、ここでは英単語を想定している。

    Args:
        text: 解析対象のテキスト。
        case_insensitive: True の場合、単語を小文字に正規化する。

    Returns:
        抽出された単語のリスト（出現順）。
    """
    # 英数字、ハイフン、アポストロフィを含む単語パターン
    pattern = re.compile(r"[a-zA-Z0-9]+(?:[-'][a-zA-Z0-9]+)*")
    words = pattern.findall(text)
    if case_insensitive:
        words = [word.lower() for word in words]
    return words
