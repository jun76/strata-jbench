#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込んで単語の出現頻度を集計し、
上位 N 件を表形式で表示する。

使い方:
    python word_count.py <file1> <file2> ... [-n TOP_N] [--min-length MIN_LEN]

例:
    python word_count.py sample.txt -n 10
    python word_count.py file1.txt file2.txt -n 20 --min-length 3
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple, Dict, Union


def tokenize(text: str, min_length: int = 1) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト文字列。
        min_length: 単語の最小長さ（1以上）。

    Returns:
        小文字化された単語トークンのリスト。
    """
    tokens = re.findall(r'[a-zA-Z0-9]+', text)
    return [t.lower() for t in tokens if len(t) >= min_length]


def count_words(text: str, min_length: int = 1) -> Counter[str]:
    """
    テキスト内の単語出現回数をカウントする。

    Args:
        text: 入力テキスト文字列。
        min_length: 単語の最小長さ（1以上）。

    Returns:
        単語 -> 出現回数 の Counter オブジェクト。
    """
    tokens = tokenize(text, min_length)
    return Counter(tokens)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の単語とその出現回数を取得する。

    Args:
        counter: 単語 -> 出現回数 の Counter。
        n: 取得する上位件数。

    Returns:
        (単語, 出現回数) のリスト（出現回数降順）。
    """
    return counter.most_common(n)
