def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """
    単語を正規化する。

    Args:
        word: 入力された単語。
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        正規化された単語（通常は小文字）。
    """
    if not case_sensitive:
        return word.lower()
    return word


def tokenize(text: str) -> list[str]:
    """
    テキストから単語のリストを取り出す。

    正規表現 \\b[\\w]+\\b を使用して、アルファベット・数字・アンダースコア
    で構成されるトークンを抽出する。

    Args:
        text: 解析対象のテキスト。

    Returns:
        抽出された単語のリスト（順序保持）。
    """
    return re.findall(r'\b[\w]+\b', text)
