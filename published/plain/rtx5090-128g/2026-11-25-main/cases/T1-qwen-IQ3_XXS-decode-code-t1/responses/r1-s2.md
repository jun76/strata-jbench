#!/usr/bin/env python3
"""
単語出現頻度集計ツール

テキストファイルを読み込み、単語の出現頻度を集計し、上位 N 件を表形式で出力する。

Usage:
    python word_freq.py [input_file] [--top N] [--case-sensitive]
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト文字列
        case_sensitive: 大文字小文字を区別するか (True: 区別, False: 大文字化して統一)

    Returns:
        単語のリスト
    """
    # 英数字、ハイフン、アポストロフィを含む単語を抽出
    # 例: "don't" -> "don't", "e-mail" -> "e-mail"
    words = re.findall(r"[A-Za-z0-9'\-]+", text)

    if not case_sensitive:
        words = [word.lower() for word in words]

    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数 の Counter オブジェクト
    """
    return Counter(words)
