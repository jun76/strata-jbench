#!/usr/bin/env python3
"""
wordfreq.py - テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示するツール。

Usage:
    python wordfreq.py <file> [-n N]
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    英数字の連続を単語とし、大文字小文字は区別しない（小文字に変換）。
    """
    words: List[str] = []
    current_word: List[str] = []
    for char in text:
        if char.isalnum():
            current_word.append(char.lower())
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    if current_word:
        words.append(''.join(current_word))
    return words


def count_words(text: str) -> Counter:
    """
    テキスト中の単語出現頻度を Counter で返す。
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から出現頻度上位 n 件の (word, count) のリストを返す。
    頻度が同じ場合は辞書順（アルファベット順）でソートする。
    """

    # 明示的に (count, word) でソートして安定性を確保する。
    items = list(counter.items())
    # 頻度の降順、単語の昇順
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
