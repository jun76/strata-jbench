def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、すべて小文字に変換する。
        min_length: 単語の最小長。これより短い単語は除外する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()

    # 英数字とアンダースコアを単語の文字として扱い、それ以外を区切りとする
    # 簡易的なトークナイザ: 英数字の連続を単語とみなす
    words: List[str] = []
    current_word: List[str] = []
    for char in text:
        if char.isalnum() or char == '_':
            current_word.append(char)
        else:
            if current_word:
                word = ''.join(current_word)
                if len(word) >= min_length:
                    words.append(word)
                current_word = []
    # 末尾の単語
    if current_word:
        word = ''.join(current_word)
        if len(word) >= min_length:
            words.append(word)

    return words
