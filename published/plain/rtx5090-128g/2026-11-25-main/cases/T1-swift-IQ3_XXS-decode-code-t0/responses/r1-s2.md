#!/usr/bin/env python3
"""
word_freq.py: テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    大文字小文字を区別せず、アルファベットと数字のみのトークンを抽出する。
    """
    # 簡易的なトークナイザ: 英数字の連続を単語とみなす
    import re
    return [word.lower() for word in re.findall(r'[a-zA-Z0-9]+', text)]


def count_words(text: str) -> Counter:
    """
    テキスト中の単語出現頻度を Counter で返す。
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から出現頻度上位 n 件の (word, count) のリストを返す。
    頻度が同じ場合は辞書順でソートする。
    """
    # most_common は頻度順だが、同頻度時の順序は保証されないため、明示的にソート
    items = list(counter.items())
    # 頻度降順、次に単語の辞書順
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
