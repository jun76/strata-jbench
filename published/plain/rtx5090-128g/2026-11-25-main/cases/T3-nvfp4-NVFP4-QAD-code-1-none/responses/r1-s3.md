def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    """
    Merge all overlapping or touching closed intervals and return them sorted by start.
    
    Args:
        intervals: A list of intervals, where each interval is a list [start, end].
        
    Returns:
        A list of merged, non-overlapping intervals sorted by start.
    """
    if not intervals:
        return []
    
    # Create a sorted copy of intervals to avoid modifying the original
    sorted_intervals = sorted(intervals, key=lambda x: x[0])
    
    merged = []
    current_start, current_end = sorted_intervals[0][0], sorted_intervals[0][1]
    
    for i in range(1, len(sorted_intervals)):
        next_start, next_end = sorted_intervals[i][0], sorted_intervals[i][1]
        
        # If the next interval overlaps or touches the current interval
        if next_start <= current_end:
            # Extend the current interval's end
            current_end = max(current_end, next_end)
        else:
            # No overlap, add the current interval to merged and start a new one
            merged.append([current_start, current_end])
            current_start, current_end = next_start, next_end
    
    # Add the last interval
    merged.append([current_start, current_end])
    
    return merged
