テキストファイルを読み込み、単語の出現頻度を上位 N 件を表形式で表示する CLI ツール。

    --min-length: 単語の最小長（デフォルト: 1）

Output:
    表形式で上位 N 件の単語と出現回数を出力する
"""

import argparse

def read_file(path: Path) -> str:
    """ファイルを読み込んで文字列を返す。

    Args:
        path: ファイルパス

    Returns:
        ファイルの内容

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        IOError: 読み込み失敗の場合
    """
    try:
        return path.read_text(encoding="utf-8")
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {path}")
    except IOError as e:
        raise IOError(f"ファイルの読み込み失敗: {e}")


def tokenize(text: str, ignore_case: bool, min_length: int) -> List[str]:
    """テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        ignore_case: True の場合、単語を小文字に変換する
        min_length: 単語の最小長。これ未満の単語は除外する

    Returns:
        単語のリスト
    """
    words = []
    for word in text.split():
        # 句読点を除去
        cleaned = word.strip(".,!?;:\"'()[]{}<>")
        if not cleaned:
            continue
        if ignore_case:
            cleaned = cleaned.lower()
        if len(cleaned) >= min_length:
            words.append(cleaned)
    return words
