#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を
集計し、上位 N 件を表形式で表示する。

Usage:
    python word_freq.py <filepath> [--top N]

Example:
    python word_freq.py sample.txt --top 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    単語はアルファベット・数字・ハイフン・アポストロフィで構成される。
    """
    words = []
    current = []
    for ch in text:
        if ch.isalnum() or ch == '-' or ch == "'":
            current.append(ch)
        else:
            if current:
                words.append(''.join(current))
                current = []
    if current:
        words.append(''.join(current))
    return words


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度を Counter で返す。
    """
    words = tokenize(text)
    return Counter(words)


def format_table(counts: Counter, top_n: int) -> str:
    """
    上位 N 件の単語と出現回数を表形式で整形して返す。
    """
    most_common = counts.most_common(top_n)
    if not most_common:
        return "No words found."

    # カラム幅計算
    word_width = max(len(word) for word, _ in most_common)
    count_width = max(len(str(count)) for _, count in most_common)

    # ヘッダー
    header = f"{'Word':<{word_width}} {'Count':>{count_width}}"
    separator = f"{'-' * word_width} {'-' * count_width}"

    lines = [header, separator]
    for word, count in most_common:
        lines.append(f"{word:<{word_width}} {count:>{count_width}}")

    return '\n'.join(lines)
