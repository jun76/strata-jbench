```python
"""
テキストファイルを読み込み、単語出現頻度を上位 N 件を表形式で表示する CLI ツール。

使い方:
    python wordfreq.py <file> [--top N] [--case-sensitive] [--min-length L]

例:
    python wordfreq.py sample.txt --top 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import Dict, List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
        min_length: 単語の最小長。これより短い単語は除外される。

    Returns:
        単語のリスト。
    """
    # 英数字とハイフン/アポストロフィを含むトークンを抽出
    # \w+ は英数字とアンダースコアを含む。ハイフンやアポストロフィは
    # 単語の一部として扱いたいので、パターンを調整する。
    pattern = r"[A-Za-z0-9]+(?:[-'][A-Za-z0-9]+)*"
    words = re.findall(pattern, text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    # 最小長でフィルタ
    words = [w for w in words if len(w) >= min_length]
    return words


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Dict[str, int]:
    """
    テキスト内の単語出現頻度を集計する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
        min_length: 単語の最小長。

    Returns:
        単語 -> 出現回数の辞書。
    """
    words = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    counter = Counter(words)
    return dict(counter)
