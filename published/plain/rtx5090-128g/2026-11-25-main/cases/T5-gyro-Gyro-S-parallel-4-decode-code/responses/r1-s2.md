wordfreq: テキストファイルから単語出現頻度を集計し、上位 N 件を

使い方:
    python wordfreq.py <ファイル> --top 10

import argparse

from collections import Counter
from typing import Dict, List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 分割対象のテキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
            False の場合、小文字化してから分割する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字・ハイフン・アポストロフィを含む連続文字列を単語として抽出
    pattern = r"[a-zA-Z0-9'\-]+"
    return re.findall(pattern, text)


def count_words(tokens: List[str]) -> Dict[str, int]:
    """
    単語の出現頻度を集計する。

    Args:
        tokens: 単語のリスト。

    Returns:
        単語 -> 出現回数 の辞書。
    """
    return Counter(tokens)


def get_top_n(counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    出現回数が上位 n 件の (単語, 回数) のタプルを返す。

    Args:
        counts: 単語 -> 出現回数 の辞書。
        n: 取得件数。

    Returns:
        上位 n 件の (単語, 回数) のリスト（回数の降順）。
    """
    return counts.most_common(n)
