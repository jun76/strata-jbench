コマンドラインツール: テキストファイルの単語出現頻度を上位N件で表示する。

使い方:
    python word_freq.py input.txt --top 10

オプション:
    --top N        上位N件を表示 (デフォルト: 10)
    --case-sensitive 大文字小文字を区別する (デフォルト: False, 小文字に統一)
"""

import argparse

from collections import Counter
from typing import List, Tuple


def read_words_from_file(filepath: str) -> List[str]:
    """
    ファイルを読み込み、単語のリストを返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        単語のリスト（大文字小文字は小文字に統一される）。
    """
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            content = f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {filepath}")
    except Exception as e:
        raise IOError(f"ファイル読み込みエラー: {e}")

    # 単語を抽出（空白で分割し、句読点等を除去する簡易処理）
    words = []
    for line in content.splitlines():
        # 句読点や特殊文字を除去して単語のみ抽出
        cleaned = "".join(ch if ch.isalnum() or ch == "_" else " " for ch in line)
        for word in cleaned.split():
            if word:
                words.append(word.lower())
    return words


def count_word_frequency(words: List[str]) -> Counter:
    """
    単語の出現頻度を集計する。

    Args:
        words: 単語のリスト。

    Returns:
        Counter オブジェクト（単語 -> 出現回数）。
    """
    return Counter(words)
