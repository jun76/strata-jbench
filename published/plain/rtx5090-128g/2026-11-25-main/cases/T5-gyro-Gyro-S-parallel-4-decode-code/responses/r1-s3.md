text_frequency.py

コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を集計し、

import argparse
import sys
from collections import Counter
from typing import Dict, List, Tuple


def tokenize_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字/小文字を区別する。
                       False の場合、すべて小文字に正規化する。

    Returns:
        抽出された単語のリスト。
    """
    # 簡易的な単語分割: 空白・改行・句読点で分割し、アルファベットのみを残す
    words = []
    current = []
    for ch in text:
        if ch.isalpha():
            current.append(ch)
        else:
            if current:
                words.append(''.join(current))
                current = []
    if current:
        words.append(''.join(current))

    if not case_sensitive:
        words = [w.lower() for w in words]

    return words


def count_frequencies(words: List[str]) -> Dict[str, int]:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語 -> 出現回数の辞書。
    """
    return Counter(words)
