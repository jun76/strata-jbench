#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度を計算し、
上位 N 件の単語とその頻度を表形式で出力するコマンドラインツール。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    Args:
        text: 入力テキスト文字列。
        
    Returns:
        小文字化された単語のリスト。
    """
    # 英数字とハイフンを含む単語を抽出（日本語などの非ASCII文字は除外）
    # 必要に応じて正規表現を調整可能
    words = re.findall(r"[a-zA-Z0-9'-]+", text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    単語リストから頻度カウンターを作成する。
    
    Args:
        words: 単語のリスト。
        
    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度カウンターから上位 N 件の単語と頻度を取得する。
    
    Args:
        counter: 単語頻度の Counter。
        n: 取得する上位件数。
        
    Returns:
        (単語, 頻度) のタプルのリスト。頻度が高い順にソート。
        頻度が同じ場合は単語の辞書順でソート。
    """
    # most_common は頻度順に返すが、同頻度時の順序は不定。
    # 安定した出力のため、頻度降順、単語昇順でソートする。
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items
