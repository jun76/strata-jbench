def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、すべて小文字に変換する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフンを含む単語を抽出（例: "don't" -> "don't", "e-mail" -> "e-mail"）
    words = re.findall(r"[a-z0-9]+(?:[-'][a-z0-9]+)*", text)
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、すべて小文字に変換する。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)
