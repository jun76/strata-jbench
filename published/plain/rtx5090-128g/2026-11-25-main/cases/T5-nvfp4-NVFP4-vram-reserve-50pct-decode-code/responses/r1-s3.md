#!/usr/bin/env python3
"""
word_freq.py - テキストファイルの単語出現頻度を上位 N 件表示する CLI ツール。

使い方:
    python word_freq.py input.txt
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt -n 5 --case-sensitive
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字・ハイフン・アポストロフィを含む連続文字列を単語として抽出
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語ごとの出現回数を保持する Counter。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度上位 N 件の単語を返す。

    Args:
        counter: 単語頻度を保持する Counter。
        n: 取得する件数。

    Returns:
        (単語, 出現回数) のタプルを頻度降順に並べたリスト。
    """
    return counter.most_common(n)
