A command-line tool that reads a text file, counts word frequencies,

"""

    Returns:
        Parsed arguments namespace.
    """
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and display the top N words."
    )
    parser.add_argument(
        "file",

        help="Path to the input text file.",
    )
    parser.add_argument(
        "--top",
        "-n",
        type=int,

    )
    parser.add_argument(
        "--case-sensitive",

    )

        "--min-length",
        "-l",
        type=int,
        default=1,
        help="Minimum word length to include (default: 1).",
    )
    return parser.parse_args(argv)
