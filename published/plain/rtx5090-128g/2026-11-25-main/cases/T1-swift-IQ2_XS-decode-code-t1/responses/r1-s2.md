#!/usr/bin/env python3
"""word_freq.py – Count word frequencies and display top N entries.

Command-line tool that reads a text file, counts word frequencies
(case-insensitive, ignoring punctuation), and prints a table of the
top N most frequent words.

Usage:
    python word_freq.py <file> [-n TOP_N]
"""

import argparse
import sys
from collections import Counter
import re
from typing import List, Tuple, Dict


def tokenize(text: str) -> List[str]:
    """Split text into lower-case words, stripping surrounding punctuation.

    Args:
        text: The input string.

    Returns:
        A list of non-empty lower-case words.
    """
    # Extract sequences of alphanumeric characters as words
    words = re.findall(r"[a-zA-Z0-9]+", text)
    return [w.lower() for w in words]


def count_frequencies(words: List[str]) -> Dict[str, int]:
    """Count the occurrences of each word.

    Args:
        words: A list of words (already lower-cased).

    Returns:
        A dictionary mapping each word to its count.
    """
    return dict(Counter(words))


def top_n(freqs: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """Return the top N words by frequency.

    Args:
        freqs: Dictionary mapping words to counts.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then alphabetically for ties.
    """
    sorted_items = sorted(freqs.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]


def format_table(entries: List[Tuple[str, int]], file_name: str, total_words: int) -> str:
    """Format the top entries as a table.

    Args:
        entries: List of (word, count) tuples.
        file_name: Name of the file being analyzed.
        total_words: Total number of words counted.

    Returns:
        A formatted string containing the table.
    """
    if not entries:
        return f"File: {file_name}\nTotal words: {total_words}\nNo words found
