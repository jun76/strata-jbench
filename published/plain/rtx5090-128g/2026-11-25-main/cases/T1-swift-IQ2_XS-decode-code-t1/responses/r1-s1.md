#!/usr/bin/env python3
"""
wordfreq.py – Count word frequencies in a text file and display the top N words.

Usage:
    python wordfreq.py --file input.txt --top 10
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Args:
        text: Input string.

    Returns:
        List of lowercase word tokens (only alphabetic/numeric sequences).
    """
    # Split on non-word characters and convert to lowercase
    tokens = re.findall(r"[a-zA-Z0-9]+", text)
    return [tok.lower() for tok in tokens]


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: Input string.

    Returns:
        Counter mapping each word to its frequency.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def top_words(counter: Counter, n: int = 10) -> List[Tuple[str, int]]:
    """
    Return the top N most common words.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
