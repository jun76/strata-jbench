word_count CLI: Count word frequencies in a text file and display top N entries as a table.

Usage:
    python word_count.py [-n N] [-s] [-v] [--sort-by {frequency,lexicographic}] file

Options:
    -n, --top N   Number of top entries to display (default: 10)
    -s, --sort-by Sort key: 'frequency' (default) or 'lexicographic'
    -v, --verbose Include words with frequency 1
    --version     Show version
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple

def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words, stripping punctuation.

    Args:
        text: Input string.

    Returns:
        List of lowercase word tokens.
    """
    # Split on whitespace and strip surrounding punctuation from each token
    words = []
    for token in text.split():
        # Strip punctuation from beginning and end
        cleaned = token.strip('!@#$%^&*()_+-=[]{}|;:\'",.<>?/`~')
        # Keep alphanumeric, hyphens, underscores, apostrophes inside
        # Remove leading/trailing non-alphanumeric
        cleaned = re.sub(r'^[^A-Za-z0-9]+', '', cleaned)
        cleaned = re.sub(r'[^A-Za-z0-9]+$', '', cleaned)
        if cleaned:
            words.append(cleaned.lower())
    return words

def count_words(text: str) -> Counter:
    """
    Count word frequencies in text.

    Args:
        text: Input string.

    Returns:
        Counter mapping lowercase words to their frequencies.
    """
    return Counter(tokenize(text))
