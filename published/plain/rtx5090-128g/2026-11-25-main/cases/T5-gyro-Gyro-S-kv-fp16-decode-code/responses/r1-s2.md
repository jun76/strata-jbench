"""
word_count.py - テキストファイルから単語の出現頻度を集計し、上位N件を表形式で表示するコマンドラインツール。

Usage:
    python word_count.py input.txt [--top N] [--case-insensitive] [--sort-by freq|word]

Example:
    python word_count.py sample.txt --top 10 --case-insensitive
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def normalize_word(word: str, case_insensitive: bool = False) -> str:
    """単語を正規化する。

    Args:
        word: 元の単語文字列。
        case_insensitive: True の場合、小文字に変換する。

    Returns:
        正規化された単語文字列。
    """
    if case_insensitive:
        return word.lower()
    return word


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。

    Args:
        text: 入力テキスト。

    Returns:
        単語のリスト（空文字列は除外）。
    """
    # 空白・改行で分割し、空文字を除去
    words = text.split()
    return [w for w in words if w]


def count_words(
    text: str,
    case_insensitive: bool = False,
) -> Counter[str]:
    """テキスト内の単語頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、大文字・小文字を区別しない。

    Returns:
        単語をキー、出現回数値を値とする Counter。
    """
    words = tokenize(text)
    normalized = [normalize_word(w, case_insensitive) for w in words]
    return Counter(normalized)
