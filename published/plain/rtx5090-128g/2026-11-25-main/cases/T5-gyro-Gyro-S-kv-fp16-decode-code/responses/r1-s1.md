word_freq.py — コマンドラインツール：テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する。

    python word_freq.py input.txt [--top N] [--case-sensitive]

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に正規化。

    Returns:
        単語のリスト（句読点や空白は除去）
    """
    if not case_sensitive:
        text = text.lower()

    # 単語：英数字とハイフンを含む文字列
    words: List[str] = []
    current: List[str] = []
    for ch in text:
        if ch.isalnum() or ch == '-':
            current.append(ch)
        else:
            if current:
                words.append(''.join(current))
                current = []
    if current:
        words.append(''.join(current))
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    テキスト内の単語出現頻度を集計する。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        単語 -> 出現回数の Counter
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)
