import argparse

from typing import Dict, List, Tuple


def tokenize_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字/小文字を区別する。
            False の場合、すべて小文字に正規化する。

    Returns:
        抽出された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン/アポストロフィを保持し、それ以外を区切りとして扱う
    words = []
    current = []
    for char in text:
        if char.isalnum() or char in ('-', "'"):
            current.append(char)
        elif current:
            words.append(''.join(current))
            current = []
    if current:
        words.append(''.join(current))
    return words


def count_words(words: List[str], min_length: int = 1) -> Dict[str, int]:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。
        min_length: 最小単語長。これ未満の単語は除外する。

    Returns:
        単語 -> 頻度の辞書。
    """
    filtered = [w for w in words if len(w) >= min_length]
    return dict(Counter(filtered))
