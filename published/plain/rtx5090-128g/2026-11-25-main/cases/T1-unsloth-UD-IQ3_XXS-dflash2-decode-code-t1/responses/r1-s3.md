wordfreq - テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツール。

使い方:
    python wordfreq.py --file input.txt [--top 10] [--case-sensitive]

from collections import Counter

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語リストにトークン化する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別する場合は True、区別しない場合は False。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とアンダースコアのみを単語として抽出
    words = re.findall(r'[a-zA-Z0-9_]+', text)
    return words


def count_words(words: List[str]) -> Counter[str]:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語と出現回数をマッピングした Counter。
    """
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counter から出現回数の上位 n 件を取得する。

    Args:
        counter: 単語と出現回数をマッピングした Counter。
        n: 取得する件数。

    Returns:
        (単語, 出現回数) のタプルのリスト。
    """
    return counter.most_common(n)
