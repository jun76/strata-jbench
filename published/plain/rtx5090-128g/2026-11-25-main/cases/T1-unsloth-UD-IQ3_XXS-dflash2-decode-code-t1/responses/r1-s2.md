"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を集計し、上位 N 件を表形式で表示する。

使い方:
    python word_freq.py <file_path> [-n TOP_N]

例:
    python word_freq.py sample.txt
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストに分割する。

    単語は英数字（日本語文字も含む Unicode 文字）を抽出し、小文字化して返す。
    日本語の単語分割は簡易的に連続する日本語文字列を一つのトークンとして扱う。

    Args:
        text: 入力テキスト

    Returns:
        単語（小文字）のリスト
    """
    # 日本語のひらがな、カタカナ、漢字を単語として捉える
    # 英数字は単語として捉える
    # その他記号は除外
    pattern = re.compile(r'[a-zA-Z0-9\u3040-\u309F\u30A0-\u30FF\u4E00-\u9FFF]+')
    tokens = pattern.findall(text)
    return [token.lower() for token in tokens]


def compute_frequency(word_list: List[str]) -> Counter:
    """
    単語リストから出現頻度を計算する。

    Args:
        word_list: 単語のリスト

    Returns:
        単語と出現回数の Counter
    """
    return Counter(word_list)


def get_top_n(freq_counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度 Counter から上位 N 件の単語と出現回数を取り出す。

    Args:
        freq_counter: 単語と出現回数の Counter
        n: 上位何件を取得するか

    Returns:
        (単語, 出現回数) のタプルのリスト
    """
    return freq_counter.most_common(n)
