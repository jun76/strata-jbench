#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file and display the top N results."""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path


def tokenize(text: str) -> list[str]:
    """Tokenize text into lowercase words.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    return re.findall(r'\b[a-zA-Z0-9_]+\b', text.lower())


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """Get the top N most common words from a counter.

    Args:
        counter: A Counter object mapping words to counts.
        n: The number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
