```python
#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file and display top N entries as a table."""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words, stripping punctuation."""
    return re.findall(r"[a-z]+", text.lower())


def count_words(text: str) -> Counter[str]:
    """Count word frequencies from text."""
    return Counter(tokenize(text))


def format_table(counter: Counter[str], n: int) -> str:
    """Format top N word frequencies as an ASCII table."""
    top = counter.most_common(n)
    if not top:
        return "No words found."

    # Determine column widths
    word_width = max(len(word) for word, _ in top)
    count_width = max(len(str(count)) for _, count in top)

    header = f"{'Word':<{word_width}} | {'Count':<{count_width}}"
    separator = "-" * word_width + " | " + "-" * count_width

    lines = [header, separator]
    for word, count in top:
        lines.append(f"{word:<{word_width}} | {count:<{count_width}}")

    return "\n".join(lines)


def read_file(path: str) -> str:
    """Read text from a file."""
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def main() -> None:
    """Main entry point for the command-line tool."""
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and display top N entries as a table."
    )
    parser.add_argument("file", help="Path to the input text file")
    parser.add_argument("-n", "--top", type=int, default=10, help="Number of top entries to display (default: 10)")

    args = parser.parse_args()

    text = read_file(args.file)
    counter = count_words(text)
    table = format_table(counter, args.top)
    print(table)


if __name__ == "__main__":
    main()
```
