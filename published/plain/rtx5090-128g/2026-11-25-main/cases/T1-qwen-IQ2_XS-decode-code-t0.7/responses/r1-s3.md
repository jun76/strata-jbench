#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file and display top N words."""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def read_words_from_file(filepath: str) -> List[str]:
    """Read a text file and return a list of words (lowercased, stripped)."""
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            content = f.read()
    except FileNotFoundError:
        print(f"Error: File '{filepath}' not found.", file=sys.stderr)
        sys.exit(1)

    # Split into words, convert to lowercase, strip non-alphanumeric characters
    words = []
    for word in content.split():
        # Strip leading/trailing non-alphanumeric characters (punctuation)
        cleaned = word.strip(".,;:!?'\"()[]{}<>-")
        if cleaned:
            words.append(cleaned.lower())

    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """Count the frequency of each word in the list."""
    return Counter(words)


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N words by frequency, as a list of (word, count) tuples."""
    return counter.most_common(n)


def format_table(top_words: List[Tuple[str, int]]) -> str:
    """Format the top words as a table string."""
    if not top_words:
        return "No words found."

    # Determine column widths
    max_word_len = max(len(word) for word, _ in top_words)
    max_count_len = max(len(str(count)) for _, count in top_words)

    # Build header
    header_word = "Word".ljust(max_word_len)
    header_count = "Count".rjust(max_count_len)
    separator = "-" * (max_word_len + 2 + max_count_len)

    lines = [
        f"{header_word}  {header_count}",
        separator,
    ]

    for word, count in top_words:
        lines.append(f"{word.ljust(max_word_len)}  {str(count).rjust(max_count_len)}")

    return "\n".join(lines)
