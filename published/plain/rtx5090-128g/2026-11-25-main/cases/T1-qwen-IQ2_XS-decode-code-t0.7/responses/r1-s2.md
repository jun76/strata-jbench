#!/usr/bin/env python3
"""Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N
most frequent words in a table format.
"""

from __future__ import annotations

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Split text into lowercase word tokens.

    Words are defined as sequences of alphanumeric characters.
    Punctuation and whitespace are treated as separators.

    Args:
        text: Input text string.

    Returns:
        List of lowercase word tokens.
    """
    words: List[str] = []
    current: List[str] = []

    for char in text:
        if char.isalnum():
            current.append(char.lower())
        else:
            if current:
                words.append("".join(current))
                current = []

    if current:
        words.append("".join(current))

    return words


def count_frequencies(text: str) -> Counter[str]:
    """Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        Counter mapping each word to its occurrence count.
    """
    return Counter(tokenize(text))


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words from the counter.

    Args:
        counter: Counter of word frequencies.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
