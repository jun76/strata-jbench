#!/usr/bin/env python3
"""
Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words in a table format.
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def read_text_file(filepath: str) -> str:
    """Read the contents of a text file."""
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except Exception as e:
        raise RuntimeError(f"Error reading file {filepath}: {e}")


def extract_words(text: str) -> List[str]:
    """Extract lowercase words from text, splitting on whitespace and punctuation."""
    # Split on whitespace, then strip non-alphanumeric characters from each token
    words: List[str] = []
    for token in text.split():
        # Strip leading/trailing non-alphanumeric characters
        stripped = token.strip('.,;!?()[]{}<>"\'`-_~#&$@^*+/\\')
        if stripped:
            words.append(stripped.lower())
    return words


def count_words(words: List[str]) -> Counter[str]:
    """Count the frequency of each word."""
    return Counter(words)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top N words by frequency as a list of (word, count) tuples."""
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]]) -> str:
    """Format the word frequency entries as a table string."""
    if not entries:
        return "No words found."

    # Determine max word length for alignment
    max_word_len = max(len(word) for word, _ in entries)
    header = f"{'Word':<{max_word_len}} | {'Count':>5}"
    separator = '-' * (max_word_len + 8)
    lines = [header, separator]

    for word, count in entries:
        lines.append(f"{word:<{max_word_len}} | {count:>5}")

    return '\n'.join(lines)
