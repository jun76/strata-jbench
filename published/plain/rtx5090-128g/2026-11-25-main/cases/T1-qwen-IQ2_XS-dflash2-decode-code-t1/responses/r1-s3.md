#!/usr/bin/env python3
"""CLI tool to read a text file, count word frequencies, and display the top N entries in a table format."""

import argparse
from collections import Counter
from pathlib import Path
import sys
from typing import Optional


def count_words(text: str) -> Counter:
    """Count word frequencies from a string.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter mapping each word to its occurrence count.
    """
    # Normalize to lowercase for case-insensitive counting
    words = text.lower().split()
    # Strip common punctuation from words
    cleaned = []
    for w in words:
        stripped = w.strip('.,!?;:\'"()[]{}')
        if stripped:
            cleaned.append(stripped)
    return Counter(cleaned)


def format_table(counts: Counter, top_n: int) -> str:
    """Format the top N word counts as a plain-text table.

    Args:
        counts: A Counter of word frequencies.
        top_n: Number of top entries to display.

    Returns:
        A string containing the formatted table.
    """
    top_items = counts.most_common(top_n)
    if not top_items:
        return "No words found."

    # Determine column widths
    word_width = max(len(w) for w, _ in top_items)
    count_width = max(len(str(c)) for _, c in top_items)

    header = f"{'Word':<{word_width}} | {'Count':>{count_width}}"
    separator = f"{'-' * word_width}-+-{'-' * count_width}"
    lines = [header, separator]
    for word, count in top_items:
        lines.append(f"{word:<{word_width}} | {count:>{count_width}}")
    return "\n".join(lines)

        Exit code: 0 on success, 1 on failure.
