#!/usr/bin/env python3
"""
Text word frequency counting tool.

Reads a text file, counts word occurrences, and displays the top N results
in a table format.
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path


def tokenize(text: str) -> list[str]:
    """
    Tokenize text into lowercase words.

    Uses a regex to extract sequences of alphabetic characters as words.
    Non-alphabetic characters are ignored.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase word tokens.
    """
    return [word.lower() for word in re.findall(r"[a-zA-Z]+", text)]


def count_words(text: str) -> Counter[str]:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text to count words from.

    Returns:
        A Counter mapping each word to its occurrence count.
    """
    words = tokenize(text)
    return Counter(words)


def top_words(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """
    Return the top N most frequent words and their counts.

    Args:
        counter: A Counter mapping words to counts.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
