#!/usr/bin/env python3
"""
word_counter.py

A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.

Usage:
    python word_counter.py <file> [--top N] [--case-sensitive] [--min-length L]
"""

import argparse
import sys
from collections import Counter
from typing import TextIO


def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """
    Normalize a word by stripping punctuation and optionally lowering case.

    Args:
        word: The raw word string to normalize.
        case_sensitive: If False, convert to lowercase. If True, preserve case.

    Returns:
        The normalized word string.
    """
    # Strip common punctuation from both ends
    cleaned = word.strip(".,;:!?\"'()[]{}-–—")
    if not case_sensitive:
        cleaned = cleaned.lower()
    return cleaned


def extract_words(text: str, case_sensitive: bool = False) -> list[str]:
    """
    Extract words from text by splitting on whitespace and normalizing each token.

    Args:
        text: The input text string.
        case_sensitive: Whether to preserve case during normalization.

    Returns:
        A list of normalized, non-empty words.
    """
    tokens = text.split()
    words = []
    for token in tokens:
        normalized = normalize_word(token, case_sensitive)
        if normalized:
            words.append(normalized)
    return words


def count_frequencies(words: list[str]) -> Counter[str]:
    """
    Count the frequency of each word in the list.

    Args:
        words: A list of words to count.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)
