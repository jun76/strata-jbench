#!/usr/bin/env python3
"""
Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words
in a tabular format.
"""

import argparse

from collections import Counter

def read_file(filepath: str) -> str:
    """
    Read the contents of a text file.

    Args:
        filepath: Path to the text file to read.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into words.

    Words are defined as sequences of alphanumeric characters and underscores.
    Words are converted to lowercase for case-insensitive counting.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercased words.
    """
    words = []
    current_word = []
    for char in text:
        if char.isalnum() or char == '_':
            current_word.append(char)
        else:
            if current_word:
                words.append(''.join(current_word).lower())
                current_word = []
    if current_word:
        words.append(''.join(current_word).lower())
    return words


def count_words(text: str) -> Counter:
    """
    Count the frequency of each word in the text.

    Args:
        text: The input text.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    words = tokenize(text)
    return Counter(words)
