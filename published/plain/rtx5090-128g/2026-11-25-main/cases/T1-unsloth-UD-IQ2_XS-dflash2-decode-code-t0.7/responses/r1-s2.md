def load_text(file_path: str) -> str:
    """
    指定されたパスのテキストファイルを読み込み、内容を文字列として返す。

    Args:
        file_path (str): 読み込むテキストファイルのパス

    Returns:
        str: ファイルの内容

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        PermissionError: 読み取り権限がない場合
    """
    with open(file_path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語リストに分割する。
    単語は英数字・ハイフン・アポストロフィを含む連続文字列として抽出される。

    Args:
        text (str): 入力テキスト
        case_sensitive (bool): 大文字小文字を区別するかどうか。
                                False の場合、すべて小文字に変換する。

    Returns:
        List[str]: 単語のリスト
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字、ハイフン、アポストロフィを含む連続文字列を単語として抽出
    words = re.findall(r"[a-z0-9'-]+", text)
    return words

        Counter: 単語をキー、出現回数を値とする Counter オブジェクト
    """
