"""
word_freq.py - テキストファイルから単語の出現頻度を計算し、
上位 N 件を表形式で表示するコマンドラインツール。

使用方法:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt --top 5 --sort-by count
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_text(file_path: Path) -> str:
    """
    テキストファイルを UTF-8 で読み込み、内容を返す。

    Args:
        file_path: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を格納した文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        UnicodeDecodeError: UTF-8 でデコードできない場合。
    """
    return file_path.read_text(encoding="utf-8")


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    大文字小文字を区別せず、非英数字文字で区切られたトークンを単語として扱う。

    Args:
        text: 入力テキスト。

    Returns:
        正規化された単語のリスト（大文字小文字無視、小文字に変換済み）。
    """
    # 非英数字文字で分割し、空文字列を除外
    words = []
    current_word = []
    for char in text:
        if char.isalnum():
            current_word.append(char.lower())
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))
    return words
