#!/usr/bin/env python3
"""テキストファイルの単語出現頻度を上位 N 件表示するコマンドラインツール."""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを単語リストに分割する.

    単語は、英数字とハイフン、アポストロフィを含む文字列として抽出される。
    すべて小文字に変換される。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    # 英数字、ハイフン、アポストロフィを含むトークンを抽出
    return re.findall(r"[a-z0-9]+(?:['-][a-z0-9]+)*", text.lower())


def count_words(text: str) -> Counter:
    """テキスト中の単語出現頻度をカウントする.

    Args:
        text: 入力テキスト。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (単語, 回数) を取得する.

    Args:
        counter: 単語出現頻度の Counter。
        n: 取得する上位件数。

    Returns:
        上位 n 件の (単語, 回数) のリスト。回数の降順、同率時は辞書順。
    """
    # most_common は回数降順、同率は出現順だが、テスト再現性のため辞書順で安定化
    items = counter.items()
    sorted_items = sorted(items, key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
