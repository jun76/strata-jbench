#!/usr/bin/env python3
"""
text_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位 N 件表示する。

使い方:
    python text_freq.py input.txt -n 10
    python text_freq.py input.txt --top 5 --case-insensitive
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Optional


def tokenize(text: str, case_insensitive: bool = True) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_insensitive: 大文字小文字を区別しない場合 True。

    Returns:
        単語のリスト。
    """
    if case_insensitive:
        text = text.lower()
    # 英数字からなる単語を抽出（正規表現で単語境界を使用）
    words = re.findall(r"[a-zA-Z0-9]+", text)
    return words


def count_words(text: str, case_insensitive: bool = True) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_insensitive: 大文字小文字を区別しない場合 True。

    Returns:
        単語と出現回数の Counter。
    """
    words = tokenize(text, case_insensitive)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の単語と頻度を取得する。

    Args:
        counter: 単語頻度の Counter。
        n: 上位取得する件数。

    Returns:
        (単語, 頻度) のタプルのリスト。頻度の降順、同頻度なら単語の辞書順。
    """
    # most_common は頻度順だが、同頻度時の順序は保証されない。
    # 頻度降順、単語昇順でソートする。
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
