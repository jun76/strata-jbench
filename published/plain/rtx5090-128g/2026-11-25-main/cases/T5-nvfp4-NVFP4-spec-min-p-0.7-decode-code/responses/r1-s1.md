def tokenize(text: str, case_insensitive: bool = True) -> List[str]:
    """
    テキストから単語のリストを抽出する。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、単語を小文字に変換して比較する。

    Returns:
        単語のリスト。非英数字は区切りとして扱われる。
    """
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    pattern = r"[a-zA-Z0-9]+(?:[-''][a-zA-Z0-9]+)*"
    words = re.findall(pattern, text)
    if case_insensitive:
        words = [w.lower() for w in words]
    return words

    Returns:
        単語と出現回数の Counter。
    """
    return Counter(words)
