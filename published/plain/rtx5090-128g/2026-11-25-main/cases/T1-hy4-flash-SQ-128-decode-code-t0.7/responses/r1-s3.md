#!/usr/bin/env python3
"""テキストファイル内の単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Sequence


def tokenize(text: str) -> list[str]:
    """テキストを小文字化し、英数字単語に分割して返す。

    Args:
        text: 入力テキスト。

    Returns:
        抽出された単語のリスト。
    """
    return re.findall(r"\b[a-zA-Z0-9]+\b", text.lower())


def count_words(text: str) -> Counter[str]:
    """テキストから単語頻度を集計する。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(tokenize(text))


def format_table(counter: Counter[str], top_n: int) -> str:
    """頻度上位 N 件を表形式文字列に整形する。

    Args:
        counter: 単語頻度。
        top_n: 表示する上位件数。

    Returns:
        整形された表（ヘッダー付き）。
    """
    rows = counter.most_common(top_n)
    width_word = max((len(w) for w, _ in rows), default=5)
    width_word = max(width_word, len("Word"))
    width_count = len("Count")

    lines = [
        f"{'Word':<{width_word}}  {'Count':>{width_count}}",
        f"{'-' * width_word}  {'-' * width_count}",
    ]
    for word, count in rows:
        lines.append(f"{word:<{width_word}}  {count:>{width_count}}")

    return "\n".join(lines)
