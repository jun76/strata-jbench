def tokenize(text: str, case_insensitive: bool = True, min_len: int = 1) -> List[str]:
    """
    テキストを単語リストに分割する。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、単語を小文字に正規化する。
        min_len: この長さ未満の単語を除外する。

    Returns:
        単語のリスト。

    >>> tokenize("Hello, World! hello.", case_insensitive=True)
    ['hello', 'world', 'hello']

    >>> tokenize("a bb ccc dddd", min_len=2)
    ['bb', 'ccc', 'dddd']
    """
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    raw_tokens = re.findall(r"[A-Za-z0-9]+(?:[-'][A-Za-z0-9]+)*", text)
    if case_insensitive:
        raw_tokens = [t.lower() for t in raw_tokens]
    return [t for t in raw_tokens if len(t) >= min_len]
