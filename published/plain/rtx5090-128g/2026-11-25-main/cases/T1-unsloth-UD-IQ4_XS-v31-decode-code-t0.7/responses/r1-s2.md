#!/usr/bin/env python3
"""
Word Frequency Counter

A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a tabular format.

Usage:
    python word_freq.py input.txt [-n 10]
    python word_freq.py input.txt --top 20
    python word_freq.py input.txt --case-sensitive
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Iterable


def tokenize(text: str, case_sensitive: bool = False) -> list[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If True, preserve case. If False, convert to lowercase.

    Returns:
        A list of tokens (words) extracted from the text.
    """
    if not case_sensitive:
        text = text.lower()
    # Match sequences of alphanumeric characters and underscores as words
    tokens = re.findall(r'\b[\w]+\b', text)
    return tokens


def count_word_frequencies(tokens: Iterable[str]) -> Counter[str]:
    """
    Count the frequency of each token.

    Args:
        tokens: An iterable of tokens (words).

    Returns:
        A Counter object mapping each token to its frequency.
    """
    return Counter(tokens)


def get_top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """
    Get the top N most common items from a Counter.

    Args:
        counter: The Counter object containing word frequencies.
        n: The number of top items to return.

    Returns:
        A list of tuples (word, count), sorted by count descending,
        then by word ascending for ties.
    """
    # most_common() returns items sorted by count descending
    # For ties, we need secondary sort by word ascending
    items = sorted(
        counter.items(),
        key=lambda x: (-x[1], x[0])
    )
    return items[:n]
