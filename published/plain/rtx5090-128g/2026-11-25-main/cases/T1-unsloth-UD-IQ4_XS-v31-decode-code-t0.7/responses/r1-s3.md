#!/usr/bin/env python3
"""
Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words
in a formatted table.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase words (alphabetic strings).
    """
    return re.findall(r'[a-zA-Z]+', text.lower())


def count_words(text: str) -> Counter[str]:
    """
    Count word occurrences in the given text.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter mapping each word to its occurrence count.
    """
    words = tokenize(text)
    return Counter(words)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then alphabetically ascending for ties.
    """
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]]) -> str:
    """
    Format word-frequency entries as a readable table.

    Args:
        entries: A list of (word, count) tuples.

    Returns:
        A formatted string representing the table.
    """
    if not entries:
        return "No words found."

    max_word_len = max(len(word) for word, _ in entries)
    word_col_width = max(max_word_len, len("Word")) + 2
    count_col_width = max(6, len("Count")) + 2

    header = f"{'Word':<{word_col_width}}{'Count':>{count_col_width}}"
    separator = "-" * (word_col_width + count_col_width)

    lines = [header, separator]
    for word, count in entries:
