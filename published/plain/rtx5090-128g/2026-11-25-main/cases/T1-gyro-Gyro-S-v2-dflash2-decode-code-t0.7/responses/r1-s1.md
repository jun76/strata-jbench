def load_words_from_file(path: str) -> List[str]:
    """Read a text file and return a list of lowercase words.

    Words are defined as sequences of alphanumeric characters and underscores,
    separated by whitespace or punctuation.

    Args:
        path: Path to the text file.

    Returns:
        A list of words found in the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    try:
        with open(path, 'r', encoding='utf-8') as f:
            content = f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {path}")
    except IOError as e:
        raise IOError(f"Error reading file '{path}': {e}")

    # Split by whitespace and remove punctuation
    words = []
    for word in content.split():
        # Remove leading/trailing punctuation
        cleaned = word.strip()
        if cleaned:
            words.append(cleaned.lower())
    return words


def count_word_frequency(words: List[str]) -> Counter[str]:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    return Counter(words)


def get_top_n_frequencies(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, frequency), sorted by frequency descending.
    """
    return counter.most_common(n)
