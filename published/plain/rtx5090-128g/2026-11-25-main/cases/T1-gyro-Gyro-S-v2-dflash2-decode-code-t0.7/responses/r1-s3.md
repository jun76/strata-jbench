#!/usr/bin/env python3
"""
Top-word frequency counter command-line tool.

Reads text from a file, counts word frequencies, and displays the top N words
in a tabular format.

Usage:
    topwords [-n N] [--case-sensitive] [--min-length M] FILE

Examples:
    topwords -n 10 sample.txt
    topwords --case-sensitive -n 5 input.txt
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def read_text(filepath: str) -> str:
    """
    Read the contents of a text file.

    Args:
        filepath: Path to the text file to read.

    Returns:
        The entire contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        OSError: If the file cannot be read.
    """
    return Path(filepath).read_text(encoding="utf-8")


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words, optionally case-insensitive and filtered by length.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, words are converted to lowercase.
        min_length: Minimum character length for a word to be included.

    Returns:
        A list of tokenized words.
    """
    words = text.split()
    if not case_sensitive:
        words = [w.lower() for w in words]
    return [w for w in words if len(w) >= min_length]


def count_words(words: List[str]) -> Dict[str, int]:
    """
    Count the frequency of each word in a list.

    Args:
        words: A list of words.

    Returns:
        A dictionary mapping each word to its frequency count.
    """
    return dict(Counter(words))
