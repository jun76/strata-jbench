def read_file(path: str) -> str:
    """指定されたパスのテキストファイルを読み込み、その内容を返す。

    Args:
        path: ファイルパス

    Returns:
        ファイルの全テキスト内容

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        IOError: ファイルの読み込み失敗の場合
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def count_words(text: str) -> Counter:
    """テキスト中の単語（英字のみ）を小文字化して出現頻度を数える。

    Args:
        text: 入力テキスト

    Returns:
        単語 -> 出現回数の Counter
    """
    words = []
    for word in text.split():
        # 英字のみに絞り込み、小文字に変換
        cleaned = "".join(c for c in word.lower() if c.isalpha())
        if cleaned:
            words.append(cleaned)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (word, count) のリストを返す。

    Args:
        counter: 単語出現頻度の Counter
        n: 取得する件数

    Returns:
        (word, count) のリスト（降順）
    """
    return counter.most_common(n)
