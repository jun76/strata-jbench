"""

def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出します。
    正規表現を使用して、アルファベット、数字、およびハイフンを含むトークンを抽出します。
    すべて小文字に変換されます。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 抽出された単語のリスト。
    """
    # \b\w+\b では日本語や特殊文字が扱いにくいですが、
    # ここでは一般的な英数字単語を想定し、\w+ (アルファベット、数字、アンダースコア) を使用します。
    # より厳密に「単語」を定義するなら [a-zA-Z0-9'-]+ などを使うことも可能ですが、
    # 今回は標準的な \w+ を採用します。
    words = re.findall(r'\w+', text.lower())
    return words

    """
    単語リストから頻度をカウントします。

    Args:
        words (List[str]): 単語のリスト。

    Returns:
        Counter: 単語とその出現回数のマッピング。
    """
    return Counter(words)
