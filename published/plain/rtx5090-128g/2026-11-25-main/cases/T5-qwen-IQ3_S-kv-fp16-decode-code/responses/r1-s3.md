Options:

    --sort-by KEY       並べ替えキー: 'freq' (頻度) または 'alpha' (アルファベット順)
    -i, --ignore-case   大文字小文字を区別しない (デフォルト: 区別する)

import argparse
import re

from collections import Counter
from typing import List, Tuple, Optional


def extract_words(text: str, ignore_case: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。
    単語は、アルファベット、数字、アンダースコアを含む連続文字列とみなす。

    Args:
        text: 入力テキスト
        ignore_case: Trueの場合、大文字小文字を区別しない

    Returns:
        抽出された単語のリスト
    """
    if ignore_case:
        text = text.lower()
    # \w+ は [a-zA-Z0-9_] に一致する
    words = re.findall(r'\w+', text)
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度を集計する。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    return Counter(words)
