    -t, --type TYPE   単語の分類 (case: 大文字小文字を区別しない / original: 区別する) (default: case)

"""

def tokenize(text: str, ignore_punctuation: bool = True) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 対象テキスト文字列。
        ignore_punctuation: True の場合、句読点を除去して単語のみ抽出する。

    Returns:
        単語のリスト。
    """
    if ignore_punctuation:
        # 文字と数字を保持し、それ以外を空白に置換
        tokens = re.findall(r"[a-zA-Z0-9_]+", text)
    else:
        # 空白区切りのトークン
        tokens = text.split()
    return tokens

    """

    Args:
        words: 単語のリスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、すべて小文字化してカウントする。

    Returns:
        各単語とその出現回数の辞書 (Counter)。
    """
    if not case_sensitive:
        words = [w.lower() for w in words]
    return Counter(words)
