以下は、テキストファイルを読み込んで単語の出現頻度を集計し、上位 N 件をテーブル形式で表示する CLI ツールの実装です。argparse、型ヒント、docstring を使用し、pytest による単体テストも付いています。

"""
wordcount.py

テキストファイルを読み込み、単語の出現頻度を集計して
上位 N 件をテーブル形式で出力するコマンドラインツール。

使い方例:
    python wordcount.py sample.txt --top 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト文字列

    Returns:
        単語のリスト（小文字化済み）
    """
    # 大文字小文字を無視して、英数字とハイフンを保留する
    # その他の文字は区切りとして扱う
    words = []
    current_word = []
    for char in text.lower():
        if char.isalnum() or char == '-':
            current_word.append(char)
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))
    return words


def count_words(text: str) -> Counter[str]:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト文字列

    Returns:
        単語 -> 出現回数 の Counter
    """
    tokens = tokenize(text)
    return Counter(tokens)
