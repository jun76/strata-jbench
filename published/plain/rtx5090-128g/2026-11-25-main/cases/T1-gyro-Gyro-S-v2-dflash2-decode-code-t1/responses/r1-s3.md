def load_file(path: str) -> str:
    """
    指定されたパスのテキストファイルを UTF-8 で読み込む。

    Args:
        path: テキストファイルのパス

    Returns:
        ファイルの内容

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        UnicodeDecodeError: UTF-8 でデコードできない場合
        OSError: その他読み込みエラー
    """
    with open(path, encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, ignore_case: bool = False, min_len: int = 1) -> List[str]:
    """
    テキストを空白区切りで単語に分割する。

    Args:
        text: 入力テキスト
        ignore_case: True の場合、単語を小文字に変換する
        min_len: この長以下の単語は除外する

    Returns:
        単語のリスト
    """
    words = text.split()
    if ignore_case:
        words = [w.lower() for w in words]
    return [w for w in words if len(w) >= min_len]


def count_words(words: List[str]) -> Counter:
    """
    単語の出現頻度を集計する。

    Args:
        words: 単語のリスト

    Returns:
        単語 → 出現回数 の Counter
    """
    return Counter(words)
