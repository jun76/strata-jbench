word_freq.py

Usage:
    python word_freq.py <file_path> [--top N] [--sort-by {freq,word}]

Example:
    python word_freq.py sample.txt --top 10 --sort-by freq
"""

def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト文字列

    Returns:
        単語のリスト（小文字化、句読点除去）

    Raises:
        ValueError: 入力が文字列でない場合
    """
    if not isinstance(text, str):
        raise ValueError("Input must be a string")
    
    # 小文字化し、句読点・記号を除去
    cleaned = []
    for char in text:
        if char.isalpha() or char.isdigit():
            cleaned.append(char.lower())
        elif char in (' ', '\n', '\t'):
            cleaned.append(' ')
        # その他は除去
    
    # 空白で分割して空文字列を除く
    words = cleaned.split()
    return [w for w in words if w]


def count_words(words: List[str]) -> Dict[str, int]:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数 の辞書

    Raises:
        TypeError: 入力がリストでない場合
    """
    if not isinstance(words, list):
        raise TypeError("Input must be a list")
    
    counter: Dict[str, int] = Counter(words)
    return dict(counter)
