```python
#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する。

使い方:
    python word_counter.py <file> [--top N] [--sort {freq,alpha}] [--width W]

例:
    python word_counter.py data.txt --top 10 --sort alpha
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple

def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト文字列。

    Returns:
        単語のリスト。空白区切りで分割し、句読点や記号は除去する。
    """
    words = []
    for word in text.split():
        # 句読点・記号を除去
        cleaned = word.strip(".,!?;:'\"()[]{}-–—")
        if cleaned:
            words.append(cleaned.lower())
    return words


def count_words(text: str) -> Counter[str]:
    """
    テキスト中の単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト文字列。

    Returns:
        単語とその出現回数の Counter。
    """
    words = tokenize(text)
    return Counter(words)


def get_top_words(counter: Counter[str], n: int, sort_by: str = "freq") -> List[Tuple[str, int]]:
    """
    上位 N 件の単語とその頻度を取得する。

    Args:
        counter: 単語頻度の Counter。
        n: 取得する件数。
        sort_by: ソート順。"freq" で頻度降順、"alpha" でアルファベット昇順。

    Returns:
        (単語, 頻度) のタプルリスト。
    """
    if sort_by == "alpha":
        items = sorted(counter.items(), key=lambda x: x[0])
    else:
        # freq でソート: 頻度降順、同頻度ならアルファベット昇順
        items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
