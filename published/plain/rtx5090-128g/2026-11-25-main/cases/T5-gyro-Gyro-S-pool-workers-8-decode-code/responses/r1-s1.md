"""
word_freq CLI tool: read a text file, count word occurrences, and print the top N words in a table.

Usage:
    python word_freq.py <file> [-n N] [--case-sensitive] [--sort-by freq|alpha]

Examples:
    python word_freq.py sample.txt -n 10
    python word_freq.py sample.txt -n 5 --case-sensitive --sort-by alpha
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Split text into words. Words are sequences of alphanumeric characters.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, words are lowercased before counting.

    Returns:
        A list of word tokens.
    """
    # Match sequences of alphanumeric characters (letters and digits)
    words = re.findall(r"[a-zA-Z0-9]+", text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    Count word occurrences in the given text.

    Args:
        text: The input text.
        case_sensitive: If False, words are lowercased before counting.

    Returns:
        A Counter mapping each word to its frequency.
    """
    tokens = tokenize(text, case_sensitive=case_sensitive)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int, sort_by: str = "freq") -> List[Tuple[str, int]]:
    """
    Get the top N entries from a word frequency counter.

    Args:
        counter: The word frequency counter.
        n: Number of top entries to return.
        sort_by: Sort order: 'freq' (descending by frequency) or 'alpha' (ascending alphabetically).

    Returns:
        A list of (word, count) tuples, sorted according to sort_by.
    """
    if sort_by == "freq":
        # Sort by frequency descending, then alphabetically ascending as tie-breaker
        items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
