#!/usr/bin/env python3
"""
top_words.py - コマンドラインツール: テキストファイルから単語出現頻度を上位N件表示

Usage:
    python top_words.py input.txt --top 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        小文字化された単語リスト
    """
    # 英数字のみを単語として抽出
    return re.findall(r'\b[a-zA-Z0-9]+\b', text.lower())


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語 -> 出現回数 のCounterオブジェクト
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位n件の単語と出現回数を取得する。
    
    Args:
        counter: 単語出現頻度のカウント
        n: 上位件数
        
    Returns:
        (単語, 出現回数) のタプルリスト
    """
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]]) -> str:
    """
    表形式で結果をフォーマットする。
    
    Args:
        entries: (単語, 出現回数) のタプルリスト
        
    Returns:
        表形式の文字列
    """
    if not entries:
        return ""
    
    # 列幅を計算
    rank_width = len(str(len(entries)))
    word_width = max(len(word) for word, _ in entries)
    count_width = max(len(str(count)) for _, count in entries)
    
    lines = []
    # ヘッダー
    header_rank = "Rank".rjust(rank_width)
    header_word = "Word".ljust(word_width)
    header_count = "Count".rjust(count_width)
    lines.append(f"{header_rank} | {header_word
