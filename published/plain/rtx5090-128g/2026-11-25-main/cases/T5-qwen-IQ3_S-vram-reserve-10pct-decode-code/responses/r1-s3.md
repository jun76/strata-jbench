### 1. 実装: `word_count.py`

```python
#!/usr/bin/env python3
"""
word_count.py - テキストファイルの単語出現頻度を集計し、上位N件を表示するCLIツール。
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple

def read_text(filepath: str) -> str:
    """
    指定されたファイルパスからテキストを読み込む。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        str: ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み取りエラーが発生した場合。
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()

def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストに分割する。
    単語は、英数字とアンダースコアで構成される連続文字列として抽出される。
    すべて小文字に変換される。

    Args:
        text: 入力テキスト。

    Returns:
        List[str]: 単語のリスト。
    """
    # 正規表現を使わず、strメソッドで簡潔に実装
    # 単語境界を考慮し、非単語文字で分割
    import re
    # \w+ は [a-zA-Z0-9_] にマッチする
    return [word.lower() for word in re.findall(r'\w+', text)]

def count_words(tokens: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        tokens: 単語のリスト。

    Returns:
        Counter: 単語 -> 出現回数 のマップ。
    """
    return Counter(tokens)

def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度が高い順に上位N件の単語と回数を取得する。
    頻度が同じ場合は、辞書順で昇順に並べる。

    Args:
        counter: 単語出現頻度
