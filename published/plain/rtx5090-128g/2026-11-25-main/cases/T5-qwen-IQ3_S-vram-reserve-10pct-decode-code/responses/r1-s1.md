"""
word_freq.py - コマンドラインツール: テキストファイルから単語出現頻度を集計し、上位N件を表形式で表示する。

使い方:
    python word_freq.py input.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Optional


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。単語は英数字の連続（最小1文字）とする。

    Args:
        text: 入力テキスト

    Returns:
        抽出された単語のリスト（小文字に変換済み）
    """
    # 英数字の連続を単語として抽出
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [w.lower() for w in words]


def count_word_frequencies(words: List[str]) -> Counter:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        Counter オブジェクト（単語 -> 出現回数）
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """上位 n 件の (単語, 出現回数) のリストを返す。

    Args:
        counter: 単語出現頻度の Counter
        n: 上位何件を取得するか

    Returns:
        (単語, 出現回数) のタプルのリスト。出現回数の降順、同順位は辞書順。
    """
    if n <= 0:
        return []
    # most_common は出現回数順。同順位の場合は辞書順で安定ソートするため、
    # 先に辞書順でソートしてから most_common を使う、または key で指定。
    # Counter.most_common は出現回数の降順で、同順位は任意順。
    # 安定した結果を得るため、(count, word) でソートする。
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
