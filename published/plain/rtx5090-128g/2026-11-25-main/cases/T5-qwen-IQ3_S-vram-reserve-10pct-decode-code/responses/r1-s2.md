top_words.py - コマンドラインツール: テキストファイルから単語出現頻度を上位N件表示

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple, Optional


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するか
        min_length: 単語の最小長（これ未満の単語は除外）

    Returns:
        単語リスト
    """
    # 英数字とハイフンを単語文字として扱う
    words = re.findall(r"[a-zA-Z0-9]+(?:'[a-zA-Z0-9]+)*", text)
    
    if not case_sensitive:
        words = [w.lower() for w in words]
    
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度を集計する。

    Args:
        words: 単語リスト

    Returns:
        単語からカウントへのCounter
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位N件を取得する。同じカウントの場合は単語の辞書順で安定ソートする。

    Args:
        counter: 単語からカウントへのCounter
        n: 上位N件

    Returns:
        (単語, カウント)のリスト、上位順
    """
    # Counter.most_commonはカウント降順だが、同カウント時の順序は保証しない
    # 同カウント時は辞書順で安定させる
    items = [(word, count) for word, count in counter.items()]
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
