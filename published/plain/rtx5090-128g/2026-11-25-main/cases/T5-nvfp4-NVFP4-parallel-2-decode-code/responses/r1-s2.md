コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を上位N件表示する。

使い方:

    python word_counter.py input.txt --top 20 --case-sensitive

import argparse

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_sensitive: Trueの場合、大文字小文字を区別する

    Returns:
        単語のリスト
    """
    # 英数字の連続を単語として抽出
    tokens = re.findall(r'[A-Za-z0-9]+', text)
    if not case_sensitive:
        tokens = [token.lower() for token in tokens]
    return tokens


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_sensitive: Trueの場合、大文字小文字を区別する

    Returns:
        単語 -> 出現回数 のCounter
    """
    tokens = tokenize(text, case_sensitive)
    return Counter(tokens)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    上位N件の単語と出現回数を取得する。

    Args:
        counter: 単語出現カウント
        n: 取得する件数

    Returns:
        (単語, 回数) のタプルのリスト（降順）
    """
    return counter.most_common(n)
