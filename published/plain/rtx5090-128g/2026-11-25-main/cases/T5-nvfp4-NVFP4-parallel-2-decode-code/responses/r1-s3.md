"""
コマンドラインツール: テキストファイルを読み込んで単語の出現頻度を集計し、
上位 N 件を表形式で表示する。

使い方:
    python wordfreq.py input.txt -n 10

オプション:
    -n, --top N      表示する上位単語の数 (デフォルト: 10)
    -f, --file FILE  入力テキストファイルのパス (必須)
    --case-sensitive 大文字小文字を区別する (デフォルト: 区別しない)
"""

import argparse
import re
import sys
from collections import Counter
from typing import Dict, List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合、大文字小文字を区別する

    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字のみを単語として抽出
    words = re.findall(r'[a-z0-9]+', text, re.IGNORECASE)
    return words


def count_words(words: List[str]) -> Dict[str, int]:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とする辞書
    """
    return dict(Counter(words))

    word_counts: Dict[str, int],

        (単語, 出現回数) のタプルのリスト。出現回数の降順でソートされている。
        同数の場合は単語の辞書順でソートされる。
    """
