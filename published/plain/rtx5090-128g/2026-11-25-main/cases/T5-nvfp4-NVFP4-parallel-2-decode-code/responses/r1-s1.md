def tokenize(text: str) -> List[str]:
    """
    テキストを単語のリストに分割する。

    単語は英数字およびハイフン、アポストロフィを含む連続文字列とみなす。
    大文字小文字は無視して小文字に変換する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    # 英数字、ハイフン、アポストロフィを含むトークンを抽出
    tokens = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return [token.lower() for token in tokens]

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    tokens = tokenize(text)
    return Counter(tokens)
