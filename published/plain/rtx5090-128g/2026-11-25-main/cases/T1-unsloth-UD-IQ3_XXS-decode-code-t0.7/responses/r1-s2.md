def read_file(path: str) -> str:
    """テキストファイルを読み込み、内容を返す。

    Args:
        path: 読み込むファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み取りに失敗した場合。
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """テキストから単語（トークン）を抽出する。

    英数字とハイフンを含む連続文字列を単語とみなす。
    大文字小文字の区別や最小長を指定できる。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_len: 抽出する単語の最小文字数。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()

    tokens: List[str] = []
    current: List[str] = []
    for char in text:
        if char.isalnum() or char == "-":
            current.append(char)
        else:
            if current:
                token = "".join(current)
                if len(token) >= min_len:
                    tokens.append(token)
                current = []
    if current:
        token = "".join(current)
        if len(token) >= min_len:
            tokens.append(token)

    return tokens
