#!/usr/bin/env python3
"""
Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N most frequent words
in a formatted table.
"""

import argparse
import sys
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Args:
        text: Input text string.

    Returns:
        List of lowercase word tokens.
    """
    # Convert to lowercase and extract words (sequences of alphanumeric characters)
    return re.findall(r'\b[a-z0-9]+\b', text.lower())


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        Counter object with word frequencies.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to retrieve.

    Returns:
        List of tuples (word, count) sorted by count descending, then alphabetically.
    """
    # most_common returns list of tuples (word, count)
    return counter.most_common(n)
