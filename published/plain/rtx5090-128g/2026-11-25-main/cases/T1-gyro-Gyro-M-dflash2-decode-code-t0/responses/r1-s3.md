"""
word_freq.py - コマンドラインツール: テキストファイルから単語出現頻度を上位N件表示
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    英数字の連続を単語とし、それ以外を区切りとする。
    すべて小文字に変換する。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト
    """
    # 英数字の連続を抽出し、小文字化
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [w.lower() for w in words]


def count_words(text: str) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語をキー、出現回数を値とするCounter
    """
    words = tokenize(text)
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counterから上位n件の(単語, 回数)を返す。
    回数が同じ場合は単語の辞書順でソートする。

    Args:
        counter: 単語出現頻度のCounter
        n: 上位件数

    Returns:
        (単語, 回数)のリスト、上位順
    """
    # 回数降順、同回数なら単語昇順
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
