def normalize_words(text: str) -> List[str]:
    """テキストから単語を抽出し、小文字化してリストとして返す。

    Args:
        text: 入力テキスト文字列。

    Returns:
        小文字化された単語のリスト（句読点・空白は除去）。
    """
    words: List[str] = []
    for token in text.split():
        # 句読点や非アルファベット文字を除去
        cleaned = "".join(c for c in token if c.isalpha())
        if cleaned:
            words.append(cleaned.lower())
    return words


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """単語リストから出現頻度を Counter で集計する。

    Args:
        words: 単語のリスト。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Counter から上位 N 件の (word, count) を取得する。

    Args:
        counter: 単語頻度 Counter。
        n: 取得する件数。

    Returns:
        上位 n 件の (word, count) のリスト（頻度降順）。
    """
    return counter.most_common(n)
