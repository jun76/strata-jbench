def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    単語は英数字とハイフンの連続文字列として認識される。
    大文字/小文字は区別されず、すべて小文字に正規化する。

    Args:
        text: 入力テキスト

    Returns:
        小文字化された単語のリスト
    """
    words = []
    current = []
    for char in text:
        if char.isalnum() or char == '-':
            current.append(char.lower())
        else:
            if current:
                words.append(''.join(current))
                current = []
    if current:
        words.append(''.join(current))
    return words


def count_words(text: str) -> Counter[str]:
    """
    テキスト中の単語出現頻度を集計する。

    Args:
        text: 入力テキスト

    Returns:
        単語 -> 出現回数の Counter
    """
    words = tokenize(text)
    return Counter(words)
