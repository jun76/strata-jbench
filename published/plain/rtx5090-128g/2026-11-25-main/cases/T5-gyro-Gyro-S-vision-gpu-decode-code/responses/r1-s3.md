"""
wordfreq.py - コマンドラインツール：テキストファイルの単語出現頻度を上位N件表示する。
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def load_text(file_path: str) -> str:
    """
    テキストファイルを読み込んで、その内容を文字列として返す。

    Args:
        file_path: ファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み込み失敗。
    """
    path = Path(file_path)
    return path.read_text(encoding="utf-8")


def count_words(text: str) -> Counter:
    """
    テキストから単語（空白区切り）を抽出し、出現頻度をカウントする。
    単語は小文字化して正規化する。

    Args:
        text: テキスト内容。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    words = text.split()
    normalized = [w.lower() for w in words]
    return Counter(normalized)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (word, count) のリストを返す。
    同スコアの場合は辞書順（アルファベット順）で安定してソートする。

    Args:
        counter: 単語 -> 出現回数 の Counter。
        n: 上位件数。

    Returns:
        [(word, count), ...] のリスト。
    """
    # 上位 n 件を取得（同スコアは辞書順で安定）
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
