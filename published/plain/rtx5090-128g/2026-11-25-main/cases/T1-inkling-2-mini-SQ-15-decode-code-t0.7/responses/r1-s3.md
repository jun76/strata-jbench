"""テキストファイルの単語出現頻度を集計して上位 N 件を表示する CLI。"""
from __future__ import annotations

import argparse
from collections import Counter
from pathlib import Path
import re
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """英字・数字の単語を抽出して出現回数を返す。"""
    words = re.findall(r"[A-Za-z0-9_]+", text.lower())
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """出現頻度の高い順に上位 n 件を返す（同率は単語で昇順）。"""
    return sorted(counter.items(), key=lambda item: (-item[1], item[0]))[:n]


def read_file(path: Path) -> str:
    """UTF-8 テキストファイルを読み込む。"""
    return path.read_text(encoding="utf-8")


def format_table(rows: List[Tuple[str, int]]) -> str:
    """表形式の文字列を生成する。"""
    if not rows:
        return "No words found."
    headers = ("Rank", "Word", "Count")
    lines = [f"{headers[0]:>5} | {headers[1]:<30} | {headers[2]:>6}"]
    lines.append("-" * len(lines[0]))
    for i, (word, count) in enumerate(rows, start=1):
        lines.append(f"{i:>5} | {word:<30} | {count:>6}")
    return "\n".join(lines)
