import argparse
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def load_text(path: Path) -> str:
    """テキストファイルを読み込む。"""
    return path.read_text(encoding="utf-8")


def count_words(text: str) -> Counter:
    """テキスト中の単語を小文字化して集計する。"""
    words = text.lower().split()
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """頻度上位 N 件を返す。"""
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """上位 N 件を表形式で整形する。"""
    if not items:
        return "(no words)"

    max_word_len = max(len(word) for word, _ in items)
    max_count_len = max(len(str(count)) for _, count in items)

    header = f"{'Word'.ljust(max_word_len)}  {'Count'.rjust(max_count_len)}"
    separator = "-" * (max_word_len + max_count_len + 2)

    rows = [header, separator]
    for word, count in items:
        rows.append(f"{word.ljust(max_word_len)}  {str(count).rjust(max_count_len)}")

    return "\n".join(rows)


def main() -> None:
    parser = argparse.ArgumentParser(
        description="テキストファイルの単語出現頻度を上位 N 件表示します。"
    )
    parser.add_argument("file", type=Path, help="入力テキストファイル")
    parser.add_argument("-n", type=int, default=10, help="表示件数（デフォルト: 10）"
