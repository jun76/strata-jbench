以下は、テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で表示するコマンドラインツールです。

```python
#!/usr/bin/env python3
"""テキストファイルから単語の出現頻度を集計し、上位N件を表示するCLIツール。"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを小文字化して単語に分割する。"""
    return re.findall(r"\b\w+\b", text.lower())


def count_words(tokens: List[str]) -> Counter:
    """トークンリストから単語の出現回数を集計する。"""
    return Counter(tokens)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """出現頻度の高い上位n件を返す。同数はアルファベット順。"""
    return sorted(counter.items(), key=lambda item: (-item[1], item[0]))[:n]


def format_table(rows: List[Tuple[str, int]]) -> str:
    """上位N件を表形式の文字列として整形する。"""
    if not rows:
        return "（単語が見つかりません）"

    headers = ["Rank", "Word", "Count"]
    data = [[str(i + 1), word, str(count)] for i, (word, count) in enumerate(rows)]

    widths = [
        max(len(headers[i]), max((len(row[i]) for row in data), default=0))
        for i in range(len(headers))
    ]

    def line(values: List[str]) -> str:
        return " | ".join(value.ljust(widths[i])
