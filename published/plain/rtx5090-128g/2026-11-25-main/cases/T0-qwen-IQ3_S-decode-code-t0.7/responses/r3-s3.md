def read_file(file_path: str) -> str:
    """指定されたファイルを読み込み、その内容を文字列として返す。

    Args:
        file_path: 読み込むファイルのパス

    Returns:
        ファイルの内容 (文字列)

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        UnicodeDecodeError: ファイルがUTF-8でデコードできない場合
    """
    with open(file_path, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """テキストから単語を抽出する。

    単語は英数字（a-z, A-Z, 0-9）およびハイフン、アポストロフィを
    含む連続した文字列として定義される。
    case_sensitive が False の場合、単語は小文字に変換される。
    min_length より短い単語は除外される。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか
        min_length: 単語の最小長

    Returns:
        抽出された単語のリスト
    """
    # 英数字、ハイフン、アポストロフィを含むトークンを抽出
    tokens = re.findall(r"[a-zA-Z0-9'\-]+", text)

    if not case_sensitive:
        tokens = [token.lower() for token in tokens]

    # min_length 以上の単語のみ残す
    tokens = [token for token in tokens if len(token) >= min_length]

    return tokens
