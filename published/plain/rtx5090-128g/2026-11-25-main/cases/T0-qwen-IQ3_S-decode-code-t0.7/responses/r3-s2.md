import argparse

import re
from collections import Counter

def load_text(filepath: str) -> str:
    """テキストファイルを読み込んで文字列を返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を示す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み取りエラーが発生した場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語トークンを抽出する。

    単語は英数字（a-z, A-Z, 0-9）およびアンダースコアで構成される連続文字列として定義される。
    大文字/小文字を区別しない場合は、トークンを小文字に変換する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: True の場合、大文字・小文字を区別する。False の場合、小文字に正規化する。

    Returns:
        単語トークンのリスト。
    """
    # 単語を抽出: 英数字とアンダースコアの連続
    words = re.findall(r"[a-zA-Z0-9_]+", text)
    if not case_sensitive:
        words = [word.lower() for word in words]
    return words


def count_frequencies(words: List[str]) -> Counter[str]:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とするCounter辞書。
    """
    return Counter(words)
