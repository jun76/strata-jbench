def read_text_file(filepath: str, encoding: str = 'utf-8') -> str:
    """
    指定されたパスのテキストファイルを読み込み、内容を文字列として返す。

    Args:
        filepath (str): 読み込むファイルのパス。
        encoding (str): ファイルのエンコーディング。デフォルトは 'utf-8'。

    Returns:
        str: ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        UnicodeDecodeError: 指定エンコーディングでデコードできない場合。
    """
    with open(filepath, 'r', encoding=encoding) as f:
        return f.read()


def normalize_text(text: str) -> str:
    """
    テキストを正規化する。
    - 大文字を小文字に変換
    - アルファベット以外の文字を空白に変換（単語の境界を作る）
    
    Args:
        text (str): 正規化するテキスト。

    Returns:
        str: 正規化されたテキスト。
    """
    # 大文字を小文字に
    text = text.lower()
    # 英数字以外の文字をスペースに置き換えることで、単語を分離
    # ここでは、英数字とスペース以外の文字をスペースに変換
    normalized_chars = []
    for char in text:
        if char.isalnum() or char.isspace():
            normalized_chars.append(char)
        else:
            normalized_chars.append(' ')
    
    return ''.join(normalized_chars)
