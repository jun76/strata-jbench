Reads a text file, counts word occurrences, and displays the top N most frequent

def extract_words(text: str) -> List[str]:
    """
    Extract words from text by splitting on whitespace and removing punctuation.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase words with punctuation stripped.
    """
    # Split on whitespace and strip common punctuation from each token
    words: List[str] = []
    for token in text.split():
        # Strip leading and trailing punctuation (keep internal apostrophes for contractions)
        cleaned = token.strip(".,!?;:\"'()[]{}-–—…")
        if cleaned:
            words.append(cleaned.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word.
    
    Args:
        words: A list of words.
        
    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top items to return.
        
    Returns:
        A list of tuples (word, count) for the top N most common words.
    """
    return counter.most_common(n)
