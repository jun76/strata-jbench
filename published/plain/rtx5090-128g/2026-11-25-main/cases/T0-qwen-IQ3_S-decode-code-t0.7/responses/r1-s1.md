"""
word_freq_counter.py

A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    Count word occurrences in the given text.
    
    Words are defined as sequences of alphanumeric characters.
    All words are lowercased before counting.
    
    Args:
        text: The input text to analyze.
        
    Returns:
        A Counter object mapping each word to its occurrence count.
    """
    # Extract words: sequences of alphanumeric characters
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    # Normalize to lowercase
    words = [word.lower() for word in words]
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word counts.
        n: The number of top entries to return.
        
    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then by word ascending for ties.
    """
    # most_common returns items sorted by count descending
    # For ties, we need to sort alphabetically
    items = counter.items()
    # Sort by count descending, then by word ascending
    sorted_items = sorted(items, key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
