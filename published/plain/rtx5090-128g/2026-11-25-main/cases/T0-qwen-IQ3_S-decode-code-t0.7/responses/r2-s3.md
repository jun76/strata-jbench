#!/usr/bin/env python3
"""
wordfreq.py - テキストファイルの単語出現頻度を集計し、上位N件を表形式で表示するツール。
"""

import argparse
import re
from collections import Counter
from typing import Dict, List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    正規表現を使って英数字（およびアンダースコア）で構成されたトークンを抽出し、
    すべて小文字に変換してリストとして返す。

    Args:
        text: 入力テキスト。

    Returns:
        抽出された単語のリスト（小文字化済み）。
    """
    # \w+ は [a-zA-Z0-9_] に一致する
    words = re.findall(r'\w+', text)
    return [word.lower() for word in words]


def count_words(text: str) -> Dict[str, int]:
    """
    テキスト内の単語出現頻度を集計する。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とする辞書。
    """
    words = extract_words(text)
    return dict(Counter(words))


def get_top_n(counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の高い順に上位n件の単語と回数を取り出す。
    回数が同じ場合は辞書順（単語のアルファベット順）で並べる。

    Args:
        counts: 単語の出現頻度辞書。
        n: 取り出す件数。

    Returns:
        (単語, 回数) のタプルのリスト。上位 n 件。
    """
    # (-count, word) でソートして、回数の降順、単語の昇順（辞書順）を実現
    sorted_items = sorted(counts.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
