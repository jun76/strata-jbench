def read_text(filepath: str) -> str:
    """ファイルを読み込んでテキストを返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み取りに失敗した場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストを単語リストに分割する。

    単語は英数字とアンダースコアからなるトークンとして扱う。
    大文字小文字の区別は case_sensitive フラグで制御する。

    Args:
        text: 分割するテキスト。
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        単語（トークン）のリスト。
    """
    if not case_sensitive:
        text = text.lower()

    words: List[str] = []
    current: List[str] = []
    for ch in text:
        if ch.isalnum() or ch == '_':
            current.append(ch)
        else:
            if current:
                words.append("".join(current))
                current = []
    if current:
        words.append("".join(current))
    return words


def count_frequencies(words: List[str]) -> Dict[str, int]:
    """単語リストから出現頻度辞書を構築する。

    Args:
        words: 単語のリスト。

    Returns:
        単語 -> 出現回数 の辞書。
    """
    freq: Dict[str, int] = {}
    for word in words:
        freq[word] = freq.get(word, 0) + 1
    return freq
