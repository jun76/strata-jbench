```python
"""単語出現頻度集計 CLI ツール。

テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で表示します。
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Sequence


WORD_RE = re.compile(r"[A-Za-z0-9_]+(?:'[A-Za-z0-9]+)?")


def tokenize(text: str) -> list[str]:
    """テキストを小文字の単語トークンに分割する.

    Args:
        text: 入力テキスト.

    Returns:
        小文字化された単語のリスト.

    Examples:
        >>> tokenize("Hello, hello! World's best.")
        ['hello', 'hello', "world's", 'best']
    """
    return [m.group(0).lower() for m in WORD_RE.finditer(text)]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を集計する.

    Args:
        text: 入力テキスト.

    Returns:
        単語をキー、出現回数を値とする Counter.
    """
    return Counter(tokenize(text))


def read_file(path: Path, encoding: str = "utf-8") -> str:
    """ファイルを読み込んで文字列を返す.

    Args:
        path: 読み込むファイルパス.
        encoding: 文字エンコーディング (既定: utf-8).

    Returns:
        ファイル全体の文字列.

    Raises:
        FileNotFoundError: ファイルが存在しない場合.
        UnicodeDecodeError: エンコーディングが不正な場合.
    """
    return path.read_text(encoding=encoding)
