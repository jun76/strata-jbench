def tokenize(text: str) -> List[str]:
    """テキストを単語にトークン化する。

    単語は連続する英数字（\w+）として検出され、小文字化される。
    大文字/小文字は無視されるが、数字やアンダースコアは単語の一部として扱われる。

    Args:
        text: 入力テキスト文字列。

    Returns:
        小文字化された単語のリスト。
    """
    # \w は [a-zA-Z0-9_] に相当。単語の境界を明確にするため
    # 英字または数字を含むトークンだけを抽出。
    return re.findall(r"[a-zA-Z0-9]+", text.lower())


def count_words(text: str) -> Counter:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト文字列。

    Returns:
        単語 -> 出現回数の Counter オブジェクト。
    """
    return Counter(tokenize(text))
