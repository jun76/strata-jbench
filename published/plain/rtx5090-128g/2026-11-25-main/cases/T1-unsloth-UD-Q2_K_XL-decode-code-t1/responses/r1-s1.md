def read_text_file(file_path: str) -> str:
    """
    Read the contents of a text file and return it as a string.

    Args:
        file_path: Path to the text file to read.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        PermissionError: If the file cannot be read due to permissions.
        OSError: For other I/O errors.
    """
    with open(file_path, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """
    Split the text into words using a regex that matches alphanumeric
    sequences (including underscores). Words are lowercased for case-insensitive
    counting.

    Args:
        text: The input text string.

    Returns:
        A list of lowercased words extracted from the text.
    """
    # Match sequences of alphanumeric characters and underscores
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return [word.lower() for word in words]
