_WORD_RE = re.compile(r"[A-Za-z0-9']+")


def tokenize(text: str, *, case_sensitive: bool = False) -> list[str]:
    """テキストを単語トークンに分割する。

    Args:
        text: 解析対象の文字列。
        case_sensitive: True なら大文字小文字を区別し、False なら小文字化して
            区別しない（デフォルト）。

    Returns:
        単語トークンのリスト。句読点や空白は除去される。
    """
    words = _WORD_RE.findall(text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_words(text: str, *, case_sensitive: bool = False) -> Counter[str]:
    """テキスト中の単語出現頻度を数える。

    Args:
        text: 解析対象の文字列。
        case_sensitive: 大文字小文字を区別するか。False なら小文字化して集計。

    Returns:
        単語をキー、出現回数を値とする ``collections.Counter``。
    """
    return Counter(tokenize(text, case_sensitive=case_sensitive))
