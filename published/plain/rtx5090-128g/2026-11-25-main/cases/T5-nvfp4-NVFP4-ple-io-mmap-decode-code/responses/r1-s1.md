Usage:

def extract_words(text: str, case_insensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    Extract words from the given text.

    Words are defined as sequences of alphanumeric characters.
    Non-alphanumeric characters are treated as delimiters.

    Args:
        text: The input text to extract words from.
        case_insensitive: If True, convert all words to lowercase before counting.
        min_len: Minimum length of a word to be included (default 1).

    Returns:
        A list of extracted words.
    """
    # Find all sequences of alphanumeric characters as words
    words = re.findall(r'[A-Za-z0-9]+', text)
    if case_insensitive:
        words = [w.lower() for w in words]
    if min_len > 1:
        words = [w for w in words if len(w) >= min_len]
    return words


def count_words(words: List[str]) -> Dict[str, int]:
    """
    Count the frequency of each word in the list.

    Args:
        words: A list of words to count.

    Returns:
        A dictionary mapping each word to its frequency.
    """
    return dict(Counter(words))
