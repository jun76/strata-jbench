"""
word_freq - Count word frequencies in a text file and display the top N results.

Usage:
    python word_freq.py <input_file> [-n TOP_N] [--case-sensitive]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, words are lowercased before counting.

    Returns:
        A list of word tokens.
    """
    # Extract words (sequences of alphanumeric characters and apostrophes/hyphens within words)
    words = re.findall(r"[a-zA-Z0-9]+(?:['\-][a-zA-Z0-9]+)*", text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_word_frequencies(text: str, case_sensitive: bool = False) -> Counter:
    """
    Count the frequency of each word in the given text.

    Args:
        text: The input text.
        case_sensitive: If False, words are lowercased before counting.

    Returns:
        A Counter object with word frequencies.
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top results to return.

    Returns:
        A list of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)
