"""
count_words.py — 単語出現頻度集計コマンドラインツール

テキストファイルを読み込み、単語の出現頻度を集計し、上位 N 件を
表形式で表示する。

使用方法:
    python count_words.py [options] FILE

オプション:
    -n N       表示する上位件数 (デフォルト: 10)
    --sort     頻度の降順でソート (デフォルト: True)
    --case-sensitive  大文字小文字を区別 (デフォルト: False)
    --word-pattern    単語を抽出する正規表現パターン
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(
    text: str,
    pattern: str = r"\b[a-zA-Z]+(?:'\w+)?\b",
    case_sensitive: bool = False,
) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        pattern: 単語を抽出する正規表現パターン
        case_sensitive: True の場合、大文字小文字を区別する

    Returns:
        抽出された単語のリスト
    """
    matches = re.findall(pattern, text)
    if not case_sensitive:
        matches = [w.lower() for w in matches]
    return matches


def count_words(
    text: str,
    pattern: str = r"\b[a-zA-Z]+(?:'\w+)?\b",
    case_sensitive: bool = False,
) -> Counter[str]:
    """
    テキスト中の単語出現頻度を集計する。

    Args:
        text: 入力テキスト
        pattern: 単語を抽出する正規表現パターン
        case_sensitive: True の場合、大文字小文字を区別する

    Returns:
        単語とその出現回数を保持する Counter
    """
    words = extract_words(text, pattern, case_sensitive)
    return Counter(words)
