"""
word_count.py - Text word frequency counter CLI tool.

Reads a text file, tokenizes words, counts occurrences, and displays
the top N words in a formatted table.
"""

from argparse import ArgumentParser, Namespace
from collections import Counter
from typing import Dict, List, Tuple
import sys


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words (non-empty, stripped of punctuation).
    """
    # Remove punctuation and convert to lowercase
    words = []
    for word in text.lower().split():
        # Strip common punctuation from start and end
        cleaned = word.strip(".,!?;:'\"()[]{}")
        if cleaned:
            words.append(cleaned)
    return words


def count_words(text: str) -> Dict[str, int]:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text string.

    Returns:
        A dictionary mapping each word to its occurrence count.
    """
    words = tokenize(text)
    return dict(Counter(words))
