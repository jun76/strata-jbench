#!/usr/bin/env python3
"""テキストファイルから単語出現頻度を計算し、上位 N 件を表形式で出力するコマンドラインツール。"""

import argparse

def tokenize(text: str) -> List[str]:
    """テキストを単語リストに分割する。

    単語は英数字の連続（\w+）として定義され、すべて小文字に変換される。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    return re.findall(r'\w+', text.lower())


def count_words(text: str) -> Counter:
    """テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(tokenize(text))


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (word, count) を返す。

    出現回数の降順、同数時は辞書順で並び替える。

    Args:
        counter: 単語頻度 Counter。
        n: 取得する上位件数。

    Returns:
        (word, count) タプルのリスト。
    """
    return sorted(counter.items(), key=lambda x: (-x[1], x[0]))[:n]
