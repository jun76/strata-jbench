word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位 N 件の単語とその頻度を表形式で出力するコマンドラインツール。
"""

import argparse
import re

def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    単語は英数字とハイフンで構成される文字列として定義される。
    大文字小文字は無視される（すべて小文字に変換）。
    
    Args:
        text (str): 入力テキスト。
        
    Returns:
        List[str]: 抽出された単語のリスト（小文字）。
    """
    # 英数字とハイフンを許容する正規表現
    # \b は単語境界、[a-z0-9-]+ は英数字とハイフン
    words = re.findall(r'\b[a-z0-9-]+\b', text.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。
    
    Args:
        words (List[str]): 単語のリスト。
        
    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度が高い順に上位 N 件の単語と頻度を取得する。
    
    Args:
        counter (Counter): 単語の頻度カウンター。
        n (int): 取得する件数。
        
    Returns:
        List[Tuple[str, int]]: (単語, 頻度) のタプルのリスト。
    """
    return counter.most_common(n)
