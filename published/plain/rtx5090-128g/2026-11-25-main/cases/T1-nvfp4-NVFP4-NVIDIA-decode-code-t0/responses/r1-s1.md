word_freq.py - テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive] [--min-length LEN]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse

from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、すべて小文字に変換する。
        min_length: この長さ未満の単語は除外する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む単語を抽出（簡易的なトークナイザ）
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return [w for w in words if len(w) >= min_length]


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 最小単語長。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    words = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(words)
