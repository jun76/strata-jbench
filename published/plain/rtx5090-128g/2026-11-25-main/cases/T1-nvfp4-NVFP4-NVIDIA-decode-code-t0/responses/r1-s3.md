word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントし、

def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    単語は英数字とハイフンで構成される文字列として定義される。
    大文字小文字は無視される（すべて小文字に変換）。
    
    Args:
        text: 入力テキスト
        
    Returns:
        抽出された単語のリスト
    """
    # 英数字とハイフンを許容する正規表現パターン
    pattern = r'[a-zA-Z0-9-]+'
    words = re.findall(pattern, text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。
    
    Args:
        words: 単語のリスト
        
    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    return Counter(words)

    Returns:

    """
