def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語トークンを抽出する。

    句読点や記号は除去し、空白区切りで分割する。
    case_sensitive が False の場合、小文字に変換する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。

    Returns:
        単語トークンのリスト。
    """
    if not case_sensitive:
        text = text.lower()

    # 句読点や記号を空白に置き換えてから分割
    import string
    translator = str.maketrans('', '', string.punctuation)
    cleaned = text.translate(translator)
    words = cleaned.split()

    # 空文字列や数字のみを除く
    return [w for w in words if w and not w.isdigit()]


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語トークンのリスト。

    Returns:
        単語ごとの出現回数を格納した Counter。
    """
    return Counter(words)
