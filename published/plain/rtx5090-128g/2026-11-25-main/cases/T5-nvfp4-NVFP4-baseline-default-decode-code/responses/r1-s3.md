#!/usr/bin/env python3
"""
テキストファイルを読み込み、単語の出現頻度を集計して上位N件を表形式で表示するコマンドラインツール。

使い方:
    python word_freq.py <file> [-n N] [--sort-by {count,word}]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。英数字のみのトークンを抽出する。
    """
    return re.findall(r'[a-z0-9]+', text.lower())


def count_words(text: str) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    words = tokenize(text)
    return Counter(words)


def top_n(counter: Counter, n: int, sort_by: str = "count") -> List[Tuple[str, int]]:
    """
    Counterから上位n件の単語と回数を取得する。

    Args:
        counter: 単語頻度のCounter。
        n: 取得する件数。
        sort_by: 並び順の基準。"count"で回数降順、"word"でアルファベット順。

    Returns:
        (単語, 回数)のタプルのリスト。sort_byが"count"の場合は回数降順、同数ならアルファベット順。
        sort_byが"word"の場合はアルファベット順。
    """
    if sort_by == "word":
        items = sorted(counter.items(), key=lambda x: x[0])
    else:
        # count降順、同数ならアルファベット順
        items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))

    return items[:n]
