def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """
    単語を正規化する。

    Args:
        word: 元の単語文字列。
        case_sensitive: True の場合は大文字小文字を区別し、False の場合は小文字に変換。

    Returns:
        正規化された単語文字列。
    """
    if not case_sensitive:
        return word.lower()
    return word


def tokenize(text: str, min_len: int = 1) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    単語は英数字とハイフン、アポストロフィを含む連続文字列として定義される。
    指定した最小長以上の単語のみを返す。

    Args:
        text: 入力テキスト。
        min_len: 単語の最小文字数（この長さ未満のトークンは除外）。

    Returns:
        正規化前のトークンリスト。
    """
    # 英数字、ハイフン、アポストロフィを含む連続文字列
    tokens = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return [t for t in tokens if len(t) >= min_len]
