#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位N件表示する。

使い方:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt --top 20 --case-sensitive
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフンを含む単語を抽出（日本語はそのまま）
    words = re.findall(r'\b[\w-]+\b', text)
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (単語, 回数) を取得する。

    Args:
        counter: 単語頻度の Counter。
        n: 取得する件数。

    Returns:
        (単語, 回数) のタプルのリスト。回数の降順、同数時は単語の辞書順。
    """
    # most_common は回数の降順で返すが、同数の場合の順序は保証されないため、
    # 明示的にソートする
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
