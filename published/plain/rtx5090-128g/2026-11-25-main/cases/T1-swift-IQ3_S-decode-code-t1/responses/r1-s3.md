"""Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N most frequent
words in a table format.

Usage:
    python word_freq.py <file> [-n TOP_N]
"""

def read_file(filepath: str) -> str:
    """Read the entire content of a file.

    Args:
        filepath: Path to the text file.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Split text into lowercase words, stripping non-alphanumeric characters.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase word tokens.
    """
    import re
    # Match sequences of alphanumeric characters (including underscores as part of words)
    words = re.findall(r'[a-zA-Z0-9_]+', text)
    return [word.lower() for word in words]


def count_words(tokens: List[str]) -> Dict[str, int]:
    """Count the frequency of each word in the token list.

    Args:
        tokens: List of word tokens.

    Returns:
        A dictionary mapping each word to its occurrence count.
    """
    return dict(Counter(tokens))


def get_top_n(word_counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """Return the top N most frequent words.

    Words with the same frequency are sorted alphabetically.

    Args:
        word_counts: Dictionary mapping words to their counts.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending, then word ascending.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_words = sorted(word_counts.items(), key=lambda x: (-x[1], x[0]))
