"""
wordfreq.py - Text file word frequency counter CLI tool.

Reads a text file, counts word frequencies, and displays the top N words
in a table format.
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.
    
    Words are defined as sequences of alphanumeric characters.
    Returns a list of lowercase words.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase words.
    """
    return re.findall(r'[a-z0-9]+', text.lower())


def count_words(text: str) -> Counter:
    """
    Count the frequency of each word in the given text.
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter object with word frequencies.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.
        
    Returns:
        A list of tuples (word, count) for the top N words.
    """
    return counter.most_common(n)


def format_table(data: List[Tuple[str, int]]) -> str:
    """
    Format word-frequency pairs as a table string.
    
    Args:
        data: A list of tuples (word, count).
        
    Returns:
        A formatted table string.
    """
    if not data:
        return "No words found."
    
    # Determine column widths
    word_width = max(len(word) for word, _ in data)
    word_width = max(word_width, 5)  # minimum width for header "Word"
    count_width = max(len(str(count)) for _, count in data)
    count_width = max(count_width, 5)  # minimum width for header "Count"
    
    # Header
    header = f"{'Word'.ljust(word_width)} | {'Count'.rjust(count_width
