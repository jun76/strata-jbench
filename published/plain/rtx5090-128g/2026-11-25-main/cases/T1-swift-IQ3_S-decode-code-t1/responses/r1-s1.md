#!/usr/bin/env python3
"""
word_freq: Count word frequencies in a text file and display top N words.

Usage:
    python word_freq.py <input_file> [-n TOP_N] [--case-sensitive]
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = True) -> List[str]:
    """Extract words from text, lowercasing if case_sensitive is False."""
    words = re.findall(r'\b[a-zA-Z]+(?:\'[a-zA-Z]+)?\b', text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_words(words: List[str]) -> Counter:
    """Count occurrences of each word."""
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words as (word, count) tuples."""
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]], rank_width: int = 4) -> str:
    """Format a list of (word, count) tuples into a formatted table string."""
    if not items:
        return ""
    max_word_len = max(len(word) for word, _ in items)
    max_count_len = max(len(str(count)) for _, count in items)
    lines: List[str] = []
    header_rank = "Rank".rjust(rank_width)
    header_word = "Word".ljust(max_word_len)
    header_count = "Count".rjust(max_count_len)
    lines.append(f"{header_rank}  {header_word}  {header_count}")
    separator = f"{'-' * rank_width}  {'-' * max_word_len}  {'-' * max_count_len}"
    lines.append(separator)
    for rank, (word, count) in enumerate(items, 1):
        rank_str = str(rank).rjust(rank_width)
        word_str = word.ljust(max_word_len)
        count_str = str(count).rjust(max_count_len)
        lines.append(f"{rank_str}  {word_str}  {count_str}")
    return "\n".join(lines)
