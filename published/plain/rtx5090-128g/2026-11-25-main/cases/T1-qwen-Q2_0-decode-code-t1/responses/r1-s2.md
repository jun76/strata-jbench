    assert args.n == 5


def test_parse_arguments_default_n():
    parser = parse_arguments()

def test_count_words_simple():
    text = "hello world hello again"
    counts = count_words(text)
    assert counts["hello"] == 2
    assert counts["world"] == 1
    assert counts["again"] == 1


def test_count_words_case_insensitive():
    text = "Hello hello HELLO"
    counts = count_words(text)
    assert counts["hello"] == 3


def test_count_words_handles_punctuation():
    text = "don't, don't"
    counts = count_words(text)
    # Our simple splitting: words are alphanumeric sequences
    # "don't" → ['don', 't'] (since apostrophe is not in word chars)
    assert counts["don"] == 2
    assert counts["t"] == 2
