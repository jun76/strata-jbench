#!/usr/bin/env python3
"""Text file word frequency counting command-line tool."""

import argparse

from collections import Counter

def count_words(text: str) -> List[Tuple[str, int]]:
    """Count word occurrences in the given text.

    Args:
        text: The input text to analyze.

    Returns:
        A list of (word, count) tuples sorted by count descending,
        then alphabetically for ties.
    """
    # Simple word extraction: split on whitespace and punctuation-like chars
    # For simplicity, treat words as sequences of alphanumeric characters.
    words = []
    for line in text.splitlines():
        for token in line.split():
            # Strip common punctuation from edges
            stripped = token.strip('.,!?"\'()[]{}')
            if stripped:
                words.append(stripped.lower())
    counter = Counter(words)
    return sorted(counter.items(), key=lambda x: (-x[1], x[0]))


def format_table(entries: List[Tuple[str, int]], n: int) -> str:
    """Format top N entries as a simple text table.

    Args:
        entries: List of (word, count) tuples.
        n: Number of top entries to display.

    Returns:
        A formatted table string.
    """
    top = entries[:n]
    if not top:
        return "No words found."

    # Determine column widths
    word_width = max(len(w) for w, _ in top)
    count_width = max(len(str(c)) for _, c in top)

    header = f"{'Word':<{word_width}}  {'Count':<{count_width}}"
    separator = '-' * (word_width + 2 + count_width)
    lines = [header, separator]
    for word, count in top:
        lines.append(f"{word:<{word_width}}  {count:<{count_width}}")
    return "\n".join(lines)
