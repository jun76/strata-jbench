def read_text_file(filepath: str) -> str:
    """Read a text file and return its contents as a string.
    
    Args:
        filepath: Path to the text file to read.
        
    Returns:
        The contents of the file as a string.
    """
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except OSError as e:
        raise OSError(f"Error reading file {filepath}: {e}")


def count_words(text: str) -> Dict[str, int]:
    """Count word occurrences in a text string.
    
    Words are extracted by splitting on whitespace and punctuation.
    Words are case-insensitive (lowercased).
    
    Args:
        text: The input text string.
        
    Returns:
        A dictionary mapping words to their occurrence counts.
    """
    # Normalize: lower case
    text = text.lower()
    # Replace common punctuation with spaces for splitting
    for char in ".,!?;:\"\''()[]{}<>/@#$%&*~^|_`+-=":
        text = text.replace(char, " ")
    # Split on whitespace
    words = text.split()
    # Filter out empty strings
    words = [w for w in words if w]
    return dict(Counter(words))
