word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位 N 件の単語を表形式で出力するコマンドラインツール。

Usage:
    python word_counter.py <file> [-n N] [--case-sensitive] [--min-length L]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 単語の最小長（この長さ未満の単語は除外）。

    Returns:
        抽出された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    words = re.findall(r"[a-z0-9]+(?:[-'][a-z0-9]+)*", text)
    return [w for w in words if len(w) >= min_length]


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 単語の最小長。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(words)
