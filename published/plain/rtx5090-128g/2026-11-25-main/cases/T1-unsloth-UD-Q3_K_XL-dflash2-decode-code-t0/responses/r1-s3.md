テキストファイルを読み込み、単語の出現頻度を数え、
上位 N 件を表形式で出力するコマンドラインツール。

Usage:
    python wordfreq.py input.txt -n 10
    python wordfreq.py input.txt --top 20 --case-sensitive
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語トークンを抽出する.

    Args:
        text: 入力テキスト.
        case_sensitive: True の場合、大文字小文字を区別する.

    Returns:
        単語トークンのリスト.
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフンを含む単語を抽出（日本語は未対応）
    tokens = re.findall(r"[a-z0-9]+(?:[-'][a-z0-9]+)*", text)
    return tokens


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """テキスト内の単語出現頻度をカウントする.

    Args:
        text: 入力テキスト.
        case_sensitive: True の場合、大文字小文字を区別する.

    Returns:
        単語をキー、出現回数を値とする Counter.
    """
    tokens = tokenize(text, case_sensitive=case_sensitive)
    return Counter(tokens)
