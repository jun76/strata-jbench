#!/usr/bin/env python3
"""word_freq: テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

使用例:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt -n 5 --case-sensitive --no-punctuation
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def normalize_text(text: str, case_sensitive: bool = False, remove_punctuation: bool = True) -> str:
    """テキストを正規化する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。False の場合は小文字に変換。
        remove_punctuation: True の場合、句読点・記号をスペースで置換する。

    Returns:
        正規化されたテキスト。
    """
    if not case_sensitive:
        text = text.lower()
    if remove_punctuation:
        # 句読点・記号をスペースに置換（単語境界を保持）
        text = re.sub(r'[^\w\s]', ' ', text)
    return text


def tokenize(text: str) -> List[str]:
    """テキストを単語リストに分割する。

    Args:
        text: 入力テキスト。

    Returns:
        単語のリスト（空白区切り、空文字列は除外）。
    """
    words = text.split()
    return [word for word in words if word]


def count_frequencies(words: List[str]) -> Counter[str]:
    """単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
