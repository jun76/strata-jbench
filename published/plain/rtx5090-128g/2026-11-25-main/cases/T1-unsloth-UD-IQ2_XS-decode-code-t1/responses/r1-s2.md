コマンドラインツール: テキストファイルを読み込んで単語の出現頻度を集計し、上位N件を表形式で表示する。

Example:
    python word_freq.py sample.txt -n 10

import argparse

import re
from collections import Counter
from typing import List, Tuple, Optional, IO


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字・小文字を区別するかどうか。Falseの場合、小文字に変換。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 単語の区切り: 英数字を除く文字
    tokens = re.findall(r'[a-zA-Z0-9]+', text)
    return tokens


def count_words(tokens: List[str]) -> Counter:
    """
    単語の出現頻度を数える。

    Args:
        tokens: 単語のリスト。

    Returns:
        Counter: 単語をキー、出現回数を値とする。
    """
    return Counter(tokens)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位N件を取得する。

    Args:
        counter: 単語頻度カウンター。
        n: 上位件数。

    Returns:
        (word, count) のタプルのリスト。頻度が高い順に、同頻度の場合は辞書順でソート。
    """
    # Counter.most_common は頻度高い順だが、同頻度の場合の順序は不定
    # 辞書順でソートするため、(-count, word) でソート
    items = [(word, cnt) for word, cnt in counter.items()]
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
