"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を集計し、
上位 N 件を表形式で表示する。

使い方:
    python word_freq.py --file sample.txt --top 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。

    Args:
        text: 入力テキスト

    Returns:
        小書きされた単語のリスト
    """
    words = text.split()
    return [word.lower() for word in words]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を集計する。

    Args:
        text: 入力テキスト

    Returns:
        単語とその出現回数の Counter オブジェクト
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """上位 n 件を取得する。

    Args:
        counter: 単語頻度の Counter
        n: 取得する件数

    Returns:
        (単語, 回数) のタプルのリスト。回数降順で、同回数時は辞書順でソート。
    """
    items = list(counter.items())
    # 回数降順、同回数時は辞書順でソート
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
