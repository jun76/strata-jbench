```python
"""コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を上位 N 件で表示する。

Usage:
    python word_freq.py <file> [--top N] [--case-sensitive]

Example:
    python word_freq.py sample.txt --top 10
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def load_words(file_path: str, case_sensitive: bool = False) -> List[str]:
    """指定されたテキストファイルから単語のリストを読み込む。

    Args:
        file_path: テキストファイルのパス。
        case_sensitive: 大文字/小文字を区別するかどうか。

    Returns:
        単語のリスト（空白区切りで分割されたトークン）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        UnicodeDecodeError: 文字エンコーディングの問題の場合。
    """
    path = Path(file_path)
    if not path.exists():
        raise FileNotFoundError(f"File not found: {file_path}")

    with open(path, "r", encoding="utf-8") as f:
        content = f.read()

    # 空白区切りで分割し、各トークンから句読点や数字などの非アルファベットを除去
    tokens = content.split()
    words = []
    for token in tokens:
        # 句読点を除去し、アルファベットと数字のみを残す
        cleaned = "".join(ch for ch in token if ch.isalnum())
        if cleaned:
            if not case_sensitive:
                cleaned = cleaned.lower()
            words.append(cleaned)
    return words


def count_words(words: List[str]) -> Counter:
    """単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        各単語の出現回数を格納した Counter。
    """
    return Counter(words)
