#!/usr/bin/env python3
"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a tabular format.

Usage:
    python word_counter.py [--top N] [--case-sensitive] <file>
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """Count word occurrences in the given text.

    Args:
        text: The input text to analyze.
        case_sensitive: If False (default), words are lowercased before counting.

    Returns:
        A Counter mapping words to their occurrence counts.
    """
    if not case_sensitive:
        text = text.lower()

    # Split on whitespace and strip punctuation from each token
    words: List[str] = []
    for token in text.split():
        # Remove leading/trailing punctuation characters
        cleaned = token.strip(".,!?;:\"'()[]{}")
        if cleaned:
            words.append(cleaned)

    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top n most common words.

    Args:
        counter: A Counter of word frequencies.
        n: The number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then by word ascending for ties.
    """
    # Counter.most_common already sorts by count desc, but for ties
    # we want alphabetical order. We'll sort manually.
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]

    """Format entries as a simple text table.

    Args:
        entries: A list of (word, count) tuples.

    Returns:
        A formatted string with aligned columns.
    """
    if not entries:
        return ""

    # Determine column widths
    max_word_len = max(len(word) for word, _ in entries)
