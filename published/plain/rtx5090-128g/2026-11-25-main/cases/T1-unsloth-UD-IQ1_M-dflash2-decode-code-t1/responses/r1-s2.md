"""word_counter.py - コマンドラインツール：テキストファイルの単語出現頻度を上位N件で表示"""

import argparse
from collections import Counter

def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出し、小文字化する。

    Args:
        text: 入力テキスト文字列

    Returns:
        小文字化された単語のリスト（空白区切り、句読点は除去）
    """
    import re
    # 句読点や特殊文字をスペースに置き換えて分割
    cleaned = re.sub(r'[^a-zA-Z0-9\s]', ' ', text)
    words = cleaned.lower().split()
    return [w for w in words if w]


def count_words(words: List[str]) -> Counter:
    """単語リストの出現頻度をCounterで集計する。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とするCounter
    """
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counterから出現頻度上位N件を取得する。

    Args:
        counter: 単語と出現回数のCounter
        n: 取得する上位件数

    Returns:
        (単語, 出現回数)のタプルのリスト（頻度降順、同頻度なら辞書順）
    """
    # 頻度降順、同頻度なら単語の辞書順でソート
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
