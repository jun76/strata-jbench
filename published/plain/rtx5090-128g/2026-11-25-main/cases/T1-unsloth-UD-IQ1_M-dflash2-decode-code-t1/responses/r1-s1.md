def read_text(file_path: str) -> str:
    """
    指定されたパスのテキストファイルを読み込み、その内容を返す。
    UTF-8 で読み込み、BOM は自動的に処理される。

    Args:
        file_path: テキストファイルのパス

    Returns:
        ファイル内容の文字列

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        PermissionError: ファイルの読み込み権限がない場合
    """
    with open(file_path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> list[str]:
    """
    テキストから単語を抽出する。
    単語とは、アルファベット・数字・ハイフン・アポストロフィを含む連続文字列であり、
    小文字に変換される。

    Args:
        text: 解析対象のテキスト

    Returns:
        小文字化された単語のリスト
    """
    # 単語の定義: アルファベット、数字、ハイフン、アポストロフィを含む連続文字列
    words: list[str] = []
    current_word: list[str] = []

    for char in text:
        if char.isalnum() or char in ("-", "'"):
            current_word.append(char)
        else:
            if current_word:
                word = "".join(current_word).lower()
                # 先頭/末尾がハイフンやアポストロフィの場合、それらを除く
                word = word.strip("-'")
                if word:
                    words.append(word)
                current_word = []
    if current_word:
        word = "".join(current_word).lower().strip("-'")
        if word:
            words.append(word)

    return words

    """

    Args:

    Returns:
        単語ごとの出現頻度を表す Counter
    """
    return Counter(words)
