#!/usr/bin/env python3
"""
A command-line tool to read text files, count word occurrences, and display the top N words in a table format.
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple, Dict


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words, removing punctuation."""
    import re
    # Use regex to extract words (letters and digits)
    return [word.lower() for word in re.findall(r'[a-z0-9]+', text)]


def count_words(text: str) -> Counter[str]:
    """Count the frequency of each word in the text."""
    return Counter(tokenize(text))


def read_file(filepath: str) -> str:
    """Read content from a text file."""
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words as a list of (word, count) tuples."""
    return counter.most_common(n)


def format_table(words: List[Tuple[str, int]], headers: List[str] = None) -> str:
    """Format a list of (word, count) tuples into a simple ASCII table."""
    if not words:
        return "No words found."
    
    if headers is None:
        headers = ["Word", "Count"]
    
    # Calculate column widths
    word_width = max(len(headers[0]), max(len(w) for w, _ in words))
    count_width = max(len(headers[1]), max(len(str(c)) for _, c in words))
    
    # Create header row
    header_row = f"{headers[0]:<{word_width}} | {headers[1]:>{count_width}}"
    separator = "-" * word_width + "-+-" + "-" * count_width
    
    # Create data rows
    rows = []
    for word, count in words:
        row = f"{word:<{word_width}} | {count:>{count_width}}"
        rows.append(row)
    
    # Combine and return
    table_lines = [header_row, separator] + rows
    return "\n".join(table_lines)
