Word Frequency Counter CLI Tool.

Reads a text file, counts word frequencies, and displays the top N words
in a tabular format.
"""

def read_file(path: str) -> str:
    """
    Read the contents of a text file.

    Args:
        path: Path to the text file to read.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(path, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into words by splitting on whitespace.

    Args:
        text: The input text string.

    Returns:
        A list of words extracted from the text.
    """
    return text.split()


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words to count.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) sorted by frequency descending.
    """
    return counter.most_common(n)
