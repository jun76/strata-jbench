"""
word_freq CLI: Count word frequencies in a text file and display top N.

Usage:
    python word_freq.py input.txt -n 10

Features:
- Reads a text file
- Counts word frequencies (case-insensitive)
- Shows top N most frequent words in a table format
- Uses argparse, type hints, and docstrings
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def read_words(filepath: str) -> List[str]:
    """
    Read words from a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        List of words (lowercase, stripped of punctuation).
    """
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            content = f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except IOError as e:
        raise IOError(f"Error reading file: {e}")

    # Normalize: lowercase, split on whitespace, remove punctuation from edges
    words = content.split()
    cleaned_words = []
    for word in words:
        # Strip common punctuation from start and end
        cleaned = word.strip(".,!?;:'\"()[]{}")
        if cleaned:
            cleaned_words.append(cleaned.lower())
    return cleaned_words


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """
    Count the frequency of each word.

    Args:
        words: List of words.

    Returns:
        Counter mapping words to their frequencies.
    """
    return Counter(words)
