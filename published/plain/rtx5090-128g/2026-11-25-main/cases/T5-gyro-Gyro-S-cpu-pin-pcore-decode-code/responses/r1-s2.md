#!/usr/bin/env python3
"""
word_freq.py — コマンドラインツール: テキストファイルから単語頻度を集計し、上位N件を表形式で表示する。

使い方:
    python word_freq.py input.txt --top 10
    python word_freq.py input.txt --top 5 --case-sensitive
    python word_freq.py input.txt --top 10 --width 30
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: Trueなら大文字小文字を区別、Falseなら小文字に統一。

    Returns:
        単語のリスト。
    """
    if case_sensitive:
        words = text.split()
    else:
        words = [w.lower() for w in text.split()]
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: Trueなら大文字小文字を区別。

    Returns:
        Counter: 単語 -> 出現回数 のマッピング。
    """
    words = tokenize(text, case_sensitive=case_sensitive)
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位n件の (単語, 頻度) を返す。頻度降順、同頻度時は辞書順。

    Args:
        counter: 単語頻度のカウンター。
        n: 取得する件数。

    Returns:
        [(word, count), ...] のリスト。
    """
    # most_common は頻度降順で返す。同頻度時は挿入順だが、安定した辞書順にするため
    # 頻度降順 + 辞書昇順でソートする。
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
