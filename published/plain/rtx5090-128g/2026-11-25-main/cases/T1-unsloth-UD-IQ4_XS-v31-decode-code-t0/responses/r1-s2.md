def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。

    Returns:
        単語トークンのリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む連続文字列を単語とみなす
    tokens = re.findall(r"[a-z0-9]+(?:['-][a-z0-9]+)*", text)
    return tokens

    """

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    tokens = tokenize(text, case_sensitive=case_sensitive)
    return Counter(tokens)
