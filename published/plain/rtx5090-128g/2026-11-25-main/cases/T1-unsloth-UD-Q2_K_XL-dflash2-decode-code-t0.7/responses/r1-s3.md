#!/usr/bin/env python3
"""
wordfreq: Count word frequencies in text files and display top N results.

Usage:
    python wordfreq.py <file> [--top N] [--case-sensitive]
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple, Optional


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Extract words from text.
    
    Args:
        text: Input text string.
        case_sensitive: If False, words are lowercased for counting.
        
    Returns:
        List of words extracted from the text.
    """
    # Extract sequences of word characters (letters, digits, underscores)
    words = re.findall(r'\w+', text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_words(words: List[str]) -> Counter:
    """
    Count occurrences of each word.
    
    Args:
        words: List of words to count.
        
    Returns:
        Counter object with word counts.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words.
    
    Args:
        counter: Counter object with word counts.
        n: Number of top items to return.
        
    Returns:
        List of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
