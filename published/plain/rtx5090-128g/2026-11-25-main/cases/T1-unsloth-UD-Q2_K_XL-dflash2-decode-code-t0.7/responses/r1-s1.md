#!/usr/bin/env python3
"""
word_freq.py - 単語出現頻度分析ツール

テキストファイルを読み込み、単語の出現頻度を集計し、
上位 N 件を整形された表形式で出力します。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    英数字のシーケンスを単語として認識し、すべて小文字に変換する。
    
    Args:
        text (str): 入力テキスト。
        
    Returns:
        List[str]: 抽出された単語のリスト。
    """
    # 正規表現で英数字の連続部分を抽出
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    return [word.lower() for word in words]


def count_frequencies(words: List[str]) -> Counter:
    """
    単語リストから頻度をカウントする。
    
    Args:
        words (List[str]): 単語のリスト。
        
    Returns:
        Counter: 単語とその出現回数を格納した Counter オブジェクト。
    """
    return Counter(words)
