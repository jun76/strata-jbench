#!/usr/bin/env python3
"""
word_count.py

テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で表示するコマンドラインツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト文字列。

    Returns:
        小文字化された単語のリスト。
    """
    # 英数字の単語に分割し、小文字に変換
    # \b\w+\b は英数字とアンダースコアの連続を単語とみなす
    # 必要に応じて正規表現を調整可能
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text.lower())
    return words


def count_words(words: List[str]) -> Counter[str]:
    """
    単語リストから出現頻度を集計する。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    頻度上位 N 件の単語を取得する。

    Args:
        counter: 単語と出現回数の Counter。
        n: 取得する上位件の数。

    Returns:
        (単語, 出現回数) のタプルのリスト。頻度順にソート済み。
    """
    # most_common は頻度順にソートして返す
    return counter.most_common(n)
