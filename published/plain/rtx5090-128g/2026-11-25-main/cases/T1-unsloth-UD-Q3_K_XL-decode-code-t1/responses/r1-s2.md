#!/usr/bin/env python3
"""
Word frequency counter and top-N display tool.

This module provides functionality to read a text file, count word occurrences,
and display the top N most frequent words in a formatted table.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Dict


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Words are defined as sequences of alphanumeric characters (including underscores).
    The tokenization is case-insensitive.

    Args:
        text: Input text string.

    Returns:
        A list of lowercased word tokens.
    """
    # Match sequences of alphanumeric characters and underscores
    return re.findall(r'\w+', text.lower())


def count_words(text: str) -> Dict[str, int]:
    """
    Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        A dictionary mapping each word to its occurrence count.
    """
    tokens = tokenize(text)
    return dict(Counter(tokens))


def get_top_n_words(
    word_counts: Dict[str, int],
    n: int = 10
) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words.

    If there are ties in count, words are sorted alphabetically as a secondary criterion.

    Args:
        word_counts: Dictionary mapping words to their counts.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending, then word ascending.
    """
    sorted_words = sorted(
        word_counts.items(),
        key=lambda item: (-item[1], item[0])
    )
    return sorted_words[:n]
