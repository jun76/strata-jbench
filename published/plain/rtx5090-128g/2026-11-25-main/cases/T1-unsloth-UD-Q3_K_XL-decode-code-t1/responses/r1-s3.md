#!/usr/bin/env python3
"""
word_freq.py — テキストファイルから単語出現頻度を数え、上位N件を表形式で表示するCLIツール.

使い方:
    python word_freq.py input.txt [-n 10] [--case-sensitive] [--top]

依存: なし（標準ライブラリのみ）
"""

from __future__ import annotations

@dataclass(frozen=True)
class WordCount:
    """単語とその出現回数を保持する不変データクラス."""

    word: str
    count: int


def tokenize(text: str, *, case_sensitive: bool = False) -> list[str]:
    """テキストを単語トークンに分割する.

    Args:
        text: 入力テキスト.
        case_sensitive: True の場合、大文字・小文字を区別する.
                        False の場合、全て小文字に変換する.

    Returns:
        抽出された単語トークンのリスト.

    Raises:
        TypeError: text が str でない場合.

    Examples:
        >>> tokenize("Hello World! Hello.")
        ['hello', 'world', 'hello']
        >>> tokenize("Hello World! Hello.", case_sensitive=True)
        ['Hello', 'World', 'Hello']
    """
    if not isinstance(text, str):
        raise TypeError(f"text must be str, got {type(text).__name__}")

    if not case_sensitive:
        text = text.lower()

    # 英数字・ハイフン・アポストロフィで構成されるトークンを抽出
    # 例: "don't", "state-of-the-art", "Python3.12" → ['don', 't', ...] は避ける
    #   より正確には、連続する英数字/ハイフン/アポストロフィを1トークンとする
    tokens = re.findall(r"[a-zA-Z0-9]+(?:[-'][a-zA-Z0-9]+)*", text)
    return tokens
