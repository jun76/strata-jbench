def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    merged: list[list[int]] = []

    for start, end in sorted(intervals, key=lambda x: x[0]):
        if not merged:
            merged.append([start, end])
        else:
            last = merged[-1]
            if start <= last[1]:
                last[1] = max(last[1], end)
            else:
                merged.append([start, end])

    return merged
