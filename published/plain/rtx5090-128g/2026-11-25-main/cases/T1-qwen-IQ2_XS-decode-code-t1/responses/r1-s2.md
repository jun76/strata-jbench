#!/usr/bin/env python3
"""Command-line tool to count word frequencies from a text file and display top N results as a table."""

import argparse
import collections
import re
import sys
from typing import Optional


def count_words(text: str) -> collections.Counter:
    """Count word occurrences in the given text.

    Words are defined as sequences of alphanumeric characters (Unicode-aware).
    Words are converted to lowercase for consistent counting.

    Args:
        text: The input text to analyze.

    Returns:
        A collections.Counter mapping words to their occurrence counts.
    """
    words = re.findall(r'\w+', text, re.UNICODE)
    return collections.Counter(w.lower() for w in words)


def top_n(counter: collections.Counter, n: int) -> list[tuple[str, int]]:
    """Return the top n most common words from the counter.

    Args:
        counter: The word frequency counter.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples ordered by descending count.
    """
    return counter.most_common(n)


def format_table(entries: list[tuple[str, int]]) -> str:
    """Format the top entries as a simple ASCII table.

    Args:
        entries: A list of (word, count) tuples.

    Returns:
        A string representation of the table.
    """
    if not entries:
        return ""

    # Determine column widths
    word_width = max(len(word) for word, _ in entries)
    count_width = max(len(str(count)) for _, count in entries)
    count_width = max(count_width, len("Count"))

    header_line = f"{'Word'.ljust(word_width)} {'Count'.rjust(count_width)}"
    separator = '-' * (word_width + count_width + 1)
    lines = [header_line, separator]

    for word, count in entries:
        lines.append(f"{word.ljust(word_width)} {str(count).rjust(count_width)}")

    return '\n'.join(lines)
