#!/usr/bin/env python3
"""
Command-line tool to read a text file, count word occurrences,
and display the top N entries in a table format.
"""

import argparse
import sys
from collections import Counter
from pathlib import Path


def count_words(text: str) -> Counter[str]:
    """Count word occurrences in the given text.

    Args:
        text: The input text.

    Returns:
        A Counter object with word counts.
    """
    words = text.lower().split()
    # Strip non-alpha characters from each word
    clean_words = [w.strip('.,;:!?()[]{}"\u0027') for w in words]
    # Remove empty strings
    clean_words = [w for w in clean_words if w]
    return Counter(clean_words)


def top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """Return the top N most common words.

    Args:
        counter: Counter of word frequencies.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples.
    """
    return counter.most_common(n)
