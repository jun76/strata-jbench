"""
Command-line tool that reads a text file, counts word occurrences,
and displays the top N words in a table format.

Usage:
    python word_freq.py <input_file> [--top N] [--encoding UTF-8]

Example:
    python word_freq.py sample.txt --top 10
"""

import argparse

def count_words(text: str) -> Counter[str]:
    """Count the occurrence of each word in the given text.

    Words are split by whitespace and lowercased for consistent counting.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping each word to its occurrence count.
    """
    words = text.lower().split()
    return Counter(words)


def top_n_words(counter: Counter[str], n: int) -> List[Dict[str, object]]:
    """Return the top N words and their counts from a Counter.

    Args:
        counter: A Counter mapping words to counts.
        n: Number of top entries to return.

    Returns:
        A list of dicts with keys 'rank', 'word', 'count', sorted by count descending,
        then word ascending for ties.
    """
    # Get top n entries, with tie-breaking by alphabetical order of word
    top_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))[:n]
    result: List[Dict[str, object]] = []
    for rank, (word, count) in enumerate(top_items, start=1):
        result.append({"rank": rank, "word": word, "count": count})
    return result
