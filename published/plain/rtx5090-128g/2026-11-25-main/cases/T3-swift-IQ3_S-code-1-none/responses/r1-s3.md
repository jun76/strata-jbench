def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    if not intervals:
        return []
    
    # Sort by start, then by end
    sorted_intervals = sorted(intervals, key=lambda x: (x[0], x[1]))
    
    merged = [sorted_intervals[0][:]]  # copy of the first interval
    
    for current in sorted_intervals[1:]:
        last = merged[-1]
        # Check if current overlaps or touches the last merged interval
        # Touching means current[0] <= last[1] + 1? No, touching for closed intervals
        # For closed intervals, [1,3] and [3,5] touch (overlap at 3). [1,2] and [3,4] do not touch.
        # "Overlapping or touching" typically means if current_start <= last_end, they merge.
        # Touching in closed intervals context: [1,2] and [2,3] touch at 2. [1,2] and [3,4] do not touch.
        # So the condition is: current[0] <= last[1]
        if current[0] <= last[1]:
            # Merge by updating the end to the max
            last[1] = max(last[1], current[1])
        else:
            merged.append(current[:])
    
    return merged
