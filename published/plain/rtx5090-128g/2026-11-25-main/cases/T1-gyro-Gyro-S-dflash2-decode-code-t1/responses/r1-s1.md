"""
word_freq CLI tool: read a text file, count word occurrences, and show top N.

Usage:
    python word_freq.py <file> [--top N] [--case-sensitive] [--min-length L]
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Split text into words.

    Words are defined as sequences of alphanumeric characters.
    If case_sensitive is False, all words are lowercased.
    Words shorter than min_length are discarded.
    """
    if not text:
        return []
    words = re.findall(r'[A-Za-z0-9]+', text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    return words


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter[str]:
    """
    Count word frequencies in the given text.

    Returns a Counter mapping each word to its occurrence count.
    """
    tokens = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(tokens)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Return the top N most frequent words, sorted by count descending,
    then by word ascending for ties.
    """
    return sorted(counter.items(), key=lambda x: (-x[1], x[0]))[:n]
