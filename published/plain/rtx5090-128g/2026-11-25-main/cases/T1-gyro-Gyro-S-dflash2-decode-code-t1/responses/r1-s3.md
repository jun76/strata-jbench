def tokenize(text: str) -> list[str]:
    """テキストを単語に分割する。

    単語はアルファベット・数字・ハイフンを含む文字列として定義される。
    大文字小文字は区別しない（すべて小文字に正規化）。

    Args:
        text: 分割対象のテキスト文字列。

    Returns:
        小文字に正規化された単語のリスト。
    """
    # \w+ はアルファベット、数字、アンダースコアをマッチ
    # さらにハイフンを含む単語も考慮
    pattern = re.compile(r'[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*')
    matches = pattern.findall(text)
    return [word.lower() for word in matches]


def count_words(words: list[str]) -> Counter[str]:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        各単語とその出現回数を保持する Counter。
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """出現頻度の上位N件を取得する。

    Args:
        counter: 単語の出現頻度を保持する Counter。
        n: 取得する件数。

    Returns:
        (単語, 出現回数) のタプルリスト。頻度が高い順にソートされる。
        頻度が同じ場合は単語の辞書順でソートされる。
    """
    # 頻度降順、同一頻度なら辞書順
    items = counter.most_common(n)
    # most_common は頻度降順で返すため、追加の辞書順ソートは不要
    return items
