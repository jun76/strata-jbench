#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file.

Reads a UTF-8 encoded text file, tokenizes words case-insensitively,
counts occurrences, and prints the top-N results as a table.
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Sequence

TOKEN_RE = re.compile(r"[A-Za-z0-9]+(?:'[A-Za-z0-9]+)?")


def tokenize(text: str) -> list[str]:
    """Extract lowercase word tokens from text.

    Args:
        text: Raw text content.

    Returns:
        List of lowercase tokens (apostrophes preserved).
    """
    return [m.group(0).lower() for m in TOKEN_RE.finditer(text)]


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in text.

    Args:
        text: Raw text content.

    Returns:
        Counter mapping token -> frequency.
    """
    return Counter(tokenize(text))
