#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file.

Usage example:
    python wordfreq.py sample.txt --top 10
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Extract lowercase alphabetic words from text.

    Args:
        text: Input text string.

    Returns:
        List of lowercase words (only alphabetic characters).

    Example:
        >>> tokenize("Hello, hello! World.")
        ['hello', 'hello', 'world']
    """
    return re.findall(r"[A-Za-z]+", text.lower())


def count_words(tokens: List[str]) -> Counter[str]:
    """Count word frequencies.

    Args:
        tokens: List of words.

    Returns:
        Counter object mapping word -> frequency.
    """
    return Counter(tokens)


def top_words(counter: Counter[str], top_n: int) -> List[Tuple[str, int]]:
    """Return top N most common words.

    Args:
        counter: Word frequency counter.
        top_n: Number of entries to return.

    Returns:
        List of (word, count) tuples sorted by descending frequency.
        Ties are sorted alphabetically.

    Example:
        >>> top_words(Counter({'a': 2, 'b': 2, 'c': 1}), 2)
        [('a', 2), ('b', 2)]
    """
    return sorted(
        counter.most_common(),
        key=lambda item: (-item[1], item[0])
    )[:top_n]
