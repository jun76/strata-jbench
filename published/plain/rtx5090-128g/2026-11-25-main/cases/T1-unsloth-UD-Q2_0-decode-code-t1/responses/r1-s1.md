def normalize_words(text: str) -> List[str]:
    """
    テキストを小文字化し、単語単位に分割する。
    英数字のみを含むトークンを抽出する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    return [w.lower() for w in re.findall(r'\b[a-z0-9]+\b', text.lower())]


def count_words(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 N 件の (word, count) を返す。
    同数値の場合は辞書順で並べる。

    Args:
        counter: 単語頻度 Counter。
        n: 取得する件数。

    Returns:
        [(word, count), ...] のリスト。上位 N 件。
    """
    return sorted(counter.items(), key=lambda x: (-x[1], x[0]))[:n]
