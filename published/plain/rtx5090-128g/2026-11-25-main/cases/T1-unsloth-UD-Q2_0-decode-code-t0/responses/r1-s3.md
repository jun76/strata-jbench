word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントし、

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。
    
    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        
    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    
    # 英数字とハイフンを含む単語を抽出（例: "don't" -> "don't", "hello-world" -> "hello-world"）
    # 必要に応じて正規表現を調整可能
    words = re.findall(r"[a-zA-Z0-9'-]+", text)
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        
    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)
