#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位N件表示する。

使い方:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt --top 20 --case-sensitive
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。Falseの場合、すべて小文字に変換。

    Returns:
        抽出された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフンを含む単語を抽出（日本語はそのまま）
    words = re.findall(r'\b[\w\-]+\b', text)
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とするCounterオブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counterから上位N件の単語と出現回数を取得する。

    Args:
        counter: 単語の出現頻度を保持するCounter。
        n: 取得する上位件数。

    Returns:
        (単語, 出現回数)のタプルのリスト。出現回数の降順でソート。
        同数の場合は単語の辞書順でソート。
    """
    # most_commonは出現回数の降順で返すが、同数の場合の順序は保証されない
    # 辞書順も考慮してソートする
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
