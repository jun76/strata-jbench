def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストに分割する。
    
    正規表現を用いて英数字の連続した文字列を単語として抽出し、
    すべて小文字に変換する。
    
    Args:
        text (str): 入力テキスト。
        
    Returns:
        List[str]: 小文字化された単語のリスト。
    """
    # 英数字の連続した文字列を抽出
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    return [word.lower() for word in words]


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text (str): 入力テキスト。
        
    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = tokenize(text)
    return Counter(words)
