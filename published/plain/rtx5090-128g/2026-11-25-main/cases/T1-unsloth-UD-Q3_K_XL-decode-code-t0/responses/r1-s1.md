#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位 N 件の単語を表形式で表示するコマンドラインツール。

Usage:
    python word_counter.py <file> [-n N] [--case-sensitive] [--min-length LEN]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 単語の最小長（この長さ未満の単語は除外）。

    Returns:
        抽出された単語のリスト。
    """
    # 英数字とハイフンを含む単語を抽出（日本語など非ASCII文字は別途対応可能）
    # ここでは一般的な英語/ASCIIベースの単語抽出を行う
    pattern = r"[a-zA-Z0-9\-]+"
    words = re.findall(pattern, text)

    if not case_sensitive:
        words = [w.lower() for w in words]

    # 最小長でフィルタリング
    words = [w for w in words if len(w) >= min_length]

    return words


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 単語の最小長。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(words)
