word_freq.py — テキストファイルから単語出現頻度を算出し、上位N件を表形式で表示するCLIツール。

import argparse

def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        小文字化された単語のリスト（非英数字は区切りとして扱う）
    """
    return re.findall(r'[a-z0-9]+', text.lower())


def count_words(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語をキー、出現回数を値とするCounter
    """
    return Counter(tokenize(text))

    """

        (単語, 出現回数)のタプルのリスト（頻度降順、同頻度時はアルファベット昇順）

    return sorted(items, key=lambda x: (-x[1], x[0]))[:n]
