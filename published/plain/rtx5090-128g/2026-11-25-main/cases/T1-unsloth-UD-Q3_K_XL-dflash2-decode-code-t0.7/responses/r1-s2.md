def load_text(filepath: str) -> str:
    """
    テキストファイルを読み込み、その内容を文字列として返す。

    Args:
        filepath: 読み込むファイルのパス。

    Returns:
        ファイルの内容を表す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        UnicodeDecodeError: ファイルがUTF-8としてデコードできない場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str, case_insensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。単語は英数字とハイフン、アポストロフィを含む連続文字列とみなす。

    Args:
        text: 入力テキスト。
        case_insensitive: Trueの場合、単語を小文字に変換する。

    Returns:
        抽出された単語のリスト。
    """
    # 英数字、ハイフン、アポストロフィを含む連続文字列を単語として抽出
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    if case_insensitive:
        words = [w.lower() for w in words]
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    return Counter(words)
