使い方:
    python word_counter.py [OPTIONS] FILEPATH

def read_file(filepath: str) -> str:
    """
    指定されたパスのテキストファイルを読み込み、その内容を文字列として返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイル内の全テキスト内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        PermissionError: 読み取り権限がない場合。
        UnicodeDecodeError: 想定されるエンコーディングでデコードできない場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()
