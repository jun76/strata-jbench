```python
#!/usr/bin/env python3
"""Word frequency counter CLI tool.

Reads text files, counts word frequencies, and displays the top N words
in a tabular format.

Usage:
    python word_freq.py [options] [files ...]

Options:
    --top N          Show top N words (default: 10)
    --case-sensitive Preserve case (default: case-insensitive)
    --sort-by        Sort order: 'count' (default) or 'alpha'
"""

import argparse
import sys
from collections import Counter
from typing import List, Dict, Tuple


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """Count word frequencies in the given text.

    Args:
        text: The input text to analyze.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    if not case_sensitive:
        text = text.lower()
    # Split on whitespace and strip punctuation-like characters from edges
    words = []
    for raw_word in text.split():
        # Strip leading/trailing punctuation
        word = raw_word.strip('.,;:!?"\'()[]{}-_')
        if word:
            words.append(word)
    return Counter(words)


def get_top_n_words(
    counter: Counter,
    n: int,
    sort_by: str = 'count'
) -> List[Tuple[str, int]]:
    """Get the top N words sorted by count or alphabetically.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top words to return.
        sort_by: 'count' for descending frequency, 'alpha' for alphabetical order.

    Returns:
        A list of (word, count) tuples.
    """
    if sort_by == 'alpha':
        # Alphabetical sort, then by count descending as tie-breaker
        sorted_words = sorted(counter.items(), key=lambda x: (x[0].lower(), -x[1]))
    else:
        # Default: sort by count descending, then alphabetically as tie-breaker
        sorted_words = sorted(counter.items(), key=lambda x: (-x[1], x[0].lower()))
    return sorted_words
