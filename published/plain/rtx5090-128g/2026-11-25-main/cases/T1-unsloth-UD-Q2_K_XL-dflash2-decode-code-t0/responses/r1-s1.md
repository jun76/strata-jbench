#!/usr/bin/env python3
"""
wordfreq.py - テキストファイルから単語出現頻度を集計し、上位N件を表形式で表示するツール。

Usage:
    python wordfreq.py <file> [-n N] [--case-sensitive] [--min-len LEN]

Example:
    python wordfreq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。Falseの場合、すべて小文字に変換。
        min_len: 単語の最小長。これより短い単語は除外。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    
    # 英数字とハイフン、アポストロフィを含む単語を抽出（例: don't, well-known）
    # 必要に応じて正規表現を調整可能
    words = re.findall(r"[a-zA-Z0-9'-]+", text)
    
    # 最小長でフィルタリング
    if min_len > 1:
        words = [w for w in words if len(w) >= min_len]
        
    return words


def count_words(text: str, case_sensitive: bool = False, min_len: int = 1) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_len: 単語の最小長。

    Returns:
        単語をキー、出現回数を値とするCounterオブジェクト。
    """
    words = tokenize(text, case_sensitive=case_sensitive, min_len=min_len)
    return Counter(words)
