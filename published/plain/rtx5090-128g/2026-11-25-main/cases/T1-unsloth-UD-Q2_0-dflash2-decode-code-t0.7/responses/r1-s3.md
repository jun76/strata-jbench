#!/usr/bin/env python3
"""
word_freq.py

コマンドラインツールでテキストファイルを読み込み、単語出現頻度を集計し、
上位 N 件を表形式で表示します。

import argparse

def read_file_content(filepath: str) -> str:
    """
    ファイルを読み込み、内容を返す。

    Args:
        filepath: 読み込むファイルのパス。

    Returns:
        str: ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み込みでエラーが発生した場合。
    """
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {filepath}")
    except IOError as e:
        raise IOError(f"ファイルの読み込みでエラーが発生しました: {e}")


def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストに分割する。
    正規表現を使用してアルファベットと数字の連続した文字列（単語）を抽出し、
    小文字に変換する。

    Args:
        text: 入力テキスト。

    Returns:
        List[str]: 小文字化された単語のリスト。
    """
    # \w+ は [a-zA-Z0-9_] にマッチする。単語の定義として一般的。
    # 必要に応じて [a-zA-Z0-9]+ に制限することも可能だが、ここでは \w+ を採用。
    words = re.findall(r'\w+', text.lower())
    return words


def count_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度を集計する。

    Args:
        words: 単語のリスト。

    Returns:
        Counter: 単語と出現回数をキー・バリューとする Counter オブジェクト。
    """
    return Counter(words)
