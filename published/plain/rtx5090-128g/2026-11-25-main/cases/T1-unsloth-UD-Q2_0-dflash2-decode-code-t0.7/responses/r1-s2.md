def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストに分割する。
    大文字・小文字を区別せず、アルファベット・数字・ハイフンで構成される単語を抽出する。
    日本語のようなスペース区切りでないテキストは、1文字単位ではなく、
    連続する文字列を単語として扱う（簡易的な実装）。

    Args:
        text (str): 入力テキスト

    Returns:
        List[str]: 小文字化された単語のリスト
    """
    # アルファベット、数字、ハイフンを許容する正規表現
    # 日本語環境では、より高度なトークナイザ（MeCabなど）が必要だが、
    # ここでは一般的な英数字ベースのトークナイザを使用する。
    words = re.findall(r'[a-zA-Z0-9\-]+', text.lower())
    return words


def count_words(text: str) -> Dict[str, int]:
    """
    テキスト内の単語出現頻度を集計する。

    Args:
        text (str): 入力テキスト

    Returns:
        Dict[str, int]: 単語をキー、出現回数を値とする辞書
    """
    words = tokenize(text)
    return dict(Counter(words))
