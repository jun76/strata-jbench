def read_text(filepath: Path) -> str:
    """
    テキストファイルを読み込み、その内容を文字列として返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み取り中にエラーが発生した場合。
    """
    try:
        return filepath.read_text(encoding="utf-8")
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {filepath}")
    except IOError as e:
        raise IOError(f"ファイルの読み取りに失敗しました: {filepath} ({e})")


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    単語は、アルファベット、数字、および一部の記号（' , . 等）を含まない
    区切り文字で分割される。ここでは単純に空白と改行で分割し、
    前後の句読点を除去する。

    Args:
        text: 分割対象のテキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()

    # 句読点や記号を空白に置き換えて分割
    import re
    # 単語として扱うのは、アルファベット、数字、ハイフン、アポストロフィ
    # それ以外は区切りとみなす
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return words
