## 🧠 wordfreq.py

```python
"""単語出現頻度集計 CLI ツール。

テキストファイルを読み込み、単語の出現回数を集計して上位 N 件を表形式で表示する。
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Sequence


WORD_RE = re.compile(r"[A-Za-z]+(?:'[A-Za-z]+)?")


def extract_words(text: str) -> list[str]:
    """テキストから単語のみを抽出する（小文字化）。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    return [m.group(0).lower() for m in WORD_RE.finditer(text)]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を集計する。

    Args:
        text: 入力テキスト。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    return Counter(extract_words(text))


def format_table(counter: Counter[str], top_n: int) -> str:
    """集計結果を表形式文字列に変換する。

    Args:
        counter: 単語頻度。
        top_n: 表示する上位件数。

    Returns:
        表形式の文字列（ヘッダ付き）。
    """
    rows = counter.most_common(top_n)
    width_word = max((len(w) for w, _ in rows), default=4)
    width_word = max(width_word, len("word"))
    width_count = max((len(str(c)) for _, c in rows), default=5)
    width_count = max(width_count, len("count"))

    header = f"{'word':<{width_word}} | {'count':>{width_count
