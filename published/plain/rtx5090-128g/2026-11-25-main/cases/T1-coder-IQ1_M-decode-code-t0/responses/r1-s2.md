#!/usr/bin/env python3
"""
Command-line tool to count word frequencies in a text file and display top N words.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.
    
    Words are defined as sequences of alphanumeric characters (including underscores).
    Case-insensitive counting.
    
    Args:
        text: The input text string.
    
    Returns:
        A Counter object mapping words to their frequencies.
    """
    # Extract words: sequences of alphanumeric characters and underscores
    words = re.findall(r'[a-zA-Z0-9_]+', text.lower())
    return Counter(words)


def get_top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: Number of top words to return.
    
    Returns:
        A list of tuples (word, count) for the top N words.
    """
    return counter.most_common(n)

    """

        return ""

        lines.append(f"{word:<{max_word_len}} {count}")
    
    return "\n".join(lines)
