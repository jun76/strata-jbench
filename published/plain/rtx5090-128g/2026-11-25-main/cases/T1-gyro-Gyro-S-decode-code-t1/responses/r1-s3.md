"""
CLI tool to read a text file, count word frequencies, and display top N words in tabular form.
"""

from __future__ import annotations

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List


def normalize_word(word: str) -> str:
    """Normalize a word by lowercasing and removing punctuation."""
    # Remove leading/trailing whitespace and common punctuation
    cleaned = word.strip().lower()
    # Strip non-alphanumeric characters from both ends
    cleaned = cleaned.strip(' .,;:!?"()[]{}-_/\\')
    return cleaned


def extract_words(text: str) -> List[str]:
    """Extract normalized words from a given text string."""
    words = text.split()
    normalized = []
    for word in words:
        normalized_word = normalize_word(word)
        if normalized_word:  # skip empty strings
            normalized.append(normalized_word)
    return normalized


def count_words(text: str) -> Counter:
    """Count the frequency of each normalized word in the text."""
    words = extract_words(text)
    return Counter(words)


def get_top_n_words(counter: Counter, n: int) -> List[tuple[str, int]]:
    """Return the top N words by frequency, ties broken by alphabetical order."""
    # Sort by count descending, then by word ascending for deterministic ties
    items = sorted(counter.items(), key=lambda item: (-item[1], item[0]))
    return items[:n]


def format_table(entries: List[tuple[str, int]]) -> str:
    """Format the top N word-frequency pairs as a fixed-width table."""
    if not entries:
        return "No words found."
    max_word_len = max(len(word) for word, _ in entries)
    header_word_width = max(max_word_len, len("Word"))
    header_count_width = len("Count")
    lines = [
        f"{'Word'.ljust(header_word_width)}  {'Count'.rjust(header_count_width)}",
        f"{'-' * header_word_width}  {'-' * header_count_width}",
    ]
    for word, count in entries:
        lines.append(f"{word.ljust(header_word_width)}  {str(count).rjust(header_count_width)}")
    return "\n".
