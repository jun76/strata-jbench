#!/usr/bin/env python3

import argparse

def read_words_from_file(file_path: Path) -> List[str]:
    """テキストファイルから単語を抽出する。

    Args:
        file_path: 読み込むファイルのパス。

    Returns:
        単語のリスト。空白・改行で分割し、小文字に正規化される。
    """
    try:
        content = file_path.read_text(encoding="utf-8")
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {file_path}")
    except UnicodeDecodeError:
        raise ValueError(f"UTF-8 でデコードできません: {file_path}")

    # 空白・改行で分割し、小文字に正規化
    words = content.split()
    return [word.lower() for word in words if word.strip()]


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        各単語とその出現回数の Counter。
    """
    return Counter(words)


def get_top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """上位 N 件の単語と出現回数を取得する。

    Args:
        counter: 単語の出現頻度カウンター。
        n: 取得する件数。

    Returns:
        (単語, 出現回数) のタプルを降順に並べたリスト。
        同じ回数では辞書順でソートされる。
    """
    # 出現回数降順、同回数なら単語の辞書順
    sorted_items = sorted(
        counter.items(),
        key=lambda item: (-item[1], item[0])
    )
    return sorted_items[:n]
