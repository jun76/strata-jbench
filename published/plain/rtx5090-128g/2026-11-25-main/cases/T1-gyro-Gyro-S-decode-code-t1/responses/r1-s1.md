"""CLI tool that reads a text file, counts word frequencies, and displays the top N results."""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """Count occurrences of each word in the given text.

    Words are defined as sequences of characters separated by whitespace.
    Words are converted to lowercase for consistent counting.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter mapping each word (lowercase) to its frequency.
    """
    words = text.lower().split()
    return Counter(words)


def top_n_frequencies(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top N word frequencies sorted by count (descending), then by word (ascending).

    Args:
        counter: A Counter mapping words to their frequencies.
        n: The number of top entries to return.

    Returns:
        A list of (word, count) tuples sorted by count descending, then word ascending.
    """
    items = counter.items()
    # Sort by count descending, then by word ascending for deterministic tie-breaking
    sorted_items = sorted(items, key=lambda item: (-item[1], item[0]))
    return sorted_items[:n]


def format_table(entries: List[Tuple[str, int]]) -> str:
    """Format word-frequency entries as a simple table string.

    Args:
        entries: A list of (word, count) tuples.

    Returns:
        A formatted table string with aligned columns.
    """
    if not entries:
        return "No entries."
    
    # Determine column widths
    word_width = max(len(word) for word, _ in entries)
    count_width = max(len(str(count)) for _, count in entries)
    
    # Header
    header = f"{'WORD'.ljust(word_width)} | {'COUNT'.rjust(count_width)}"
    separator = '-' * (word_width + 3 + count_width)
    
    lines = [header, separator]
    for word, count in entries:
        lines.append(f"{word.ljust(word_width)} | {str(count).rjust(count_width)}")
    
    return '\n'.join(lines)
