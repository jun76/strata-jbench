"""
Word Frequency Counter CLI Tool

This module provides a command-line tool to read a text file, 
count word frequencies, and display the top N most frequent words 
in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import TextIO


def extract_words(text: str) -> list[str]:
    """
    Extract words from text by splitting on non-alphanumeric characters.
    
    Args:
        text: Input text string.
        
    Returns:
        List of words (lowercased).
    """
    # Split on non-alphanumeric characters, convert to lowercase
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_words(words: list[str]) -> Counter[str]:
    """
    Count the frequency of each word.
    
    Args:
        words: List of words to count.
        
    Returns:
        Counter object mapping words to their frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """
    Get the top N most common words.
    
    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.
        
    Returns:
        List of tuples (word, count) sorted by count descending.
    """
    return counter.most_common(n)
