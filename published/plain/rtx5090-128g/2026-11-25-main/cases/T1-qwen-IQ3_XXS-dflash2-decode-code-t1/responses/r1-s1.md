"""
word_freq.py - 単語出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツール.
"""

import argparse

from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_text(file_path: Path) -> str:
    """テキストファイルを読み込んで文字列を返す.

    Args:
        file_path: 読み込むテキストファイルのパス.

    Returns:
        ファイルの内容.

    Raises:
        FileNotFoundError: ファイルが存在しない場合.
        PermissionError: ファイルの読み取り権限がない場合.
    """
    if not file_path.exists():
        raise FileNotFoundError(f"ファイルが見つかりません: {file_path}")
    return file_path.read_text(encoding="utf-8")


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する.

    単語は、文字（アルファベット・数字）とアンダースコアで構成される文字列とし、
    大文字小文字を区別せず小文字に変換する.

    Args:
        text: 分割対象のテキスト.

    Returns:
        小文字化した単語のリスト.
    """
    import re
    # 単語: 文字・数字・アンダースコアの連続
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return [w.lower() for w in words]


def count_words(tokens: List[str]) -> Counter:
    """単語リストの出現頻度をカウントする.

    Args:
        tokens: 単語のリスト.

    Returns:
        単語をキー、出現回数を値とする Counter.
    """
    return Counter(tokens)
