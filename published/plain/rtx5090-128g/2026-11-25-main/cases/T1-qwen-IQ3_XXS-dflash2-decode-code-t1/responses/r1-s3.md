#!/usr/bin/env python3
"""
テキストファイルを読み込み、単語の出現頻度を上位N件集計して表形式で出力するCLIツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(file_path: str) -> List[Tuple[str, int]]:
    """
    テキストファイルから単語を抽出し、出現頻度を降順でソートして返す。

    Args:
        file_path: 読み込むテキストファイルのパス。

    Returns:
        単語と出現回数のタプルを要素とするリスト。頻度が高い順。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        PermissionError: ファイル読み取り権限がない場合。
        UnicodeDecodeError: ファイルのデコードに失敗した場合。
    """
    path = Path(file_path)
    if not path.is_file():
        raise FileNotFoundError(f"ファイルが見つかりません: {file_path}")

    try:
        text = path.read_text(encoding="utf-8")
    except PermissionError:
        raise PermissionError(f"ファイル読み取り権限がありません: {file_path}")
    except UnicodeDecodeError:
        raise UnicodeDecodeError("utf-8", b"", 0, 0, f"UTF-8デコードに失敗しました: {file_path}")

    # 単語を抽出（英数字を含む文字列、ハイフンやアポストロフィを許容）
    # 例: "hello-world" -> "hello-world", "don't" -> "don't"
    words = re.findall(r"[a-zA-Z0-9]+(?:[-'][a-zA-Z0-9]+)*", text, re.IGNORECASE)
    
    # 単語を小文字に変換して集計
    counter = Counter(word.lower() for word in words)
    
    # 頻度が高い順にソートし、同頻度の場合は辞書順で安定化
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    
    return sorted_items
