def load_text(filepath: str) -> str:
    """
    ファイルを読み込んでその内容（文字列）を返す。

    Args:
        filepath: 読み込むファイルのパス。

    Returns:
        ファイルの内容を表す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み取りエラーが発生した場合。
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()
