#!/usr/bin/env python3
"""wordfreq.py: Text file word frequency counter and top-N reporter."""

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from pathlib import Path
from typing import Iterator


@dataclass
class WordStat:
    """A word and its occurrence count."""
    word: str
    count: int


def tokenize(text: str) -> Iterator[str]:
    """Split text into lowercase alphanumeric word tokens.

    Args:
        text: Input text string.

    Yields:
        Normalized word tokens (lowercase, alphanumeric only).
    """
    # Extract words consisting of alphanumeric characters (including underscores and unicode letters/digits)
    # Using \w+ matches [a-zA-Z0-9_] and unicode word characters
    for match in re.finditer(r'\w+', text, re.UNICODE):
        yield match.group().lower()


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        Counter object with words as keys and counts as values.
    """
    return Counter(tokenize(text))


def get_top_n(counter: Counter[str], n: int) -> list[WordStat]:
    """Get the top N most common words.

    Args:
        counter: Counter object with word counts.
        n: Number of top words to return.

    Returns:
        List of WordStat objects sorted by count descending, then word ascending.
    """
    if n <= 0:
        return []
    # most_common returns items sorted by count descending.
    # For ties, we sort by word ascending to ensure deterministic output.
    items = sorted(counter.items(), key=lambda item: (-item[1], item[0]))
    return [WordStat(word=w, count=c) for w, c in items[:n]]
