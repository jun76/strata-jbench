def tokenize(text: str, *, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語トークンリストを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
        min_length: この長さ未満のトークンは除外する。

    Returns:
        正規化された単語トークンのリスト。
    """
    if not case_sensitive:
        text = text.lower()

    # Unicode-aware word boundary using \w+ (letters, digits, underscores, and CJK)
    tokens = re.findall(r"\b\w+\b", text, re.UNICODE)
    return [t for t in tokens if len(t) >= min_length]
