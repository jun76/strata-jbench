#!/usr/bin/env python3
"""
CLI tool to count word frequencies from a text file and display the top N entries.

Usage:
    python word_count.py input.txt --top 10
    python word_count.py input.txt --top 20 --min-length 3

Features:
- Reads a text file
- Counts word occurrences (case-insensitive by default)
- Displays the top N most frequent words in a formatted table
- Supports configurable options via argparse
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    Count the frequency of each word in the given text.

    Args:
        text: The input text string.
        case_sensitive: If True, words are counted as-is. If False, they are
                        lowercased before counting.

    Returns:
        A Counter mapping words to their occurrence counts.
    """
    if not case_sensitive:
        text = text.lower()

    # Split on whitespace and remove empty strings
    words = text.split()
    return Counter(words)


def format_table(items: List[Tuple[str, int]], width: int = 50) -> str:
    """
    Format a list of (word, count) tuples as an aligned table.

    Args:
        items: List of (word, count) tuples, already sorted by count descending.
        width: Maximum width for the word column.

    Returns:
        A formatted string table.
    """
    if not items:
        return ""

    # Determine column widths
    word_width = max(len(word) for word, _ in items) + 2
    count_width = max(len(str(count)) for _, count in items) + 2

    lines = []
    header = f"{'Word':<{word_width}}{'Count':>{count_width}}"
    separator = f"{'-' * word_width}{'-' * count_width}"
    lines.append(header)
    lines.append(separator)

    for word, count in items:
        lines.append(f"{word:<{word_width}}{count:>{count_width}}")

    return "\n".join(lines)
