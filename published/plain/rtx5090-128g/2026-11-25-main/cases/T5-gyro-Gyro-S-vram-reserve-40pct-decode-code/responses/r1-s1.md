```python
"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を集計し、
上位 N 件を表形式で表示する。

使い方:
    python word_freq.py --file sample.txt --top 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。

    Args:
        text: 入力テキスト

    Returns:
        小書きされた単語のリスト
    """
    words = text.split()
    return [word.lower() for word in words]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語とその出現回数の Counter
    """
    return Counter(tokenize(text))


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """上位 N 件の (単語, 頻度) のリストを取得する。

    Args:
        counter: 単語頻度の Counter
        n: 取得する件数

    Returns:
        頻度順に並んだ (単語, 頻度) のリスト（長さは最大 n）
    """
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """表形式で結果を出力する。

    Args:
        items: (単語, 頻度) のリスト

    Returns:
        表形式の文字列
    """
    if not items:
        return "No words found."

    # ヘッダー
    header = "Rank\tWord\tCount"
    separator = "-" * 30

    rows = []
    for rank, (word, count) in enumerate(items, start=1):
        rows.append(f"{rank}\t{word}\t{count}")

    return "\n".join([header, separator] + rows)
