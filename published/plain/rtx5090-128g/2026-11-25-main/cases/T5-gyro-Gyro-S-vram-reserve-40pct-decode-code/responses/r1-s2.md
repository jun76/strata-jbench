"""
コマンドラインツール: テキストファイルから単語の出現頻度を集計し、
上位 N 件を表形式で表示する。

使い方:
    python word_frequency.py input.txt --top 10

オプション:
    --top N       : 上位 N 件を表示（デフォルト 10）
    --ignore-case : 大文字・小文字を無視して集計（デフォルト False）
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def read_file(path: str) -> str:
    """ファイルを読み込み、テキストを返す。"""
    try:
        with open(path, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {path}")
    except OSError as e:
        raise OSError(f"ファイル読み込みエラー: {e}")


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    単語はアルファベット・数字・ハイフンを含む連続文字列とする。
    """
    words = []
    current = []
    for ch in text:
        if ch.isalnum() or ch == "-":
            current.append(ch)
        else:
            if current:
                words.append("".join(current))
                current = []
    if current:
        words.append("".join(current))
    return words


def count_words(text: str, ignore_case: bool = False) -> Counter[str]:
    """
    単語の出現頻度を集計する。

    Args:
        text: 入力テキスト
        ignore_case: True の場合、小文字に変換して集計

    Returns:
        単語 -> 出現回数 の Counter
    """
    words = tokenize(text)
    if ignore_case:
        words = [w.lower() for w in words]
    return Counter(words)
