_WORD_PATTERN = re.compile(r"[a-zA-Z0-9]+(?:['\-][a-zA-Z0-9]+)*")


def normalize_text(text: str) -> str:
    """テキストを単語抽出可能な形に正規化する。

    Args:
        text: 元のテキスト文字列。

    Returns:
        正規化されたテキスト（ここではそのまま返す。抽出は ``extract_words`` で行う）。
    """
    return text


def extract_words(text: str) -> List[str]:
    """テキストから単語リストを抽出する。

    単語は以下のルールで正規化される:
    - すべて小文字に変換
    - 英数字、ハイフン、アポストロフィを含む連続文字列を単語として認識
    - 句読点や特殊文字は除去

    Args:
        text: 解析対象のテキスト文字列。

    Returns:
        正規化された単語のリスト（出現順）。
    """
    normalized = normalize_text(text)
    words = _WORD_PATTERN.findall(normalized)
    # すべて小文字に変換
    return [word.lower() for word in words]
