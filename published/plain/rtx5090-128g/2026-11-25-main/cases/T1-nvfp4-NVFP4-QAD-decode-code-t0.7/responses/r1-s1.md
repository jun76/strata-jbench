#!/usr/bin/env python3
"""テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール."""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


# ---------------------------------------------------------------------------
# 型エイリアス
# ---------------------------------------------------------------------------
WordFrequency = Tuple[str, int]  # (単語, 出現回数)


# ---------------------------------------------------------------------------
# 純粋関数群（テスト容易性のため CLI ロジックから分離）
# ---------------------------------------------------------------------------
def tokenize(text: str) -> List[str]:
    """テキストを単語リストに分割する.

    英数字とアンダースコアを含む連続文字列を単語とみなす。
    大文字・小文字は区別しない（すべて小文字に変換）。

    Args:
        text: 入力テキスト。

    Returns:
        単語のリスト（順序は原文の出現順、重複あり）。
    """
    return re.findall(r"[a-zA-Z0-9_]+", text.lower())


def count_frequencies(tokens: List[str]) -> Counter:
    """トークンリストから単語出現頻度を計算する.

    Args:
        tokens: 単語のリスト。

    Returns:
        単語 → 出現回数 の Counter。
    """
    return Counter(tokens)


def top_n_frequencies(
    freq: Counter,
    n: int,
) -> List[WordFrequency]:
    """頻度 Counter から上位 N 件を返す.

    同率の場合は辞書順（アルファベット順）でソートされる。

    Args:
        freq: 単語 → 出現回数 の Counter。
        n: 取得したい上位件数。n <= 0 の場合は空リストを返す。

    Returns:
        [(単語, 回数), ...] のリスト。
    """
    if n <= 0:
        return []
