#!/usr/bin/env python3
"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位 N 件の単語とその頻度を表形式で出力するコマンドラインツール。

使い方:
    python word_freq.py input.txt --top 10
    python word_freq.py input.txt -n 5 --case-sensitive
"""

from __future__ import annotations

import argparse

import re
from collections import Counter

def read_file(filepath: str) -> str:
    """
    指定されたパスのテキストファイルを読み込み、その内容を返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を示す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み込み中にエラーが発生した場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def normalize_text(text: str, case_sensitive: bool = False) -> str:
    """
    テキストを正規化する。

    - case_sensitive が False の場合、テキストを小文字に変換する。
    - 単語の区切りとして、英数字以外の文字を空白に置換する。

    Args:
        text: 正規化対象のテキスト。
        case_sensitive: 大文字・小文字を区別するかどうか。

    Returns:
        正規化されたテキスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字以外の文字を空白に置換（単語の区切りを整える）
    text = re.sub(r"[^\w\s]", " ", text)
    return text
