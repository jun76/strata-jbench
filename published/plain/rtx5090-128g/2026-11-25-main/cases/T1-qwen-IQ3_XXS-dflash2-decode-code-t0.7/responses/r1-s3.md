def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    
    単語は、英数字とアンダースコアを含む連続した文字列として定義される。
    句読点や空白は区切りとして扱われる。

    Args:
        text: 入力テキスト。

    Returns:
        単語のリスト。
    """
    words: List[str] = []
    current: List[str] = []
    for ch in text:
        if ch.isalnum() or ch == '_':
            current.append(ch)
        else:
            if current:
                words.append(''.join(current))
                current = []
    if current:
        words.append(''.join(current))
    return words


def count_words(
    text: str,
    case_insensitive: bool = True,
) -> Counter[str]:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_insensitive: 大文字小文字を区別しない場合 True（デフォルト）。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    tokens = tokenize(text)
    if case_insensitive:
        tokens = [w.lower() for w in tokens]
    return Counter(tokens)
