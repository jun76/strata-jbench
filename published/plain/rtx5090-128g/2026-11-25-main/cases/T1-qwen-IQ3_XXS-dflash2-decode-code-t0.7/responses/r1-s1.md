wordfreq.py - 単語出現頻度集計ツール

テキストファイルを読み込み、単語の出現頻度を集計し、

import argparse

from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト

    Returns:
        小文字化された単語のリスト
    """
    # 英数字とハイフンのみを単語として扱い、それ以外を区切りとする
    words = re.findall(r'[a-zA-Z0-9\-]+', text.lower())
    return words


def count_words(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語 -> 出現回数 の Counter
    """
    words = extract_words(text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 N 件の (単語, 回数) を取得する。

    Args:
        counter: 単語出現回数の Counter
        n: 取得する上位件数

    Returns:
        [(単語, 回数), ...] のリスト
    """
    return counter.most_common(n)
