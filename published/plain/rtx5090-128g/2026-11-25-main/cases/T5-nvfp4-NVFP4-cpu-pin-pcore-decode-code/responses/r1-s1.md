def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    単語は空白区切りのトークンとし、句読点（.,!?;:'"()[]{}<>）を除去する。
    大文字小文字は無視して小文字に変換する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        正規化された単語のリスト
    """
    import re
    # 英数字とハイフン、アポストロフィを含むトークンを抽出
    # 句読点を除去し、小文字化
    words = re.findall(r"[a-zA-Z0-9'-]+", text.lower())
    return words


def count_word_frequencies(text: str) -> Counter:
    """
    テキスト中の単語出現頻度をCounterで返す。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語をキー、出現回数を値とするCounter
    """
    words = extract_words(text)
    return Counter(words)
