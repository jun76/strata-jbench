def tokenize(text: str, case_insensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、すべて小文字に変換して単語を抽出する。

    Returns:
        単語のリスト。
    """
    if case_insensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを単語の一部として扱う
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return words

    """

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    words = tokenize(text, case_insensitive)
    return Counter(words)
