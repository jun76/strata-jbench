def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """単語を正規化する。

    Args:

                        False の場合、小文字に変換する。

    Returns:
        正規化された単語。
    """
    if not case_sensitive:
        return word.lower()
    return word


def tokenize(text: str) -> list[str]:
    """テキストを単語に分割する。

    Args:
        text: 分割対象のテキスト。

    Returns:
        単語のリスト。
    """
    words = []
    current_word = []
    for char in text:
        if char.isalnum() or char in ("-", "'"):
            current_word.append(char)
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))
    return words
