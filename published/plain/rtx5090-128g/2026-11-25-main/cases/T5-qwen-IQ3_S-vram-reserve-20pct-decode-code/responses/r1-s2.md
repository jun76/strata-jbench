"""
word_freq.py - A command-line tool to count word frequencies in a text file.

Reads a text file, counts occurrences of each word (case-insensitive,
stripped of punctuation), and displays the top N most frequent words
in a table format.

Usage:
    python word_freq.py <input_file> [--top N] [--case-sensitive]

Example:
    python word_freq.py sample.txt --top 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """Tokenize text into words, removing punctuation and optionally normalizing case.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, convert all words to lowercase.

    Returns:
        A list of word tokens.
    """
    if not case_sensitive:
        text = text.lower()
    # Match sequences of alphanumeric characters and apostrophes (for contractions like "don't")
    words = re.findall(r"[a-zA-Z0-9']+", text)
    return words


def count_words(words: List[str]) -> Counter:
    """Count occurrences of each word in the list.

    Args:
        words: A list of word tokens.

    Returns:
        A Counter mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words and their counts.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top items to return.

    Returns:
        A list of tuples (word, count), sorted by count descending.
    """
    return counter.most_common(n)
