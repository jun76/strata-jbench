def normalize_word(word: str, case_sensitive: bool, min_length: int) -> Optional[str]:
    """
    単語を正規化する。

    Args:
        word (str): 元の単語。
        case_sensitive (bool): 大文字小文字を区別するかどうか。
        min_length (int): 最低限の単語長。

    Returns:
        Optional[str]: 正規化された単語。条件を満たさない場合は None。
    """
    if not case_sensitive:
        word = word.lower()
    
    # 非英数字を除去して単語部分だけを取り出す（簡易的な正規化）
    # 例: "hello," -> "hello", "don't" -> "don" (apostropheは除外)
    # より厳密な正規化が必要な場合は、正規表現を使用することをお勧めします。
    cleaned = ''.join(char for char in word if char.isalnum())
    
    if len(cleaned) < min_length:
        return None
    
    return cleaned
