word_freq.py – Count word frequencies in a text file and display the top N words.

Usage:
    python word_freq.py <input_file> [-n N] [-c/--case-sensitive]
"""

import argparse

def read_file(filepath: str) -> str:
    """Read and return the entire contents of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The file contents as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        OSError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """Tokenize text into words.

    Words are sequences of alphanumeric characters. By default, all words
    are lowercased for case-insensitive counting.

    Args:
        text: Input text string.
        case_sensitive: If True, preserve original case; otherwise, lowercase.

    Returns:
        A list of word tokens.
    """
    if not case_sensitive:
        text = text.lower()
    # Extract alphanumeric sequences as words
    words: List[str] = []
    current: List[str] = []
    for char in text:
        if char.isalnum():
            current.append(char)
        else:
            if current:
                words.append("".join(current))
                current = []
    if current:
        words.append("".join(current))
    return words


def count_words(words: List[str]) -> Counter:
    """Count occurrences of each word.

    Args:
        words: List of word tokens.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)
