### 1. コマンドラインツール: `word_freq.py`

```python
#!/usr/bin/env python3
"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度を集計し、
上位 N 件を表形式で表示するコマンドラインツール。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    大文字・小文字は無視し、アルファベットと数字のみの連続を単語とみなす。
    """
    # 正規表現で英数字の連続を抽出
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。
    """
    return Counter(words)


def get_top_n_frequencies(freq_counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度が高い上位 N 件の (単語, 頻度) を返す。
    同頻度の場合は辞書順（アルファベット順）で並ぶようにする。
    """
    # (-count, word) でソートすることで、頻度降順、同頻度時は単語昇順
    sorted_items = sorted(
        freq_counter.items(),
        key=lambda x: (-x[1], x[0])
    )
    return sorted_items[:n]


def format_table(data: List[Tuple[str, int]]) -> str:
    """
    (単語, 頻度) のリストを表形式の文字列に変換する。
    """
    if not data:
        return "No data available."

    # 列の最大幅を計算
    word_width = max(len(word) for word, _ in data) if data else 0
    freq_width = max(len(str(freq)) for _, freq in data) if data else 0

    # ヘッダー
    header_word = "Word".ljust(word_width)
    header_freq = "Count".rjust(freq_width)
