"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度を集計し、
上位 N 件を表形式で表示するコマンドラインツール。

from __future__ import annotations

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語のリストに分割する。

    単語は英数字とハイフンを許容し、それ以外の文字を区切りとして分割する。
    すべて小文字に変換する。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト
    """
    import re
    # 英数字とハイフンを単語として抽出
    tokens = re.findall(r'[a-zA-Z0-9\-]+', text.lower())
    return tokens


def count_words(text: str) -> Counter:
    """
    テキスト中の単語の出現回数をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (単語, 出現回数) を取得する。

    Args:
        counter: 単語出現回数の Counter
        n: 取得する件数

    Returns:
        (単語, 出現回数) のリスト
    """
    return counter.most_common(n)
