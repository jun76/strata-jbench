#!/usr/bin/env python3
"""
wordfreq: テキストファイルの単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from typing import TextIO


@dataclass(frozen=True)
class WordCount:
    """単語と出現回数を保持する不変データ構造。"""
    word: str
    count: int


def tokenize(text: str) -> list[str]:
    """テキストから単語トークンを取り出す。

    正規表現で英数字・アンダースコアで構成されるトークンを抽出し、
    すべて小文字に正規化する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語トークンのリスト。
    """
    # 英数字とアンダースコアを含む連続文字列を単語とみなす
    tokens = re.findall(r'[A-Za-z0-9_]+', text)
    return [token.lower() for token in tokens]


def count_words(text: str) -> Counter[str]:
    """テキスト中の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(tokenize(text))
