#!/usr/bin/env python3

word_freq.py - コマンドラインツールでテキストファイルから単語出現頻度の上位N件を表示する。

import argparse

from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出し、小文字化してリストで返す。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語のリスト（小文字化済み）
    """
    # 英数字以外の文字を区切りとして単語を分割
    words: List[str] = []
    current_word = ""
    for char in text:
        if char.isalnum():
            current_word += char
        else:
            if current_word:
                words.append(current_word.lower())
                current_word = ""
    if current_word:
        words.append(current_word.lower())
    return words


def count_word_frequencies(words: List[str]) -> List[Tuple[str, int]]:
    """単語リストから出現頻度を計算し、頻度降順・アルファベット昇順でソートしたリストを返す。
    
    Args:
        words: 単語のリスト
        
    Returns:
        (単語, 出現回数) のタプルのリスト。頻度降順、同頻度なら単語の昇順でソート。
    """
    counter = Counter(words)
    # 頻度降順、同頻度なら単語の昇順
    sorted_items: List[Tuple[str, int]] = sorted(
        counter.items(), key=lambda x: (-x[1], x[0])
    )
    return sorted_items


def get_top_n(frequencies: List[Tuple[str, int]], n: int) -> List[Tuple[str, int]]:
    """上位N件の頻度を取得する。
    
    Args:
        frequencies: 頻度リスト
        n: 上位N件
        
    Returns:
        上位N件の (単語, 回数) リスト
    """
    return frequencies[:n]
