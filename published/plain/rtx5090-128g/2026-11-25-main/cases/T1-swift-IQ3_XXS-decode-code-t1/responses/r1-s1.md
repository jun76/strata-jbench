#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語の出現頻度を集計し、上位 N 件を表示するコマンドラインツール

Usage:
    python word_freq.py <input_file> [-n N] [-s] [-o output_file]

Args:
    <input_file>  読み込むテキストファイル
    -n N          上位件数 (デフォルト: 10)
    -s            大文字・小文字を区別しない (デフォルト: 区別する)
    -o output     結果を出力するファイル (指定なし: 標準出力)
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = True) -> List[str]:
    """
    テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合大文字小文字を区別し、False の場合は小文字に変換

    Returns:
        単語のリスト
    """
    # 英数字・ハイフン・アポストロフィを単語として認識
    words: List[str] = re.findall(r"[a-zA-Z0-9'-]+", text)
    if not case_sensitive:
        words = [word.lower() for word in words]
    return words


def count_frequencies(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数の Counter
    """
    return Counter(words)


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度が高い上位 n 件の (単語, 回数) のリストを返す。

    Args:
        counter: 単語 -> 出現回数の Counter
        n: 上位件数

    Returns:
        (単語, 回数) のリスト
    """
    return counter.most_common(n)
