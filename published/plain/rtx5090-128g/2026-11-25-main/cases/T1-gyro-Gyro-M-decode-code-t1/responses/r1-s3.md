def count_words(text: str) -> Counter[str]:
    """
    Count word occurrences in the given text.

    Words are defined as sequences of alphanumeric characters and underscores
    (matching \\b\\w+\\b). Counting is case-insensitive.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter mapping lowercase words to their occurrence counts.

    >>> count_words("Hello hello WORLD")
    Counter({'hello': 2, 'world': 1})
    >>> count_words("")
    Counter()
    """
    words = re.findall(r"\b\w+\b", text.lower())
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Return the top N most common items from a counter.

    Args:
        counter: A Counter of words and counts.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples sorted by count descending,
        then by word ascending for ties.

    >>> top_n(Counter({'a': 5, 'b': 3, 'c': 5}), 2)
    [('a', 5), ('c', 5)]
    """
    # Counter.most_common() sorts by count descending; for ties,
    # we need to also sort by word ascending.
    items = counter.most_common()
    # Re-sort: primary key is -count (descending), secondary is word (ascending)
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
