def tokenize(text: str) -> list[str]:
    """
    Tokenize the given text into lowercase words.

    Extracts sequences of alphanumeric characters and apostrophes,
    converts to lowercase, and returns them as a list.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens.
    """
    # Match sequences of alphanumeric characters and apostrophes (e.g., "don't")
    words = re.findall(r"[a-zA-Z0-9']+", text)
    return [word.lower() for word in words]


def count_words(text: str) -> Counter[str]:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter mapping each word to its occurrence count.
    """
    tokens = tokenize(text)
    return Counter(tokens)
