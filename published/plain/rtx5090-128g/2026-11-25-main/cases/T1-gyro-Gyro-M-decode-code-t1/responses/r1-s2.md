"""word_count.py - Count word frequencies from a text file and display top N words in a table."""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def normalize_word(word: str) -> str:
    """Normalize a word by converting to lowercase and stripping non-alphanumeric characters."""
    return re.sub(r'[^a-zA-Z0-9]', '', word.lower())


def extract_words(text: str) -> List[str]:
    """Extract normalized words from the given text.

    Words are split on whitespace and then normalized (lowercased, stripped of
    non-alphanumeric characters). Empty results are discarded.

    Args:
        text: The input text string.

    Returns:
        A list of normalized words.
    """
    raw_words = text.split()
    normalized = []
    for w in raw_words:
        norm = normalize_word(w)
        if norm:
            normalized.append(norm)
    return normalized


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter mapping each normalized word to its occurrence count.
    """
    words = extract_words(text)
    return Counter(words)


def top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words from the counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples sorted by descending frequency.
    """
    return counter.most_common(n)
