def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割します。
    英数字の連続を単語として抽出し、小文字に変換します。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト（小文字化済み）
    """
    import re
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_words(words: List[str]) -> List[Tuple[str, int]]:
    """
    単語リストから出現頻度を集計し、頻度降順でソートします。
    同頻度の場合は単語の昇順でソートします。

    Args:
        words: 単語のリスト

    Returns:
        (単語, 頻度) のタプルのリスト
    """
    counter = Counter(words)
    # 頻度降順、同頻度なら単語の昇順
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items
