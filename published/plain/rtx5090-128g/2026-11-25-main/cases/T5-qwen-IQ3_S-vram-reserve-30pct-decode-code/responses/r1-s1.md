#!/usr/bin/env python3
"""
Top Words Counter

A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def read_file(filepath: str) -> str:
    """Read and return the contents of a text file.
    
    Args:
        filepath: Path to the text file to read.
        
    Returns:
        The text content of the file.
        
    Raises:
        FileNotFoundError: If the file does not exist.
        OSError: If there is an error reading the file.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.
    
    Extracts sequences of alphanumeric characters (including underscores)
    and converts them to lowercase.
    
    Args:
        text: The input text to tokenize.
        
    Returns:
        A list of lowercase word tokens.
    """
    # Match words consisting of alphanumeric characters and underscores
    return [word.lower() for word in re.findall(r'\b\w+\b', text)]


def count_words(text: str) -> Counter:
    """Count the frequency of each word in the text.
    
    Args:
        text: The input text.
        
    Returns:
        A Counter object with word frequencies.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.
        
    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then alphabetically for ties.
    """
    # most_common returns sorted by count descending
    # For ties, we want alphabetical order, so we sort manually
    items = list(counter.items())
    # Sort by count descending, then by word ascending
    items.sort(key=lambda x: (-x[1], x[0]))
    return items
