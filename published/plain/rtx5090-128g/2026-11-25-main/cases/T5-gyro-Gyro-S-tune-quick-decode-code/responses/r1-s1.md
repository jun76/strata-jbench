"""
wordfreq.py — コマンドラインツール：テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する。

使い方:
    python wordfreq.py input.txt --top 10

オプション:
    --top N     : 上位 N 件を表示（デフォルト: 10）
    --case      : 大文字小文字を区別する（デフォルト: False, 小文字化して集計）
"""

import argparse

from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字化して集計する。

    Returns:
        単語のリスト
    """
    # 単語を抽出（アルファベット・数字・ハイフン・アポストロフィを含む連続文字列）
    words: List[str] = []
    current = ""
    for ch in text:
        if ch.isalnum() or ch in ("-", "'"):
            current += ch
        else:
            if current:
                words.append(current)
                current = ""
    if current:
        words.append(current)

    if not case_sensitive:
        words = [w.lower() for w in words]

    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト内の単語出現頻度を集計する。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        単語 -> 出現回数の Counter
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)
