```python
"""
wordfreq.py - コマンドラインツール：テキストファイルの単語出現頻度を上位 N 件表示するツール。

使い方:
    python wordfreq.py input.txt -n 10
"""

import argparse
import sys
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 対象テキスト。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    # 英字・数字・ハイフン・アポストロフィを含む連続文字列を単語として抽出
    words = re.findall(r"[a-zA-Z0-9'-]+", text.lower())
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    頻度上位 n 件の (word, count) を返す。

    Args:
        counter: 単語 -> 出現回数 の Counter。
        n: 上位件数。

    Returns:
        (word, count) のリスト。頻度降順。同頻度時は辞書順で安定。
    """
    return sorted(counter.items(), key=lambda item: (-item[1], item[0]))[:n]


def format_table(items: List[Tuple[str, int]]) -> str:
    """
    (word, count) のリストを表形式でフォーマットする。

    Args:
        items: (word, count) のリスト。

    Returns:
        表形式の文字列。
    """
    if not items:
        return ""
    max_word_len = max(len(word) for word, _ in items)
    lines = []
    header = f"{'Word'.ljust(max_word_len)} | Count"
    separator = "-" * (max_word_len + 8)  # "|" + 1 space + "Count" approx
    lines.append(header)
    lines.append(separator)
    for word, count in items:
        lines.append(f"{word.ljust(max_word_len)} | {count}")
    return "\n".join(lines)
