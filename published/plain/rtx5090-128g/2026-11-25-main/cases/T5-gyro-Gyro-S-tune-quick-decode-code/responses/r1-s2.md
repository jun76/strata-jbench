#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルから単語出現頻度を集計し上位N件を表示する。

入力: テキストファイルパス
出力: 単語とその出現回数を表形式で表示

オプション:
  --top N: 上位N件を表示 (デフォルト: 10)
  --ignore-case: 大文字/小文字を区別しない (デフォルト: False)
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_text(file_path: str) -> str:
    """指定されたファイルからテキストを読み込む。"""
    try:
        return Path(file_path).read_text(encoding="utf-8")
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {file_path}")
    except UnicodeDecodeError as e:
        raise ValueError(f"UTF-8デコードエラー: {e}")


def tokenize(text: str, ignore_case: bool = False) -> List[str]:
    """テキストを単語に分割する。"""
    # 空白区切りで分割し、句読点や記号を除去
    words = []
    for word in text.split():
        # 単語から句読点や記号を除去
        clean_word = "".join(c for c in word if c.isalnum())
        if clean_word:
            if ignore_case:
                clean_word = clean_word.lower()
            words.append(clean_word)
    return words


def count_words(words: List[str]) -> Counter:
    """単語の出現頻度をカウントする。"""
    return Counter(words)


def get_top_words(counter: Counter, top_n: int) -> List[Tuple[str, int]]:
    """上位N件の単語と出現回数を取得する。"""
    # 出現回数で降順ソート、同じ回数は単語で昇順ソート
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:top_n]
