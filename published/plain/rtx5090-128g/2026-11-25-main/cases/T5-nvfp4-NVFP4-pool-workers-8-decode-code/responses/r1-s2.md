word_freq.py

from __future__ import annotations

# 単語の正規表現: 英数字とハイフン、アポストロフィを含む連続文字列
# 例: "don't", "state-of-the-art", "word123"
_WORD_PATTERN = re.compile(r"[a-zA-Z0-9'\-]+", re.IGNORECASE)


def tokenize(text: str) -> List[str]:
    """
    テキストを小文字化し、単語トークンのリストに分割する。

    Args:
        text: 入力テキスト文字列。

    Returns:
        小文字化された単語トークンのリスト。
    """
    return [word.lower() for word in _WORD_PATTERN.findall(text)]


def count_words(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト文字列。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(tokenize(text))
