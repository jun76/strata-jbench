"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a tabular format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    Count word occurrences in the given text.
    
    Words are extracted using a regex that matches sequences of word characters.
    The comparison is case-insensitive.
    
    Args:
        text: The input text to analyze.
        
    Returns:
        A Counter mapping each lowercase word to its occurrence count.
    """
    words = re.findall(r'\b\w+\b', text, re.IGNORECASE)
    return Counter(word.lower() for word in words)


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.
    
    Args:
        counter: A Counter mapping words to counts.
        n: The number of top words to return.
        
    Returns:
        A list of (word, count) tuples sorted by count descending,
        then by word ascending for ties.
    """
    items = counter.items()
    # Sort by count descending, then by word ascending for deterministic output
    sorted_items = sorted(items, key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
