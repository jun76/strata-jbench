# 英数字とハイフン/アポストロフィを含む連続文字列を単語とみなす
_DEFAULT_TOKEN_RE = re.compile(r"[A-Za-z0-9]+(?:['\-][A-Za-z0-9]+)*")


def tokenize(text: str, *, case_sensitive: bool = False, min_len: int = 1) -> list[str]:
    """テキストを単語トークン列に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: True なら大文字小文字を区別する。
        min_len: この長さ未満のトークンは除外する。

    Returns:
        単語トークンのリスト (順序は原文出現順)。
    """
    tokens = _DEFAULT_TOKEN_RE.findall(text)
    if not case_sensitive:
        tokens = [t.lower() for t in tokens]
    if min_len > 1:
        tokens = [t for t in tokens if len(t) >= min_len]
    return tokens
