"""
wordfreq.py — テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で出力する CLI ツール。

Usage:
    python wordfreq.py <input_file> [--top N] [--case-sensitive] [--min-len L]
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    テキストを単語トークンに分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False なら小文字化する。
        min_len: 最小単語長。この長さ未満のトークンは除外する。

    Returns:
        単語トークンのリスト。
    """
    if not case_sensitive:
        text = text.lower()

    # 英数字とハイフン、アポストロフィを含む連続文字列を単語とみなす
    # 例: don't -> don't, e-mail -> e-mail, hello -> hello
    tokens = re.findall(r"[a-z0-9]+(?:[-'][a-z0-9]+)*", text) if not case_sensitive else \
             re.findall(r"[A-Za-z0-9]+(?:[-'][A-Za-z0-9]+)*", text)

    return [tok for tok in tokens if len(tok) >= min_len]
