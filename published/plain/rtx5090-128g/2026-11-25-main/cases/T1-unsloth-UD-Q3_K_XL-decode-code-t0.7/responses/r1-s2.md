word_freq.py: テキストファイルから単語の出現頻度を計算し、上位 N 件を表示するツール。

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive] [--sort-by {count,word}]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text (str): 入力テキスト。
        case_sensitive (bool): 大文字小文字を区別するかどうか。デフォルトは False。

    Returns:
        List[str]: 単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフンを含む単語を抽出する正規表現
    # 必要に応じて \w+ や [a-z]+ などに調整可能
    words = re.findall(r'[a-zA-Z0-9\-]+', text)
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト内の単語の出現頻度をカウントする。

    Args:
        text (str): 入力テキスト。
        case_sensitive (bool): 大文字小文字を区別するかどうか。デフォルトは False。

    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)
