def read_text_file(filepath: Path) -> str:
    """テキストファイルを読み込んで内容を返す。

    Args:
        filepath: 読み込むファイルのパス

    Returns:
        ファイルの内容（文字列）

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        OSError: ファイルの読み込み中にエラーが発生した場合
    """
    return filepath.read_text(encoding="utf-8")


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。

    単語は英数字とハイフンで構成され、大文字小文字を区別しない。
    句読点や記号は単語の一部から除外される。

    Args:
        text: 解析対象のテキスト

    Returns:
        抽出された単語のリスト（小文字に変換済み）
    """
    # 単語区切り: 英数字、ハイフン、アポストロフィを単語文字として扱う
    # それ以外は区切りとして扱う
    import re
    # 英数字、ハイフン、アポストロフィを含むトークンを抽出
    tokens = re.findall(r"[a-zA-Z0-9'\-]+", text)
    # 前後のハイフンやアポストロフィを除去し、小文字に変換
    words = []
    for token in tokens:
        # 前後のハイフンとアポストロフィを除去
        word = token.strip("-'")
        if word:  # 空文字でない場合のみ追加
            words.append(word.lower())
    return words
