#!/usr/bin/env python3
"""
Command-line tool: Count word frequencies in a text file and display the top N.

Usage:
    python wordfreq.py <file> [--top N] [--case-sensitive] [--min-length L]

Options:
    --top N            Show top N words (default: 10)
    --case-sensitive   Preserve case when counting (default: False)
    --min-length L     Minimum word length to include (default: 1)
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Split text into words, treating sequences of alphanumeric characters as words.

    Args:
        text: Input text string.
        case_sensitive: If False, words are lowercased before counting.

    Returns:
        A list of word strings (tokens).
    """
    # Split on non-alphanumeric characters
    words = re.findall(r"[A-Za-z0-9]+", text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    Count word frequencies in the given text.

    Args:
        text: Input text string.
        case_sensitive: If False, words are lowercased before counting.

    Returns:
        A Counter mapping each word to its frequency.
    """
    tokens = tokenize(text, case_sensitive)
    return Counter(tokens)


def top_n(counter: Counter[str], n: int, min_length: int = 1) -> List[Tuple[str, int]]:
    """
    Return the top N words by frequency, filtered by minimum word length.

    Args:
        counter: Counter of word frequencies.
        n: Number of top entries to return.
        min_length: Minimum length of words to include.

    Returns:
        A list of (word, count) tuples sorted by descending count,
        then by word for ties.
    """
    filtered = [(word, count) for word, count in counter.items() if len(word) >= min_length]
    # Sort by count descending, then word ascending for stable ordering
