```python
#!/usr/bin/env python3
"""
word_freq.py — コマンドラインツール: テキストファイルから単語の出現頻度を集計し、
上位 N 件を表形式で表示する。

使用例:
  python word_freq.py input.txt -n 10
  python word_freq.py input.txt --top 5 --case-sensitive
"""

from __future__ import annotations

import argparse
import sys
from collections import Counter
from typing import TextIO


def read_words_from_file(filepath: str, case_sensitive: bool = False) -> list[str]:
    """テキストファイルを読み込み、単語のリストを返す。

    Args:
        filepath: 読み込むテキストファイルのパス。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。

    Returns:
        単語文字列のリスト。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        content = f.read()
    words = content.split()
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_word_frequencies(words: list[str]) -> Counter[str]:
    """単語の出現頻度を Counter で集計する。

    Args:
        words: 単語文字列のリスト。

    Returns:
        各単語とその出現回数を保持する Counter。
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """Counter から上位 n 件の (word, count) タプルを返す。

    Args:
        counter: 単語頻度の Counter。
        n: 取得する上位件数。

    Returns:
        上位 n 件の (単語, 出現回数) のリスト。頻度が高い順に並び、
        同頻度の場合は単語の辞書順でソートされる。
    """
    # most_common は頻度順だが、同頻度では辞書順ではない。
    # 頻度降順 → 単語の辞書順（昇順）でソートする。
    items = sorted(counter.items(), key=lambda item: (-item[1], item[0]))
    return items
