```python
"""
word_freq: コマンドラインツールでテキストファイルから単語出現頻度を集計し、
上位 N 件を表形式で表示する。

使い方:
    python word_freq.py <file> --top N [--sort-by count|word] [--case-sensitive]
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple, Optional


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。
    
    Args:
        text: 入力テキスト
        case_sensitive: True なら大文字/小文字を区別
    
    Returns:
        単語のリスト
    """
    words = text.split()
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_word_frequencies(text: str, case_sensitive: bool = False) -> Counter:
    """
    単語の出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        case_sensitive: True なら大文字/小文字を区別
    
    Returns:
        単語 -> 出現回数の Counter
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)


def get_top_n(counter: Counter, n: int, sort_by: str = "count") -> List[Tuple[str, int]]:
    """
    上位 N 件の (word, count) を取得する。
    
    Args:
        counter: 頻度 Counter
        n: 取得件数
        sort_by: "count" で頻度順、"word" でアルファベット順
    
    Returns:
        (word, count) のリスト
    """
    items = list(counter.items())
    if sort_by == "count":
        # 頻度降順、同頻度ならアルファベット昇順
        items.sort(key=lambda x: (-x[1], x[0]))
    elif sort_by == "word":
        # アルファベット昇順
        items.sort(key=lambda x: x[0])
    else:
        raise ValueError(f"Unknown sort_by: {sort_by}")
    return items[:n]
