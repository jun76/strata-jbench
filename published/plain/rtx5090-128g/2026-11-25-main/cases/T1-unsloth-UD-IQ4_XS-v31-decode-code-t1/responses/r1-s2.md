"""word_freq.py - コマンドラインでテキストファイルの単語頻度を集計し上位 N 件を表示するツール."""

from __future__ import annotations

import argparse
import re

from collections import Counter
from pathlib import Path

def tokenize(text: str) -> List[str]:
    """テキストから単語リストを抽出する.

    英数字のみで構成されるトークンを抽出し、すべて小文字に変換する。

    Args:
        text: 入力テキスト。

    Returns:
        単語のリスト。
    """
    return re.findall(r"[a-zA-Z0-9]+", text.lower())


def count_words(words: List[str]) -> Counter[str]:
    """単語リストから出現頻度をカウントする.

    Args:
        words: 単語のリスト。

    Returns:
        単語 -> 出現回数の Counter オブジェクト。
    """
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """上位 n 件の単語と出現回数を返す.

    出現回数の多い順にソートし、同数なら辞書順で整列する。

    Args:
        counter: 単語の出現頻度を表す Counter。
        n: 取得したい上位件数。

    Returns:
        (単語, 出現回数) のタプルのリスト。
    """
    # most_common は回数降順だが、同数時の順序は保証されないため明示的にソート
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
