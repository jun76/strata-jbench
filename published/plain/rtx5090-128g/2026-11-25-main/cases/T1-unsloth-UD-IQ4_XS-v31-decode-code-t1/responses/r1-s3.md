word_counter.py

テキストファイルを読み込み、単語の出現頻度を集計して

@dataclass(frozen=True)

    """単語とその出現頻度を保持する不変データクラス。"""

    word: str
    count: int


def read_text(file_path: Path, encoding: str = "utf-8") -> str:
    """指定されたテキストファイルを読み込む。

    Args:
        file_path: 読み込むファイルのパス。
        encoding: ファイルのエンコーディング（デフォルト: utf-8）。

    Returns:
        ファイルの内容を表す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        PermissionError: 読み取り権限がない場合。
        UnicodeDecodeError: デコーディングに失敗した場合。
    """
    return file_path.read_text(encoding=encoding)


def tokenize(text: str) -> list[str]:
    """テキストから単語リストを抽出する。

    英数字の連続を単語とみなし、それ以外を区切り文字として扱う。
    単語はすべて小文字に変換される。

    Args:
        text: 分割対象のテキスト。

    Returns:
        小文字化された単語のリスト。
    """
    # \b[\w]+\b は英数字とアンダースコアを含む単語を抽出する
    return re.findall(r"\b[\w]+\b", text.lower())
