以下は、テキストファイルを読み込んで単語の出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツールです。

**`word_freq.py`**

```python
#!/usr/bin/env python3
"""テキストファイルから単語の出現頻度を集計して上位 N 件を表示する。"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """テキストから単語を抽出して出現回数を集計する。"""
    words = re.findall(r"\b\w+\b", text.lower())
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """出現頻度の高い上位 n 件を返す。"""
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """単語と頻度のリストを表形式の文字列に整形する。"""
    if not items:
        return "(データなし)"
    word_width = max(len(word) for word, _ in items)
    lines = [f"{'単語':<{word_width}}  {'頻度':>6}",
             f"{'-' * word_width}  {'-' * 6}"]
    for word, count in items:
        lines.append(f"{word:<{word_width}}  {count:>6}")
    return "\n".join(lines)


def main() -> None:
    parser = argparse.ArgumentParser(
        description="テキストファイルから単語の出現頻度を集計し、上位 N 件を表示します。"
    )
    parser.add_argument("filepath", type=Path, help="対象のテキストファイル")
