#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルから単語出現頻度を集計し、
上位N件を表形式で表示する。

Usage:
    python word_freq.py <file> [-n TOP_N] [--case-sensitive]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: Trueの場合、大文字・小文字を区別する。

    Returns:
        単語のリスト（正規表現 [a-zA-Z0-9']+ にマッチするトークン）。
    """
    if not case_sensitive:
        text = text.lower()
    pattern = re.compile(r"[a-zA-Z0-9']+")
    return pattern.findall(text)


def count_frequencies(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: Trueの場合、大文字・小文字を区別する。

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counterから上位n件の単語と頻度を返す。

    Args:
        counter: 単語頻度のCounter。
        n: 上位n件。

    Returns:
        (word, count) のタプルのリスト（頻度降順）。
        頻度が同じ場合は単語の辞書順。
    """
    # most_common は頻度降順、同頻度の場合は挿入順。
    # 辞書順に安定させるため、頻度で降順、単語で昇順のキーでソートする。

    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
