def read_file(path: str) -> str:
    """Read the contents of a text file.

    Args:
        path: Path to the text file.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def count_words(text: str, case_insensitive: bool = False) -> Counter:
    """Count the frequency of each word in the given text.

    Words are split on whitespace. If case_insensitive is True, words
    are lowercased before counting.

    Args:
        text: The input text.
        case_insensitive: Whether to ignore case when counting words.

    Returns:
        A Counter mapping words to their frequencies.
    """
    words = text.split()
    if case_insensitive:
        words = [word.lower() for word in words]
    return Counter(words)
