    -n, --top  : 上位件数 (default: 10)
    -w, --width: 単語列の表示幅 (default: 20)

import argparse

from pathlib import Path

def normalize_word(word: str) -> str:
    """単語を小文字化し、非アルファベット文字を除去する.

    Args:
        word: 元の単語文字列

    Returns:
        正規化された単語 (空文字列の場合は '' を返す)
    """
    # 小文字化 + アルファベットのみ残す
    cleaned = ''.join(c for c in word.lower() if c.isalpha())
    return cleaned


def count_words(text: str) -> Counter:
    """テキスト内の単語出現頻度を Counter で集計する.

    Args:
        text: 対象のテキスト文字列

    Returns:
        各正規化単語の出現回数を含む collections.Counter
    """
    words = text.split()
    normalized = [normalize_word(w) for w in words]
    # 空文字列は除外
    return Counter(w for w in normalized if w)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (word, count) を取得する.

    Args:
        counter: 単語頻度 Counter
        n: 取得件数

    Returns:
        [(word, count), ...] のリスト (count 降順, word 昇順の tie-break)
    """
    if n <= 0:
        return []
    # most_common は count 降順, 同じ count の場合は key 昇順で安定
    return counter.most_common(n)
