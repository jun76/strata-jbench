#!/usr/bin/env python3
"""
Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Args:
        text: Input text string.

    Returns:
        List of lowercase word tokens.
    """
    return [word.lower() for word in text.split()]


def count_words(text: str) -> Counter[str]:
    """
    Count word occurrences in text.

    Args:
        text: Input text string.

    Returns:
        Counter mapping words to their occurrence counts.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def format_table(
    word_counts: Counter[str],
    top_n: int,
    title: str = "Top Words",
) -> str:
    """
    Format word counts as a table string.

    Args:
        word_counts: Counter of word frequencies.
        top_n: Number of top words to display.
        title: Table title.

    Returns:
        Formatted table string.
    """
    if not word_counts:
        return f"{title}: (empty)"

    # Get top N words sorted by count descending, then alphabetically
    items: List[Tuple[str, int]] = sorted(
        word_counts.items(),
        key=lambda item: (-item[1], item[0]),
    )[:top_n]

    if not items:
        return f"{title}: (empty)"

    # Find max word length for alignment
    max_word_len = max(len(word) for word, _ in items)
    max_count_len = max(len(str(count)) for _, count in items)

    lines = [title, "-" * (max_word_len + max_count_len + 3)]

    for word, count in items:
        lines.append(f"  {word:<{max_word_len}}  {count:> {max_count_len}}")

    return "\n".join(lines)
