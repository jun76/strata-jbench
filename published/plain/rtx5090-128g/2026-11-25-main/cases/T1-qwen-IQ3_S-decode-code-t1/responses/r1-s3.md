word_counter.py

テキストファイルを読み込み、単語の出現頻度を集計して
上位 N 件の単語とそのカウントを表形式で出力するコマンドラインツール。

from collections import Counter
from typing import List, Tuple, Optional

# 単語を抽出するための正規表現 (英数字、ハイフン、アポストロフィを含む)
# 例: "don't" -> "don't", "e-mail" -> "e-mail"
WORD_PATTERN = re.compile(r"\b[\w'\-]+\b", re.IGNORECASE)


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出し、小文字化してリストとして返す。

    Args:
        text: 入力テキスト。

    Returns:
        抽出された単語のリスト（小文字）。
    """
    matches = WORD_PATTERN.findall(text)
    # 正規表現の大文字小文字無視フラグ (IGNORECASE) はマッチには使われるが、
    # 結果の文字列そのものの大文字小文字は保たれる。
    # 頻度集計のため、統一して小文字に変換する。
    return [word.lower() for word in matches]


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから頻度を集計する。

    Args:
        words: 単語のリスト。

    Returns:
        各単語とその出現回数をキー・値とするCounterオブジェクト。
    """
    return Counter(words)
