"""
top_words.py - テキストファイルから単語出現頻度の上位 N 件を集計するコマンドラインツール。
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    テキストから単語の出現頻度をカウントする。

    単語は英字と数字の連続とする（正規表現 \\b[a-zA-Z0-9]+\\b）。
    すべて小文字に変換して集計する。

    Args:
        text: 解析対象のテキスト文字列。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    # 英数字の単語を抽出
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    # 小文字に統一
    lower_words = [w.lower() for w in words]
    return Counter(lower_words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 N 件の (単語, 回数) タプルを返す。

    Args:
        counter: 単語頻度 Counter。
        n: 取得する件数。

    Returns:
        上位 N 件の (単語, 回数) のリスト。
    """
    return counter.most_common(n)

    """
    if not filepath.exists():
        raise FileNotFoundError(f"File not found: {filepath}")
    return filepath.read_text(encoding='utf-8')
