#!/usr/bin/env python3
"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度をカウントして上位 N 件を表示するコマンドラインツール。

Features:
- 英単語のみを認識し、大文字小文字は無視 (lowercase)
- 句読点や数字を除去
- argparse で CLI インターフェース提供
- 出力は整形されたテーブル形式
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple, Dict


def extract_words(text: str) -> List[str]:
    """
    テキストから英単語のみを抽出し、正規化 (小文字化) する。
    
    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 抽出された単語リスト。
    """
    # 英単語のみ抽出 (a-z, A-Z)。他の文字は区切りとみなす。
    words = re.findall(r'[a-zA-Z]+', text)
    # 全て小文字に変換
    return [w.lower() for w in words]


def count_frequencies(text: str) -> Dict[str, int]:
    """
    テキスト内の単語頻度をカウントする。
    
    Args:
        text (str): 入力テキスト。

    Returns:
        Dict[str, int]: 単語 -> 出現回数の辞書。
    """
    words = extract_words(text)
    return dict(Counter(words))
