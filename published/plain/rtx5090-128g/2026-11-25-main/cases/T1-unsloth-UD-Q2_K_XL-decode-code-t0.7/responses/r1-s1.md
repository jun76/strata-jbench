#!/usr/bin/env python3
"""
word_counter.py - テキストファイル内の単語出現頻度を上位 N 件で表示するコマンドラインツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから英数字の単語を抽出し、小文字に変換して返す。

    Args:
        text (str): 解析対象のテキスト。

    Returns:
        List[str]: 小文字化された単語のリスト。
    """
    # 英数字の連続した文字列を単語として抽出
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    return [word.lower() for word in words]


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text (str): 解析対象のテキスト。

    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = extract_words(text)
    return Counter(words)


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    最も出現頻度の高い上位 n 件の単語とその回数を取得する。

    Args:
        counter (Counter): 単語頻度の Counter オブジェクト。
        n (int): 取得する上位件数。

    Returns:
        List[Tuple[str, int]]: (単語, 出現回数) のタプルのリスト。
    """
    # most_common はデフォルトで降順にソートされる
    return counter.most_common(n)
