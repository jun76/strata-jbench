    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=10,
        help="表示する単語の最大数 (デフォルト: 10)"
    )
    parser.add_argument(
        "--ignore-case",
        action="store_true",
        help="大文字・小文字を区別せず集計する"
    )
    return parser.parse_args()


def read_file_content(filepath: str) -> str:
    """
    テキストファイルを読み込み、内容を返す。

    Args:
        filepath (str): 読み込むファイルのパス。

    Returns:
        str: ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み込み中にエラーが発生した場合。
    """
    path = Path(filepath)
    if not path.exists():
        raise FileNotFoundError(f"ファイルが見つかりません: {filepath}")
    if not path.is_file():
        raise IOError(f"指定されたパスはファイルではありません: {filepath}")
    
    try:
        return path.read_text(encoding="utf-8")
    except Exception as e:
        raise IOError(f"ファイルの読み込みに失敗しました: {e}")
