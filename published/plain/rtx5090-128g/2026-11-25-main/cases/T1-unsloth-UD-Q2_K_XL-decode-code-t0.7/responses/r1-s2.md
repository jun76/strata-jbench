#!/usr/bin/env python3
"""
wordcount.py - テキストファイル内の単語出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツール。

使用例:
    python wordcount.py input.txt --top 10
    python wordcount.py input.txt --top 5 --ignore-case --sort order
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, ignore_case: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト文字列。
        ignore_case: True の場合、単語を小文字に正規化する。

    Returns:
        抽出された単語のリスト。
    """
    # 英数字とハイフンを含むトークンを抽出（例: "don't" -> "don't", "hello" -> "hello"）
    words = re.findall(r"[A-Za-z0-9]+(?:-[A-Za-z0-9]+)*", text)
    if ignore_case:
        words = [w.lower() for w in words]
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)
