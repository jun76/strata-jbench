# 単語区切り: 英数字 + ハイフン/アポストロフィを単語とし、それ以外は区切りにする
_DEFAULT_WORD_RE = re.compile(r"[0-9A-Za-z'\-]+")


def read_text(path: Path | str) -> str:
    """読み込み失敗時に FileNotFoundError を素直に伝播させるヘルパー。"""
    p = Path(path)
    return p.read_text(encoding="utf-8")


def tokenize(text: str, pattern: re.Pattern[str] | None = None) -> list[str]:
    """
    テキストを単語リストに分割する。

    Args:
        text: 対象テキスト。
        pattern: 単語マッチ用正規表現。省略時 `_DEFAULT_WORD_RE`。

    Returns:
        単語(小文字化済み・順序保持)のリスト。
    """
    p = pattern or _DEFAULT_WORD_RE
    return [m.group(0).lower() for m in p.finditer(text)]


def count_words(text: str, pattern: re.Pattern[str] | None = None) -> Counter[str]:
    """
    テキスト中の単語出現回数を数える。

    Args:
        text: 対象テキスト。
        pattern: 単語マッチ用正規表現。省略時 `_DEFAULT_WORD_RE`。

    Returns:
        単語 -> 回数の Counter。
    """
    words = tokenize(text, pattern)
    return Counter(words)
