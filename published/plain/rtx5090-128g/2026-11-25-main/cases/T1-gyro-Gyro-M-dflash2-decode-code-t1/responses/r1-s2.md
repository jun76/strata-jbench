Features:

def parse_words(text: str, ignore_case: bool = False, min_length: int = 1) -> List[str]:
    """
    Parse text into words, optionally ignoring case and filtering by minimum length.

    Args:
        text: The input text to parse.
        ignore_case: If True, words are lowercased before counting.
        min_length: Minimum length of words to include. Words shorter than this are excluded.

    Returns:
        A list of words that meet the criteria.
    """
    # Extract words using regex: sequences of alphanumeric characters
    words = re.findall(r'\b\w+\b', text)
    
    if ignore_case:
        words = [word.lower() for word in words]
    
    words = [word for word in words if len(word) >= min_length]
    
    return words


def count_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter mapping words to their counts.
    """
    return Counter(words)
