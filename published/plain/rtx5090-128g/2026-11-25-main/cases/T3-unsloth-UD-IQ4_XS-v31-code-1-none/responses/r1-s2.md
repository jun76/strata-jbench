def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    """
    Merge all overlapping or touching closed intervals.
    
    Args:
        intervals: A list of intervals, where each interval is a list [start, end].
                   The list may be unsorted and may be empty.
    
    Returns:
        A sorted list of non-overlapping, non-touching intervals.
    """
    if not intervals:
        return []
    
    # Sort intervals by start time
    sorted_intervals = sorted(intervals, key=lambda x: x[0])
    
    merged = [sorted_intervals[0][:]]  # Copy of the first interval
    
    for current in sorted_intervals[1:]:
        last = merged[-1]
        # If current interval overlaps or touches the last merged interval
        if current[0] <= last[1]:
            # Extend the end of the last interval if needed
            if current[1] > last[1]:
                last[1] = current[1]
        else:
            # No overlap, add the current interval
            merged.append(current[:])  # Copy to avoid modifying input
    
    return merged
