#!/usr/bin/env python3
"""
wordfreq.py - テキストファイルの単語出現頻度を集計し、上位 N 件を表形式で表示するツール。

Usage:
    python wordfreq.py <input_file> [-n N] [--sort-by freq|alpha]

Example:
    python wordfreq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    # 英数字とハイフンを含む単語として抽出
    return re.findall(r"[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*", text.lower())


def count_words(words: List[str]) -> Counter:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int, sort_by: str = "freq") -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (word, count) を返す。

    Args:
        counter: 単語頻度の Counter。
        n: 取得する件数。
        sort_by: 並び順キー。'freq' (頻度降順) または 'alpha' (単語昇順)。

    Returns:
        (word, count) のタプルのリスト。
    """
    if sort_by == "freq":
        # 頻度降順、同じ頻度なら辞書順
        items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    elif sort_by == "alpha":
        items = sorted(counter.items(), key=lambda x: x[0])
    else:
        raise ValueError(f"Unknown sort_by: {sort_by}")
    return items[:n]
