#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込んで単語の出現頻度を
集計し、上位 N 件を表形式で表示する。

使い方:
    python word_frequency.py <file_path> [--top N] [--case-sensitive] [--min-length M]

例:
    python word_frequency.py sample.txt --top 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def read_file(file_path: str) -> str:
    """指定されたファイルを読み込み、その内容を文字列として返す。

    Args:
        file_path: 読み込むファイルのパス

    Returns:
        ファイルの内容を示す文字列

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        UnicodeDecodeError: ファイルがUTF-8でエンコードされていない場合
    """
    with open(file_path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """テキストから単語トークンを抽出する。

    Args:
        text: 解析対象のテキスト
        case_sensitive: 大文字と小文字を区別するかどうか
        min_length: 抽出する単語の最小長（これより短い単語は無視）

    Returns:
        単語トークンのリスト
    """
    # 英数字とハイフンを含む単語を抽出（日本語などの文字も考慮）
    # 英数字、ハイフン、アポストロフィを含む連続文字列を単語とみなす
    pattern = r"[a-zA-Z0-9'-]+"
    words = re.findall(pattern, text)

    if not case_sensitive:
        words = [w.lower() for w in words]

    # 最小長でフィルタリング
    words = [w for w in words if len(w) >= min_length]

    return words


def count_frequencies(words: List[str]) -> Counter[str]:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語とその出現回数をキー・値とするCounterオブジェクト
    """
    return Counter(words)
