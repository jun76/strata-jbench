#!/usr/bin/env python3
"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a tabular format.
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize the input text into words.

    Words are defined as sequences of alphanumeric characters.
    The tokenization is case-insensitive (all words are lowercased).

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    return re.findall(r'[a-z0-9]+', text.lower())


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word in the list.

    Args:
        words: A list of words.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) sorted by count descending,
        then alphabetically ascending for ties.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]


def format_table(items: List[Tuple[str, int]]) -> str:
    """
    Format the word frequency items as a table.

    Args:
        items: A list of tuples (word, count).

    Returns:
        A formatted string representing the table.
    """
    if not items:
        return "No words found."

    # Calculate column widths
    word_width = max(len(word) for word, _ in items)
    word_width = max(word_width, 4)  # Minimum header width for "Word"
    count_width = max(len(str(count)) for _, count in items)
    count_width
