def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    正規表現 [a-zA-Z0-9']+ を使用して、英数字とアポストロフィを含む連続した文字列を単語として扱う。
    すべて小文字に変換して大文字小文字を区別しない。

    Args:
        text (str): 入力テキスト

    Returns:
        List[str]: 単語のリスト
    """
    # 正規表現で単語を抽出 (英数字とアポストロフィを含む)
    words = re.findall(r"[a-zA-Z0-9']+", text)
    # すべて小文字に変換
    return [word.lower() for word in words]

        Counter: 単語ごとの出現回数を格納した Counter オブジェクト
    """
    words = tokenize(text)
