"""
Text frequency counter command line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into words.

    Words are defined as sequences of alphanumeric characters.
    The text is lowercased and split on whitespace.

    Args:
        text: Input text string.

    Returns:
        List of tokens (words).
    """
    # Lowercase the text
    text = text.lower()
    # Split on whitespace and filter empty strings
    tokens = text.split()
    # Remove punctuation-like characters that might be attached to words
    cleaned_tokens = []
    for token in tokens:
        # Strip leading and trailing punctuation
        stripped = token.strip(".,!?;:\"'()[]{}")
        if stripped:
            cleaned_tokens.append(stripped)
    return cleaned_tokens


def count_words(tokens: List[str]) -> Counter:
    """
    Count occurrences of each word.

    Args:
        tokens: List of words.

    Returns:
        Counter object mapping words to their counts.
    """
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        List of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
