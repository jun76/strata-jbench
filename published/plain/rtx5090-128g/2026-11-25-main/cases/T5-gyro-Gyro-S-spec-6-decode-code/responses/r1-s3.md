#!/usr/bin/env python3
"""Command-line tool to read a text file and display the top N most frequent words."""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count word occurrences in the given text.

    Words are split by whitespace and non-alphabetic characters are stripped.
    Words are converted to lowercase for case-insensitive counting.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object mapping each word to its occurrence count.
    """
    words = []
    for word in text.split():
        # Strip non-alphabetic characters from start and end
        stripped = word.strip()
        if stripped:
            # Remove punctuation from beginning and end
            cleaned = ''.join(c for c in stripped if c.isalpha() or c.isdigit())
            if cleaned:
                words.append(cleaned.lower())
    return Counter(words)


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples containing (word, count), sorted by count descending,
        then alphabetically for ties.
    """
    items = list(counter.items())
    # Sort by count descending, then by word alphabetically for ties
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(top_words: List[Tuple[str, int]]) -> str:
    """Format the top words into a table string.

    Args:
        top_words: A list of tuples containing (word, count).

    Returns:
        A formatted table as a string.
    """
    if not top_words:
        return "No words found."

    lines = []
    lines.append(f"{'Word':<20} {'Count':>5}")
    lines.append("-" * 25)
    for word, count in top_words:
        lines.append(f"{word:<20} {count:>5}")
    return "\n".join(lines)
