def count_words(text: str, *, min_len: int = 1, ignore_case: bool = True) -> Counter[str]:
    """
    テキストから単語を抽出し、出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        min_len: 単語の最小長。これより短い単語は除外される。
        ignore_case: True の場合、単語は小文字に正規化される。

    Returns:
        Counter[str]: 単語 -> 出現回数 のマッピング。
    """
    if ignore_case:
        text = text.lower()
    words = [w for w in text.split() if len(w) >= min_len]
    return Counter(words)
