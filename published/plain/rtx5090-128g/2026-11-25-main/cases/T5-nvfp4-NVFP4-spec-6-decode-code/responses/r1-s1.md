word_freq.py - Count word frequencies from a text file and display the top N words.

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse

from collections import Counter

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: Input text string.
        case_sensitive: If False, convert words to lowercase before counting.

    Returns:
        A list of word tokens (strings).
    """
    # Extract words using regex: sequences of alphanumeric characters
    words = re.findall(r'[a-zA-Z0-9]+', text)
    if not case_sensitive:
        words = [word.lower() for word in words]
    return words


def count_words(text: str, case_sensitive: bool = False) -> Dict[str, int]:
    """
    Count the frequency of each word in the given text.

    Args:
        text: Input text string.
        case_sensitive: If False, treat words as case-insensitive.

    Returns:
        A dictionary mapping each word to its count.
    """
    tokens = tokenize(text, case_sensitive)
    return dict(Counter(tokens))


def get_top_n(counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N words by frequency.

    Ties are broken alphabetically.

    Args:
        counts: Dictionary mapping words to their counts.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending, then word ascending.
    """
    # Sort by count descending, then by word ascending for tie-breaking
    sorted_items = sorted(counts.items(), key=lambda item: (-item[1], item[0]))
    return sorted_items[:n]
