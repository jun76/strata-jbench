word_freq.py - テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示するツール。

    python word_freq.py input.txt [--top N] [--case-sensitive]

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: True の場合、大文字・小文字を区別する。False の場合、小文字に変換する。

    Returns:
        抽出された単語のリスト。単語はアルファベット、数字、アンダースコアを含む連続した文字列として定義される。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とアンダースコアを含む連続文字列を単語として抽出
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return words

    """

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)
