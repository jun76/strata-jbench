#!/usr/bin/env python3

テキストファイルを読み込み、単語の出現頻度を集計し、

import argparse
import re

from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。

    単語はアルファベット（a-z, A-Z）と数字（0-9）で構成される文字列として
    認識される。大文字小文字は区別せず、小文字に正規化される。

    Args:
        text: 入力テキスト文字列。

    Returns:
        小文字化された単語のリスト。
    """
    # アルファベットと数字のみを含む単語を抽出
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から出現頻度の上位 n 件の単語を返す。

    頻度が同じ場合は、辞書順（アルファベット順）でソートされる。

    Args:
        counter: 単語の出現頻度を保持する Counter。
        n: 返す上位単語の数。

    Returns:
        (単語, 出現回数) のタプルのリスト。上位 n 件。
    """
    # most_common は頻度順だが、同頻度時の順序は不定なので、
    # 頻度降順、単語昇順で再ソートする
    items = list(counter.items())
    items.sort(key=lambda item: (-item[1], item[0]))
    return items[:n]
