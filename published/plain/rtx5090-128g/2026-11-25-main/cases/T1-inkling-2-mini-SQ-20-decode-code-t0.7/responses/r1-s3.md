#!/usr/bin/env python3
"""テキストファイルから単語の出現頻度を集計し、上位 N 件を表示するCLIツール。"""

from __future__ import annotations

import argparse
import re
from collections import Counter
from pathlib import Path


def tokenize(text: str) -> list[str]:
    """テキストを単語トークン列に分割する。

    英数字とアンダースコアを含む連続部分を単語とみなし、小文字化する。
    """
    return re.findall(r"[A-Za-z0-9_]+", text.lower())


def count_words(text: str) -> Counter[str]:
    """テキストから単語頻度を数える。"""
    return Counter(tokenize(text))


def top_words(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """頻度降順・同頻度時は単語昇順で上位 N 件を返す。"""
    return sorted(counter.items(), key=lambda item: (-item[1], item[0]))[:n]


def format_table(rows: list[tuple[str, int]]) -> str:
    """上位単語を表形式の文字列として整形する。"""
    if not rows:
        return "（単語が見つかりません）"
    word_width = max(len(word) for word, _ in rows)
    lines = [f"{'Word'.ljust(word_width)}  Count", f"{'-' * word_width}  -----"]
    for word, count in rows:
        lines.append(f"{word.ljust(word_width)}  {count}")
    return "\n".join(lines)
