# word_freq.py
from __future__ import annotations

import argparse
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    テキストから単語を抽出して出現回数を集計する。

    単語は英数字とアンダースコアの連続部分として扱い、
    大文字小文字は区別しない。
    """
    import re
    words = re.findall(r"[A-Za-z0-9_]+", text.lower())
    return Counter(words)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    集計結果から上位 n 件を返す。
    同数の場合は単語のアルファベット順で安定化する。
    """
    if n < 0:
        raise ValueError("n は 0 以上である必要があります。")
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(rows: List[Tuple[str, int]]) -> str:
    """
    上位 n 件を表形式の文字列として返す。
    """
    if not rows:
        return "（単語が見つかりません）"
    name_width = max(len(word) for word, _ in rows)
    lines = [f"{'単語'.ljust(name_width)} | 回数",
             f"{'-' * name_width}-+------"]
    for word, count in rows:
        lines.append(f"{word.ljust(name_width)} | {count}")
    return "\n".join(lines)
