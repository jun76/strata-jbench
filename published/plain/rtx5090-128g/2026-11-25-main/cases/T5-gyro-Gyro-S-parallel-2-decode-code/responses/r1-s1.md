#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルの単語出現頻度を上位N件で表示する。

使い方:
    python word_freq.py input.txt --top 10

オプション:
    --top N        上位N件を表示 (デフォルト: 10)
    --case-sensitive 大文字小文字を区別する (デフォルト: False, 小文字に統一)
"""

import argparse

from collections import Counter
from typing import List, Tuple


def read_words_from_file(filepath: str) -> List[str]:
    """
    ファイルを読み込み、単語のリストを返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        単語のリスト（小文字に統一される）。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        content = f.read()
    words = content.split()
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str], case_sensitive: bool = False) -> Counter[str]:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。
        case_sensitive: Trueの場合、大文字小文字を区別する。

    Returns:
        単語とその出現回数のCounter。
    """
    if case_sensitive:
        return Counter(words)
    else:
        # 小文字に統一してカウント
        lower_words = [word.lower() for word in words]
        return Counter(lower_words)


def get_top_n_frequencies(
    counter: Counter[str], n: int
) -> List[Tuple[str, int]]:
    """
    出現頻度が高い上位n件の単語とその回数を取得する。

    Args:
        counter: 単語の出現頻度Counter。
        n: 取得する上位件数。

    Returns:
        (単語, 回数)のタプルのリスト。頻度降順。
    """
    return counter.most_common(n)
