#!/usr/bin/env python3
"""
CLI tool to read a text file, count word frequencies, and display top N results.

Usage:
    python word_freq.py --file <path> --top <N>
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(file_path: str) -> Counter[str]:
    """Count word frequencies in a text file.

    Words are split on whitespace and normalized to lowercase.
    Punctuation is stripped from the beginning and end of each token.

    Args:
        file_path: Path to the text file to read.

    Returns:
        A Counter mapping normalized words to their occurrence counts.
    """
    with open(file_path, "r", encoding="utf-8") as f:
        content = f.read()

    # Split on whitespace, normalize, strip punctuation
    tokens = content.split()
    normalized: List[str] = []
    for token in tokens:
        word = token.strip(".,!?;:\"'()[]{}")
        if word:
            normalized.append(word.lower())

    return Counter(normalized)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Get the top N words by frequency from a Counter.

    Args:
        counter: A Counter mapping words to their counts.
        n: Number of top entries to retrieve.

    Returns:
        A list of (word, count) tuples sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by count descending, then by word ascending for stable ordering
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(items: List[Tuple[str, int]], width: int = 20) -> str:
    """Format a list of (word, count) tuples as a simple text table.

    Args:
        items: List of (word, count) tuples.
        width: Maximum width of the word column.

    Returns:
        A formatted string representing the table.
    """
