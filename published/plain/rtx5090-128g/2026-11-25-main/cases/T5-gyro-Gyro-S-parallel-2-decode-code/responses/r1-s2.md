"""
word_counter.py - 単語出現頻度を集計し上位N件を表形式で表示するCLIツール。

使い方:
    python word_counter.py input.txt --top 10
    python word_counter.py input.txt --top 5 --min-length 3
"""

import argparse
import sys
from collections import Counter
from typing import Dict, List, Tuple


def tokenize(text: str, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        min_length: 単語の最小文字数（これ未満の単語は除外）

    Returns:
        小文字化された単語リスト
    """
    # 非英数字を区切りとして扱い、単語のみ抽出
    words = []
    current = []
    for char in text:
        if char.isalnum():
            current.append(char.lower())
        else:
            if current:
                word = "".join(current)
                if len(word) >= min_length:
                    words.append(word)
                current = []
    if current:
        word = "".join(current)
        if len(word) >= min_length:
            words.append(word)
    return words


def count_words(text: str, min_length: int = 1) -> Dict[str, int]:
    """
    テキスト内の単語出現頻度を集計する。

    Args:
        text: 入力テキスト
        min_length: 単語の最小文字数

    Returns:
        単語とその出現回数の辞書
    """
    words = tokenize(text, min_length)
    return dict(Counter(words))
