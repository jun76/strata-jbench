word_freq - コマンドラインツールでテキストファイルの単語頻度を集計し、
上位N件を表形式で表示する。

Usage:
    python word_freq.py input.txt [-n TOP_N] [--case-sensitive]
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple, TextIO


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト内の単語の出現頻度をカウントする。

    Args:
        text: 解析対象のテキスト。
        case_sensitive: 単語の大小文字を区別するか。False の場合、すべて小文字に変換してカウントする。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字の連続を単語とみなす
    words = re.findall(r'[a-zA-Z0-9_]+', text)
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度の高い順に上位N件の (word, count) のリストを返す。
    頻度が同じ場合は単語のアルファベット順（辞書順）でソートする。

    Args:
        counter: 単語と出現回数の Counter。
        n: 取得する上位件数。

    Returns:
        (word, count) のタプルのリスト。
    """
    # まず頻度で降順、次に単語名で昇順ソート
    sorted_items = sorted(counter.items(), key=lambda item: (-item[1], item[0]))
    return sorted_items[:n]
