WORD_PATTERN = re.compile(r'\w+')

def tokenize(text: str) -> List[str]:
    """与えられたテキストから小文字化された単語のリストを抽出する。

    Args:
        text: 解析対象の文字列。

    Returns:
        正規表現 \\w+ で抽出され、すべて小文字に変換された単語のリスト。
    """
    return WORD_PATTERN.findall(text)

def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 解析対象の文字列。

    Returns:
        単語 (小文字) をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(tokenize(text))
