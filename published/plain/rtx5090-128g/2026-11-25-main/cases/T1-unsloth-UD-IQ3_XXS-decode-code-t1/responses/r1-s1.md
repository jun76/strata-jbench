使い方:
    python wordcount.py input.txt -n 10
    python wordcount.py input.txt --sort-by count --case-sensitive

依存:
    - argparse, collections, pathlib, re, textwrap (標準ライブラリのみ)
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語リストに分割する。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合、大文字・小文字を区別する。False の場合、全て小文字に変換する。

    Returns:
        単語のリスト（空文字列や空白は除外）
    """
    # 正規表現で単語を抽出（アルファベット、数字、ハイフンを含む連続文字列）
    pattern = r'[a-zA-Z0-9\-]+'
    words = re.findall(pattern, text)
    if not case_sensitive:
        words = [word.lower() for word in words]
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト中の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合、大文字・小文字を区別する。False の場合、全て小文字に変換する。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)
