def normalize_words(text: str) -> List[str]:
    """
    テキストを小文字化し、非英数字を区切りとして単語リストを生成する。

    Args:
        text: 入力テキスト。

    Returns:
        正規化された単語のリスト。
    """
    # 英数字（a-z, 0-9）のみを抽出し、それ以外を空白として分割
    words = re.findall(r'[a-z0-9]+', text.lower())
    return words


def count_frequency(words: List[str]) -> Counter[str]:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、頻度を値とする Counter。
    """
    return Counter(words)
