#!/usr/bin/env python3
"""
word_freq CLI tool: Read a text file, count word frequencies, and display top N entries as a table.
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.

    Args:
        text: Input text string.

    Returns:
        List of lowercase word tokens (only alphabetic characters considered).
    """
    return re.findall(r'[a-zA-Z]+', text.lower())


def count_words(text: str) -> Counter:
    """Count the frequency of each word in the given text.

    Args:
        text: Input text string.

    Returns:
        Counter object with word frequencies.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples, sorted by frequency descending.
    """
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]], width_word: int = 20, width_count: int = 10) -> str:
    """Format a list of (word, count) tuples as a simple text table.

    Args:
        items: List of (word, count) tuples.
        width_word: Column width for the word column.
        width_count: Column width for the count column.

    Returns:
        Formatted string representing the table.
    """
    if not items:
        return "No words found."

    lines = []
    header = f"{'Word':<{width_word}} | {'Count':>{width_count}}"
    separator = "-" * width_word + "-+-" + "-" * width_count
    lines.append(header)
    lines.append(separator)
    for word, count in items:
        lines.append(f"{word:<{width_word}} | {count:>{width_count}}")
    return "\n".join(lines)
