#!/usr/bin/env python3
"""テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する。"""

from __future__ import annotations

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import Iterable


def tokenize(text: str) -> list[str]:
    """テキストを小文字化し、単語らしいトークンに分割する。"""
    return re.findall(r"[a-z0-9]+", text.lower())


def count_words(path: Path) -> Counter:
    """ファイルを読み込み、単語出現頻度を集計する。"""
    text = path.read_text(encoding="utf-8")
    return Counter(tokenize(text))


def top_n(counter: Counter, n: int) -> list[tuple[str, int]]:
    """上位 N 件を (word, count) のリストで返す。"""
    return counter.most_common(n)


def format_table(rows: Iterable[tuple[str, int]]) -> str:
    """表形式の文字列を生成する。"""
    rows = list(rows)
    if not rows:
        return "(no words)"

    word_width = max(len(word) for word, _ in rows)
    lines = [f"{'word'.ljust(word_width)}  count"]
    lines.append(f"{'-' * word_width}  {'-' * 5}")
    for word, count in rows:
        lines.append(f"{word.ljust(word_width)}  {count}")
    return "\n".join(lines)
