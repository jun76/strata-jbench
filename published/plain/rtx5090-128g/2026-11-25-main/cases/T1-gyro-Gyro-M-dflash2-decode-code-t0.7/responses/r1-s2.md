"""
Top N Word Frequency Counter

A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a table format.
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    Count word occurrences in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping words to their occurrence counts.
        Words are normalized to lowercase and non-alphanumeric characters
        are treated as delimiters.
    """
    # Normalize to lowercase and extract words (alphanumeric sequences)
    words = []
    current_word = []
    for char in text.lower():
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    if current_word:
        words.append(''.join(current_word))

    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object mapping words to their occurrence counts.
        n: The number of top entries to return.

    Returns:
        A list of (word, count) tuples, ordered by descending count.
        If there are fewer than n distinct words, returns all available entries.
    """
    return counter.most_common(n)
