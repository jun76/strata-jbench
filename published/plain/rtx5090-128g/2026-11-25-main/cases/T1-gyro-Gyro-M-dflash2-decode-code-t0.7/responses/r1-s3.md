"""Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.

Usage:
    python wordfreq.py <file> [--top N] [--case-sensitive]
"""

from typing import List, Tuple


def read_text(file_path: str) -> str:
    """Read the content of a text file.

    Args:
        file_path: Path to the text file.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
    """
    with open(file_path, "r", encoding="utf-8") as f:
        return f.read()


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """Count word occurrences in the given text.

    Words are defined as sequences of alphanumeric characters (including underscores),
    separated by non-alphanumeric characters.

    Args:
        text: The input text to analyze.
        case_sensitive: If True, words are counted as-is. If False, words are
                        lowercased before counting.

    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    if not case_sensitive:
        text = text.lower()
    # Split on non-alphanumeric/underscore characters
    words: List[str] = []
    current_word: List[str] = []
    for char in text:
        if char.isalnum() or char == "_":
            current_word.append(char)
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))
    return Counter(words)
