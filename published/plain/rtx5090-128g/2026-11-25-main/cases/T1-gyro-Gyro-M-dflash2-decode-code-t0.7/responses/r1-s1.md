"""
text_word_counter.py

A command-line tool that reads a text file, counts word frequencies,
and displays the top N words in a table format.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def read_text_file(file_path: str) -> str:
    """
    Read the contents of a text file.

    Args:
        file_path: Path to the text file to read.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(file_path, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """
    Extract words from text, converting to lowercase.

    Words are defined as sequences of alphanumeric characters.
    Non-alphanumeric characters are treated as separators.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Use regex to find all words (sequences of alphanumeric characters)
    words = re.findall(r"[a-zA-Z0-9]+", text)
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words to count.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) sorted by frequency descending.
    """
    return counter.most_common(n)
