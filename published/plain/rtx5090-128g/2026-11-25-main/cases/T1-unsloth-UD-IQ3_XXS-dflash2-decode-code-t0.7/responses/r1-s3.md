#!/usr/bin/env python3
"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N most frequent
words in a formatted table.
"""

import argparse
import re

from collections import Counter

def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Words are defined as sequences of alphanumeric characters and underscores.
    Punctuation and other characters are treated as separators.

    Args:
        text: Input text to tokenize.

    Returns:
        List of lowercase word tokens.
    """
    return re.findall(r'\b\w+\b', text.lower())


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: Input text to analyze.

    Returns:
        Counter object with word counts.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a counter.

    In case of ties in frequency, words are sorted alphabetically.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to retrieve.

    Returns:
        List of tuples (word, count) sorted by count descending, then word ascending.
    """
    # Counter.most_common sorts by count descending, but for ties we want
    # alphabetical order. We need to handle this explicitly.
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
