def extract_words(text: str, case_sensitive: bool = False) -> list[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。

    Returns:
        単語のリスト。正規表現 \\b[\\w']+\\b に一致するトークンを抽出する。
    """
    if not case_sensitive:
        text = text.lower()
    # 単語は英数字とアポストロフィを含む連続文字列とする
    words = re.findall(r"\b[\w']+\b", text)
    return words

        Counter オブジェクト。キーが単語、値が出現回数。
    """
    return Counter(words)
