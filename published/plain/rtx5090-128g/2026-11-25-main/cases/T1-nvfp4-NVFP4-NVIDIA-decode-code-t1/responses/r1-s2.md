def load_text(filepath: Path, case_insensitive: bool = False) -> List[str]:
    """
    テキストファイルを読み込み、単語のリストを返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:

    """

        content = filepath.read_text(encoding="utf-8")

        raise PermissionError(f"ファイルの読み取り権限がありません: {filepath}")

        # UTF-8 で失敗した場合は cp932 で再試行（日本語ファイル対応）
        content = filepath.read_text(encoding="cp932")

    words: List[str] = content.split()
    if case_insensitive:
        words = [word.lower() for word in words]
    return words
