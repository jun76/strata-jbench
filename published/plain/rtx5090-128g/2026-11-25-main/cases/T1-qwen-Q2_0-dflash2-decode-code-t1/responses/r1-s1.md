import argparse
import collections
import sys
from typing import List, Tuple


def parse_args(argv: List[str] = None) -> argparse.Namespace:
    """
    Parse command line arguments.

    Args:
        argv: List of argument strings. Defaults to sys.argv[1:].

    Returns:
        Namespace object with arguments: filename, top_n, output_format
    """
    parser = argparse.ArgumentParser(
        description="Read a text file and print top N word frequencies."
    )
    parser.add_argument(
        "filename",
        type=str,
        help="Path to the text file to read",
    )
    parser.add_argument(
        "--top-n",
        type=int,
        default=10,
        help="Number of top words to display (default: 10)",
    )
    parser.add_argument(
        "--output-format",
        type=str,
        choices=["table", "csv"],
        default="table",
        help="Output format: 'table' or 'csv' (default: table)",
    )
    return parser.parse_args(argv)


def count_word_frequencies(text: str) -> collections.Counter:
    """
    Count word frequencies in a given text.

    Words are split by whitespace, lowercased, and stripped of punctuation
    from word edges. Only alphanumeric characters are kept in words.

    Args:
        text: Input text string.

    Returns:
        Counter mapping lowercase words to their occurrence counts.
    """
    words: List[str] = []
    for word in text.split():
        # Strip non-alphanumeric characters from edges
        cleaned = "".join(c for c in word.lower() if c.isalnum())
        if cleaned:
            words.append(cleaned)
    return collections.Counter(words)


def format_as_table(frequencies: collections.Counter, top_n: int) -> str:
    """
    Format word frequencies as a plain text table.

    Args:
        frequencies: Counter object of word frequencies.
        top_n: Number of top words to display.

    Returns:
        String representation of the table.
    """
    top_words = frequencies.most_common(top_n)
