"""テキストファイルから単語の出現頻度を集計し、上位 N 件を表示する CLI ツール。"""

from __future__ import annotations

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import Iterable


def tokenize(text: str) -> list[str]:
    """テキストを小文字の単語リストに分解する。"""
    return re.findall(r"[a-z0-9]+", text.lower())


def count_words(text: str) -> Counter:
    """テキストから単語出現頻度を集計する。"""
    return Counter(tokenize(text))


def read_file(path: str | Path) -> str:
    """テキストファイルを読み込む。"""
    return Path(path).read_text(encoding="utf-8")


def top_n(counter: Counter, n: int) -> list[tuple[str, int]]:
    """上位 N 件を (単語, 頻度) のリストで返す。"""
    return counter.most_common(n)


def format_table(rows: Iterable[tuple[str, int]]) -> str:
    """(単語, 頻度) の行を表形式で整形する。"""
    rows = list(rows)
    if not rows:
        return "(結果なし)"
    word_width = max(len(word) for word, _ in rows)
