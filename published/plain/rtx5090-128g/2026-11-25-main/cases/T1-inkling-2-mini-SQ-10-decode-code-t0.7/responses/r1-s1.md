#!/usr/bin/env python3
"""テキストファイルから単語の出現頻度を集計して上位 N 件を表形式で表示する。"""

from __future__ import annotations

import argparse
import re
from collections import Counter
from pathlib import Path


def tokenize(text: str) -> list[str]:
    """テキストを単語トークン列に変換する。"""
    return re.findall(r"[a-zA-Z0-9_]+", text.lower())


def count_words(path: Path) -> Counter:
    """テキストファイルを読み込み、単語出現頻度を集計する。"""
    text = path.read_text(encoding="utf-8")
    return Counter(tokenize(text))


def top_n(counter: Counter, n: int) -> list[tuple[str, int]]:
    """出現頻度の高い上位 N 件を返す。"""
    return counter.most_common(n)


def format_table(rows: list[tuple[str, int]]) -> str:
    """上位 N 件を表形式の文字列にする。"""
    if not rows:
        return "(no words)"

    header = f"{'rank':<5} {'word':<30} {'count':>10}"
    lines = [header, "-" * len(header)]
    for i, (word, count) in enumerate(rows, start=1):
        lines.append(f"{i:<5} {word:<30} {count:>10}")
    return "\n".join(lines)
