#!/usr/bin/env python3
"""テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する。"""
from __future__ import annotations

import argparse
from collections import Counter
import re
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを小文字化し、単語（英数字）として抽出する。"""
    return re.findall(r"[a-z0-9]+", text.lower())


def count_words(path: Path) -> Counter:
    """ファイルを読み込み、単語出現頻度を返す。"""
    text = path.read_text(encoding="utf-8")
    return Counter(tokenize(text))


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """上位 N 件を (word, count) のリストで返す。"""
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """表形式の文字列を生成する。"""
    if not rows:
        return "(no words)"
    header = "word\tcount"
    lines = [header]
    for word, count in rows:
        lines.append(f"{word}\t{count}")
    return "\n".join(lines)
