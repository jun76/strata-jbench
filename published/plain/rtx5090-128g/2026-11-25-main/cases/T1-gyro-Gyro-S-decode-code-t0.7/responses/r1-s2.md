"""
コマンドラインツール: テキストファイルから単語の出現頻度を集計し、上位N件を表形式で表示する。

使用例:
    python word_freq.py input.txt --top 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def load_text(filepath: str) -> str:
    """指定されたファイルパスからテキストを読み込む。"""
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。非英数字の文字は区切りに使用される。"""
    words = []
    current_word = []
    for char in text:
        if char.isalnum():
            current_word.append(char.lower())
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))
    return words


def count_words(words: List[str]) -> Counter:
    """単語の出現頻度をカウントする。"""
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """頻度の上位N件の単語とその出現回数を取得する。"""
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """単語と出現回数を表形式でフォーマットする。"""
    if not items:
        return "No words found."

    # 最大単語長を計算して列幅を決定
    max_word_len = max(len(word) for word, _ in items)
    header = "  {:<{max_len}} {:>8}".format("word", "count", max_len=max_word_len)
    separator = "  {:<{max_len}} {:>8}".format("-" * max_word_len, "--------", max_len=max_word_len)

    lines = [header, separator]
    for word, count in items:
        line = "  {:<{max_len}} {:>8}".format(word, count, max_len=max_word_len)
        lines.append(line)

    return "\n".join(lines)
