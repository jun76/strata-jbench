def count_words(text: str, ignore_case: bool = True) -> Counter[str]:
    """
    テキスト中の単語を抽出し、出現頻度をカウントする。

    Args:
        text: 対象のテキスト文字列。
        ignore_case: True の場合、単語を小文字化して集計する。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    # 単語を分割: 空白区切り、句読点や記号は除去
    words = text.split()
    if ignore_case:
        words = [w.lower() for w in words]

    # 句読点や記号を除去して純粋な単語のみを残す
    cleaned: List[str] = []
    for word in words:
        # 先頭・末尾の句読点や記号を除去
        stripped = word.strip(".,!?;:\"'()[]{}")
        if stripped:
            cleaned.append(stripped)

    return Counter(cleaned)
