def read_text(path: str) -> str:
    """指定されたパスのテキストファイルを読み込み、その内容を文字列として返す。

    Args:
        path: テキストファイルのパス。

    Returns:
        ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        OSError: ファイル読み込み失敗時。
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語を抽出する。

    単語は空白や改行で区切られたトークンとし、句読点は除去する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。

    Returns:
        単語のリスト。
    """
    # 句読点や特殊文字を除去し、単語のみ抽出
    # 簡易な方法: 英数字のみを含むトークンを抽出
    words = []
    current = []
    for ch in text:
        if ch.isalnum():
            current.append(ch)
        else:
            if current:
                word = "".join(current)
                if not case_sensitive:
                    word = word.lower()
                words.append(word)
                current = []
    if current:
        word = "".join(current)
        if not case_sensitive:
            word = word.lower()
        words.append(word)
    return words
