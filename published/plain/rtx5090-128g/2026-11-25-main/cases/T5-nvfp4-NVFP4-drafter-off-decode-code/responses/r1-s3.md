#!/usr/bin/env python3
"""Text file word frequency analyzer.

Reads a text file, counts the occurrence of each word, and displays the top N
most frequent words in a table format.
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """Extract words from text, converting to lowercase.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase words found in the text.
    """
    # Match sequences of alphanumeric characters and apostrophes
    words = re.findall(r"[a-zA-Z0-9']+", text)
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """Count the frequency of each word in the list.

    Args:
        words: A list of words.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a counter.

    Args:
        counter: A Counter object containing word frequencies.
        n: The number of top items to return.

    Returns:
        A list of (word, count) tuples sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by count descending, then by word ascending for consistent ordering
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
