def extract_words(text: str, ignore_case: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    単語は、英数字、アンダースコア、ハイフン、アポストロフィを含む
    連続した文字列として定義される。

    if ignore_case:
        words = [w.lower() for w in words]
    return words
