def load_words(file_path: str, case_sensitive: bool = False) -> List[str]:
    """
    指定されたテキストファイルを読み込み、単語のリストを返す。

    Args:
        file_path: 読み込むテキストファイルのパス。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、
                        すべて小文字に変換する。

    Returns:
        単語のリスト。非英数字は単語の区切りとして扱われ、単語自体から
        非英数字は取り除かれる。
    """
    with open(file_path, 'r', encoding='utf-8') as f:
        text = f.read()

    if not case_sensitive:
        text = text.lower()

    # 非英数字をスペースで置換し、単語を分割
    cleaned_chars: List[str] = []
    for ch in text:
        if ch.isalnum():
            cleaned_chars.append(ch)
        else:
            cleaned_chars.append(' ')

    cleaned_text = ''.join(cleaned_chars)
    words = cleaned_text.split()
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度を集計する。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。

    return Counter(words)
