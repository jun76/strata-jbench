word_freq.py - コマンドラインツール：テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する。

Usage:
    python word_freq.py <file> [--top N]
"""

import argparse

import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストに分割する。
    単語はアルファベットと数字の連続とした。
    大文字・小文字は区別せず、すべて小文字に変換する。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト
    """
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (word, count) を取得する。

    Args:
        counter: 単語出現頻度の Counter
        n: 取得する上位件数

    Returns:
        (word, count) のタプルのリスト
    """
    return counter.most_common(n)
