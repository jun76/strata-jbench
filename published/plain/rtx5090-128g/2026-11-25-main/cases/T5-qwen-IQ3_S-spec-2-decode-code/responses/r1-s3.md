#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルの単語出現頻度を集計し、
上位 N 件を表形式で表示します。

Usage:
    python word_counter.py <input_file> [-n N] [--case-insensitive] [--min-len MIN_LEN]
"""

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from typing import List, Tuple


@dataclass
class WordCount:
    """単語とその出現回数を保持するデータクラス。"""
    word: str
    count: int

    def __str__(self) -> str:
        return f"{self.word}: {self.count}"


def tokenize(text: str, case_insensitive: bool = True, min_len: int = 1) -> List[str]:
    """
    テキストを単語に分割してリストとして返します。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、単語を小文字化します。
        min_len: 考慮する単語の最小長。

    Returns:
        単語のリスト。
    """
    # 英数字のみを単語として抽出（日本語やその他のUnicode文字は除外）
    words = re.findall(r'\b[a-zA-Z]+\b', text)
    if case_insensitive:
        words = [w.lower() for w in words]
    return [w for w in words if len(w) >= min_len]
