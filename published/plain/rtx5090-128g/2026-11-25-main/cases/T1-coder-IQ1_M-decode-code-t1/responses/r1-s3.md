def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.

    Splits text into sequences of word characters and converts them to lowercase.

    Args:
        text: Input text string.

    Returns:
        List of lowercase tokens.
    """
    return re.findall(r"[A-Za-z0-9]+", text)


def count_words(tokenizer: callable, texts: List[str]) -> Dict[str, int]:
    """Count word occurrences in a list of texts.

    Args:
        tokenizer: Function that takes text and returns list of tokens.
        texts: List of input text strings.

    Returns:
        Dictionary mapping each token to its occurrence count.
    """
    counter: Counter = Counter()
    for text in texts:
        counter.update(tokenize(text))
    return dict(counter)


def top_n(freq: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """Return top N frequent words sorted by count descending, then lexicographically.

    Args:
        freq: Dictionary of word -> count.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples.
    """
    items = sorted(freq.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
