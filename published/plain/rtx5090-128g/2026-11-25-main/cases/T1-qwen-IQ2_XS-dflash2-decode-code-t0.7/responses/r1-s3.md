テキストファイルを読み込んで単語の出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツール。

Example:
    python wordfreq.py sample.txt -n 10

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_text_file(filepath: Path) -> str:
    """テキストファイルを読み込んで内容を返す。

    Args:
        filepath: 読み込むファイルのパス。

    Returns:
        ファイルの内容 (str)。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み込みエラーの場合。
    """
    try:
        return filepath.read_text(encoding="utf-8")
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except IOError as e:
        raise IOError(f"Error reading file: {e}")


def count_words(text: str) -> Counter:
    """テキストから単語の出現頻度をカウントする。

    単語は空白・改行・句読点で区切られ、小文字化される。

    Args:
        text: カウント対象のテキスト。

    Returns:
        単語とその頻度の Counter。
    """
    import string
    # 句読点を削除し、小文字化
    cleaned = text.lower()
    # 句読点を空白に置き換える
    translator = str.maketrans("", "", string.punctuation)
    cleaned = cleaned.translate(translator)
    # 空白・改行で分割
    words = cleaned.split()
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (word, count) を返す。

    Args:
        counter: 単語頻度の Counter。
        n: 上位件数。

    Returns:
        上位 n 件の (word, count) のリスト。
    """
    return counter.most_common(n)
