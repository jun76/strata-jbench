"""Command-line tool to count word frequencies in a text file and display top N entries."""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words using regex.

    Args:
        text: Input text string.

    Returns:
        List of lowercase word tokens.
    """
    # Extract words using regex pattern (letters, digits, underscores as word chars)
    tokens = re.findall(r'\b[a-z0-9_]+\b', text.lower())
    return tokens


def count_words(text: str) -> Counter:
    """Count word frequencies in text.

    Args:
        text: Input text string.

    Returns:
        Counter mapping words to their frequencies.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get top N most common words.

    Args:
        counter: Counter with word frequencies.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples, sorted by frequency descending.
    """
    return counter.most_common(n)
