#!/usr/bin/env python3
"""
Text File Word Frequency Analyzer.

A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a table format.
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    Extract words from text and convert them to lowercase.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Extract words using regex: sequences of alphabetic characters
    words = re.findall(r'[a-zA-Z]+', text)
    return [word.lower() for word in words]


def count_word_frequencies(text: str) -> Counter[str]:
    """
    Count the frequency of each word in the given text.

    Args:
        text: Input text string.

    Returns:
        A Counter object with word frequencies.
    """
    words = extract_words(text)
    return Counter(words)


def get_top_n_words(
    word_counts: Counter[str],
    n: int
) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.

    Args:
        word_counts: A Counter object with word frequencies.
        n: Number of top words to retrieve.

    Returns:
        A list of tuples (word, count) sorted by frequency descending.
    """
    return word_counts.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """
    Format the word frequency data as a table string.

    Args:
        rows: A list of tuples (word, count).

    Returns:
        A formatted table string.
    """
    if not rows:
        return "No words found."

    # Calculate column widths
    word_width = max(len("Word") + 2, max(len(word) for word, _ in rows) + 2)
    count_width = max(len("Count") + 2, max(len(str(count)) for _, count in rows) + 2)

    # Build header
    header = f"{'Word':<{word_width}} | {'Count':<{count_width
