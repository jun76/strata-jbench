"""
word_freq: A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def normalize_words(text: str) -> List[str]:
    """
    Normalize text into lowercase words by extracting only alphabetic sequences.
    
    Args:
        text: The input text to normalize.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    return re.findall(r'[a-zA-Z]+', text.lower())


def count_words(words: List[str]) -> Counter:
    """
    Count occurrences of each word.
    
    Args:
        words: A list of words.
        
    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their counts.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.
        
    Returns:
        A list of (word, count) tuples sorted by frequency descending.
        If there are ties in frequency, words are sorted alphabetically.
    """
    # Counter.most_common returns elements sorted by count descending,
    # and for equal counts, insertion order (which we'll handle by sorting)
    # To ensure deterministic output, we sort by (-count, word)
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
