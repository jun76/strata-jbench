def tokenize(text: str, ignore_case: bool = True, min_length: int = 1) -> list[str]:
    """テキストを単語トークンに分割する。

    Args:
        text (str): 入力テキスト。
        ignore_case (bool): 大文字・小文字を区別しない場合は True。
        min_length (int): 単語の最小文字数。これ未満のトークンは除外する。

    Returns:
        list[str]: 抽出された単語のリスト（重複含む）。
    """
    import re

    # 英数字、ハイフン、アポストロフィを含む単語を抽出
    pattern = r"[a-zA-Z0-9'-]+"
    words = re.findall(pattern, text)

    if ignore_case:
        words = [w.lower() for w in words]

    # 最小長フィルターと空文字除去
    words = [w for w in words if len(w) >= min_length]

    return words
