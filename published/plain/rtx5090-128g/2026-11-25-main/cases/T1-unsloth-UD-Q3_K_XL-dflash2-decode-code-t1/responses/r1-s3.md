#!/usr/bin/env python3
"""Text frequency counter command-line tool.

Reads text from a file or stdin, counts word occurrences, and displays
the top N words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from pathlib import Path
from typing import List, Tuple


@dataclass
class WordFrequency:
    """Represents a single word and its occurrence count."""

    word: str
    count: int


def extract_words(text: str) -> List[str]:
    """Extract words from text.

    Converts text to lowercase and splits on non-alphabetic characters.
    Only sequences of alphabetic characters are considered words.

    Args:
        text: The input text to process.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Extract sequences of alphabetic characters, convert to lowercase
    return [w.lower() for w in re.findall(r'[a-zA-Z]+', text)]


def count_words(text: str) -> Counter:
    """Count the occurrence of each word in the given text.

    Args:
        text: The input text to count words from.

    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    words = extract_words(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top entries to return.

    Returns:
        A list of tuples, each containing a word and its count,
        sorted by count descending, then by word ascending.
    """
    # Sort by count descending, then by word ascending for consistent ordering
    sorted_items = sorted(
        counter.items(),
        key=lambda item: (-item[1], item[0])
    )
    return sorted_items[:n]
