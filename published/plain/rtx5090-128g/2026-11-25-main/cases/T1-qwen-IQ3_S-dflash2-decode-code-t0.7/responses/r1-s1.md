#!/usr/bin/env python3
"""
word_counter.py: テキストファイルから単語の出現頻度をカウントし、上位N件を表示するツール。
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple

def count_words(text: str) -> Counter:
    """
    テキストから単語を抽出し、出現頻度をカウントする。
    
    Args:
        text (str): 入力テキスト
        
    Returns:
        Counter: 単語とその出現回数の辞書
    """
    # 単語の分割と正規化（小文字化、句読点の除去）
    # 簡易的な実装：スペースと改行で分割し、句読点を除去
    words = []
    for word in text.split():
        # 句読点や特殊文字を除去
        clean_word = ''.join(c for c in word if c.isalnum())
        if clean_word:
            words.append(clean_word.lower())
    
    return Counter(words)

def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    カウンターから上位N件の単語を取得する。
    
    Args:
        counter (Counter): 単語の出現頻度カウンター
        n (int): 取得する件数
        
    Returns:
        List[Tuple[str, int]]: (単語, 出現回数) のリスト
    """
    return counter.most_common(n)

    """
