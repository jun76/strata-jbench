"""
word_freq_counter.py

A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    Extract words from the given text.

    Words are sequences of alphanumeric characters (including underscores),
    converted to lowercase for case-insensitive counting.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Use regex to find all sequences of word characters (\w+)
    words = re.findall(r'\w+', text.lower())
    return words


def count_word_frequencies(text: str) -> Counter:
    """
    Count the frequency of each word in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    words = extract_words(text)
    return Counter(words)


def get_top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to retrieve.

    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by count descending, then by word ascending for stable ordering
    sorted_items = sorted(
        counter.items(),
        key=lambda x: (-x[1], x[0])
    )
    return sorted_items[:n]
