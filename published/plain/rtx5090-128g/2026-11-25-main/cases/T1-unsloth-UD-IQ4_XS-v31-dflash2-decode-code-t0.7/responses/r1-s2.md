"""
word_freq.py - テキストファイルから単語の出現頻度を分析する CLI ツール

このツールは、指定されたテキストファイルを読み込み、単語の出現頻度を
集計して上位 N 件を表形式で出力します。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語のリストに分割する。

    単語を英数字の連続文字列として抽出し、小文字に変換する。
    日本語のような非 ASCII 文字列は、各文字を個別のトークンとして扱う。

    Args:
        text: 入力テキスト

    Returns:
        小文字化された単語のリスト
    """
    # 英数字の連続文字列を抽出し、小文字に変換
    # 日本語などの文字も一文字ずつ抽出
    tokens = re.findall(r"[a-zA-Z0-9]+|[\u3040-\u30FF\u4E00-\u9FFF]", text)
    return [token.lower() for token in tokens]


def count_words(text: str) -> Counter[str]:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    tokens = tokenize(text)
    return Counter(tokens)
