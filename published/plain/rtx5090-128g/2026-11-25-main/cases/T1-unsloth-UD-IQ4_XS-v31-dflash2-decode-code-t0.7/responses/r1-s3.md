"""
wordfreq: A command-line tool to count word frequencies in a text file.

Usage:
    python wordfreq.py <input_file> [-n TOP_N] [--min-len MIN_LEN] [--ignore-case/--no-ignore-case]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Optional


def tokenize(text: str, min_len: int = 1, ignore_case: bool = True) -> List[str]:
    """Tokenize text into words, filtering by minimum length and optionally lowercasing.

    Args:
        text: The input text to tokenize.
        min_len: Minimum length of a word to include.
        ignore_case: Whether to convert words to lowercase.

    Returns:
        A list of tokenized words.
    """
    # Extract words: sequences of alphanumeric characters and underscores
    words = re.findall(r'\b\w+\b', text)
    if min_len > 1:
        words = [w for w in words if len(w) >= min_len]
    if ignore_case:
        words = [w.lower() for w in words]
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a counter.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top items to return.

    Returns:
        A list of tuples (word, count) for the top N most common words.
    """
    return counter.most_common(n)
