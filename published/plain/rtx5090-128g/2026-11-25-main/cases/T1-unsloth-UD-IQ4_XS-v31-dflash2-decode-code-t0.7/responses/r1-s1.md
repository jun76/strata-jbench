import argparse

def tokenize(text: str, case_insensitive: bool = True) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、単語を小文字に変換して比較する。

    Returns:
        単語のリスト。
    """
    if case_insensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを単語の一部として扱い、それ以外を区切りとする
    words: List[str] = []
    current_word: List[str] = []
    for char in text:
        if char.isalnum() or char in ("-", "'"):
            current_word.append(char)
        else:
            if current_word:
                # 先頭/末尾のハイフンやアポストロフィを除く
                word = "".join(current_word).strip("-'")
                if word:
                    words.append(word)
                current_word = []
    if current_word:
        word = "".join(current_word).strip("-'")
        if word:
            words.append(word)
    return words


def count_frequencies(text: str, case_insensitive: bool = True) -> Counter:
    """
    テキスト中の単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、単語を小文字に変換して比較する。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    words = tokenize(text, case_insensitive=case_insensitive)
    return Counter(words)
