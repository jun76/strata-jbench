#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file.

Usage:
    python wordfreq.py <path> [--top N]
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


WORD_RE = re.compile(r"\b[a-zA-Z]+(?:'[a-zA-Z]+)?\b")


def read_text(path: str) -> str:
    """Read the entire content of a text file.

    Args:
        path: Path to the text file.

    Returns:
        File contents as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
    """
    return Path(path).read_text(encoding="utf-8")


def tokenize(text: str) -> List[str]:
    """Extract words from text, ignoring punctuation and case.

    Args:
        text: Input text.

    Returns:
        List of lowercase words.
    """
    return [m.group(0).lower() for m in WORD_RE.finditer(text)]


def count_words(words: List[str]) -> Counter:
    """Count occurrences of each word.

    Args:
        words: List of words.

    Returns:
        Counter mapping word → frequency.
    """
    return Counter(words)
