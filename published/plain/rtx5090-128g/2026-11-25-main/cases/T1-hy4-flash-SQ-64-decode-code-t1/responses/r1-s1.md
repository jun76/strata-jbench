"""Command-line tool to count word frequencies and display the top-N results."""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Sequence


TOKEN_RE = re.compile(r"[A-Za-z0-9]+(?:'[A-Za-z0-9]+)?")


def tokenize(text: str) -> list[str]:
    """Split text into normalized lowercase word tokens.

    Args:
        text: Raw text content.

    Returns:
        List of tokens (lowercased, apostrophes preserved inside words).

    Example:
        >>> tokenize("Hello, hello! Don't stop.")
        ['hello', 'hello', "don't", 'stop']
    """
    return TOKEN_RE.findall(text.lower())


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in the given text.

    Args:
        text: Raw text content.

    Returns:
        A ``Counter`` mapping tokens to counts.
    """
    return Counter(tokenize(text))


def read_text(path: Path) -> str:
    """Read a UTF-8 text file.

    Args:
        path: Path to the input file.

    Returns:
        File contents as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
    """
    return path.read_text(encoding="utf-8")
