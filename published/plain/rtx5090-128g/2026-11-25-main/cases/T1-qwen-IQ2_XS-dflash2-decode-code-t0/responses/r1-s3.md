def read_file(filepath: str) -> str:
    """Read the contents of a text file.

    Args:
        filepath: Path to the input text file.

    Returns:
        The full text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except IOError as e:
        raise IOError(f"Error reading file {filepath}: {e}")


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """Tokenize text into words.

    Args:
        text: Input text string.
        case_sensitive: If True, preserve case; otherwise, lowercase all words.
        min_length: Minimum word length to include.

    Returns:
        List of word tokens.
    """
    # Split on whitespace and strip punctuation from each token
    raw_tokens = text.split()
    words: List[str] = []
    for token in raw_tokens:
        # Strip leading/trailing punctuation
        stripped = token.strip(".,;:!?\"'()[]{}<>")
        if not stripped:
            continue
        if len(stripped) < min_length:
            continue
        if not case_sensitive:
            stripped = stripped.lower()
        words.append(stripped)
    return words
