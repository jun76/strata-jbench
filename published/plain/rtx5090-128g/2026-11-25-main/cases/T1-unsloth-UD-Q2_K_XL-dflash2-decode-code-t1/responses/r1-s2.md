import argparse

def load_text(filepath: str) -> str:
    """
    指定されたパスのテキストファイルを読み込み、内容を文字列として返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を格納した文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        UnicodeDecodeError: ファイルが UTF-8 でデコードできない場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語リストに分割する。

    Args:
        text: 分割対象のテキスト文字列。
        case_sensitive: 大文字・小文字を区別するかどうか。False の場合、単語を小文字に変換する。

    Returns:
        単語のリスト。
    """
    # 英数字の単語を取得
    words = re.findall(r"[a-zA-Z0-9]+", text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_words(words: List[str]) -> Counter[str]:
    """
    単語リストから出現頻度を集計する。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)
