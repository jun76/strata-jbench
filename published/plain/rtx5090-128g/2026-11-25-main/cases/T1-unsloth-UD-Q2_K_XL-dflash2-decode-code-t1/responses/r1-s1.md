#!/usr/bin/env python3
"""
TopN Words Command Line Tool

Reads a text file, counts word frequencies, and displays the top N words
in a formatted table.

Usage:
    python topn.py [options] <input_file>

Examples:
    python topn.py data.txt
    python topn.py -n 10 --case-sensitive data.txt
    python topn.py --max-len 5 input.txt
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False, max_len: int = 0) -> List[str]:
    """Extract words from text.

    Args:
        text: Input text string.
        case_sensitive: If True, preserve case. If False, convert to lowercase.
        max_len: Maximum word length. If 0, no limit. Words longer than this are excluded.

    Returns:
        List of extracted words.
    """
    # Use regex to extract word-like tokens (letters, digits, apostrophes within words)
    words = re.findall(r"[a-zA-Z0-9']+", text)
    
    if not case_sensitive:
        words = [word.lower() for word in words]
    
    if max_len > 0:
        words = [word for word in words if len(word) <= max_len]
    
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: List of words.

    Returns:
        Counter object with word counts.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a counter.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        List of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
