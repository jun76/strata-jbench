const lerp=(a,b,t)=>a+(b-a)*t;

function hullHalf(g,x0,x1,face){
  for(let x=x0;x<=x1;x++){
    const t=(x-x0)/(x1-x0);
    const hw=Math.max(1,Math.round(lerp(5.5,1.0,t)));
    const bow=Math.pow(t,0.6);
    const top=Math.round(lerp(7,3.0,bow));
    const bot=Math.round(lerp(0,1.4,bow));
    for(let y=bot;y<=top;y++){
      let col=C.hullMid;
      if(y<=1)col=C.hullDark; else if(y>=top-1)col=pick(C.hullLight,C.hullMid,0.4);
      else col=pick(C.hullMid,C.hullLight,0.25);
      V(g,x,y,face*hw,col);
    }
    V(g,x,top+1,face*hw,C.trim); // gunwale trim line
  }
}
function deckPlane(g,x0,x1,face,ymax){
  for(let x=x0;x<=x1;x++){

}

A few things you can try once it's open:

- **Drag** to orbit, **scroll** to zoom, **right‑drag** to pan. It auto‑orbits until you touch it.
- **"Send her into the harbour"** walks the barque from its anchorage, past the quay, into the berth under the lighthouse beam.
- **Harbour view / Bridge / Deck / Aerial** ease the camera to preset angles.
- Toggle **Sea state** (waves + sun‑glitter), **Harbour lights** (lamps go dark), and **Seabirds**; the **Wind** slider speeds up the swell and sways the rigging.

The whole thing is one file, uses Three.js via CDN, and renders immediately on load — no build step, no assets to fetch besides the font.
