  const clamp = (v,a,b)=>Math.max(a,Math.min(b,v));

    return '#'+[c(r),c(g),c(b)].map(v=>v.toString(16).padStart(2,'0')).join('');}
  function pick(arr,rng){return arr[Math.floor(rng()*arr.length)];}
  function mulberry32(a){return function(){a|=0;a=a+0x6D2B79F5|0;let t=Math.imul(a^a>>>15,1|a);
    t=t+Math.imul(t^t>>>7,61|t)^t;return((t^t>>>14)>>>0)/4294967296;};}

  const boxGeo = new THREE.BoxGeometry(1,1,1);
  boxGeo.translate(0,0.5,0); // block rests on its base cell
  const cache = new Map();
  function mat(hex,opts={}){
    const key = hex+JSON.stringify(opts);
    if(cache.has(key)) return cache.get(key);
    const m = opts.emissive
      ? new THREE.MeshStandardMaterial({color:hex,emissive:opts.emissive,emissiveIntensity:opts.emissiveIntensity??1,roughness:opts.roughness??0.5,metalness:opts.metalness??0})
      : new THREE.MeshStandardMaterial({color:hex,roughness:opts.roughness??0.85,metalness:opts.metalness??0});
    cache.set(key,m); return m;
  }
  class VoxelBuilder{
    constructor(){this.map=new Map();this.total=0;}
    add(x,y,z,hex,opts={}){const k=hex+'|'+JSON.stringify(opts);
      let a=this.map.get(k); if(!a){a=[];this.map.set(k,a);} a.push([x,y,z]); this.total++;}
    build(parent,cast=true,recv=true){
      for(const [k,arr] of this.map){
        const [hex,optsS]=k.split('|'); const opts=JSON.parse(optsS);
        const m=mat(hex,opts);
        const inst=new THREE.InstancedMesh(boxGeo,m,arr.length);
        const mtx=new THREE.Matrix4();
        for(let i=0;i<arr.length;i++){const[x,y,z]=arr[i];mtx.makeTranslation(x,y,z);inst.setMatrixAt(i,mtx);}
        inst.instanceMatrix.needsUpdate=true; inst.castShadow=cast; inst.receiveShadow=recv; parent.add(inst);
      }
    }
  }

  const rng = mulberry32(1234);

  function house(x,z,w,d){
    const wall=pick(['#c9b79a','#b7a07f','#d8cbb0','#c2a988'],rng);
    const roof=pick(ROOF,rng);
    const h=3+Math.floor(rng()*3);
    for(let dz=0;dz<d;dz++) for(let dx=0;dx<w;dx++) for(let y=1;y<=h;y++) harbor.add(x+dx,y,z+dz,wall);
    const sh=1+Math.floor(rng()*2);
    for(let dz=-1;dz<d+1;dz++) for(let dx=-1;dx<w+1;dx++) harbor.add(x+dx,h+1,z+dz,roof);
    for(let dz=0;dz<d;dz++) for(let dx=0;dx<w;dx++) harbor.add(x+dx,h+2,z+dz,shade(roof,0.8));
    const wcol=pick(['#3a5a7a','#2c3a4a','#4a6a8a'],rng);
    for(let i=0;i<2;i++) harbor.add(x+1+i*(w-2),2,z+(d-1),wcol);
    harbor.add(x,1,z+(d-1),'#5a3a2a');
    harbor.add(x+1,h+2,z+(d>>1),'#4a4a4a');
  }

  // pier planks
  for(let z=-6;z<=12;z++) for(let x=-4;x<=4;x++) harbor.add(x,0,z,pick(PLANKS,rng));
  // houses
  house(-4,-6,3,3); house(3,-6,4,3); house(-4,7,3,4);
  // crates + barrels
  for(let i=0;i<5;i++){ const cx=clamp(Math.round(-3+Math.floor(rng()*7)),-3,3), cz=clamp(Math.round(-4+Math.floor(rng()*15)),-5,11);

    harbor.add(cx,1,cz,pick(CRATES,rng));
    if(rng()>0.5) harbor.add(cx,2,cz,pick(CRATES,rng));

  for(let i=0;i<3;i++){ const cx=clamp(Math.round(-3+Math.floor(rng()*7)),-3,3), cz=clamp(Math.round(-4+Math.floor(rng()*15)),-5,11);

  // bollards + rope
  for(const [x,z] of [[-3,-1],[3,-1],[-3,9],[3,9]]) harbor.add(x,1,z,'#3a3a3a');

  const glow = new VoxelBuilder();

  glow.add(0,12,1,LAMP,{emissive:LAMP,emissiveIntensity:1.4});
  glow.add(1,12,1,LAMP,{emissive:LAMP,emissiveIntensity:1.4});

  const seaGeo=new THREE.PlaneGeometry(180,180,120,120); seaGeo.rotateX(-Math.PI/2);
  const seaMat=new THREE.MeshStandardMaterial({color:0x123a52,roughness:0.35,metalness:0.55,emissive:0x06202e,emissiveIntensity:0.35,side:THREE.DoubleSide});
  const sea=new THREE.Mesh(seaGeo,seaMat); sea.position.y=-0.15; sea.receiveShadow=true; scene.add(sea);
  const seaBase=seaGeo.attributes.position.array.slice();
  function wave(x,z,t){ return Math.sin(x*0.16+t*1.1)*0.28 + Math.cos(z*0.13+t*0.9)*0.22 + Math.sin((x+z)*0.07+t*0.7)*0.18; }
  function updateSea(t){ const p=seaGeo.attributes.position.array;
    for(let i=0;i<p.length;i+=3){ p[i+1]=wave(p[i],p[i+2],t); } seaGeo.attributes.position.needsUpdate=true; }
  function updateSeaColor(tod){
    const c=tod==='night'?0x0a1c2b:tod==='dusk'?0x274a63:tod==='dawn'?0x2c4f68:0x123a52;
    const e=tod==='night'?0x03101a:0x06202e;
    seaMat.color.setHex(c); seaMat.emissive.setHex(e);
  }

  /* ---------- sky (gradient dome) ---------- */
  const skyGeo=new THREE.SphereGeometry(500,32,16);
  const skyMat=new THREE.ShaderMaterial({side:THREE.BackSide,depthWrite:false,
    uniforms:{uTop:{value:new THREE.Color()},uBot:{value:new THREE.Color()},uSun:{value:new THREE.Color()}},
    vertexShader:`varying vec3 vP; void main(){vP=position; gl_Position=projectionMatrix*modelViewMatrix*vec4(position,1.0);}`,
    fragmentShader:`varying vec3 vP; uniform vec3 uTop; uniform vec3 uBot; uniform vec3 uSun;
      void main(){ float h=clamp(normalize(vP).y*0.5+0.5,0.0,1.0); h=pow(h,0.85);
        vec3 col=mix(uBot,uTop,h); col=mix(col,uSun,pow(1.0-abs(normalize(vP).y),3.0)*0.45);
        gl_FragColor=vec4(col,1.0);}`});
  const sky=new THREE.Mesh(skyGeo,skyMat); sky.renderOrder=-1; scene.add(sky);

  /* ---------- clouds ---------- */
  const cloudGroup=new THREE.Group();
  function makeCloud(w){ const g=new THREE.Group(); const cm=new THREE.MeshStandardMaterial({color:0xffffff,roughness:1,metalness:0,emissive:0x000000,emissiveIntensity:0});
    const n=5+Math.floor(Math.random()*4);
    for(let i=0;i<n;i++){ const s=1+Math.random()*2.2;
      const m=new THREE.Mesh(new THREE.BoxGeometry(s,s*0.7,s),cm);
      m.position.set((Math.random()-0.5)*w,(Math.random()-0.5)*1.2,(Math.random()-0.5)*w*0.5);
      m.scale.setScalar(1); g.add(m); }
    return g; }
  for(let i=0;i<6;i++){ const c=makeCloud(4+Math.random()*4);
    c.position.set(-90+Math.random()*180,42+Math.random()*22,-90+Math.random()*180);
    cloudGroup.add(c); }
  scene.add(cloudGroup);

  /* ---------- birds ---------- */
  const birds=[];
  const birdMat=new THREE.MeshStandardMaterial({color:0x101010,roughness:0.6});
  for(let i=0;i<3;i++){ const g=new THREE.Group();
    for(const s of [-1,1]){ const w=new THREE.Mesh(new THREE.BoxGeometry(0.9,0.12,0.35),birdMat); w.position.set(s*0.55,0,0); w.name='w'; g.add(w); }
    const b=new THREE.Mesh(new THREE.BoxGeometry(0.4,0.25,0.7),birdMat); g.add(b);
    g.position.set(-18+i*16,17+i*2,-6+i*3); birds.push(g); scene.add(g); }

  /* ---------- time of day ---------- */
  const TOD = {
    noon: {top:'#1f6fb0',bot:'#cfe6f5',sun:'#ffffff',sunI:2.2,hemiI:0.7,
      sunPos:[18,52,22],fog:'#cfe6f5',fogD:0.006,exp:1.0,night:false},
    dawn: {top:'#5a7fb0',bot:'#f6c9a0',sun:'#ffd9b0',sunI:1.6,hemiI:0.55,
      sunPos:[-46,16,26],fog:'#e9c9a8',fogD:0.009,exp:1.02,night:false},
    dusk: {top:'#3a5a8a',bot:'#f0955e',sun:'#ff9d5a',sunI:1.4,hemiI:0.5,
      sunPos:[46,12,-24],fog:'#e0a06f',fogD:0.010,exp:1.05,night:true},
    night:{top:'#0a1424',bot:'#22344a',sun:'#8fb0d8',sunI:0.35,hemiI:0.25,
      sunPos:[-28,40,-34],fog:'#1a2740',fogD:0.012,exp:0.95,night:true},
  };
  const COL={top:new THREE.Color(),bot:new THREE.Color(),sun:new THREE.Color(),fog:new THREE.Color()};
  let A={...TOD.noon}, B={...TOD.noon}, to=0, toStart=0, toDur=1.2;
  const clockEl=document.getElementById('clock'), toastEl=document.getElementById('toast');
  function fmt(t){const hh=Math.floor(t),mm=Math.floor((t-hh)*60);return String(hh).padStart(2,'0')+':'+String(mm).padStart(2,'0');}
  function setClock(){ const hh=(A.sunPos[2]>0?12:22)+(A.sunPos[1]>30?0:2); clockEl.textContent=fmt(clamp(hh,0,23)); }
  function applyLerp(t){
    COL.top.set(A.top).lerp(B.top,t); COL.bot.set(A.bot).lerp(B.bot,t);
    COL.sun.set(A.sun).lerp(B.sun,t); COL.fog.set(A.fog).lerp(B.fog,t);
    skyMat.uniforms.uTop.value.copy(COL.top); skyMat.uniforms.uBot.value.copy(COL.bot); skyMat.uniforms.uSun.value.copy(COL.sun);
    scene.fog.color.copy(COL.fog); scene.fog.density=lerp(A.fogD,B.fogD,t);
    renderer.toneMappingExposure=lerp(A.exp,B.exp,t);
    sun.color.copy(COL.sun); sun.intensity=lerp(A.sunI,B.sunI,t); hemi.intensity=lerp(A.hemiI,B.hemiI,t);
    sun.position.set(lerp(A.sunPos[0],B.sunPos[0],t),lerp(A.sunPos[1],B.sunPos[1],t),lerp(A.sunPos[2],B.sunPos[2],t));
    sun.target.position.set(0,0,0);
    const nightNow=lerp(A.night?1:0,B.night?1:0,t);
    lampLight.intensity=nightNow*1.4; flagLight.intensity=nightNow*0.35;
    for(const [x,z] of LANTERNS){} // lantern lights handled below
    lightMeshes.forEach(l=>{ if(l!==lampLight&&l!==flagLight) l.intensity=nightNow*0.5; });
    beam.rotation.y+=0.0; // handled in loop
    cloudGroup.children.forEach(g=>g.children.forEach(m=>{m.material.color.setHex(nightNow>0.5?0x4a5568:0xffffff);m.material.emissiveIntensity=nightNow*0.05;}));
    birdMat.color.setHex(nightNow>0.5?0x0a0a0a:0x101010);
    seaMat.emissiveIntensity=nightNow>0.5?0.15:0.35;
    setClock();
  }
  function setTOD(name){ A={...A}; B={...TOD[name]}; to=0; toStart=performance.now()/1000;
    document.querySelectorAll('.chip').forEach(c=>c.classList.toggle('active',c.dataset.tod===name));
    const n=TOD[name].night; if(n){toastEl.classList.remove('show');}
    else{toastEl.classList.add('show'); clearTimeout(setTOD._t); setTOD._t=setTimeout(()=>toastEl.classList.remove('show'),2600);} }
  document.querySelectorAll('.chip').forEach(c=>c.addEventListener('click',()=>setTOD(c.dataset.tod)));
  scene.fog=new THREE.FogExp2(new THREE.Color(TOD.noon.fog),TOD.noon.fogD);
  applyLerp(1);

  /* ---------- animate ---------- */
  const clock=new THREE.Clock();
  let introT=0;
  function animate(){
    requestAnimationFrame(animate);
    const dt=clock.getDelta(), t=clock.elapsedTime;

    if(introT<1){ introT=Math.min(1,introT+dt/2.4);
      const e=1-Math.pow(1-introT,3);
      const s=lerp(96,46,e);
      camera.position.set(Math.sin(0.5)*s,lerp(40,20,e),Math.cos(0.5)*s);
      camera.lookAt(0,3,0);
    }

    if(to<1){ const now=performance.now()/1000; to=Math.min(1,(now-toStart)/toDur); applyLerp(to); }

    updateSea(t);
    const sh=SHIP_Y+wave(SHIP_X,SHIP_Z,t);
    scene.position.y=sh;
    scene.rotation.z=Math.sin(t*0.9)*0.012;
    scene.rotation.x=Math.cos(t*0.75)*0.010;
    beam.rotation.y=t*0.6;

    const fp=flagMesh.geometry.attributes.position.array;
    for(let i=0;i<fp.length;i+=3){ const bx=flagBase[i],by=flagBase[i+1]; fp[i+2]=Math.sin(bx*3+t*7+by*2)*0.12; }
    flagMesh.geometry.attributes.position.needsUpdate=true;

    cloudGroup.children.forEach((c,i)=>{ c.position.x+=dt*(0.5+i*0.15); if(c.position.x>100)c.position.x=-100; });

    birds.forEach((g,i)=>{ const a=t*0.3+i*2.1;
      g.position.set(Math.cos(a)*40,18+Math.sin(t*0.6+i)*3,Math.sin(a)*40);
      g.rotation.y=-a; g.rotation.x=Math.PI/2;
      g.children.forEach(w=>{ if(w.name==='w') w.rotation.x=Math.sin(t*10+i)*0.5; });
    });

    controls.update();
    renderer.render(scene,camera);
  }
  animate();

  /* ---------- resize ---------- */
  addEventListener('resize',()=>{ camera.aspect=innerWidth/innerHeight; camera.updateProjectionMatrix(); renderer.setSize(innerWidth,innerHeight); });

  /* ---------- reveal ---------- */
  setTimeout(()=>{ document.getElementById('loader').classList.add('hide'); document.body.classList.add('ready'); },450);
  </script>
</body>
