/* ---------------- voxel builder ---------------- */
class VoxelBuilder{
  constructor(){
    this.map=new Map();
    this.set=()=>{};
    this._c=(()=>{const t=new THREE.Color();return(x,y,z,col)=>{t.set(col);this.map.set(x+','+y+','+z,t.getHex());};})();
    this.set=this._c;
  }
  get(x,y,z){return this.map.get(x+','+y+','+z);}
  has(x,y,z){return this.map.has(x+','+y+','+z);}
  del(x,y,z){this.map.delete(x+','+y+','+z);}
  box(x0,y0,z0,x1,y1,z1,c){for(let x=x0;x<=x1;x++)for(let y=y0;y<=y1;y++)for(let z=z0;z<=z1;z++)this.set(x,y,z,c);}
  shell(x0,y0,z0,x1,y1,z1,c){for(let x=x0;x<=x1;x++)for(let y=y0;y<=y1;y++)for(let z=z0;z<=z1;z++){const b=(x===x0||x===x1||y===y0||y===y1||z===z0||z===z1);if(b)this.set(x,y,z,c);}}
  cyl(cx,cz,rad,y0,y1,c){const r2=rad*rad;for(let y=y0;y<=y1;y++)for(let x=Math.floor(cx-rad);x<=Math.ceil(cx+rad);x++)for(let z=Math.floor(cz-rad);z<=Math.ceil(cz+rad);z++){const dx=x-cx,dz=z-cz;if(dx*dx+dz*dz<=r2)this.set(x,y,z,c);}}
  disc(cx,cz,rad,y,c){const r2=rad*rad;for(let x=Math.floor(cx-rad);x<=Math.ceil(cx+rad);x++)for(let z=Math.floor(cz-rad);z<=Math.ceil(cz+rad);z++){const dx=x-cx,dz=z-cz;if(dx*dx+dz*dz<=r2)this.set(x,y,z,c);}}
  blob(cx,cy,cz,rx,ry,rz,c){for(let x=Math.ceil(cx-rx);x<=Math.floor(cx+rx);x++)for(let y=Math.ceil(cy-ry);y<=Math.floor(cy+ry);y++)for(let z=Math.ceil(cz-rz);z<=Math.floor(cz+rz);z++){const dx=(x-cx)/rx,dy=(y-cy)/ry,dz=(z-cz)/rz;if(dx*dx+dy*dy+dz*dz<=1)this.set(x,y,z,c);}}
  line(x0,y0,z0,x1,y1,z1,c){let x=x0,y=y0,z=z0;const dx=x1-x0,dy=y1-y0,dz=z1-z0;const n=Math.max(Math.abs(dx),Math.abs(dy),Math.abs(dz),1);for(let i=0;i<=n;i++){this.set(Math.round(x0+dx*i/n),Math.round(y0+dy*i/n),Math.round(z0+dz*i/n),c);}}
  clear(){this.map.clear();}
  trim(){
    for(const [k,col] of [...this.map]){
      const [x,y,z]=k.split(',').map(Number);
      if(!this.has(x,y+1,z)&&!this.has(x,y-1,z)&&!this.has(x+1,y,z)&&!this.has(x-1,y,z)&&!this.has(x,y,z+1)&&!this.has(x,y,z-1))this.map.delete(k);
    }
  }
  toMesh(mats){
    const g=new THREE.BoxGeometry(1,1,1);
    const out=new THREE.Group();
    for(const col of mats){
      const geo=new THREE.BoxGeometry(1,1,1);
      const pos=geo.attributes.position;
      let ci=0;
      for(const [k] of this.map){const [x,y,z]=k.split(',').map(Number);pos.setXYZ(ci++,x,y,z);}
      geo.setDrawRange(0,ci);
      geo.computeVertexNormals();
      const m=new THREE.InstancedMesh(geo,mats[0],ci);
      m.material=col; m.castShadow=true; m.receiveShadow=true;
      const d=new THREE.Object3D();
      for(let i=0;i<ci;i++){d.position.set(pos.getX(i),pos.getY(i),pos.getZ(i));d.updateMatrix();m.setMatrixAt(i,d.matrix);}
      m.instanceMatrix.needsUpdate=true;
      out.add(m);
    }
    return out;
  }
}

function updateWater(t){
  const p=waterGeo.attributes.position,n=waterGeo.attributes.normal;
  for(let i=0;i<p.count;i++){
    const x=waterBase[i*3],z=waterBase[i*3+2];

}
