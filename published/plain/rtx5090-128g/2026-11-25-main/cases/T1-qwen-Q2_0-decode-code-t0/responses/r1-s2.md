#!/usr/bin/env python3
"""
word_freq.py - テキストファイルの単語出現頻度上位 N 件を表形式で表示する CLI 工具。

Usage:
    python word_freq.py <file> [--top N] [--case-sensitive] [--min-len L]

Examples:
    python word_freq.py sample.txt --top 10
    python word_freq.py sample.txt --top 5 --case-sensitive --min-len 3
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大小写を区別するか。False なら小写に統一。
        min_len: 単語の最小長。この値未満の単語は除外。

    Returns:
        抽出された単語のリスト。
    """
    # 単語を抽出: 字母、数字、および内部の hyphen/apostrophe を含む
    # 例: "don't", "well-known", "123"
    pattern = r"[A-Za-z0-9]+(?:[-'][A-Za-z0-9]+)*"
    words = re.findall(pattern, text)

    if not case_sensitive:
        words = [w.lower() for w in words]

    if min_len > 1:
        words = [w for w in words if len(w) >= min_len]

    return words
