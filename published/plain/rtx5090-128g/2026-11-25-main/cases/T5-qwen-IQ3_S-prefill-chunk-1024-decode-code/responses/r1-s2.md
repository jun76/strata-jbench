#!/usr/bin/env python3
"""Text file word frequency counter command-line tool.

This tool reads a text file, counts the occurrences of each word,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple, Dict


def read_file(filepath: str) -> str:
    """Read and return the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """Extract words from the given text.

    Words are defined as sequences of alphanumeric characters, lowercased.

    Args:
        text: The input text.

    Returns:
        A list of lowercase words extracted from the text.
    """
    return re.findall(r'[a-z0-9]+', text.lower())


def count_words(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        A list of tuples (word, count) for the top N most common words.
    """
    return counter.most_common(n)


def format_table(top_words: List[Tuple[str, int]], title: str = "Top Words") -> str:
    """Format the top words into a table string.

    Args:
        top_words: A list of tuples (word, count).
        title: The title of the table.

    Returns:
        A formatted table string.
    """
    if not top_words:
        return f"{title}\n(No words found)"
