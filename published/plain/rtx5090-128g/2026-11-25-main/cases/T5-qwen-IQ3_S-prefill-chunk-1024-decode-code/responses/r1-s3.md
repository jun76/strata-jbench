"""
wordfreq - Count word frequencies in a text file and display top N results.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words (alphanumeric sequences)."""
    return re.findall(r'[a-z0-9]+', text.lower())


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text."""
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words as (word, count) tuples."""
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]]) -> str:
    """Format word frequency entries as a table string."""
    if not entries:
        return "No words found."

    # Determine column widths
    max_word_len = max(len(word) for word, _ in entries)
    max_count_len = max(len(str(count)) for _, count in entries)

    header_word = "Word".ljust(max_word_len)
    header_count = "Count".ljust(max_count_len)
    separator = "-" * (max_word_len + max_count_len + 1)

    lines = [f"{header_word} | {header_count}", separator]
    for word, count in entries:
        lines.append(f"{word.ljust(max_word_len)} | {str(count).rjust(max_count_len)}")
    return "\n".join(lines)


def read_file(filepath: str) -> str:
    """Read and return the content of a text file."""
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()
