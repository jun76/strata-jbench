#!/usr/bin/env python3
"""
word_freq: A command-line tool to count word frequencies in text files
and display the top N most frequent words in a table format.
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    Extract words from text, treating sequences of alphanumeric characters as words.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    # Match sequences of word characters (letters, digits, underscores)
    words = re.findall(r'\b\w+\b', text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word in the list.
    
    Args:
        words: A list of words.
        
    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their frequencies.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.
        
    Returns:
        A list of tuples, each containing a word and its count, sorted by
        descending frequency. Ties are broken by alphabetical order.
    """
    # Counter.most_common() already sorts by frequency descending,
    # but for ties, we want alphabetical order.
    # We'll sort manually to ensure consistent tie-breaking.
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
