#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を集計し、
上位 N 件を表形式で表示する。

Usage:
    python word_freq.py <file> [-n TOP_N] [--case-sensitive] [--min-len MIN_LEN]

Examples:
    python word_freq.py sample.txt
    python word_freq.py sample.txt -n 20 --case-sensitive
    python word_freq.py sample.txt --min-len 3
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    テキストを単語トークンに分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。
        min_len: 単語の最小長。これより短いトークンは除外する。

    Returns:
        単語トークンのリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 単語は英数字、アンダースコア、ハイフン、アポストロフィを許容する
    tokens = re.findall(r"[a-zA-Z0-9_'\-]+", text)
    # min_len 以上の長さを持つトークンのみを返す
    return [t for t in tokens if len(t) >= min_len]


def count_words(text: str, case_sensitive: bool = False, min_len: int = 1) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。
        min_len: 単語の最小長。これより短いトークンは除外する。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    tokens = tokenize(text, case_sensitive=case_sensitive, min_len=min_len)
    return Counter(tokens)
