#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度を上位N件まで集計して
表形式で表示するコマンドラインツールです。

Usage:
    python word_counter.py <input_file> [-n N] [--case-sensitive]
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple, Dict
import re


def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """
    単語を正規化する。
    大文字小文字を無視する場合は小文字に変換し、
    非英数字・ハイフン・アポストロフィを除く文字を除去する。
    
    Args:
        word (str): 入力単語
        case_sensitive (bool): 大文字小文字を区別するかどうか
    
    Returns:
        str: 正規化された単語
    """
    if not case_sensitive:
        word = word.lower()
    # 単語の境界として、アルファベット、数字、ハイフン、アポストロフィ以外は除去
    # 例: "hello," -> "hello", "don't" -> "don't", "e-mail" -> "e-mail"
    cleaned = re.sub(r"[^\w\-']", '', word)
    return cleaned


def count_words(text: str, case_sensitive: bool = False) -> Dict[str, int]:
    """
    テキストから単語出現頻度をカウントする。
    
    Args:
        text (str): 入力テキスト
        case_sensitive (bool): 大文字小文字を区別するかどうか
    
    Returns:
        Dict[str, int]: 単語をキー、出現回数を値とする辞書
    """
    words = text.split()
    normalized_words = [normalize_word(w, case_sensitive) for w in words]
    # 空文字列を除外
    normalized_words = [w for w in normalized_words if w]
    return dict(Counter(normalized_words))
