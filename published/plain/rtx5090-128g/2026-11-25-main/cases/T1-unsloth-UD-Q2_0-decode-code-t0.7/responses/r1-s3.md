"""
コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を集計し、
上位 N 件を表形式で表示します。
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語リストを取り出します。

    大文字は小文字に変換し、英数字の連続を単語とみなします。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    # 英数字の連続を抽出（それ以外は区切りとみなす）
    return re.findall(r"[a-z0-9]+", text.lower())


def count_words(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度を数えます。

    Args:
        text: 入力テキスト。

    Returns:
        単語 -> 出現回数の Counter。
    """
    return Counter(tokenize(text))


def top_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    出現頻度上位 N 件を返します。同頻度の場合は辞書順でソートします。

    Args:
        counter: 単語 -> 出現回数の Counter。
        n: 取得する件数。

    Returns:
        (単語, 出現回数) のリスト。上位から順。
    """
    # (-count, word) でソート: 頻度降順、同頻度なら単語昇順
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
