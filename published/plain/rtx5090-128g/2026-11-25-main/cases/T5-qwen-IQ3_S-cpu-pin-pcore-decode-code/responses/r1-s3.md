#!/usr/bin/env python3
"""
text_word_counter: Count word frequencies in a text file and display the top N results.

Usage:
    python text_word_counter.py <input_file> [-n TOP_N]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize the input text into words.
    
    Words are defined as sequences of alphanumeric characters (including underscores).
    All words are converted to lowercase for case-insensitive counting.
    
    Args:
        text: The input text to tokenize.
        
    Returns:
        A list of lowercase words.
    """
    return re.findall(r'\w+', text.lower())


def count_words(words: List[str]) -> Counter[str]:
    """
    Count the frequency of each word in the given list.
    
    Args:
        words: A list of words to count.
        
    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top items to return.
        
    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
