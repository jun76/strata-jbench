"""
word_counter: テキストファイルから単語の出現頻度をカウントし、
上位 N 件を表形式で出力するコマンドラインツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを小文字の単語リストに分割する。

    Args:
        text: 入力テキスト文字列。

    Returns:
        正規化された単語のリスト（小文字、アルファベット+数字+ハイフン+アポストロフィ）。
    """
    # 単語は [a-z0-9'-]+ に一致する。大文字は小文字に変換。
    return re.findall(r"[a-z0-9'-]+", text.lower())


def count_words(text: str) -> Counter[str]:
    """
    テキスト中の単語出現回数を Counter として返す。

    Args:
        text: 入力テキスト文字列。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    words = tokenize(text)
    return Counter(words)
