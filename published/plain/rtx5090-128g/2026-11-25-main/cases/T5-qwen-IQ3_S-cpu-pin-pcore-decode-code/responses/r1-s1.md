def extract_words(text: str, ignore_case: bool = True) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text (str): 入力テキスト。
        ignore_case (bool): 大文字小文字を区別しない場合は True。

    Returns:
        List[str]: 抽出された単語のリスト。
    """
    # 正規表現で単語(英数字とハイフンを含む)を抽出

    # 必要に応じてパターンを調整してください。
    pattern = r"\w+"
    words = re.findall(pattern, text)
    
    if ignore_case:
        words = [word.lower() for word in words]
    
    return words
