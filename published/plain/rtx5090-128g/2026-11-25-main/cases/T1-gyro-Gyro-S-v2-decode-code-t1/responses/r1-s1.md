def tokenize(text: str, min_len: int = 1) -> List[str]:
    """テキストを単語に分割し、最小長未満のトークンを除去する。

    Args:
        text: 入力テキスト
        min_len: 単語の最小文字数（この未満のトークンは除外）

    Returns:
        分割された単語のリスト
    """
    words = text.split()
    return [word for word in words if len(word) >= min_len]


def count_words(words: List[str]) -> Counter:
    """単語出現回数を集計する。

    Args:
        words: 単語のリスト

    Returns:
        Counter: {単語: 出現回数}
    """
    return Counter(words)

        (word, count) のタプルリスト
    """
