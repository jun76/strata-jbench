"""
Top-N word frequency counter command-line tool.

Reads a text file, counts word frequencies, and displays the top N words
in a table format with word, count, and percentage columns.
"""
import argparse

from collections import Counter
from pathlib import Path

def normalize_word(word: str) -> str:
    """Normalize a word by removing non-alphabetic characters and lowercasing."""
    return word.strip().lower().strip()


def read_words_from_file(filepath: str) -> List[str]:
    """
    Read all words from a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        A list of normalized words (lowercase, stripped of punctuation).

    Raises:
        FileNotFoundError: If the file does not exist.
        ValueError: If the file cannot be read.
    """
    try:
        content = Path(filepath).read_text(encoding="utf-8", errors="replace")
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except Exception as e:
        raise ValueError(f"Error reading file '{filepath}': {e}")

    # Split on whitespace and normalize each token
    words = [normalize_word(w) for w in content.split() if w]
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words to count.

    Returns:
        A Counter mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(
    counter: Counter, n: int
) -> List[Tuple[str, int]]:
    """
    Return the top N most common words and their counts.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then alphabetically ascending for ties.
    """
    # Sort by count descending, then alphabetically for tie-breaking
    items = sorted(counter.items(), key=lambda item: (-item[1], item[0]))
    return items[:n]
