def normalize_text(text: str) -> str:
    """テキストを小文字に変換し、英字と数字とハイフン・アポストロフィ以外の文字を除去する。"""
    text = text.lower()
    # 単語として残すのは英字・数字・ハイフン・アポストロフィ
    return re.sub(r"[^a-z0-9'\\-]+", " ", text)


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト（空文字は除く）
    """
    normalized = normalize_text(text)
    words = [w for w in normalized.split() if w]
    return words


def count_words(words: List[str]) -> Counter[str]:
    """
    単語出現頻度を Counter で集計する。

    Args:
        words: 単語リスト

    Returns:
        Counter[str]
    """
    counter = Counter()
    for word in words:
        counter[word] += 1
    return counter


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 n 件を返す。同頻度の場合は辞書順（アルファベット順）で安定する。

    Args:
        counter: 単語と頻度の Counter
        n: 上位何件までか

    Returns:
        (word, count) のリスト（頻度降順、同頻度はアルファベット順）
    """
    return sorted(counter.items(), key=lambda item: (-item[1], item[0]))[:n]
