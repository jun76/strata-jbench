    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか

    Returns:
        単語のリスト（アルファベット/数字/ハイフン/アポストロフィを含む連続文字列）
    """
    # 単語パターン: アルファベット、数字、ハイフン、アポストロフィを含む連続文字列
    pattern = r"[a-zA-Z0-9'\-]+"
    words = re.findall(pattern, text)
    if not case_sensitive:
        words = [word.lower() for word in words]
    return words
