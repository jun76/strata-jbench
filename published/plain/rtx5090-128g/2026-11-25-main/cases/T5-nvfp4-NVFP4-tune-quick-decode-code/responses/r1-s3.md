"""Word frequency counter CLI tool.

This module provides a command-line tool that reads a text file,
counts word occurrences, and displays the top N words in a table format.
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from pathlib import Path
from typing import List, Tuple


@dataclass(frozen=True)
class WordCount:
    """Represents a word and its occurrence count."""

    word: str
    count: int


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase words (alphanumeric sequences only).
    """
    # Extract words: sequences of alphanumeric characters (including underscores)
    return re.findall(r'\b\w+\b', text.lower())


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in the given text.

    Args:
        text: The input text.

    Returns:
        A Counter object mapping words to their counts.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[WordCount]:
    """Get the top N most common words from a counter.

    Args:
        counter: A Counter object with word counts.
        n: The number of top words to return.

    Returns:
        A list of WordCount objects sorted by count descending,
        then by word ascending for ties.
    """
    if n <= 0:
        return []
    # most_common sorts by count descending; for ties, we sort alphabetically
    items: List[Tuple[str, int]] = counter.most_common()
    # Sort by count descending, then word ascending
    items.sort(key=lambda x: (-x[1], x[0]))
    return [WordCount(word=w, count=c) for w, c in items[:n]]
