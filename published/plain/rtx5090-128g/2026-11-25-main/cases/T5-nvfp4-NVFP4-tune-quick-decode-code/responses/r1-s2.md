#!/usr/bin/env python3
"""
wordfreq: テキストファイルから単語頻度を集計し、上位N件を表形式で表示するCLIツール。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト文字列。

    Returns:
        小文字化された単語トークンのリスト。
    """
    # 英数字、ハイフン、アポストロフィを単語の一部として扱う
    return re.findall(r"[a-zA-Z0-9'-]+", text.lower())


def count_words(text: str) -> Counter[str]:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト文字列。

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counterから上位n件の（単語, 回数）タプルを返す。

    Args:
        counter: 単語頻度のCounter。
        n: 取得する上位件数。

    Returns:
        (単語, 回数) のタプルのリスト（回数の降順、同回数時は辞書順）。
    """
    # most_commonは回数の降順。同回数については、Counter内部の挿入順に依存するため、
    # 安定した辞書順を得るためにソートキーを調整する。
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
