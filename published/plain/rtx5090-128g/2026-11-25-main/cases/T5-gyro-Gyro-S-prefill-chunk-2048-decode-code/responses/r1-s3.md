#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file.

Usage:
    python word_freq.py <file> [-n TOP_N] [--case-sensitive]

Features:
    - Reads text from a file
    - Splits into words (whitespace-delimited, punctuation stripped)
    - Counts occurrences (case-insensitive by default)
    - Shows top N words with their counts in tabular format
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """Split text into words, stripping punctuation.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, words are normalized to lowercase.

    Returns:
        A list of words (cleaned tokens).
    """
    words = []
    for raw_word in text.split():
        # Strip leading/trailing punctuation characters
        cleaned = raw_word.strip(".,;:!?\"'()[]{}")
        if cleaned:
            if not case_sensitive:
                cleaned = cleaned.lower()
            words.append(cleaned)
    return words


def count_words(words: List[str]) -> Counter[str]:
    """Count occurrences of each word.

    Args:
        words: List of words to count.

    Returns:
        A Counter mapping words to their frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top N words by frequency.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples, sorted by count descending,
        then alphabetically for ties.
    """
    # Sort by: (-count, word) to get descending count and alphabetical tie-break
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
