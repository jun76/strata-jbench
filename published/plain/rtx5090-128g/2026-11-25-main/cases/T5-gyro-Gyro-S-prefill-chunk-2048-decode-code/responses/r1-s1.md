Top-N word frequency counter for text files.

Usage:
    python word_freq.py <file> [--top N] [--case-sensitive] [--min-length L]

Examples:
    python word_freq.py sample.txt --top 10
    python word_freq.py sample.txt --top 5 --case-sensitive --min-length 3
"""

import argparse

def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: Input text string.
        case_sensitive: If False, words are lowercased. If True, original case is preserved.
        min_length: Minimum word length to include (default: 1).

    Returns:
        List of tokens (words) meeting the criteria.
    """
    if not case_sensitive:
        text = text.lower()
    # Split on whitespace and filter by min_length
    tokens = []
    for word in text.split():
        # Remove punctuation attached to words (simple approach)
        cleaned = word.strip(".,;:!?"'"")
        if len(cleaned) >= min_length:
            tokens.append(cleaned)
    return tokens


def count_words(tokens: List[str]) -> Counter:
    """
    Count occurrences of each token.

    Args:
        tokens: List of word tokens.

    Returns:
        Counter object with word frequencies.
    """
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples, sorted by count descending, then word ascending.
    """
    return counter.most_common(n)
