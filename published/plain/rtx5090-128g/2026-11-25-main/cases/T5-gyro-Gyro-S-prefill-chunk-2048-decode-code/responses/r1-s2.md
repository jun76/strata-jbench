"""コマンドラインツール: テキストファイルから単語出現頻度を集計し上位 N 件を表形式で表示する."""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """テキストを単語単位に分割し、出現頻度を Counter で集計する.

    Args:
        text: 入力テキスト.

    Returns:
        各単語の出現回数を格納する Counter.
    """
    words = text.split()
    return Counter(words)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (word, count) を返す.

    Args:
        counter: 単語出現頻度の Counter.
        n: 取得する件数.

    Returns:
        上位 n 件の (word, count) のリスト.
    """
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """(word, count) のリストを表形式の文字列に整形する.

    Args:
        rows: 表示対象の (word, count) リスト.

    Returns:
        表形式の文字列.
    """
    if not rows:
        return ""
    max_word_len = max(len(word) for word, _ in rows)
    header = f"{'Word':<{max_word_len}} {'Count':>10}"
    separator = "-" * (max_word_len + 11)
    lines = [header, separator]
    for word, count in rows:
        lines.append(f"{word:<{max_word_len}} {count:>10}")
    return "\n".join(lines)


def main(argv: List[str] | None = None) -> int:
    """コマンドラインのメイン関数.

    Args:
        argv: 引数のリスト (None の場合は sys.argv[1:] を使用).

    Returns:
        終了コード (0 で成功).
    """
    parser = argparse.ArgumentParser(
        description="テキストファイルから単語出現頻度を集計し上位 N 件を表形式で表示します。"
    )
