"""
word_freq CLI tool: read a text file, count word occurrences, and print the top N words.

Usage:
    python word_freq.py <file> [--top N] [--case-sensitive] [--min-length L]

Output format (tab-separated table):
    WORD\tCOUNT
    ...
"""

import argparse
import sys
from collections import Counter
from typing import Dict, List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: Input text string.
        case_sensitive: If False, words are lowercased before counting.
        min_length: Minimum length of a word to be counted.

    Returns:
        List of tokens (words) after filtering by min_length.
    """
    if not case_sensitive:
        text = text.lower()
    # Split on whitespace and strip punctuation-like characters from tokens
    tokens = []
    for token in text.split():
        # Remove leading/trailing non-alphanumeric characters
        stripped = token.strip()
        # Filter out empty tokens or tokens shorter than min_length
        if len(stripped) >= min_length and any(c.isalnum() for c in stripped):
            tokens.append(stripped)
    return tokens


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Dict[str, int]:
    """
    Count word occurrences in the given text.

    Args:
        text: Input text string.
        case_sensitive: If False, words are lowercased before counting.
        min_length: Minimum length of a word to be counted.

    Returns:
        Dictionary mapping each word to its occurrence count.
    """
    tokens = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return dict(Counter(tokens))
