"""
コマンドラインツール: テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する。

使い方:
    python word_count.py <file> [--top N] [--ignore-case] [--min-length L]

例:
    python word_count.py sample.txt --top 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, ignore_case: bool = True, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        ignore_case: True の場合、単語を小文字に変換して集計する
        min_length: 単語の最小長（これ未満の単語は除外）

    Returns:
        単語のリスト
    """
    # 単語は英数字とハイフンを含む連続文字列として抽出
    pattern = r"[A-Za-z0-9\-]+"
    words = re.findall(pattern, text)
    if ignore_case:
        words = [w.lower() for w in words]
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    return words


def count_words(words: List[str]) -> Counter[str]:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数 の Counter
    """
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    出現回数が高い上位 N 件の単語を取得する。

    Args:
        counter: 単語 -> 出現回数 の Counter
        n: 取得する件数

    Returns:
        (単語, 回数) のタプルのリスト（回数降順）
    """
    # Counter.most_common は回数降順で返す
    return counter.most_common(n)
