Top Word Frequency Counter - A command-line tool that reads a text file,
counts word frequencies, and displays the top N words in a tabular format.

Usage:
    python word_freq.py <file> [-n N] [--case-sensitive]

    python word_freq.py sample.txt -n 10

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, words are converted to lowercase before counting.

    Returns:
        A list of word tokens.
    """
    words = text.split()
    if not case_sensitive:
        words = [word.lower() for word in words]
    return words


def count_word_frequencies(text: str, case_sensitive: bool = False) -> Counter:
    """
    Count the frequency of each word in the given text.

    Args:
        text: The input text.
        case_sensitive: If False, words are counted in lowercase.

    Returns:
        A Counter object with word frequencies.
    """
    tokens = tokenize(text, case_sensitive)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to retrieve.

    Returns:
        A list of tuples (word, count) sorted by frequency descending.
    """
    return counter.most_common(n)
