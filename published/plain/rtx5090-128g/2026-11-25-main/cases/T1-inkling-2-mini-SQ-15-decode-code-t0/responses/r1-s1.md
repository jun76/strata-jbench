#!/usr/bin/env python3
"""テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する。"""

from __future__ import annotations

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import Iterable


def tokenize(text: str) -> list[str]:
    """英数字の連続を単語として抽出する。"""
    return re.findall(r"[A-Za-z0-9_]+", text.lower())


def count_words(text: str) -> Counter[str]:
    """テキストから単語頻度を集計する。"""
    return Counter(tokenize(text))


def top_words(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """頻度降順・同頻度は単語昇順で上位 N 件を返す。"""
    return sorted(counter.items(), key=lambda item: (-item[1], item[0]))[:n]


def format_table(rows: list[tuple[str, int]]) -> str:
    """上位単語を表形式に整形する。"""
    if not rows:
        return "(単語なし)"
    width = max(len(word) for word, _ in rows)
    lines = [f"{'Word'.ljust(width)}  Count", f"{'-' * width}  -----"]
    for word, count in rows:
        lines.append(f"{word.ljust(width)}  {count}")
    return "\n".join(lines)


def read_text(path: Path) -> str:
    """UTF-8 テキストファイルを読み込む。"""
    return path.read_text(encoding="utf-8")
