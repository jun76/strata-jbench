"""
Text frequency analyzer.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    Count word occurrences in the given text.

    Args:
        text: Input text string.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    words = re.findall(r"\b\w+\b", text.lower())
    return Counter(words)


def read_file(file_path: Path) -> str:
    """
    Read text from a file.

    Args:
        file_path: Path to the text file.

    Returns:
        The file contents as a string.
    """
    return file_path.read_text(encoding="utf-8")


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Return the top N words by frequency.

    Args:
        counter: Counter of word frequencies.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples sorted by frequency descending.
    """
    return counter.most_common(n)
