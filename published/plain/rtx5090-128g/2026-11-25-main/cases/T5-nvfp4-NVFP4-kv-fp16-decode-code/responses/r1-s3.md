#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語の出現頻度をカウントし、上位N件を表形式で表示するCLIツール。
"""

import argparse
import sys
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストに分割する。
    英数字とハイフン、アポストロフィを含むトークンを単語とみなす。
    すべて小文字に変換する。

    Args:
        text: 入力テキスト

    Returns:
        単語（小文字）のリスト
    """
    # 英数字、ハイフン、アポストロフィを含むトークンを抽出
    tokens = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return [token.lower() for token in tokens]


def count_words(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (単語, 回数) のリストを返す。
    回数が同じ場合は単語の辞書順でソートされる。

    Args:
        counter: 単語と出現回数の Counter
        n: 取得する上位件数

    Returns:
        (単語, 出現回数) のタプルのリスト（上位順）
    """
    if n <= 0:
        return []
    # 回数の降順、同じ回数なら単語の昇順でソート
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
