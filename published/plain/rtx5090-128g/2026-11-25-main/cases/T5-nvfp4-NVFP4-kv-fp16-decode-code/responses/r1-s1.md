Usage:

def tokenize(text: str, min_length: int = 1, ignore_case: bool = True) -> list[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        min_length: 抽出する単語の最小長（デフォルト: 1）。
        ignore_case: 大文字小文字を区別しない場合 True（デフォルト: True）。

    Returns:
        抽出された単語のリスト。
    """
    if ignore_case:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む文字列を単語として抽出
    tokens = re.findall(r"[a-z0-9'\-]+", text)
    return [token for token in tokens if len(token) >= min_length]


def count_words(text: str, min_length: int = 1, ignore_case: bool = True) -> Counter[str]:
    """
    テキスト内の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト。
        min_length: 抽出する単語の最小長。
        ignore_case: 大文字小文字を区別しない場合 True。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    tokens = tokenize(text, min_length=min_length, ignore_case=ignore_case)
    return Counter(tokens)
