def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    
    単語は英数字とハイフンを含む文字列として定義される。
    大文字小文字は無視して小文字に統一する。
    
    Args:
        text: 分割対象のテキスト
        
    Returns:
        単語のリスト（正規化済み）
    """
    # 英数字、ハイフン、アポストロフィを含むトークンを抽出
    # 例: "don't" -> ["don't"], "hello-world" -> ["hello-world"]
    pattern = r"[a-zA-Z0-9']+"
    words = re.findall(pattern, text.lower())
    return words


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: カウント対象のテキスト
        
    Returns:
        単語をキー、出現回数を値とする Counter
    """
    words = tokenize(text)
    return Counter(words)
