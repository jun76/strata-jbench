### 1. 核心模块：`word_freq.py`

```python
# word_freq.py
from collections import Counter
from pathlib import Path
from typing import CounterType

def count_word_frequency(text: str) -> CounterType:
    """
    给定文本字符串，返回单词出现的频率。
    
    Args:
        text: 输入文本
    
    Returns:
        一个 Counter 对象，映射单词到其出现次数
    """
    # 简单 tokenizer: 小写化并只保留 alphanumeric characters
    words = [w for w in text.lower().split() if w.isalnum()]
    return Counter(words)

def top_n_words(freq: CounterType, n: int = 10) -> list[tuple[str, int]]:
    """
    从 word frequency counter 返回 top N most common words.
    
    Args:
        freq: Counter object with word counts
        n: number of top items to return
    
    Returns:
        List of tuples (word, count) sorted by count descending
    """
    return freq.most_common(n)
```
