def tokenize(text: str, min_length: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        min_length: 単語の最小長。これより短い単語は除外される。

    Returns:
        小文字化された単語のリスト
    """
    words: List[str] = []
    for word in text.split():
        # 句読点や特殊文字を除去
        cleaned_word = word.strip(".,;:!?\"'()[]{}<>-")
        if len(cleaned_word) >= min_length:
            words.append(cleaned_word.lower())
    return words


def count_words(words: List[str]) -> Dict[str, int]:
    """
    単語の出現頻度を集計する。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数 の辞書
    """
    counter: Counter[str] = Counter(words)
    return dict(counter)
