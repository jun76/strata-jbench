"""
word_freq.py - テキストファイルから単語の出現頻度を集計し、上位N件を表形式で表示する CLI ツール.

使い方:
    python word_freq.py input.txt --top 10 --ignore-case

依存:
    Python 3.9+
"""

import argparse

from collections import Counter
from typing import List, Tuple


def read_text(filepath: str) -> str:
    """指定されたファイルからテキストを読み込む.

    Args:
        filepath: 読み込むファイルのパス.

    Returns:
        ファイルの内容.

    Raises:
        FileNotFoundError: ファイルが存在しない場合.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def count_words(text: str, ignore_case: bool = False) -> Counter[str]:
    """テキスト中の単語の出現頻度をカウントする.

    Args:
        text: 対象テキスト.
        ignore_case: True の場合、大文字・小文字を無視して集計する.

    Returns:
        単語をキー、出現回数値を値とする Counter.
    """
    # 単純な分割: 空白・改行で区切る
    words = text.split()
    if ignore_case:
        words = [w.lower() for w in words]
    return Counter(words)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """カウンターから上位 N 件の (word, count) を降順で返す.

    Args:
        counter: 単語の出現頻度 Counter.
        n: 取得する件数.

    Returns:
        (word, count) のリスト. 同点時は辞書順で安定した順序を保つ.
    """
    # most_common(n) は内部で heapq を使用し、同点時は辞書順で安定
    return counter.most_common(n)
