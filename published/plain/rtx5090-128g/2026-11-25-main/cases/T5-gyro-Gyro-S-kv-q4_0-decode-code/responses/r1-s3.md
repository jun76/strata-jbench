#!/usr/bin/env python3
"""
text_frequency.py

コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を集計し、

使用方法:
    python text_frequency.py <file> [--top N] [--case-sensitive] [--min-length L]
"""

import argparse
import sys
from collections import Counter
from typing import Dict, List, Tuple


def tokenize_text(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するか
        min_length: 単語の最小長（これより短い単語は除外）

    Returns:
        単語のリスト
    """
    words = []
    for word in text.split():
        # 句読点や特殊文字を除去
        cleaned = word.strip(".,!?;:\"'()[]{}<>~`@#$%^&*-_+=/\\|")
        if not cleaned:
            continue
        if not case_sensitive:
            cleaned = cleaned.lower()
        if len(cleaned) >= min_length:
            words.append(cleaned)
    return words


def count_word_frequencies(words: List[str]) -> Dict[str, int]:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数の辞書
    """
    return dict(Counter(words))


def get_top_n(frequencies: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の頻度を取得する。頻度が高い順、同頻度の場合は辞書順。

    Args:
        frequencies: 単語 -> 頻度の辞書
        n: 上位件数

    Returns:
        (単語, 頻度) のタプルのリスト（頻度降順、同頻度時は単語の辞書順）
    """
    items = sorted(frequencies.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
