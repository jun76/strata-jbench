def tokenize(text: str, case_insensitive: bool = False, exclude_punctuation: bool = False) -> List[str]:
    """
    テキストを単語リストに分割する。

    Args:
        text: 入力テキスト文字列。
        case_insensitive: True の場合、単語を小文字に変換して比較する。
        exclude_punctuation: True の場合、英数字以外の文字を除去し、
                             単語を英数字のみで構成する。False の場合、
                             空白区切りで単語を抽出する。

    Returns:
        単語のリスト。
    """
    if case_insensitive:
        text = text.lower()

    if exclude_punctuation:
        # 英数字とスペースのみを残し、連続する非英数字をスペースに変換
        cleaned = re.sub(r'[^a-zA-Z0-9\s]', '', text)
        # 複数スペースを単一スペースに正規化
        cleaned = re.sub(r'\s+', ' ', cleaned).strip()
        if not cleaned:
            return []
        return cleaned.split(' ')
    else:
        # 空白文字で分割し、空文字列を除外
        tokens = text.split()
        return [t for t in tokens if t]
