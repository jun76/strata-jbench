以下に、要件を満たす Python スクリプトと pytest テストを提示します。

## 1. メインスクリプト: `word_freq.py`

```python
#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

使い方:
    python word_freq.py <input_file> [-n TOP_N] [--case-sensitive]

例:
    python word_freq.py sample.txt -n 10
    python word_freq.py sample.txt --case-sensitive
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


# 単語を抽出するための正規表現パターン（英数字・アンダースコア・ハイフンを含む）
_WORD_PATTERN = re.compile(r"\b[a-zA-Z0-9_'\-]+\b")


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。
                        False の場合、すべて小文字に変換する。

    Returns:
        抽出された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    return _WORD_PATTERN.findall(text)


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """単語リストから各単語の出現回数をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)


def get_top_n(freq_counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """出現頻度の上位 n 件を (単語, 回数) のタプルリストとして返す。

    同頻度の場合は辞書順（アルファベット順）でソートする。

    Args:
        freq_counter: 単語頻度の Counter。
        n: 取得する件数。

    Returns:
        上位 n 件の (単語, 回数)
