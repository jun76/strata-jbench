## 1. メインスクリプト `word_counter.py`

```python
#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度を集計して上位N件を
表形式で表示するコマンドラインツール。

Usage:
    python word_counter.py <file> [-n N] [--ignore-case] [--min-length L]
"""

import argparse
import sys
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple, Optional


def tokenize(text: str, ignore_case: bool = True, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        ignore_case: 大文字小文字を区別しない場合は True。
        min_length: 単語の最小文字数（これより短い単語は除外）。

    Returns:
        単語のリスト。
    """
    # 英数字を単語として抽出（日本語などマルチバイト文字はそのまま）
    words = re.findall(r'\w+', text)
    
    if ignore_case:
        words = [w.lower() for w in words]
    
    # 最小長より短い単語を除外
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    
    return words


def count_words(text: str, ignore_case: bool = True, min_length: int = 1) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        ignore_case: 大文字小文字を区別しない場合は True。
        min_length: 単語の最小文字数。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = tokenize(text, ignore_case=ignore_case, min_length=min_length)
