import argparse
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(file_path: Path) -> Counter:
    """テキストファイルを読み込み、単語出現頻度を集計します。"""
    text = file_path.read_text(encoding="utf-8")
    # 簡易的に空白と句読点で分割
    words = []
    for line in text.splitlines():
        for token in line.replace(",", " ").replace(".", " ").replace("!", " ").replace("?", " ").split():
            clean = token.strip("\"'()[]{}<>")
            if clean:
                words.append(clean.lower())
    return Counter(words)


def top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """頻度が高い順に上位 N 件を返します。"""
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """表形式で文字列を返します。"""
    if not rows:
        return "(no words)"
    word_width = max(len(word) for word, _ in rows)
    lines = [f"{'Word':<{word_width}}  {'Count':>5}", f"{'-' * word_width}  {'-' * 5}"]
    for word, count in rows:
        lines.append(f"{word:<{word_width}}  {count:>5}")
    return "\n".join(lines)
