"""
word_freq.py: テキストファイルの単語出現頻度を集計して上位 N 件を表形式で表示するCLI。

使い方:
    python word_freq.py <path> [-n N] [--ignore-case]
"""

from __future__ import annotations

import argparse
import re
from collections import Counter
from pathlib import Path


def tokenize(text: str, ignore_case: bool = False) -> list[str]:
    """テキストを単語（英数字とアポストロフィ）に分割する。"""
    tokens = re.findall(r"[A-Za-z0-9']+", text)
    if ignore_case:
        tokens = [t.lower() for t in tokens]
    return tokens


def count_words(text: str, ignore_case: bool = False) -> Counter:
    """テキスト中の単語出現回数を返す。"""
    return Counter(tokenize(text, ignore_case=ignore_case))


def format_table(counter: Counter, top_n: int) -> str:
    """上位 N 件を表形式（2列）で整形して返す。"""
    if top_n < 1:
        return ""
    items = counter.most_common(top_n)
    if not items:
        return "(no words)"
    width = max(len(word) for word, _ in items)
    lines = [f"{'WORD'.ljust(width)}  COUNT"]
    lines.append("-" * (width + 6))
    for word, freq in items:
        lines.append(f"{word.ljust(width)}  {freq}")
    return "\n".join(lines)
