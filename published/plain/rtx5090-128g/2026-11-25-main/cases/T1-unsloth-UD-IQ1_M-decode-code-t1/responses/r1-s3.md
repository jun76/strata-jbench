def normalize_word(word: str) -> str:
    """Normalize a word by lowercasing and stripping non-alphabetic characters.

    Args:
        word: A single token extracted from text.

    Returns:
        The normalized word with non-alphabetic characters removed and lowercased.
        If the word contains no alphabetic characters, returns an empty string.
    """
    # Lowercase first
    lower_word = word.lower()
    # Remove non-alphabetic characters (Unicode-aware)
    cleaned = ''.join(ch for ch in lower_word if ch.isalpha())
    return cleaned
