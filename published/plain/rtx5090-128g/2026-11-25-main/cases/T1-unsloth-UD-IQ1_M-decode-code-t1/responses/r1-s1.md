# 単語トークン化のための正規表現。
# Unicodeの文字・数字・ハイフン・アポストロフィを含む。
_WORD_RE = re.compile(r"[\w\-']+", re.UNICODE)


@dataclass(frozen=True)

def tokenize(text: str, case_sensitive: bool = False) -> list[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 対象テキスト。
        case_sensitive: Trueなら大文字小文字を区別、Falseならlowercase化する。

    Returns:
        抽出された単語のリスト。
    """
    tokens = _WORD_RE.findall(text)
    if not case_sensitive:
        tokens = [tok.lower() for tok in tokens]
    return tokens


def count_frequencies(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    テキスト中の単語出現頻度を Counter として返す。

    Args:
        text: 対象テキスト。
        case_sensitive: 大文字小文字の区別有無。

    Returns:
        {word: count} の Counter。
    """
    tokens = tokenize(text, case_sensitive=case_sensitive)
    return Counter(tokens)
