#!/usr/bin/env python3
"""テキストファイルから単語頻度を集計し、上位 N 件を表形式で表示するCLIツール。"""

import argparse
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを単純分割して小文字の単語列を返す。"""
    return [w.lower() for w in text.replace(",", " ").replace(".", " ").replace("!", " ").replace("?", " ").split() if w]


def count_words(text: str) -> Counter:
    """テキストから単語頻度を集計する。"""
    return Counter(tokenize(text))


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """上位 N 件の (単語, 頻度) を返す。同率は単語順で安定化する。"""
    return sorted(counter.items(), key=lambda x: (-x[1], x[0]))[:n]


def render_table(items: List[Tuple[str, int]]) -> str:
    """表形式文字列を生成する。"""
    if not items:
        return "(結果なし)"
    width = max(len(word) for word, _ in items)
    header = f"{'word':<{width}}  count"
    sep = "-" * len(header)
    rows = [f"{word:<{width}}  {count}" for word, count in items]
    return "\n".join([header, sep] + rows)
