import argparse
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def count_words(text: str) -> Counter:
    """テキストから単語を抽出して出現頻度を数える。"""
    words = text.lower().split()
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """頻度の高い順に上位 n 件を返す。同順位は辞書順。"""
    return sorted(counter.items(), key=lambda item: (-item[1], item[0]))[:n]


def display_table(items: List[Tuple[str, int]]) -> str:
    """上位 N 件を表形式の文字列として返す。"""
    if not items:
        return "No words found."

    rows = [["Rank", "Word", "Count"]]
    for i, (word, count) in enumerate(items, start=1):
        rows.append([str(i), word, str(count)])

    widths = [max(len(row[j]) for row in rows) for j in range(len(rows[0]))]
    lines = []
    for i, row in enumerate(rows):
        lines.append(" | ".join(cell.ljust(widths[j]) for j, cell in enumerate(row)))
        if i == 0:
            lines.append("-+-".join("-" * widths[j] for j in range(len(rows[0]))))
    return "\n".join(lines)
