#!/usr/bin/env python3
"""
wordfreq - Count word frequencies in text files and display top-N results.

This command-line tool reads one or more text files, tokenizes the content
into words (alphanumeric sequences), counts their frequencies, and displays
the top-N most frequent words in a formatted table.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Dict


def extract_words(text: str) -> List[str]:
    """
    Extract words from a given text string.
    
    Words are defined as sequences of alphanumeric characters (letters and digits).
    All words are converted to lowercase for case-insensitive counting.
    
    Args:
        text: Input text string.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    return re.findall(r'[a-z0-9]+', text.lower())


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word in a list of words.
    
    Args:
        words: A list of words.
        
    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top-N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top entries to return.
        
    Returns:
        A list of tuples (word, count) sorted by count descending, then by word ascending.
    """
    # Sort by count descending, then by word ascending for deterministic ordering
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
