#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込み、単語出現頻度を上位 N 件表示する。
"""

import argparse
import sys
from collections import Counter

def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    非英数字を区切り文字として扱い、単語は小文字に変換する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語のリスト（小文字、非英数字除外済み）
    """
    # 非英数字を空白に置き換えて分割
    cleaned = ''.join(c if c.isalnum() else ' ' for c in text)
    return [word.lower() for word in cleaned.split() if word]

def count_words(words: List[str]) -> Counter:
    """
    単語リストから頻度を集計する。
    
    Args:
        words: 単語のリスト
        
    Returns:
        単語 -> 頻度の Counter オブジェクト
    """
    return Counter(words)

def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度が高い順に上位 N 件の (単語, 頻度) を返す。
    同頻度の場合は辞書順で安定ソート。
    
    Args:
        counter: 単語頻度の Counter
        n: 取得件数
        
    Returns:
        [(単語, 頻度), ...] のリスト
    """
    # most_common は頻度降順、同頻度時は挿入順だが、ここでは辞書順で安定化
    items = sorted(
        counter.items(),
        key=lambda x: (-x[1], x[0])
    )
    return items[:n]
