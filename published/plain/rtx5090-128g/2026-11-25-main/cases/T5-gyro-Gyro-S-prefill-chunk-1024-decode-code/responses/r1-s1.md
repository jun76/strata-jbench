Top-N word frequency counter from a text file.

Usage:
    python word_freq.py <input_file> [--top N] [--case-sensitive] [--min-len L]

Arguments:
    input_file: Path to the text file to analyze.
    --top N:    Number of top words to display (default: 10).
    --case-sensitive: Preserve case (default: case-insensitive).
    --min-len L: Minimum word length to include (default: 1).

Output:
    A table of the top N words with their counts, sorted by count descending.
"""

import argparse

from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    Tokenize text into words, filtering by minimum length and optionally preserving case.

    Args:
        text: The input text string.
        case_sensitive: If False, words are lowercased before counting.
        min_len: Minimum word length to include (default 1).

    Returns:
        List of tokens (words) after filtering.
    """
    words = text.split()
    if not case_sensitive:
        words = [w.lower() for w in words]
    # Filter by minimum length
    words = [w for w in words if len(w) >= min_len]
    return words


def count_words(text: str, case_sensitive: bool = False, min_len: int = 1) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text string.
        case_sensitive: If False, words are lowercased before counting.
        min_len: Minimum word length to include (default 1).

    Returns:
        Counter object mapping words to their frequencies.
    """
    tokens = tokenize(text, case_sensitive=case_sensitive, min_len=min_len)
    return Counter(tokens)
