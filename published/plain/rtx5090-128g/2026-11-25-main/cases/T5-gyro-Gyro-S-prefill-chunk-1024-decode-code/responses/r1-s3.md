Text word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N most frequent words
in a tabular format.

def count_words(text: str) -> Counter[str]:
    """
    Count occurrences of words in the given text.

    Words are defined as sequences of characters separated by whitespace.
    Punctuation attached to words is stripped, and words are case-insensitive.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping each word to its occurrence count.
    """
    words = []
    for token in text.split():
        # Strip leading and trailing punctuation from each token
        stripped = token.strip(".,!?;:\"'()[]{}")
        if stripped:
            words.append(stripped.lower())
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from a counter.

    Args:
        counter: A Counter object mapping words to their counts.
        n: The number of top words to return.

    Returns:
        A list of tuples, each containing a word and its count, sorted by descending count.
    """
    return counter.most_common(n)
