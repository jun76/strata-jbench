"""Command-line tool to count word frequencies in text files and display top N results.

Usage:
    python word_counter.py --input <file> --top <N> [--case-insensitive] [--min-length <len>]

Example:
    python word_counter.py --input sample.txt --top 10
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, *, case_insensitive: bool = True, min_length: int = 1) -> List[str]:
    """Tokenize text into words.

    Args:
        text: Input text to tokenize.
        case_insensitive: If True, convert words to lowercase before counting.
        min_length: Minimum length of a word to be considered valid.

    Returns:
        A list of tokenized words (filtered by min_length).
    """
    # Simple whitespace-based tokenization; strips punctuation-like characters
    tokens: List[str] = []
    for raw_token in text.split():
        # Remove leading/trailing non-alphanumeric characters
        cleaned = raw_token.strip(".,!?;:\"'()[]{}")
        if len(cleaned) >= min_length:
            if case_insensitive:
                cleaned = cleaned.lower()
            tokens.append(cleaned)
    return tokens


def count_words(
    tokens: List[str],
    *,
    top_n: int = 10,
) -> List[Tuple[str, int]]:
    """Count word frequencies and return the top N most frequent words.

    Args:
        tokens: List of tokenized words.
        top_n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending, then alphabetically.
    """
    counter = Counter(tokens)
    # Sort by count descending, then by word ascending for tie-breaking
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:top_n]
