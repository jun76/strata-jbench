import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def read_text(filepath: str) -> str:
    """指定されたファイルからテキストを読み込む。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。

    単語は英数字（Unicode対応）の連続として定義され、すべて小文字に変換される。

    Args:
        text: 解析対象のテキスト。

    Returns:
        抽出された単語のリスト。
    """
    # Unicodeの文字や数字をマッチさせる
    words = re.findall(r"\b[\w]+\b", text, re.UNICODE)
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とするCounterオブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Counterから上位N個の単語と頻度のペアを返す。

        n: 取得する上位件数。

    Returns:
        (単語, 頻度)のタプルのリスト（頻度が高い順）。
    """
    return counter.most_common(n)
