Usage:

import argparse

from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """Tokenize text into words, filtering by minimum length and optionally normalizing case.

    Args:
        text: Input text to tokenize.
        case_sensitive: If True, preserve original case. If False, convert to lowercase.
        min_length: Minimum word length to include.

    Returns:
        A list of tokens (words) extracted from the text.
    """
    if not case_sensitive:
        text = text.lower()
    # Match sequences of alphanumeric characters and underscores (i.e., word characters)
    words = re.findall(r'\b[\w]+\b', text)
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    return words


def count_frequencies(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """Count the frequency of each word in the text.

    Args:
        text: Input text to analyze.
        case_sensitive: If True, preserve original case. If False, convert to lowercase.
        min_length: Minimum word length to include.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    words = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(words)
