text_freq.py - 単語出現頻度を集計し、上位N件を表形式で表示するコマンドラインツール。

Usage:
    python text_freq.py <file> [--top N] [--case-sensitive] [--sort-by count|alpha]
"""

def read_file(filepath: str) -> str:
    """指定されたファイルのテキスト内容を読み取る。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストを単語に分割する。

    Args:
        text: 分割対象のテキスト。
        case_sensitive: True の場合、大文字小文字を区別して分割する。
                        False の場合、すべて小文字に変換してから分割する。

    Returns:
        単語のリスト（非空白文字列のリスト）。
    """
    if not case_sensitive:
        text = text.lower()
    return text.split()


def count_frequencies(tokens: List[str]) -> Counter:
    """トークンの出現頻度をカウントする。

    Args:
        tokens: 単語のリスト。

    Returns:
        各単語とその出現回数を含む Counter オブジェクト。
    """
    return Counter(tokens)
