import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def read_text(filepath: str) -> str:
    """指定されたファイルからテキストを読み込む。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。

    単語は英数字（Unicode の alphanumeric）で構成される文字列として定義され、
    それ以外は区切りとみなす。すべて小文字に変換する。

    Args:
        text: 入力テキスト。

    Returns:
        抽出された単語のリスト。
    """
    # \w は Unicode では文字と数字とアンダースコアを含む。単語定義として \w+ を使う。
    words = re.findall(r"\w+", text, re.UNICODE)
    return [w.lower() for w in words]


def count_word_frequencies(words: List[str]) -> Counter:
    """単語リストから頻度カウンタを作成する。

    Args:
        words: 単語のリスト。

    Returns:
        単語から出現回数をマッピングする Counter オブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """カウンタから上位 N 件の (word, count) を返す。

    Args:
        counter: 単語から出現回数をマッピングする Counter。
        n: 取得する上位件数。

    Returns:
        上位 N 件の (word, count) のリスト。回数が同じ場合は辞書順で安定ソートする。
    """
    # 回数の降順、同数の場合は単語の昇順
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
