"""
word_freq: 指定されたテキストファイルから単語の出現頻度を集計し、
上位 N 件を表形式で表示するコマンドラインツールです。

使用方法:
    python word_freq.py <input_file> [-n NUM]

オプション:
    input_file    : 分析対象のテキストファイルパス
    -n, --num     : 表示する上位単語の数 (デフォルト: 10)

出力例:
    Rank | Word    | Count
    -----+---------+------
       1 | hello   |     5
       2 | world   |     3
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストに分割します。
    大文字・小文字は区別せず、アルファベットと数字のみを単語と見なします。
    
    Args:
        text: 入力テキスト

    Returns:
        単語リスト (すべて小文字)
    """
    # 英数字のみを抽出し、正規化（小文字化）
    return [word.lower() for word in re.findall(r'[a-zA-Z0-9]+', text)]


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度を集計します。

    Args:
        words: 単語リスト

    Returns:
        Counter オブジェクト (単語 -> 出現回数)
    """
    return Counter(words)
