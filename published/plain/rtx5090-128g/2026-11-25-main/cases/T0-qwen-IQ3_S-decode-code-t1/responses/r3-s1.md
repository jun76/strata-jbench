"""
text_freq.py - Count word frequencies in a text file and display top N words.
"""
import argparse
import re
from collections import Counter
from typing import Dict, List, Tuple


def read_file(filepath: str) -> str:
    """Read text file and return its content as string.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        PermissionError: If the file cannot be read due to permissions.
        UnicodeDecodeError: If the file contains invalid UTF-8 characters.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """Extract words from text by splitting on non-alphanumeric characters.

    Words are lowercased and consist of alphanumeric characters.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """Count occurrences of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count), sorted by count descending, then word ascending.
    """
    # Counter.most_common already sorts by count descending, then insertion order.
    # To break ties by word alphabetically, we sort explicitly.
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
