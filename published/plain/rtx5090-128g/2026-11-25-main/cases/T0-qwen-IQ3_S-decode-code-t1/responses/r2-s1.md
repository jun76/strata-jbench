"""
text_freq.py - 単語出現頻度を集計し、上位N件を表形式で表示するコマンドラインツール。

Usage:
    python text_freq.py <input_file> [-n N] [--case-sensitive]

Example:
    python text_freq.py sample.txt -n 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def normalize_word(word: str) -> str:
    """単語を正規化して、カウント対象のキーを返す。

    Args:
        word: 元の単語文字列。

    Returns:
        正規化された単語（大文字の場合はそのまま、それ以外は小文字に変換）。
    """
    # 前後の空白を除去し、内部の空白で分割されないようにする
    return word.strip()


def tokenize(text: str) -> List[str]:
    """テキストから単語を抽出する。

    Args:
        text: 入力テキスト。

    Returns:
        単語のリスト。
    """
    # 単純に空白区切りで分割。句読点は除外される（str.split()では含まれるが、ここでは簡略化）
    # より正確には、正規表現で単語を抽出しても良いが、今回は空白区切り
    return text.split()


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。False の場合、小文字に正規化する。

    Returns:
        単語をキー、頻度を値とする Counter。
    """
    words = tokenize(text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    else:
        # 前後の空白を除去
        words = [w.strip() for w in words]
        # 空文字列を除外
        words = [w for w in words if w]
    return Counter(words)
