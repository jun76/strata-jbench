#!/usr/bin/env python3
"""コマンドラインツール：テキストファイルを読み込んで単語の出現頻度を上位 N 件をテーブル表示する."""

import argparse

from typing import List, Tuple, Optional


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。アルファベットと数字を単語として扱い、大文字は小文字に変換して正規化。
    
    Args:
        text: 入力テキスト
        
    Returns:
        正規化された単語のリスト
    """
    words = []
    current_word = []
    for char in text:
        if char.isalnum() or char == '_':
            current_word.append(char.lower())
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    if current_word:
        words.append(''.join(current_word))
    return words


def count_words(words: List[str]) -> Counter:
    """単語リストから出現頻度をカウントする。
    
    Args:
        words: 単語のリスト
        
    Returns:
        単語とその出現回数をまとめた Counter オブジェクト
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の単語とその頻度を返す。
    
    Args:
        counter: 単語頻度の Counter
        n: 取得する上位件数
        
    Returns:
        (単語, 頻度) のタプルのリスト
    """
    if n <= 0:
        return []
    return counter.most_common(n)
