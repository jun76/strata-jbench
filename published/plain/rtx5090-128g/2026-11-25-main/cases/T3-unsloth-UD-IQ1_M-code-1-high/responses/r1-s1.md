def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    if not intervals:
        return []

    result: list[list[int]] = []

    for start, end in sorted(intervals, key=lambda interval: interval[0]):
        if result and start <= result[-1][1]:
            if end > result[-1][1]:
                result[-1][1] = end
        else:
            result.append([start, end])

    return result
