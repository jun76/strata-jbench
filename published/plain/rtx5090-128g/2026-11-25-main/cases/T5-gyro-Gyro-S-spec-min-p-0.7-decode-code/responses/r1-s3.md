"""
word_freq: コマンドラインツールでテキストファイルから単語の出現頻度を
集計し、上位 N 件を表形式で表示する。

使用例:
    python word_freq.py input.txt --top 10

依存:
    - argparse
    - collections.Counter
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。

    単語は英数字とハイフンを含む連続文字列として定義される。
    大文字小文字は区別しない（小文字に統一）。

    Args:
        text: 入力テキスト

    Returns:
        小文字化された単語のリスト
    """
    words: List[str] = []
    current: List[str] = []

    for char in text:
        if char.isalnum() or char == '-':
            current.append(char.lower())
        else:
            if current:
                words.append(''.join(current))
                current = []

    if current:
        words.append(''.join(current))

    return words


def count_words(text: str) -> Counter[str]:
    """
    テキスト中の単語出現頻度を集計する。

    Args:
        text: 入力テキスト

    Returns:
        単語 -> 出現回数の Counter
    """
    return Counter(tokenize(text))


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    上位 n 件の (単語, 出現回数) を返す。

    Args:
        counter: 単語 -> 出現回数の Counter
        n: 上位件数

    Returns:
        (単語, 出現回数) のリスト。出現回数が同じ場合は単語の辞書順でソート。
    """
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
