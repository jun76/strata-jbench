    python word_count.py [--top N] [--case-sensitive] [--ignore-punctuation] <file>

オプション:
    --top N          表示する上位単語の件数 (デフォルト: 10)
    --case-sensitive 大文字小文字を区別する (デフォルト: False)
    --ignore-punctuation 句読点・記号を除去して単語を抽出する (デフォルト: False)
    <file>           対象のテキストファイル
"""

import argparse

from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False, ignore_punctuation: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合、大文字小文字を区別
        ignore_punctuation: True の場合、句読点・記号を除去

    Returns:
        抽出された単語のリスト
    """
    if not case_sensitive:
        text = text.lower()

    if ignore_punctuation:
        # 句読点・記号を空白に置き換える
        import re
        # 単語の定義: 英数字と一部の文字（アポストロフィなど）
        # 句読点・記号を除去
        text = re.sub(r'[^\w\s]', '', text)

    words = text.split()
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語の出現頻度を集計する。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数の Counter
    """
    return Counter(words)
