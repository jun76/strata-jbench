Command-line tool: read a text file, count word frequencies,
and display the top N most frequent words in a tabular format.

Example:

def load_text(path: str) -> str:
    """Read the entire contents of a text file.

    Args:
        path: Path to the text file.

    Returns:
        The full text content as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        OSError: If the file cannot be read.
    """
    try:
        with open(path, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError as exc:
        raise FileNotFoundError(f"File not found: {path}") from exc


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """Split text into words, optionally lowercasing and filtering by length.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, words are lowercased before counting.
        min_len: Minimum word length to include (default 1).

    Returns:
        A list of word tokens (strings).
    """
    # Simple whitespace split; strip punctuation from each token
    raw_tokens = text.split()
    tokens = []
    for token in raw_tokens:
        # Remove leading/trailing punctuation characters
        cleaned = token.strip(".,;:?!)\"'`~@#$%^&*()[]{}<>/\\|+-=")
        if not case_sensitive:
            cleaned = cleaned.lower()
        if len(cleaned) >= min_len:
            tokens.append(cleaned)
    return tokens


def count_words(tokens: List[str]) -> Counter[str]:
    """Count the frequency of each word token.

    Args:
        tokens: List of word tokens.

    Returns:
        A Counter mapping each word to its frequency.
    """
    return Counter(tokens)
