```python
"""word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を集計する。

使い方:
    python word_freq.py input.txt --top 10 --min-length 2

オプション:
    --top N        上位 N 件を表示 (デフォルト: 10)
    --min-length N 最小単語長を指定。この長さ未満の単語は除外 (デフォルト: 1)
    --ignore-case  大文字小文字を区別しない (デフォルト: True)
    --case-sensitive 大文字小文字を区別する
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, ignore_case: bool = True, min_length: int = 1) -> List[str]:
    """テキストを単語リストに分解する。

    Args:
        text: 入力テキスト文字列。
        ignore_case: True の場合、大文字小文字を正規化する。
        min_length: この長さ未満の単語は除外する。

    Returns:
        単語のリスト。
    """
    # 英数字の単語を抽出（日本語も考慮したい場合は適応可能）
    # 単純化のため、[A-Za-z0-9]+ パターンを使用。必要に応じてUnicode対応可。
    words = re.findall(r"[A-Za-zÀ-ÿ0-9]+", text)
    if ignore_case:
        words = [w.lower() for w in words]
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    return words


def count_words(text: str, ignore_case: bool = True, min_length: int = 1) -> Counter:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト文字列。
        ignore_case: True の場合、大文字小文字を正規化する。
        min_length: この長さ未満の単語は除外する。

    Returns:
        Counter: 単語 -> 出現回数のマッピング。
    """
    words = tokenize(text, ignore_case=ignore_case, min_length=min_length)
    return Counter(words)
