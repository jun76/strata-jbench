#!/usr/bin/env python3
"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を
表形式で表示するコマンドラインツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_text(filepath: str) -> str:
    """
    テキストファイルを読み込んで内容の文字列を返す。

    Args:
        filepath: 読み込むテキストファイルのパス

    Returns:
        ファイルの内容を表す文字列

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        PermissionError: 読み込み権限がない場合
        UnicodeDecodeError: 文字コードデコードに失敗した場合
    """
    path = Path(filepath)
    if not path.exists():
        raise FileNotFoundError(f"ファイルが見つかりません: {filepath}")
    return path.read_text(encoding="utf-8")


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    アルファベット、数字、ハイフンを含む連続した文字列を単語として扱う。
    すべて小文字に正規化する。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト
    """
    # \bで区切られた単語を抽出。[a-zA-Z0-9'-]+ は英語単語向け。
    words = re.findall(r"[a-zA-Z0-9'-]+", text)
    return [word.lower() for word in words]
