#!/usr/bin/env python3
"""
A command-line tool to count word frequencies in text files and display
the top N most frequent words in a tabular format.

Usage:
    python wordfreq.py [-n TOP_N] [--sort-by {freq,word}] [-v] FILE [FILE ...]
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def normalize_words(text: str) -> List[str]:
    """
    Normalize text into a list of lowercase words.
    
    Args:
        text: Input text string.
        
    Returns:
        List of lowercase words extracted from the text.
    """
    # Extract words (sequences of letters and digits) and convert to lowercase
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_words(text: str) -> Dict[str, int]:
    """
    Count the frequency of each word in the given text.
    
    Args:
        text: Input text string.
        
    Returns:
        Dictionary mapping words to their occurrence counts.
    """
    words = normalize_words(text)
    return dict(Counter(words))


def read_file(file_path: Path) -> str:
    """
    Read the contents of a text file.
    
    Args:
        file_path: Path to the text file.
        
    Returns:
        The text content of the file.
        
    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If an I/O error occurs while reading the file.
    """
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError as e:
        raise FileNotFoundError(f"File not found: {file_path}") from e
    except IOError as e:
        raise IOError(f"Error reading file {file_path}: {e}") from e
