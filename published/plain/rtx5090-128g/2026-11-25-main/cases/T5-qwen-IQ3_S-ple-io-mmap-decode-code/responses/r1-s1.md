#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語出現頻度を計算し、上位N件を表示するツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        小文字化された単語のリスト
    """
    # 英数字とハイフンを含む単語を抽出（日本語などの多言語対応は簡略化）
    # 必要に応じて正規表現を調整可能
    words = re.findall(r'\b[a-zA-Z0-9-]+\b', text)
    return [word.lower() for word in words]


def count_word_frequency(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語とその出現回数の Counter オブジェクト
    """
    words = extract_words(text)
    return Counter(words)


def get_top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位 n 件の単語と頻度を取得する。
    頻度が同じ場合は単語の辞書順で安定したソートを行う。
    
    Args:
        counter: 単語頻度の Counter
        n: 取得する件数
        
    Returns:
        (単語, 頻度) のタプルリスト
    """
    # most_common は頻度降順だが、同頻度時の順序は不定
    # 安定したソートのため、頻度降順 -> 単語昇順でソート
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
