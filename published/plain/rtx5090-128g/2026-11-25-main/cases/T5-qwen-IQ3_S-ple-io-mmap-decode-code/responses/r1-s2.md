"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度を集計して
上位 N 件を表形式で出力するコマンドラインツール。

from __future__ import annotations

import argparse
import re

from collections import Counter
from pathlib import Path

def tokenize(text: str) -> List[str]:
    """テキストから単語リストを取り出す。

    英数字のみを単語として扱い、大文字は小文字に正規化する。

    Args:
        text: 入力テキスト。

    Returns:
        正規化された単語リスト。
    """
    return re.findall(r'[a-z0-9]+', text.lower())


def count_words(words: List[str]) -> Counter[str]:
    """単語リストから出現頻度を計算する。

    Args:
        words: 単語リスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)

    """Counter から上位 n 件の (word, count) を取り出す。

    同数の場合は辞書順で安定ソートする。

    Args:

        n: 取得する件数。

    Returns:
        (word, count) のリスト。上位から降順。
    """
    # 頻度で降順、同頻度なら単語で昇順
    return sorted(counter.items(), key=lambda x: (-x[1], x[0]))[:n]
