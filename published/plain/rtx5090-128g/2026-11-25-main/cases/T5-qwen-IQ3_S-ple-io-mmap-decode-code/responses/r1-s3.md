Word Frequency Counter CLI Tool.

Reads a text file, counts word occurrences, and displays the top N most frequent words

def read_text_file(file_path: Path) -> str:
    """
    Read text from a file with UTF-8 encoding.

    Args:
        file_path: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        OSError: If the file cannot be read.
    """
    try:
        return file_path.read_text(encoding='utf-8')
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {file_path}")
    except Exception as e:
        raise OSError(f"Error reading file {file_path}: {e}")

def normalize_and_tokenize(text: str) -> List[str]:
    """
    Normalize text to lowercase and split into words.

    Args:
        text: The input text string.

    Returns:
        A list of normalized words.
    """
    import re
    # Find all sequences of word characters (letters, digits, underscores)
    return re.findall(r'\w+', text.lower())

    """
    Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object containing word frequencies.
    """
    return Counter(words)
