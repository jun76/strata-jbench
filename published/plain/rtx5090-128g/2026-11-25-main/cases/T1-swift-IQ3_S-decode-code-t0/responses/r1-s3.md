#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位N件表示する。

使い方:
    python word_freq.py <file> [-n N]

例:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストから単語を抽出する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    # 英数字の連続を単語として抽出
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [w.lower() for w in words]


def count_words(text: str) -> Counter:
    """テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    words = tokenize(text)
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counterから上位n件の(単語, 回数)のリストを返す。

    Args:
        counter: 単語頻度のCounter。
        n: 取得する件数。

    Returns:
        (単語, 出現回数)のタプルのリスト。回数の降順、同数時は単語の辞書順。
    """
    # most_commonは回数の降順で返すが、同数時の順序は保証されない。
    # 辞書順で安定させるため、(-count, word)でソートする。
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
