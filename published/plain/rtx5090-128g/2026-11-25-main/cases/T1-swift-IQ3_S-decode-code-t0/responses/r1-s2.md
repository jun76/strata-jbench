#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位N件表示する。

使い方:
    python word_freq.py <file> [-n N]

例:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    英数字の連続を単語とし、すべて小文字に変換する。
    """
    return re.findall(r'[a-z0-9]+', text.lower())


def count_words(text: str) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。
    """
    words = tokenize(text)
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 n 件を (word, count) のリストで返す。
    頻度が同じ場合は辞書順でソートする。
    """
    # Counter.most_common は頻度順だが、同頻度時の順序は保証されない。
    # 頻度降順、同頻度時は単語の辞書順でソートする。
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]

    """
