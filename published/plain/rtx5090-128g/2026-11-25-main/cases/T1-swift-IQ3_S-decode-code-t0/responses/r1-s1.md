```python
#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位N件表示する。

使い方:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt --top 20 --case-sensitive
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか

    Returns:
        単語のリスト
    """
    # 英数字とハイフンを含む単語を抽出（日本語はそのまま）
    # 英単語: [a-zA-Z0-9'-]+
    # 日本語文字: [\u3040-\u309F\u30A0-\u30FF\u4E00-\u9FFF]
    pattern = r"[a-zA-Z0-9'-]+|[\u3040-\u309F\u30A0-\u30FF\u4E00-\u9FFF]"
    words = re.findall(pattern, text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか

    Returns:
        単語をキー、出現回数を値とするCounter
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counterから上位n件の単語と出現回数を取得する。

    Args:
        counter: 単語出現頻度のCounter
        n: 取得する件数

    Returns:
        (単語, 出現回数) のタプルのリスト（出現回数降順）
    """
    return counter.most_common(n)
