#!/usr/bin/env python3
"""
wordfreq.py - Text file word frequency counter CLI tool.

Reads a text file, tokenizes words (case-insensitive, punctuation stripped),
counts occurrences, and displays the top N words in a formatted table.

Usage:
    python wordfreq.py <input_file> [-n N] [--sort {freq,alpha}] [--output FORMAT]
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from pathlib import Path
from typing import TextIO


@dataclass(frozen=True)
class WordFrequency:
    """A single word-frequency pair."""

    word: str
    count: int


def tokenize(text: str) -> list[str]:
    """
    Tokenize text into lowercase words, stripping surrounding punctuation.

    Args:
        text: Input text string.

    Returns:
        List of lowercase word tokens (empty strings filtered out).
    """
    # Convert to lowercase
    text_lower = text.lower()
    # Extract words: sequences of word characters (letters, digits, underscores)
    # This also handles hyphenated words as separate tokens if desired,
    # but here we treat them as single tokens by including '-' in the pattern
    # Adjust regex as needed for specific requirements.
    words = re.findall(r"[a-z0-9]+(?:'[a-z]+)?", text_lower)
    return words


def count_frequencies(text: str) -> Counter[str]:
    """
    Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        Counter mapping words to their occurrence counts.
    """
    words = tokenize(text)
    return Counter(words)
