# 単語頻度カウンター CLI ツール

以下に、要件を満たす実装を提供します。

## 1. 本体スクリプト: `word_counter.py`

```python
#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度を数え、
上位 N 件を表形式で出力するコマンドラインツール。

Usage:
    python word_counter.py FILE [-n TOP_N] [--case-sensitive] [--min-len MIN_LEN]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
        min_len: この長さ未満のトークンは除外する。

    Returns:
        単語トークンのリスト。
    """
    # 英数字とハイフン・アポストロフィを許容するパターン
    pattern = r"[a-zA-Z0-9]+(?:['-][a-zA-Z0-9]+)*"
    tokens = re.findall(pattern, text)
    if not case_sensitive:
        tokens = [token.lower() for token in tokens]
    tokens = [token for token in tokens if len(token) >= min_len]
    return tokens


def count_frequencies(tokens: List[str]) -> Counter:
    """トークンリストから出現頻度を計算する。

    Args:
        tokens: 単語トークンのリスト。

    Returns:
        各単語とその出現回数を保持する Counter。
    """
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (word, count) を返す。

    Args:
        counter: 単語頻度 Counter。
        n: 取得する件数。

    Returns:
        (word, count) のタプルのリスト。降順ソート済み。
    """
    return counter.most_common(n)
