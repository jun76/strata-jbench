#!/usr/bin/env python3
"""A command-line tool that reads a text file, counts word frequencies, and displays the top N words in a table format."""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count the frequency of each word in the given text.

    Words are split by whitespace and converted to lowercase. Punctuation
    is stripped from the beginning and end of each token.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    words = []
    for token in text.split():
        # Strip punctuation from both ends of the token
        cleaned = token.strip(".,;:!?\"'()[]{}").lower()
        if cleaned:
            words.append(cleaned)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) sorted by descending frequency,
        then alphabetically for ties.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
