def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。

    正規表現を用いて、英数字の連続を単語とみなし、それ以外を区切りとする。
    すべて小文字に変換して比較する。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 単語のリスト（小文字）。
    """
    # 英数字の連続を抽出し、小文字に変換
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words

    """

        Counter: 単語をキー、頻度を値とするカウント。
    """
