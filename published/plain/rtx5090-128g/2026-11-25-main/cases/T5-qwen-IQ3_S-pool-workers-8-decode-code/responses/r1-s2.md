"""
word_counter: テキストファイルから単語出現頻度を集計し、上位N件を表形式で表示するコマンドラインツール。
"""

from __future__ import annotations

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。

    単語は英数字の連続として定義し、大文字は小文字に変換する。
    非英数字文字（句読点、空白、記号など）は区切りとして扱う。

    Args:
        text: 入力テキスト

    Returns:
        小文字化された単語のリスト
    """
    words: List[str] = []
    current: List[str] = []
    for char in text:
        if char.isalnum():
            current.append(char.lower())
        else:
            if current:
                words.append("".join(current))
                current = []
    if current:
        words.append("".join(current))
    return words


def count_words(text: str) -> Counter[str]:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語から出現回数へのCounter
    """
    return Counter(tokenize(text))
