#!/usr/bin/env python3
"""
テキストファイルを読み込み、単語の出現頻度を集計して上位N件を表形式で表示するコマンドラインツール。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    英数字からなるトークンを抽出し、小文字に変換する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語のリスト（小文字化済み）
    """
    # 英数字の連続を単語として抽出し、小文字に変換
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_word_frequencies(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語から出現回数をマッピングするCounter
    """
    words = extract_words(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現回数の上位N件を返す。同数の場合は単語のアルファベット順でソート。
    
    Args:
        counter: 単語出現頻度のCounter
        n: 取得する上位件数
        
    Returns:
        (単語, 出現回数) のタプルのリスト
    """
    # 出現回数降順、同数なら単語の昇順
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
