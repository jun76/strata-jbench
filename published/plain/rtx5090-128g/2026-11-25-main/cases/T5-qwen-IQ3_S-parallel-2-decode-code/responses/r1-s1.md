#!/usr/bin/env python3
"""
word_freq.py - Count word frequencies in text files and display top N results.

Usage:
    python word_freq.py <file_path> [--top N] [--case-sensitive] [--min-length MIN_LEN]

This tool reads a text file, extracts words, counts their occurrences,
and displays the top N most frequent words in a table format.
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple, Dict


def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Extract words from text, optionally filtering by case sensitivity and minimum length.
    
    Args:
        text: The input text to process.
        case_sensitive: If True, preserve original case. If False, convert to lowercase.
        min_length: Minimum word length to include (default 1).
    
    Returns:
        A list of extracted words.
    """
    # Extract sequences of alphanumeric characters and underscores as words
    words = re.findall(r'[a-zA-Z0-9_]+', text)
    
    if not case_sensitive:
        words = [w.lower() for w in words]
    
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    
    return words


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word in the given list.
    
    Args:
        words: A list of words to count.
    
    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: Number of top entries to return.
    
    Returns:
        A list of tuples (word, count) for the top N most common words.
    """
    return counter.most_common(n)
