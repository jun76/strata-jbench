Word Frequency Counter CLI Tool.

This module provides a command-line interface to read a text file,
count the frequency of each word, and display the top N most frequent
words in a table format.
"""

def extract_words(text: str) -> List[str]:
    """
    Extract words from the given text.
    
    Words are defined as sequences of alphanumeric characters (including underscores).
    All words are converted to lowercase for case-insensitive counting.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    # \w+ matches [a-zA-Z0-9_]
    return re.findall(r'\w+', text.lower())


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word in the list.
    
    Args:
        words: A list of words.
        
    Returns:
        A Counter object mapping words to their frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their counts.
    
    Args:
        counter: A Counter object containing word frequencies.
        n: The number of top items to retrieve.
        
    Returns:
        A list of tuples, where each tuple contains (word, count).
        If there are ties, the order may vary but counts will be correct.
    """
    return counter.most_common(n)
