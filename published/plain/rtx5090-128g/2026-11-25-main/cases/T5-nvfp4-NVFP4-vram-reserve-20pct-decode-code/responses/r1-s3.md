#!/usr/bin/env python3
"""
Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words
in a tabular format.
"""

import argparse

from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    Count word occurrences in the given text.

    Words are defined as sequences of alphanumeric characters, separated by
    whitespace or punctuation. Case-insensitive counting is used.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter mapping each lowercase word to its occurrence count.
    """
    # Split on non-alphanumeric characters and filter empty strings
    words = []
    current_word = []
    for char in text:
        if char.isalnum():
            current_word.append(char.lower())
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    if current_word:
        words.append(''.join(current_word))

    return Counter(words)


def get_top_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from a counter.

    Args:
        counter: A Counter mapping words to their frequencies.
        n: The number of top words to retrieve.

    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by count descending, then by word ascending for deterministic output
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
