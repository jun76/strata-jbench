def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    英数字のみからなるトークンを抽出し、小文字に変換する。
    
    Args:
        text: 入力テキスト文字列
        
    Returns:
        抽出された単語のリスト
    """
    # 英数字の連続した部分を単語として抽出
    # 日本語など非ASCII文字は除外される仕様です（要件に応じて正規表現を調整可能）
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_frequencies(words: List[str]) -> Counter[str]:
    """
    単語リストから出現頻度を計算する。
    
    Args:
        words: 単語のリスト
        
    Returns:
        単語をキー、出現回数を値とするCounter
    """
    return Counter(words)

    """

        n: 取得する上位件の数
        
    Returns:
        (単語, 出現回数) のタプルのリスト（出現回数の降順）
    """
    # most_commonは内部でソートされる
    return counter.most_common(n)
