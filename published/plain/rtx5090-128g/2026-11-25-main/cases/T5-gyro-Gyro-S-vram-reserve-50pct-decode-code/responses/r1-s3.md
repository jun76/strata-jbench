def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するか。False の場合は小文字に統一する。

    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()
    # 単語を抽出: 英数字およびハイフンを含むトークンを単語として扱う
    # 簡易的な方法: 空白で分割し、句読点を除去
    words = []
    for token in text.split():
        # 句読点や特殊文字を除去
        cleaned = ''.join(ch for ch in token if ch.isalnum() or ch == '-')
        if cleaned:
            words.append(cleaned)
    return words

    Returns:
        単語 -> 出現回数の Counter
    """
    tokens = tokenize(text, case_sensitive)
    return Counter(tokens)
