"""コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を上位 N 件で表示する。"""

from __future__ import annotations

import argparse
import collections
import re
import sys
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。単語は [a-zA-Z]+ にマッチするもの。"""
    return re.findall(r"[a-zA-Z]+", text)


def count_words(words: List[str]) -> collections.Counter[str]:
    """単語リストの出現頻度を Counter で返す。"""
    counter = collections.Counter(words)
    return counter


def top_n(counter: collections.Counter[str], n: int) -> List[Tuple[str, int]]:
    """上位 n 件の (単語, 頻度) を頻度降順で返す。同頻度時は辞書順（安定）。"""
    items = counter.items()
    # 頻度降順、同頻度時は単語昇順で安定ソート
    sorted_items = sorted(items, key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]


def format_table(items: List[Tuple[str, int]], title: str = "Top N Words") -> str:
    """表形式で表示する文字列を返す。"""
    if not items:
        return f"{title}: no words found"
    lines = [title]
    lines.append("-" * 40)
    lines.append(f"{'Rank':<6}{'Word':<20}{'Count':<10}")
    lines.append("-" * 40)
    for rank, (word, count) in enumerate(items, start=1):
        lines.append(f"{rank:<6}{word:<20}{count:<10}")
    lines.append("-" * 40)
    return "\n".join(lines)
