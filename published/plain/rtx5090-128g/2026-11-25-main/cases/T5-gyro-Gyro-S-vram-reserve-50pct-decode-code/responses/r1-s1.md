"""
word_freq.py — コマンドラインツール: テキストファイルから単語出現頻度を集計し上位N件を表示する。
"""

import argparse
import sys
from collections import Counter
from typing import Iterable


def read_file(path: str) -> str:
    """指定されたファイルを読み、その内容を文字列として返す。"""
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> Iterable[str]:
    """テキストを単語に分割する。空白・改行・句読点で区切る。"""
    # 句読点を除去し、空白で分割
    import re
    # 単語として扱う: アルファベットと数字の連続
    words = re.findall(r"[a-zA-Z0-9]+", text)
    return words


def count_words(words: Iterable[str]) -> Counter[str]:
    """単語の出現回数をカウントする。"""
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """出現頻度の上位N件を返す。同頻度の場合は辞書順（アルファベット順）でソート。"""
    # Counter.most_common は頻度順に並べるが、同頻度での安定性は保証しない。
    # 同頻度で辞書順にするため、(-count, word) でソートする。
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(rows: list[tuple[str, int]], word_width: int = 20) -> str:
    """表形式で出力する。"""
    lines = []
    header = f"{'Word':<{word_width}} {'Count':>6}"
    separator = f"{'-' * word_width} {'-' * 6}"
    lines.append(header)
    lines.append(separator)
    for word, count in rows:
        lines.append(f"{word:<{word_width}} {count:>6}")
    return "\n".join(lines)
