def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """テキストから単語トークンを抽出する。

    単語は英数字・アンダースコア・ハイフン・アポストロフィを含む文字列として認識される。
    句読点や空白は区切り文字として扱われる。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。
        min_len: この長さ未満の単語は無視する。

    Returns:
        単語トークンのリスト。
    """
    # 英数字、アンダースコア、ハイフン、アポストロフィを含む連続文字列を単語として抽出
    pattern = r"[a-zA-Z0-9_'\-]+"
    tokens = re.findall(pattern, text)
    if not case_sensitive:
        tokens = [tok.lower() for tok in tokens]
    if min_len > 1:
        tokens = [tok for tok in tokens if len(tok) >= min_len]
    return tokens
