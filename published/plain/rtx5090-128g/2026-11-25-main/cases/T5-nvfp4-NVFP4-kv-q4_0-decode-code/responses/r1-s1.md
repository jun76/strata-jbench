"""
word_freq.py - A command-line tool to count word frequencies in text files.

Usage:
    python word_freq.py <file> [--top N] [--case-sensitive] [--min-length L]
"""

def read_file(filepath: str) -> str:
    """Read the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        PermissionError: If the file cannot be read due to permissions.
        OSError: For other I/O errors.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """Tokenize text into words.

    Words are defined as sequences of alphanumeric characters and underscores
    (similar to word characters in regex).

    Args:
        text: The input text to tokenize.
        case_sensitive: If True, preserve case. If False, convert to lowercase.
        min_length: Minimum length of a word to be included.

    Returns:
        A list of tokens (words).
    """
    if not case_sensitive:
        text = text.lower()
    # Use regex to find word characters (letters, digits, underscores)
    words = re.findall(r'\w+', text)
    # Filter by minimum length
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    return words


def count_words(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)
