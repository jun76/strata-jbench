#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位N件表示する。

使い方:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt --top 20 --case-sensitive
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: Trueの場合、大文字小文字を区別する。Falseの場合、すべて小文字に変換する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフンを含む単語を抽出（日本語などのマルチバイト文字も考慮）
    words = re.findall(r'\b[\w\-]+\b', text, re.UNICODE)
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: Trueの場合、大文字小文字を区別する。

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counterから上位n件の単語と出現回数を取得する。

    Args:
        counter: 単語出現頻度のCounter。
        n: 取得する件数。

    Returns:
        (単語, 出現回数)のタプルのリスト。出現回数の降順、同数時は辞書順。
    """
    # most_commonは出現回数の降順で返すが、同数時の順序は不定なので
    # 辞書順でソートし直す
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
