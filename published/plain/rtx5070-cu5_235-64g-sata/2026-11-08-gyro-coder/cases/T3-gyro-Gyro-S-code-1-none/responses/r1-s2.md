def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    """
    Merge all overlapping or touching closed intervals.
    
    Args:
        intervals: A list of closed intervals, each represented as [start, end].
                   The input may be unsorted and may be empty. It is not modified.
    
    Returns:
        A sorted list of merged, non-overlapping (and non-touching) intervals.
        Touching intervals (where one interval's end equals another's start) are merged.
    """
    if not intervals:
        return []
    
    # Sort by start, then by end
    sorted_intervals = sorted(intervals, key=lambda x: (x[0], x[1]))
    
    merged = []
    for interval in sorted_intervals:
        start, end = interval
        if not merged or merged[-1][1] < start:
            merged.append([start, end])
        else:
            # Merge with the last interval
            merged[-1][1] = max(merged[-1][1], end)
    
    return merged
