以下は Python 3 によるコマンドラインツールでテキストファイルの単語出現頻度上位 N 件表形式で表示する。

```python
#!/usr/bin/env python3
"""
word_freq.py - 上位 N 件表形式で表示するコマンドラインツール

Usage:
    python word_freq.py <file> [-n N]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    Extract words from text.
    
    Args:
        text: Input text string.
    
    Returns:
        List of lowercase words (alphabetic only).
    """
    return re.findall(r'[a-zA-Z]+', text.lower())


def count_words(words: List[str]) -> Counter:
    """
    Count word frequencies.
    
    Args:
        words: List of words.
    
    Returns:
        Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get top N most common words.
    
    Args:
        counter: Counter object with word frequencies.
        n: Number of top items to return.
    
    Returns:
        List of (word, count) tuples.
    """
    return counter.most_common(n)


def format_table(top_items: List[Tuple[str, int]]) -> str:
    """
    Format top items as a table string.
    
    Args:
        top_items: List of (word, count) tuples.
    
    Returns:
        Formatted table string.
    """
    if not top_items:
        return ""
    
    max_word_len = max(len(word) for word, _ in top_items)
    max_count_len = max(len(str(count)) for _, count in top_items)
    
    header = f"{'Word':<{max_word_len}} {'Count':>{max_count_len}}"
    separator = f"{'-' * max_word_len} {'-' * max_count_len}"
    
    lines = [header, separator]
    for word, count in top_items:
        lines.append(f"{word:<{max_word_len
