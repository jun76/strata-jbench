# word_count.py
"""
word_count.py — 上位 N 件の word frequency table CLI tool.

Usage:
    python word_count.py <file> [-n N] [--case-insensitive] [--min-len L] [--max-len L]
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter


def tokenize(text: str, *, case_insensitive: bool = False,
             min_len: int = 1, max_len: int = 1000) -> list[str]:
    """Split text into words, filter by length, optionally lowercase.

    Args:
        text: raw text string.
        case_insensitive: if True, lower-case all tokens before counting.
        min_len: minimum token length to keep.
        max_len: maximum token length to keep.

    Returns:
        list of tokens (strings).
    """
    if case_insensitive:
        text = text.lower()
    # Unicode-aware word characters (letters, digits, underscore)
    tokens = re.findall(r"\w+", text, flags=re.UNICODE)
    return [t for t in tokens if min_len <= len(t) <= max_len]


def count_words(text: str, *, case_insensitive: bool = False,
                min_len: int = 1, max_len: int = 1000) -> Counter[str]:
    """Return a Counter of word frequencies in text."""
    tokens = tokenize(text, case_insensitive=case_insensitive,
                      min_len=min_len, max_len=max_len)
    return Counter(tokens)


def top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """Return top-n (word, count) pairs, ties broken by word lexicographic order."""
    # Sort by (-count, word) so ties are deterministic
    items = sorted(counter.items(), key=lambda kv: (-kv[1], kv[0]))
    return items[:n]
