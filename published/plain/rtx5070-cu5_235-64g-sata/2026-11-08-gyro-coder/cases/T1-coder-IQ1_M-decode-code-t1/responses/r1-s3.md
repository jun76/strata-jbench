wordfreq.py - Top N Word Frequency Counter

A command-line tool that reads a text file, counts the frequency

def tokenize(text: str) -> list[str]:
    """
    Tokenize text into a list of lowercase words.

    Words are defined as sequences of Unicode letters (including digits for simplicity).
    Punctuation and whitespace are removed. All words are lowercased.

    Args:
        text: The raw input text.

    Returns:
        A list of tokenized words (lowercase, no punctuation).
    """
    # Match sequences of word characters (letters, digits, underscores)
    # For pure "words" (letters only), you can use r"[a-zA-Z0-9']+" or Unicode letters:
    # \w+ includes [a-zA-Z0-9_] in ASCII mode; here we use \p{L}+ with regex module?
    # But standard `re` does not support \p{L} without regex module.
    # Use Unicode letter matching via re.UNICODE + \w which matches [a-zA-Z0-9_]
    # If strict letters-only is required, use: re.findall(r"[^\W\d_]+", text, flags=re.UNICODE)
    # Here we'll accept alphanumeric words (common convention):
    return [w.lower() for w in re.findall(r"\w+", text)]


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: The raw text to analyze.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    words = tokenize(text)
    return Counter(words)
