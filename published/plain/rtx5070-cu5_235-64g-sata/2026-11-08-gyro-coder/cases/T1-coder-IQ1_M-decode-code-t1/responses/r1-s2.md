#!/usr/bin/env python3
"""
word_count.py — Text file word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N
most frequent words in table form to the console.
"""

def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase alphanumeric words.

    Args:
        text: Raw input string.

    Returns:
        A list of cleaned, lowercased word tokens.
    """
    return re.findall(r"\b[a-z0-9']+\b", text.lower())


def count_words(text: str) -> dict:
    """
    Count frequency of each word in the given text.

    Args:
        text: Input string to analyze.

    Returns:
        Dictionary mapping each word to its occurrence count.
    """
    tokens = tokenize(text)
    freq: dict = {}
    for token in tokens:
        freq[token] = freq.get(token, 0) + 1
    return freq


def top_n_words(frequency: dict, n: int) -> List[Tuple[str, int]]:
    """
    Return the top N most frequent words.

    Ties are broken by alphabetical order (ascending) for stable results.

    Args:
        frequency: Mapping from word -> count.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples, sorted by count desc, then word asc.
    """
    sorted_items: List[Tuple[str, int]] = sorted(
        frequency.items(),
        key=lambda item: (-item[1], item[0]),
    )
    return sorted_items[:n]
