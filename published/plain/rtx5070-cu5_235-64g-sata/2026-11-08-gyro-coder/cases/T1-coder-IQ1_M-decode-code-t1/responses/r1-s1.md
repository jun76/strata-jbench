# word_freq.py
"""
命令ライン工具：读取 a text file, count word frequencies, and display top N entries.
"""

import argparse
import re
from collections import Counter


def clean_text(text: str) -> list[str]:
    """Normalize text to lowercase and extract words."""
    return re.findall(r"[a-z]+", text.lower())


def count_words(file_path: str) -> dict[str, int]:
    """Read a file and return word frequency dictionary."""
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            text = f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {file_path}")
    words = clean_text(text)
    return dict(Counter(words))


def get_top_n(frequencies: dict[str, int], n: int) -> list[tuple[str, int]]:
    """Return top N words sorted by frequency (descending)."""
    return sorted(frequencies.items(), key=lambda x: -x[1])[:n]


def parse_args() -> argparse.Namespace:
    """Parse command line arguments."""
    parser = argparse.ArgumentParser(description="Count word frequencies in a text file.")
    parser.add_argument("file", help="Path to the text file")
    parser.add_argument("-n", type=int, default=10, help="Number of top entries to display")
    return parser.parse_args()


def main() -> None:
    """Entry point for CLI."""
    args = parse_args()
    freq = count_words(args.file)
    top = get_top_n(freq, args.n)
    print("Top words:")
    for word, count in top:
        print(f"  {word}: {count}")


if __name__ == "__main__":
    main()
