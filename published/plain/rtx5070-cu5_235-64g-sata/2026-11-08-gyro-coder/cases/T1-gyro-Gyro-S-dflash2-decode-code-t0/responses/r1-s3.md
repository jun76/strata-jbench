def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 単語は英数字とハイフン/アポストロフィを含む連続文字列として抽出
    # 簡易的な分割: 空白・改行・句読点で区切る
    import re
    # 単語パターン: 英数字、ハイフン、アポストロフィを含む
    pattern = r"[a-zA-Z0-9]+(?:[-'][a-zA-Z0-9]+)*"
    return re.findall(pattern, text)


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト中の単語出現頻度を集計する。

    Args:
        text: 入力テキスト。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)
