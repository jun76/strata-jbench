#!/usr/bin/env python3
"""
wordfreq - Count word frequencies from a text file and display top N words.

Usage:
    python wordfreq.py <input_file> [-n TOP_N] [--case-sensitive] [--ignore-punctuation]
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple, Optional


def tokenize(text: str, case_sensitive: bool = False, ignore_punctuation: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, convert to lowercase.
        ignore_punctuation: If True, remove punctuation characters from words.

    Returns:
        A list of tokens (words).
    """
    if not case_sensitive:
        text = text.lower()

    # Split on whitespace
    tokens = text.split()

    if ignore_punctuation:
        # Remove punctuation characters from each token
        cleaned_tokens = []
        for token in tokens:
            # Remove leading/trailing punctuation, keep internal punctuation? 
            # Simpler: remove all punctuation characters
            cleaned = re.sub(r'[^\w]', '', token)
            if cleaned:
                cleaned_tokens.append(cleaned)
        tokens = cleaned_tokens

    return tokens


def count_words(tokens: List[str]) -> Counter[str]:
    """
    Count occurrences of each word.

    Args:
        tokens: A list of word tokens.

    Returns:
        A Counter mapping words to their counts.
    """
    return Counter(tokens)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending, then by word ascending.
    """
    # Sort by count descending, then by word ascending for tie-breaking
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
