コマンドラインツール: テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する。

import argparse

def read_file(path: str) -> str:
    """指定されたパスのテキストファイルを読み込み、その内容を文字列として返す。

    Args:
        path: ファイルパス

    Returns:
        ファイルのテキスト内容

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        OSError: ファイル読み込み時にエラーが発生した場合
    """
    try:
        with open(path, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {path}")
    except OSError as e:
        raise OSError(f"ファイル読み込みエラー: {e}")


def tokenize(text: str, case_insensitive: bool = False) -> List[str]:
    """テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_insensitive: True の場合、単語を小文字に変換して集計する

    Returns:
        単語のリスト
    """
    # 単純な空白区切りで単語を抽出
    # 必要に応じて正規表現でより厳密なトークン化も可能だが、ここではシンプルに
    words = text.split()
    if case_insensitive:
        words = [w.lower() for w in words]
    return words


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """単語の出現頻度を Counter で集計する。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数の Counter
    """
    return Counter(words)
