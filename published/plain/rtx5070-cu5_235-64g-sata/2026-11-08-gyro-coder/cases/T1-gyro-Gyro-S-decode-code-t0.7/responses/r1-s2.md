import argparse

from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    与えられたテキストから単語の出現頻度をカウントします。

    Args:
        text: 入力テキスト文字列。

    Returns:
        各単語とその出現回数を格納した Counter。
    """
    # 小文字に正規化し、非アルファベット文字を空白で区切ることで単語を抽出
    normalized = text.lower()
    # アルファベットと数字のみを単語として扱い、それ以外は区切りとして扱う
    words = []
    current_word = []
    for char in normalized:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counter から出現回数順に上位 N 件の (word, count) のリストを返す。
    同回数時は辞書順でソートする。

    Args:
        counter: 単語頻度の Counter。
        n: 取得する上位件数。

    Returns:
        [(word, count), ...] のリスト。
    """
    items = list(counter.items())
    # 回数を降順、同回数時は単語を昇順でソート
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
