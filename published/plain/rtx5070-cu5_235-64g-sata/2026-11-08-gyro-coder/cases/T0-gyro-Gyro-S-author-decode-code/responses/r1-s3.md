#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を
集計し、上位 N 件を表形式で表示する。

使い方:
    python word_freq.py <input_file> [--top N] [--case-sensitive]

例:
    python word_freq.py sample.txt --top 10
"""

import argparse

from collections import Counter
from typing import List, Tuple


def load_text(path: str) -> str:
    """指定されたパスのテキストファイルを読み、文字列を返す。

    Args:
        path: ファイルパス

    Returns:
        ファイルの内容

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        OSError: 読み込み失敗の場合
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストを単語に分割する。

    単語は空白区切りのトークンとし、句読点や数字は除去しない。
    case_sensitive が False の場合、小文字化して比較する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか

    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()

    # 空白で分割し、空文字列を除外
    tokens = text.split()
    return [token for token in tokens if token]


def count_words(tokens: List[str]) -> Counter:
    """単語の出現頻度をカウントする。

    Args:
        tokens: 単語のリスト

    Returns:
        単語 -> 出現回数の Counter
    """
    return Counter(tokens)
