"""
Text frequency counter CLI tool.

Reads a text file, counts word occurrences (case-insensitive, punctuation stripped),
and prints the top N words in a tabular format.

Usage:
    python word_freq.py input.txt --top 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into words.

    Words are split on non-alphanumeric characters and converted to lowercase.
    Punctuation and special characters are stripped.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens.
    """
    # Split on non-word characters (letters, digits, underscores)
    # and convert to lowercase
    words = re.findall(r'\b[a-z0-9]+\b', text.lower())
    return words


def count_words(text: str) -> Counter[str]:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter mapping each word to its occurrence count.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words.

    Args:
        counter: A Counter of word frequencies.
        n: The number of top words to retrieve.

    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by count descending, then by word ascending
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
