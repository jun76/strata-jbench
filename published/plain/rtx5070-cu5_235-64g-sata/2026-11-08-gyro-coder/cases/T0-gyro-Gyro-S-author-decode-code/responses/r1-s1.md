"""
Top-N Word Frequency Counter CLI.

Reads a text file, counts word frequencies, and prints the top N words
in a tabular format.

Usage:
    python wordfreq.py input.txt --top 10
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_words(path: Path) -> List[str]:
    """
    Read words from a text file.

    Words are split by whitespace and stripped of surrounding punctuation.
    The file is read as UTF-8 text.

    Args:
        path: Path to the text file.

    Returns:
        List of lowercase words.
    """
    text = path.read_text(encoding="utf-8")
    words = []
    for raw in text.split():
        # Strip leading/trailing punctuation characters
        stripped = raw.strip(".,!?;:\'\"()[]{}<>")
        if stripped:
            words.append(stripped.lower())
    return words


def count_words(words: List[str]) -> Counter[str]:
    """
    Count the frequency of each word.

    Args:
        words: List of words.

    Returns:
        Counter mapping words to their frequencies.
    """
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Return the top N words by frequency, sorted descending by count
    then alphabetically for ties.

    Args:
        counter: Word frequency counter.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples.
    """
    items = list(counter.items())
    # Sort by count descending, then word ascending for stable tie-breaking
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
