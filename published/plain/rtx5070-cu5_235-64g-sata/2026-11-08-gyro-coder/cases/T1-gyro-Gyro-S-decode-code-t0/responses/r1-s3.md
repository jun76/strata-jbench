def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合、大文字小文字を区別する
        min_length: 単語の最小長（この長さ未満の単語は除外）

    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()

    # 単語を抽出: 英数字とハイフン/アポストロフィを含む連続文字列
    words = []
    current = []
    for char in text:
        if char.isalnum() or char in ("-", "'"):
            current.append(char)
        else:
            if current:
                word = "".join(current)
                if len(word) >= min_length:
                    words.append(word)
                current = []
    if current:
        word = "".join(current)
        if len(word) >= min_length:
            words.append(word)

    return words


def count_words(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数 の Counter
    """
    return Counter(words)
