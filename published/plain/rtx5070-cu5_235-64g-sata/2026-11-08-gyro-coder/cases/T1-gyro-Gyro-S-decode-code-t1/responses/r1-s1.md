"""Top-N word frequency counter CLI.

Reads a text file, counts word occurrences (case-insensitive), and prints
the top N words with their counts in a table format.
"""

import argparse
import sys
from collections import Counter
from typing import Dict, List


def count_words(text: str) -> Dict[str, int]:
    """Count word occurrences in text, case-insensitive, ignoring punctuation.

    Args:
        text: Input text to analyze.

    Returns:
        Dictionary mapping lowercase words to their occurrence counts.
    """
    words = text.lower().split()
    # Strip non-alphanumeric characters from each word
    cleaned: List[str] = []
    for word in words:
        cleaned_word = "".join(ch for ch in word if ch.isalnum())
        if cleaned_word:
            cleaned.append(cleaned_word)
    return dict(Counter(cleaned))


def top_n(counter: Dict[str, int], n: int) -> List[tuple[str, int]]:
    """Return the top n most frequent words.

    Args:
        counter: Mapping of words to counts.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples sorted by count descending, then alphabetically.
    """
    items = sorted(counter.items(), key=lambda kv: (-kv[1], kv[0]))
    return items[:n]


def format_table(rows: List[tuple[str, int]]) -> str:
    """Format word-count pairs as a fixed-width table.

    Args:
        rows: List of (word, count) tuples.

    Returns:
        Multi-line string with header and formatted rows.
    """
    if not rows:
        return "No words found."

    word_width = max(len(w) for w, _ in rows)
    header = f"{'Word':<{word_width}} {'Count':>6}"
    separator = f"{'-' * word_width} {'-' * 6}"
    lines = [header, separator]
    for word, count in rows:
        lines.append(f"{word:<{word_width}} {count:>6}")
    return "\n".join(lines)
