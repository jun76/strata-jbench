Top-word frequency analyzer CLI tool.

Reads a text file, counts word frequencies, and displays the top N words
in a tabular format.
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple, Optional


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Extract words (sequences of letters/digits/underscores) and convert to lowercase
    words = re.findall(r'[a-zA-Z0-9_]+', text)
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter mapping each word to its occurrence count.
    """
    return Counter(words)


def get_top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words, sorted by count descending.

    For ties, sort alphabetically.

    Args:
        counter: A Counter mapping words to counts.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples sorted by count descending,
        then by word ascending for tie-breaking.
    """
    # Sort by count descending, then by word ascending for tie-breaking
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
