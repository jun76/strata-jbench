A command-line tool that reads a text file, counts word frequencies,
and displays the top N words in a table format.

Usage:
    python word_freq.py <input_file> [--top N] [--case-sensitive] [--no-sort]

Arguments:
    input_file: Path to the text file to analyze.
    --top N:    Show only the top N most frequent words (default: 10).
    --case-sensitive: Preserve case distinctions (default: case-insensitive).
    --no-sort: Do not sort results; preserve order of first appearance.
"""

import argparse

from collections import Counter
from typing import List, Tuple


def tokenize_text(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize text into words by splitting on whitespace and stripping punctuation-like characters.

    Args:
        text: Input text string.
        case_sensitive: If False, words are lowercased for counting.

    Returns:
        A list of words (tokens).
    """
    words = []
    for word in text.split():
        # Strip leading/trailing non-alphanumeric characters to treat "hello," as "hello"
        cleaned = word.strip(".,;!?\"'()[]{}<>`~#@$%^&-")
        if cleaned:
            if not case_sensitive:
                cleaned = cleaned.lower()
            words.append(cleaned)
    return words
