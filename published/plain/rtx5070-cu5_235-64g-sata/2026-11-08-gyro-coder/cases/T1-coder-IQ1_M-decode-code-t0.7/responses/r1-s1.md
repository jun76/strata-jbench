"""
word_freq.py - A command-line tool to calculate word frequency from text files.
Usage: python word_freq.py <file> [-n N] [-c]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """Extract words from text, converting to lowercase and removing punctuation."""
    # Match sequences of word characters (letters, digits, underscores)
    return [word.lower() for word in re.findall(r'\w+', text)]


def count_frequencies(words: List[str]) -> Counter:
    """Count the frequency of each word."""
    return Counter(words)


def get_top_n(frequencies: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words.
    
    Args:
        frequencies: A Counter object with word frequencies.
        n: The number of top items to return.
    
    Returns:
        A list of tuples (word, count), sorted by count descending.
        If counts are equal, words are sorted lexicographically.
    """
    # Counter.most_common sorts by count descending, then insertion order.
    # To ensure deterministic lexicographic sort for ties, we sort explicitly.
    items = list(frequencies.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]


def main() -> None:
    """Main function to parse arguments and process the file."""
    parser = argparse.ArgumentParser(
        description="Calculate word frequency from a text file and display top N words."
    )
    parser.add_argument(
        "file",
        help="Path to the input text file"
    )
    parser.add_argument(
        "-n",
        type=int,
        default=10,
        help="Number of top words to display (default: 10)"
    )
    parser.add_argument(
        "-c",
        action="store_true",
        help="Include case-sensitive counting (default is case-insensitive)"
    )
    
    args = parser.parse_args()
