"""Command-line tool that counts word frequencies in a text file
and prints the top-N words as a table.

Usage:
    python word_count.py <file> [-n N]

The tool reads a text file, tokenizes it into words (case-insensitive,
punctuation stripped), counts how often each word appears, and prints
the ``N`` most frequent words (default N = 10) as an aligned table.
"""

def tokenize(text: str) -> list[str]:
    """Split *text* into lowercase alphabetic word tokens.

    Punctuation is stripped from the ends of each whitespace-delimited token
    (e.g. ``"don't"`` -> ``"don't"``; ``"hello,"`` -> ``"hello"``;
    ``"(hello)"`` -> ``"hello"``). Non-alphabetic tokens are discarded.

    Parameters:
        text: the raw text to tokenize.

    Returns:
        A list of lowercase word tokens (may be empty).
    """
    tokens: list[str] = []
    for raw in text.split():
        word = raw.strip(string.punctuation)
        if word and word.isalpha():
            tokens.append(word.lower())
    return tokens


def count_words(text: str) -> Counter[str]:
    """Return a frequency ``Counter`` of the words in *text*.

    Parameters:
        text: the raw text to analyze.

    Returns:
        A :class:`collections.Counter` mapping each lowercase word to its
        occurrence count.
    """
    return Counter(tokenize(text))
