#!/usr/bin/env python3
"""
word_freq.py

A command-line tool that reads a text file, counts word frequencies,
and prints the top N most frequent words in table form.

Usage:
    python word_freq.py [-h] [-n N] [-w WIDTH] [--case-sensitive] input_file

Examples:
    python word_freq.py sample.txt
    python word_freq.py -n 10 sample.txt
    python word_freq.py --case-sensitive -w 15 sample.txt

Dependencies:
    Python standard library only.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Parameters:
        text: The input text.
        case_sensitive: If False, convert all words to lowercase.

    Returns:
        A list of words.
    """
    if not case_sensitive:
        text = text.lower()
    # Extract sequences of alphanumeric characters and underscores as words.
    words = re.findall(r"[a-zA-Z0-9_]+", text)
    return words


def count_frequencies(words: List[str]) -> Counter:
    """
    Count occurrences of each word in the list.

    Parameters:
        words: List of words.

    Returns:
        A Counter mapping each word to its frequency.
    """
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Return the top N most common words with their counts.

    Parameters:
        counter: A Counter object.
        n: Number of top items to return.

    Returns:
        A list of (word, count) tuples, sorted by descending frequency.
    """
    return counter.most_common(n)
