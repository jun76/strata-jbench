    """

    """
    return re.findall(r"[a-z0-9]+", text.lower())


def count_words(iterable: Iterable[str]) -> collections.Counter[str]:
    """
    単語の出現頻度を Counter で集計する。
    """
    return collections.Counter(iterable)
