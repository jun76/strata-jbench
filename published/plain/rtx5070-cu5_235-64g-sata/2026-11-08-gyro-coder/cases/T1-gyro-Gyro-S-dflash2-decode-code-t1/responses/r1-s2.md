def count_words(text: str, ignore_case: bool = False) -> List[Tuple[str, int]]:
    """テキストから単語出現頻度を集計する。

    Args:
        text: 対象テキスト
        ignore_case: True の場合、大文字小文字を無視して集計する

    Returns:
        (word, count) のタブルのリスト。頻度が高い順、同頻度の場合は辞書順でソート。

    Raises:
        None
    """
    words = []
    if ignore_case:
        # 大文字小文字を無視して集計
        normalized = text.lower()
        # 単語を分割（空白・改行・区切り文字で分割）
        raw_tokens = normalized.split()
        words = raw_tokens
    else:
        raw_tokens = text.split()
        words = raw_tokens

    counter = Counter(words)
    # 頻度が高い順、同頻度の場合は辞書順（昇順）でソート
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items
