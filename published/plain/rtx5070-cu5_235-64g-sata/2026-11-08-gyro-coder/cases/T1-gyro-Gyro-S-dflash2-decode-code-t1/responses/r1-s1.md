"""単語出現頻度を集計し、上位N件を表形式で表示するコマンドラインツール。

Usage:
    python word_counter.py <file_path> [--top N] [--case-sensitive]
"""
import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple

# 単語抽出パターン（英数字とハイフン/アポストローフを含む）
_WORD_PATTERN = re.compile(r"[a-z0-9]+(?:['][a-z]+)?(?:-[a-z]+)*", re.IGNORECASE)


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: Trueなら大文字/小文字を区別して抽出

    Returns:
        抽出された単語のリスト
    """
    words = _WORD_PATTERN.findall(text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_word_frequencies(text: str, case_sensitive: bool = False) -> Counter[str]:
    """テキスト内の単語出現頻度をCounterで集計する。

    Args:
        text: 入力テキスト
        case_sensitive: Trueなら大文字/小文字を区別して集計

    Returns:
        単語 -> 出現回数 のCounter
    """
    words = extract_words(text, case_sensitive)
    return Counter(words)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """頻度上位N件の単語と回数を返す。

    Args:
        counter: 頻度Counter
        n: 上位件数

    Returns:
        (word, count) のリスト（頻度降順）
    """
    return counter.most_common(n)
