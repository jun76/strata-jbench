def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    """
    Merge overlapping or touching closed intervals.

    Intervals are considered touching if one interval's start equals
    another interval's end, e.g. [1, 2] and [2, 3] merge to [1, 3].

    The input is not modified.
    """
    if not intervals:
        return []

    sorted_intervals = sorted(intervals, key=lambda interval: (interval[0], interval[1]))

    merged: list[list[int]] = []

    for start, end in sorted_intervals:
        if not merged:
            merged.append([start, end])
        else:
            last_start, last_end = merged[-1]

            if start <= last_end:
                merged[-1][1] = max(last_end, end)
            else:
                merged.append([start, end])

    return merged
