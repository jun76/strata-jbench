def read_text_file(path: str) -> str:
    """
    テキストファイルを読み込み、その内容を文字列として返す。

    Args:
        path: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        OSError: ファイル読み取りに失敗した場合。
    """
    if not os.path.isfile(path):
        raise FileNotFoundError(f"File not found: {path}")
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 分割対象のテキスト。
        case_sensitive: True の場合、大文字/小文字を区別する。
                        False の場合、すべて小文字に正規化して区別しない。

    Returns:
        単語のリスト。
    """
    # 非アルファベット・数字を区切りとして扱う（簡易的なトークン化）
    # 単語は英数字からなる連続部分列
    words = []
    current = []
    for ch in text:
        if ch.isalnum():
            current.append(ch)
        else:
            if current:
                words.append("".join(current))
                current = []
    if current:
        words.append("".join(current))

    if not case_sensitive:
        words = [w.lower() for w in words]

    return words
