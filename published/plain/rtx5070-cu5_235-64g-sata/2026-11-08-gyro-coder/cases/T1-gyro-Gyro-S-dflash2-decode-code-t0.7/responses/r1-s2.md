"""
コマンドラインツール: テキストファイルを読み込み、単語出現頻度を上位N件を表形式で表示する。

Usage:
    python word_frequency.py <file> [-n N] [--case-insensitive]

Example:
    python word_frequency.py sample.txt -n 10
"""

import argparse

from collections import Counter
from typing import List, Tuple

# 型ヒント用の別名
FrequencyTuple = Tuple[str, int]


def tokenize(text: str, case_insensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_insensitive: True の場合、単語を小文字化して集計する

    Returns:
        単語のリスト
    """
    if case_insensitive:
        text = text.lower()
    # 基本的な分割: 空白区切り。非アルファベット文字を除去して単語抽出
    words = []
    for token in text.split():
        # 句読点等を除去して純粋な単語部分を取得
        word = "".join(ch for ch in token if ch.isalnum())
        if word:
            words.append(word)
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語出現頻度を集計する。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数の Counter
    """
    return Counter(words)
