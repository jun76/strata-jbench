def tokenize(text: str, *, case_sensitive: bool = False) -> list[str]:
    """テキストを空白区切りの単語リストに分割する。

    Args:
        text: 対象テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。
                        False の場合、小文字に正規化する。

    Returns:
        単語のリスト（正規化済み）。
    """
    if not case_sensitive:
        text = text.lower()
    # 空白・改行で分割し、空文字を除外
    words = text.split()
    return words
