def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    単語は英数字（およびハイフン・アポストロフィを含む）の連続として定義される。
    日本語などの CJK 文字は 1 文字ずつ単語として扱う。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。

    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()

    # 英数字・ハイフン・アポストロフィの連続、または CJK 文字 1 文字
    pattern = r"[a-zA-Z0-9]+(?:[-'][a-zA-Z0-9]+)*|[\u4e00-\u9fff\u3040-\u309f\u30a0-\u30ff]"
    tokens = re.findall(pattern, text)
    return tokens


def count_words(text: str, case_sensitive: bool = False) -> Dict[str, int]:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか

    Returns:
        単語 -> 出現回数の辞書
    """
    tokens = tokenize(text, case_sensitive=case_sensitive)
    return dict(Counter(tokens))
