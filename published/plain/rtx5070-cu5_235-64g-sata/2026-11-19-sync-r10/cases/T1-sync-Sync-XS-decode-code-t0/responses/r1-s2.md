# 英数字・ハイフン・アポストロフィを単語として扱う（例: don't, e-mail, 2024）
_WORD_RE = re.compile(r"[A-Za-z0-9]+(?:[-'][A-Za-z0-9]+)*")


def extract_words(text: str, *, case_sensitive: bool = False) -> List[str]:
    """テキストから単語のリストを抽出する。

    Args:
        text: 対象テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
                        False の場合、すべて小文字に正規化する。

    Returns:
        抽出された単語のリスト（出現順）。
    """
    words = _WORD_RE.findall(text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


# ---------------------------------------------------------------------------
# 頻度集計
# ---------------------------------------------------------------------------

def count_words(words: List[str]) -> Counter[str]:
    """単語リストから出現頻度を数える。

    Args:
        words: 単語のリスト。

    Returns:
        Counter[str] — 単語をキー、出現回数を値とする。
    """
    return Counter(words)
