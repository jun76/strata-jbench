wordfreq.py — テキストファイルを読み込み、単語の出現頻度を上位 N 件まで表形式で表示する CLI ツール。

使い方:
    python wordfreq.py input.txt --top 10

    python wordfreq.py input.txt --top 10 --output csv
"""

import argparse

from collections import Counter

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字化して集計する。

    Returns:
        単語トークンのリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    tokens = re.findall(r"[a-zA-Z0-9]+(?:[-'][a-zA-Z0-9]+)*", text)
    return tokens


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """テキスト内の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字の区別。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    tokens = tokenize(text, case_sensitive=case_sensitive)
    return Counter(tokens)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (word, count) を返す。

    Args:
        counter: 単語頻度の Counter。
        n: 上位件数。

    Returns:
        (word, count) のタプルリスト。
    """
    return counter.most_common(n)
