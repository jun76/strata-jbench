# apostrophes or hyphens (e.g., "don't", "state-of-the-art")
WORD_PATTERN = re.compile(r"\b[\w'-]+\b")


def normalize_word(token: str) -> str:
    """Normalize a token to lowercase for counting.

    Args:
        token: A raw word token extracted from text.

    Returns:
        The lowercased token.
    """
    return token.lower()

    """Split text into normalized words.

    Args:

        ignore_punctuation: If True, strip leading/trailing punctuation from
            each token before counting. If False, use raw tokens as-is.

    Returns:
        A list of normalized word strings.
    """
    tokens = WORD_PATTERN.findall(text)
    if not ignore_punctuation:
        return [normalize_word(t) for t in tokens]

    cleaned = []

        stripped = t.strip(".,;:!?\u201C\u201D\u2018\u2019-")  # common edge punctuation
        if stripped:
            cleaned.append(normalize_word(stripped))
    return cleaned
