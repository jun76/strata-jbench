import argparse
import sys
from collections import Counter
from typing import List, Tuple
from pathlib import Path
import re


def extract_words(text: str) -> List[str]:
    """
    Extract words from text. Words are sequences of alphanumeric characters
    (letters and digits), separated by non-word characters.
    Words are converted to lowercase for consistent counting.
    
    Args:
        text: Input text string
        
    Returns:
        List of extracted words (lowercase)
    """
    # Extract sequences of word characters (letters, digits, underscore)
    # Convert to lowercase for consistent counting
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    Count occurrences of each word.
    
    Args:
        words: List of words
        
    Returns:
        Counter object with word frequencies
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get top N most common words.
    
    Args:
        counter: Counter object with word frequencies
        n: Number of top words to retrieve
        
    Returns:
        List of tuples (word, count) sorted by count descending
    """
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]]) -> str:
    """
    Format word frequency entries as a table string.
    
    Args:
        entries: List of tuples (word, count)
        
    Returns:
        Formatted table as string
    """
    if not entries:
        return ""
    
    words = [word for word, _ in entries]
    counts = [count for _, count in entries]
    
    max_word_len = max(len(w) for w in words) if words else 0
    max_count_len = max(len(str(c)) for c in counts) if counts else 0
    
    header_word = "Word".ljust(max_word_len)
    header_count = "Count".rjust(max_count_len)
    
    lines = []
    lines.append(f"{header_word}  {header_count}")
    lines.append(f"{header_word}  {header_count}")
