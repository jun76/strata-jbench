Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words in a table format.
"""

def count_words(text: str) -> Counter[str]:
    """
    Count the frequency of each word in the given text.

    Words are extracted by splitting on whitespace and stripping common punctuation.
    The comparison is case-insensitive.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter mapping each word (lowercase) to its occurrence count.
    """
    # Normalize case and split on whitespace
    words = text.lower().split()

    # Strip common punctuation from each word
    cleaned_words = []
    for word in words:
        # Strip leading and trailing punctuation characters
        stripped = word.strip(".,;:!?\"'()[]{}<>-_~`@#$%^&*+=/\\|")
        if stripped:
            cleaned_words.append(stripped)

    return Counter(cleaned_words)
