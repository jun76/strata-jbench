"""wordfreq: Count word frequencies in a text file and display top N results as a table."""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Extract words from text by splitting on non-alphanumeric characters.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    return [word.lower() for word in re.findall(r'[a-zA-Z]+', text)]


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter object mapping each word to its frequency.
    """
    words = tokenize(text)
    return Counter(words)
