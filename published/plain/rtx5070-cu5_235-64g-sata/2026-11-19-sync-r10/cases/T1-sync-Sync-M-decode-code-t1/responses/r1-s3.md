#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルから単語出現頻度を計算し、上位 N 件を表形式で表示する。

Usage:
    python wordfreq.py <input_file> [-n TOP_N] [--width WIDTH]

Example:
    python wordfreq.py sample.txt -n 10
"""

import argparse
import sys
from collections import Counter
from pathlib import Path


def load_text(path: Path) -> str:
    """
    テキストファイルを読み込んで文字列を返す。

    Args:
        path: 読み込むテキストファイルのパス

    Returns:
        ファイルの内容を文字列として

    Raises:
        FileNotFoundError: ファイルが存在しない場合
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> list[str]:
    """
    テキストを単語に分割する。
    小文字化し、アルファベットと数字のみを単語として抽出する。

    Args:
        text: 分割対象のテキスト

    Returns:
        単語のリスト（重複含む）
    """
    words: list[str] = []
    current_word: list[str] = []

    for char in text:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                words.append("".join(current_word).lower())
                current_word = []

    if current_word:
        words.append("".join(current_word).lower())

    return words


def count_words(text: str) -> Counter:
    """
    テキストから単語出現頻度をカウントする。

    Args:
        text: 分析対象のテキスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    return Counter(tokenize(text))
