def normalize_word(word: str) -> str:
    """Normalize a word to lowercase and strip surrounding punctuation."""
    return word.lower().strip().strip(".,;:!?\"'()[]{}")


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Parameters
    ----------
    text : str
        Input text to count words from.

    Returns
    -------
    Counter
        A collections.Counter mapping normalized words to their counts.
    """
    words = text.split()
    normalized = [normalize_word(w) for w in words]
    # Filter out empty strings resulting from stripping
    filtered = [w for w in normalized if w]
    return Counter(filtered)
