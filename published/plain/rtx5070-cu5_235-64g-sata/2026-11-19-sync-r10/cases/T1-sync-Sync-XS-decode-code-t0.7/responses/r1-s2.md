def read_file(filepath: str) -> str:
    """Read a text file and return its contents.

    Args:
        filepath: Path to the text file.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If there is an error reading the file.
    """
    path = Path(filepath)
    if not path.exists():
        raise FileNotFoundError(f"File not found: {filepath}")
    try:
        return path.read_text(encoding="utf-8")
    except UnicodeDecodeError:
        # Try latin-1 as a fallback
        return path.read_text(encoding="latin-1")


def count_words(text: str, ignore_case: bool = True) -> Counter:
    """Count word occurrences in the given text.

    Words are defined as sequences of alphanumeric characters (including underscores).
    Non-alphanumeric characters (except underscores) are treated as separators.

    Args:
        text: The input text to count words in.
        ignore_case: If True, convert all words to lowercase before counting.

    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    import re

    # Extract words: sequences of alphanumeric characters and underscores
    words = re.findall(r"[a-zA-Z0-9_]+", text)

    if ignore_case:
        words = [w.lower() for w in words]

    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.

    Args:
        counter: A Counter object containing word counts.
        n: The number of top entries to retrieve.

    Returns:
        A list of (word, count) tuples sorted by descending count.
    """
    return counter.most_common(n)
