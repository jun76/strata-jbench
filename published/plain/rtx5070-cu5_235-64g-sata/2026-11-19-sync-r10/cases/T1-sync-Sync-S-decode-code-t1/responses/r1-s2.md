_DEFAULT_SPLIT_RE = re.compile(r"[^\w']+")


def tokenize(text: str, *, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """テキストを単語列に分解する。

    Args:
        text: 解析対象のテキスト。
        case_sensitive: 真なら大文字小文字を区別する。デフォルトは False(全て小文字化)。
        min_len: 保持する最小単語長。1 未満は 1 に丸める。

    Returns:
        分解された単語リスト。句読点は除去され、連続空白・改行は区切り扱い。

    Example:
        >>> tokenize("Hello, World! Hello again")
        ['hello', 'world', 'hello', 'again']
    """
    if min_len < 1:
        min_len = 1
    lowered = text if case_sensitive else text.lower()
    tokens = _DEFAULT_SPLIT_RE.split(lowered)
    # 先頭末尾のハイフン・アポストロフィも無視し、最短長をフィルタ
    cleaned = [t.strip("'") for t in tokens if len(t.strip("'")) >= min_len]
    return cleaned
