#!/usr/bin/env python3
"""
wordfreq.py - コマンドラインツール：テキストファイルの単語出現頻度を上位N件表示。

使い方:
    python wordfreq.py input.txt -n 10
    python wordfreq.py input.txt -n 5 --case-sensitive
"""

from __future__ import annotations

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple

import re


def tokenize_text(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストを単語にトークン化する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: True の場合、大文字小文字を区別する。
                      False（デフォルト）の場合、小文字に変換して区別しない。

    Returns:
        単語のリスト（非英数字の区切りで分割、前後の空白除去）。
    """
    # 英数字のみを含む連続部分単語を抽出（区切りは非英数字）
    tokens = re.findall(r"[A-Za-z0-9]+", text)
    if not case_sensitive:
        tokens = [token.lower() for token in tokens]
    return tokens


def count_word_frequencies(tokens: List[str]) -> Counter:
    """トークンリストの出現頻度を Counter で集計する。

    Args:
        tokens: 単語のリスト。

    Returns:
        各単語の出現回数を格納した Counter。
    """
    return Counter(tokens)

    """
