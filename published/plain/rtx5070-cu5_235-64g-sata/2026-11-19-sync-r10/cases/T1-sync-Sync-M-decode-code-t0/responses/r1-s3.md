#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を上位 N 件表示する。

Usage:
    python word_freq.py <file> [-n N] [--case-sensitive] [--min-length L]
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するか
        min_length: 単語の最小長

    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()

    words: List[str] = []
    current_word: List[str] = []

    for char in text:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                word = ''.join(current_word)
                if len(word) >= min_length:
                    words.append(word)
                current_word = []

    if current_word:
        word = ''.join(current_word)
        if len(word) >= min_length:
            words.append(word)

    return words


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するか
        min_length: 単語の最小長

    Returns:
        単語と出現回数の Counter
    """
    words = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(words)
