def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text:           対象テキスト
        case_sensitive: True の場合、大文字・小文字を区別して集計する。
                        False の場合、すべて小文字に正規化してから集計する。
        min_len:        この長さ未満のトークンを除外する最小文字数。

    Returns:
        抽出された単語トークンのリスト（出現順）。

    Examples:
        >>> tokenize("Hello, world! hello World.")
        ['hello', 'world', 'hello', 'world']
        >>> tokenize("Hello, world! hello World.", case_sensitive=True)
        ['Hello', 'world', 'hello', 'World']
        >>> tokenize("a bb ccc", min_len=2)
        ['bb', 'ccc']
    """
    if not case_sensitive:
        text = text.lower()

    # 英数字のみを単語として抽出（アンダースコア・ハイフンは区切り）
    tokens = re.findall(r"[a-zA-Z0-9]+", text)

    if min_len > 1:
        tokens = [t for t in tokens if len(t) >= min_len]

    return tokens
