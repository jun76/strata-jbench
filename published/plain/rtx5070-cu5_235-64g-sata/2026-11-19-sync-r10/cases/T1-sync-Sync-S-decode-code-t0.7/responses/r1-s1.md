def tokenize(text: str, *, case_sensitive: bool = False) -> list[str]:
    """
    テキストから単語トークンを抽出する。

    単語は、英数字とハイフン/アポストロフィで構成される連続した文字列とみなす。
    例: "don't", "state-of-the-art", "café" なども単語として扱われる。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。

    Returns:
        単語のリスト（出現順）。
    """
    # Unicode の文字分類を利用しつつ、単語境界を処理する簡易トークナイザ
    import re

    # 英数字 + ハイフン + アポストロフィ + Unicode 文字を単語として認識
    # 正規表現: [\w\-']+ は Unicode 文字も \w に含まれる (Python 3.7+)
    pattern = r"[a-zA-Z0-9\-']+" if not case_sensitive else r"[a-zA-Z0-9\-']+"

    # Unicode 文字も考慮したパターン
    # \w は Unicode 文字を含むが、ハイフンとアポストロフィを追加
    token_pattern = r"[\w\-']+"

    if not case_sensitive:
        text = text.lower()

    return re.findall(token_pattern, text)
