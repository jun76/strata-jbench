```python
"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位 N 件表示する。

使い方:
    python word_freq.py -f input.txt -n 10
    python word_freq.py --file input.txt --top 10
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def normalize_word(word: str) -> str:
    """単語を正規化する（小文字化、句読点除去）。

    Args:
        word: 元の単語文字列

    Returns:
        正規化された単語（小文字、句読点なし）。空文字の場合 None を返すように呼び出し側で処理する。
    """
    # 句読点や特殊文字を除去して小文字化
    cleaned = "".join(ch.lower() if ch.isalnum() else "" for ch in word)
    return cleaned


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割し、正規化する。

    Args:
        text: 入力テキスト

    Returns:
        正規化された単語のリスト（空文字は除外）
    """
    words: List[str] = []
    for raw_word in text.split():
        normalized = normalize_word(raw_word)
        if normalized:
            words.append(normalized)
    return words


def count_word_freq(words: List[str]) -> Counter[str]:
    """単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数 の Counter
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """上位 N 件の単語と頻度のペアを取得する。

    Args:
        counter: 単語頻度 Counter
        n: 上位件数

    Returns:
        (単語, 頻度) のタプルリスト（頻度降順、同頻度時は辞書順）
    """
    # 頻度降順、同頻度時は単語の辞書順で安定ソート
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
