from typing import List, Tuple

# 型エイリアス
WordCounts = List[Tuple[str, int]]


def count_words(text: str, case_insensitive: bool = False) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_insensitive: True の場合、大文字小文字を区別しない

    Returns:
        単語 -> 出現回数 の Counter
    """
    # 単語を分割: 空白区切り
    words = text.split()
    
    if case_insensitive:
        words = [w.lower() for w in words]
    
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> WordCounts:
    """
    Counter から上位 n 件の (word, count) のリストを取得する。
    同数値の場合、辞書順（昇順）で安定してソートする。

    Args:
        counter: 単語出現頻度の Counter
        n: 上位件数

    Returns:
        [(word, count), ...] のリスト
    """
    # 頻度降順、同頻度なら辞書順昇順で安定ソート
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
