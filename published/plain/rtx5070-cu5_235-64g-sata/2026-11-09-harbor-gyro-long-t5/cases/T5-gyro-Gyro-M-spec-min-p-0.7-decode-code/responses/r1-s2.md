def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: Input string to tokenize.
        case_sensitive: If True, preserve case. If False, lowercase all words.
        min_length: Minimum length of a word to include.

    Returns:
        List of tokens (words) meeting the criteria.
    """
    if not case_sensitive:
        text = text.lower()
    # Split on non-word characters
    tokens = re.findall(r"[a-zA-Z]+", text)
    # Filter by minimum length
    tokens = [t for t in tokens if len(t) >= min_length]
    return tokens


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: Input string.
        case_sensitive: Whether to preserve case.
        min_length: Minimum word length to count.

    Returns:
        Counter mapping words to their frequencies.
    """
    tokens = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int = 10) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: Counter of word frequencies.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples sorted by frequency descending.
    """
    return counter.most_common(n)
