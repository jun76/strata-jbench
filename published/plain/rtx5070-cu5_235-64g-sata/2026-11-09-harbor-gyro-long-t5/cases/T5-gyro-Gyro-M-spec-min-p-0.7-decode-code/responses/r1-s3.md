"""
word_freq: コマンドラインツールでテキストファイルから単語出現頻度を集計し、
上位 N 件を表形式で表示する。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。英数字および日本語文字を抽出し、
    すべて小文字に変換する。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト
    """
    # 英数字と日本語文字（ひらがな、カタカナ、漢字）を抽出
    pattern = r'[a-zA-Z0-9\u3040-\u309F\u30A0-\u30FF\u4E00-\u9FFF]+'
    words = re.findall(pattern, text.lower())
    return words


def count_frequencies(text: str) -> Dict[str, int]:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語 -> 出現回数 の辞書
    """
    words = tokenize(text)
    counter = Counter(words)
    return dict(counter)


def get_top_n(counter: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 N 件を取得する。

    Args:
        counter: 単語 -> 出現回数 の辞書
        n: 取得する件数

    Returns:
        (単語, 出現回数) のタプルのリスト、出現回数降順
    """
    sorted_items = sorted(counter.items(), key=lambda x: x[1], reverse=True)
    return sorted_items[:n]
