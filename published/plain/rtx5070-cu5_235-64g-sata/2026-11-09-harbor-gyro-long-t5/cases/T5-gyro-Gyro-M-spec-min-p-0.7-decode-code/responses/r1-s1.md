"""
Command-line tool to read a text file and count word frequencies.

Displays the top N most frequent words in a tabular format.

import argparse

def read_text_file(filepath: str) -> str:
    """
    Read the contents of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """
    Extract words from text, converting to lowercase.

    Words are defined as sequences of alphanumeric characters.
    Non-alphanumeric characters are treated as separators.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Match sequences of alphanumeric characters (letters and digits)
    words = re.findall(r"[a-zA-Z0-9]+", text.lower())
    return words


def count_word_frequencies(text: str) -> Dict[str, int]:
    """
    Count the frequency of each word in the given text.

    Args:
        text: Input text string.

    Returns:
        A dictionary mapping each word to its occurrence count.
    """
    words = extract_words(text)
    return dict(Counter(words))
