def count_words(text: str) -> Dict[str, int]:
    """Count occurrences of each word in the given text.

    Words are treated case-insensitively. Non-alphanumeric characters are
    treated as word separators.

    Args:
        text: The input text to analyze.

    Returns:
        A dictionary mapping lowercase words to their occurrence counts.
    """
    # Normalize to lowercase
    text = text.lower()

    # Replace non-alphanumeric characters with spaces, then split
    words = []
    current_word = []
    for char in text:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    if current_word:
        words.append(''.join(current_word))

    counter: Counter[str] = Counter(words)
    return dict(counter)


def get_top_n(counter: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words.

    Args:
        counter: A dictionary mapping words to their counts.
        n: The number of top entries to return.

    Returns:
        A list of (word, count) tuples sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
