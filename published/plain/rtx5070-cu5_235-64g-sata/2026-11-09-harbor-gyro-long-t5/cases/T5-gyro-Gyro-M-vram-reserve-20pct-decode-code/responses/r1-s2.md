"""
Word frequency counter command-line tool.

This module provides a CLI tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text string to analyze.

    Returns:
        A Counter object mapping each word (lowercased) to its occurrence count.
    """
    # Extract words using regex: sequences of alphanumeric characters or underscores
    words = re.findall(r'\b\w+\b', text.lower())
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object containing word frequencies.
        n: The number of top entries to return.

    Returns:
        A list of tuples (word, count) sorted by count descending.
    """
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]], title: str = "Top Words") -> str:
    """
    Format word frequency entries as a simple ASCII table.

    Args:
        entries: A list of tuples (word, count).
        title: The title to display above the table.

    Returns:
        A formatted string representing the table.
    """
    if not entries:
        return "No words found."

    # Determine column widths
    max_word_len = max(len(word) for word, _ in entries) if entries else 0
    word_col_width = max(max_word_len, len("Word"))
    count_col_width = max(len(str(count)) for _, count in entries) if entries else 0
    count_col_width = max(count_col_width, len("Count"))

    # Build separator lines
    total_width = word_col_width + count_col_width + 3  # +1 space before word, +1 space between, +1 space after count
