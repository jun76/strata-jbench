"""
word_count: A command-line tool that reads a text file, counts word frequencies,
and displays the top N words in a table format.
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_text_file(filepath: Path) -> str:
    """
    Read the contents of a text file and return it as a string.

    Args:
        filepath: Path to the text file.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If there is an error reading the file.
    """
    try:
        return filepath.read_text(encoding="utf-8")
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except IOError as e:
        raise IOError(f"Error reading file {filepath}: {e}")


def count_words(text: str) -> Counter:
    """
    Count the frequency of each word in the given text.

    Words are split by whitespace, and converted to lowercase for case-insensitive counting.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    # Split by whitespace, convert to lowercase, filter out empty strings
    words = [word.lower() for word in text.split() if word]
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top n most common words from the counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
