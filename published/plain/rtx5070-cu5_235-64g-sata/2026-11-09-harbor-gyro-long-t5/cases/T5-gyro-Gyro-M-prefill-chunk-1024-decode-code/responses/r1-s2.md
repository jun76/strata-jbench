```python
"""
Command-line tool to count word frequencies from a text file and display top N results.

Usage:
    python word_freq_counter.py <file> [-n N] [-c CASE_INSENSITIVE]

Example:
    python word_freq_counter.py sample.txt -n 10
"""

import argparse
import collections
import re
import sys
from typing import Dict, List, Tuple


def count_words(text: str, case_insensitive: bool = False) -> Dict[str, int]:
    """Count the frequency of each word in the given text.

    Args:
        text: The input text string.
        case_insensitive: If True, treat words as case-insensitive.

    Returns:
        A dictionary mapping words to their occurrence counts.
    """
    # Extract words using regex (alphabetic characters, allowing apostrophes within words)
    # This matches sequences of letters possibly containing apostrophes (e.g., don't)
    word_pattern = re.compile(r"[a-zA-Z]+(?:'[a-zA-Z]+)?")
    words = word_pattern.findall(text)

    if case_insensitive:
        words = [word.lower() for word in words]

    counter: Dict[str, int] = collections.Counter(words)
    return counter


def get_top_n(counter: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words from the counter.

    Args:
        counter: Dictionary mapping words to counts.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending, then word ascending.
    """
    # Sort by count descending, then by word ascending for ties
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(items: List[Tuple[str, int]], header: str = "Word\tCount") -> str:
    """Format the top N items as a simple text table.

    Args:
        items: List of (word, count) tuples.
        header: Header line for the table.

    Returns:
        A formatted string representation of the table.
    """
    if not items:
        return header + "\n(No words found)"
