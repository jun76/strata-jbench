#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込んで単語の出現頻度を集計し、
上位 N 件を表形式で表示する。

Usage:
    python word_counter.py <file> [--top N] [--case-sensitive] [--min-length L]

import argparse
import sys
from collections import Counter
from typing import List, Tuple, Optional
import re


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 単語の最小長。これ未満の単語は除外される。

    Returns:
        単語のリスト（順序は出現順）。
    """
    # 単語を抽出: アルファベット文字の連続（英語想定）
    # 日本語対応も考慮して、Unicode文字列も許可する
    # 簡易的に: 英数字と日本語文字を含む連続文字列
    pattern = r"[a-zA-Z0-9]+|[ぁ-んァ-ン一-龥]+"
    words = re.findall(pattern, text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return [w for w in words if len(w) >= min_length]


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """
    テキスト中の単語出現頻度を集計する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 単語の最小長。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(words)
