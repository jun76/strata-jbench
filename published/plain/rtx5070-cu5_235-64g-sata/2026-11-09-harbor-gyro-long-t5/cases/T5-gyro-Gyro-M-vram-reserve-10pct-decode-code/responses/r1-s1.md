import argparse

def read_text_file(filepath: str) -> str:
    """Read the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        UnicodeDecodeError: If the file cannot be decoded.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def count_words(text: str) -> Counter:
    """Count the occurrences of each word in the given text.

    Words are split by whitespace and converted to lowercase.

    Args:
        text: The input text.

    Returns:
        A Counter object mapping words to their counts.
    """
    words = text.lower().split()
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.

    Args:
        counter: A Counter object mapping words to counts.
        n: The number of top entries to retrieve.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
