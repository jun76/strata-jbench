"""
Top N word frequency counter CLI tool.

Reads text files, counts word occurrences, and displays the top N words
in a table format.
"""

import argparse

from collections import Counter
from pathlib import Path

def count_words(text: str) -> Counter:
    """Count word occurrences in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping each lowercase word to its occurrence count.
    """
    words = text.lower().split()
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word counts.
        n: The number of top entries to return.

    Returns:
        A list of (word, count) tuples sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by count descending, then word ascending for stable tie-breaking
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
