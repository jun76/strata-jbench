"""
Command-line tool to count word frequencies in text files and display top N results.

Usage:
    python word_counter.py input_file.txt --top 10
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_text(file_path: Path) -> str:
    """Read text content from a file.

    Args:
        file_path: Path to the text file.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    return file_path.read_text(encoding="utf-8")


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.

    Args:
        text: Input text string.

    Returns:
        List of lowercase words (alphabetic sequences only).
    """
    # Extract sequences of alphabetic characters, converted to lowercase
    return re.findall(r"[a-z]+", text.lower())


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        Counter mapping each word to its frequency.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]]) -> str:
    """Format word frequency entries as a table string.

    Args:
        entries: List of (word, count) tuples.

    Returns:
        Formatted table string.
    """
    if not entries:
        return "No words found."

    lines: List[str] = []
    header = f"{'Word':<20} {'Count':>10}"
    separator = "-" * 30
    lines.append(header)
    lines.append(separator
