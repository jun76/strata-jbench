以下は、テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツールです。

```python
#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語の出現頻度を上位 N 件で表示する CLI ツール

使い方:
    python word_freq.py [options] <file>

例:
    python word_freq.py -n 10 example.txt
    python word_freq.py --ignore-case --min-length 3 --sort-by count example.txt
"""

import argparse
import sys
from collections import Counter
from typing import Dict, List, Tuple, Optional


def tokenize(text: str, ignore_case: bool = True, min_length: int = 1) -> List[str]:
    """
    テキストを単語に分割し、必要に応じて小文字化・長さ制限を適用する。

    Args:
        text: 入力テキスト
        ignore_case: True の場合、単語を小文字に統一する
        min_length: この長さ未満の単語は除外する（1以上）

    Returns:
        単語のリスト
    """
    # 簡易的なトークナイザー：空白・区切り文字で分割
    # 非アルファベット・数字を除去するため、正規表現を使わずに文字判定
    words: List[str] = []
    current: str = ""
    for char in text:
        if char.isalnum():
            current += char
        else:
            if current:
                words.append(current)
                current = ""
    if current:
        words.append(current)

    # 小文字化と長さ制限
    if ignore_case:
        words = [w.lower() for w in words]
    words = [w for w in words if len(w) >= min_length]
    return words


def count_words(words: List[str]) -> Dict[str, int]:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とする辞書
    """
    return dict(Counter(words))


def sort_counts(
    counts: Dict[str, int],
    sort_by: str = "count",
    descending: bool = True
) -> List
