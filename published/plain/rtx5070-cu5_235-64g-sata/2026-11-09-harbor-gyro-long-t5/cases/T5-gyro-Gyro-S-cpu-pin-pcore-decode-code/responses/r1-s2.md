import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """
    単語を正規化する。
    
    Args:
        word: 元の単語
        case_sensitive: 大文字・小文字を区別するかどうか
    
    Returns:
        正規化された単語
    """
    if not case_sensitive:
        return word.lower()
    return word


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    Args:
        text: 入力テキスト
    
    Returns:
        単語のリスト
    """
    # 単語を抽出: アルファベット、数字、ハイフンを含むトークン
    tokens = re.findall(r'[a-zA-Z0-9\-]+', text)
    return tokens


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    テキスト内の単語出現頻度を集計する。
    
    Args:
        text: 入力テキスト
        case_sensitive: 大文字・小文字を区別するかどうか
    
    Returns:
        単語と出現回数の Counter
    """
    tokens = tokenize(text)
    normalized = [normalize_word(token, case_sensitive) for token in tokens]
    return Counter(normalized)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    上位N件の単語と出現回数を取得する。
    
    Args:
        counter: 単語と出現回数の Counter
        n: 上位件数
    
    Returns:
        (単語, 回数) のタプルリスト
    """
    return counter.most_common(n)
