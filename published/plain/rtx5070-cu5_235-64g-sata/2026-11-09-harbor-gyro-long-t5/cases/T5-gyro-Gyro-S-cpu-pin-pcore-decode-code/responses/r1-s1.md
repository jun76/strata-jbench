"""
word_freq CLI: Count word frequencies in a text file and display top N entries.
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count occurrences of words in the given text.

    Words are split on whitespace and converted to lowercase.
    Only alphanumeric sequences are treated as words.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter mapping each word to its frequency.
    """
    import re
    # Split into words using regex to handle punctuation properly
    words = re.findall(r"[a-zA-Z0-9]+", text)
    return Counter(word.lower() for word in words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words from a Counter.

    Args:
        counter: A Counter of word frequencies.
        n: The number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by (-count, word) to get descending count and ascending word for ties
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
