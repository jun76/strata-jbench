#!/usr/bin/env python3
"""
word_freq.py — コマンドラインツール：テキストファイルから単語出現頻度を上位 N 件表示する。

使い方:
    python word_freq.py input.txt --top 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def load_text(path: str) -> str:
    """指定したパスのテキストファイルを読み込み、内容文字列を返す。"""
    try:
        with open(path, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {path}")


def tokenize(text: str) -> List[str]:
    """テキストを空白区切りの単語リストに分割する。"""
    return text.split()


def count_words(words: List[str]) -> Counter:
    """単語リストから出現頻度を Counter で集計する。"""
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counter 上位 n 件を (word, count) のタプルリストとして返す。"""
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """
    (word, count) のタプルリストを表形式の文字列に整形する。
    例:
        Word       Count
        --------   -----
        hello      3
        world      2
    """
    if not rows:
        return "（結果なし）"

    max_word_len = max(len(w) for w, _ in rows)
    header_word = "Word"
    header_count = "Count"
    word_width = max(max_word_len, len(header_word))
    count_width = 5  # 固定幅

    lines = []
    lines.append(f"{header_word:<{word_width}}   {header_count:>count_width}")
    lines.append("-" * word_width + "   " + "-" * count_width)
    for word, cnt in rows:
        lines.append(f"{word:<{word_width}}   {cnt:>count_width}")

    return "\n".join(lines)


def main(argv: List[str] | None = None) -> int
