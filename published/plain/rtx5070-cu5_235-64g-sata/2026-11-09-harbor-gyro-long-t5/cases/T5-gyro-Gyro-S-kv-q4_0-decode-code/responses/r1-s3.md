def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合、大文字小文字を区別する
        min_length: 単語の最小文字数 (これより短い単語は除外)

    Returns:
        単語のリスト
    """
    # 簡易的なトークン化: 英数字のみを含む連続文字列を単語として抽出
    words = []
    current_word = []
    for char in text:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                word = "".join(current_word)
                if len(word) >= min_length:
                    if not case_sensitive:
                        word = word.lower()
                    words.append(word)
                current_word = []
    # 最後の単語を処理
    if current_word:
        word = "".join(current_word)
        if len(word) >= min_length:
            if not case_sensitive:
                word = word.lower()
            words.append(word)
    return words
