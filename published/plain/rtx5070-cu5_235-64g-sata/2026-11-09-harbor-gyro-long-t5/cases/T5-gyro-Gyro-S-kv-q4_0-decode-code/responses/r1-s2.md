# 単語頻度集計コマンドラインツール

以下に、テキストファイルから単語の出現頻度を上位 N 件表形式で表示する Python コマンドラインツールと pytest 単体テストを提供します。

## 本体: `word_freq.py`

"""
word_freq.py - テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

使用方法:
    python word_freq.py input.txt
    python word_freq.py input.txt --top 20
    python word_freq.py input.txt --ignore-case
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def read_text_file(path: Path) -> str:
    """テキストファイルを読み込み、内容を文字列として返す。

    Args:
        path: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を文字列として返す。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        PermissionError: ファイルの読み取り権限がない場合。
        UnicodeDecodeError: ファイルが UTF-8 でデコードできない場合。
    """
    return path.read_text(encoding="utf-8")


def tokenize(text: str, ignore_case: bool = False) -> List[str]:
    """テキストを単語に分割する。

    Args:
        text: 分割対象のテキスト。
        ignore_case: True の場合、単語を小文字に変換する。

    Returns:
        単語のリスト。
    """
    words = text.split()
    if ignore_case:
        words = [w.lower() for w in words]
    return words


def count_words(words: List[str]) -> Dict[str, int]:
    """単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語 -> 出現回数の辞書。
    """
    return dict(Counter(words))
