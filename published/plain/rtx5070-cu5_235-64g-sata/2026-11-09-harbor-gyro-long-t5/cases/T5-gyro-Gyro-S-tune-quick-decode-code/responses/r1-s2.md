import argparse
import re

from collections import Counter
from typing import List, Tuple, TextIO


def read_words_from_file(path: str) -> List[str]:
    """
    テキストファイルから単語を抽出する。

    Args:
        path: テキストファイルのパス

    Returns:
        単語のリスト（正規表現で単語を分割）
    """
    with open(path, "r", encoding="utf-8") as f:
        content = f.read()
    # 単語を抽出: 英数字およびハイフンを含む語
    words = re.findall(r"[a-zA-Z0-9\-]+", content)
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語の出現頻度を集計する。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数 の Counter
    """
    counter: Counter = Counter()
    for word in words:
        counter[word.lower()] += 1
    return counter


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の単語と頻度を取得する。

    Args:
        counter: 単語頻度 Counter
        n: 取得する件数

    Returns:
        [(word, count), ...] のリスト（頻度降順）
    """
    return counter.most_common(n)
