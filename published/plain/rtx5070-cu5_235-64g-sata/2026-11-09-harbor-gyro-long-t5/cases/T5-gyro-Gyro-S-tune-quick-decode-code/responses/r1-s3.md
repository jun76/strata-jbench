"""
wordfreq.py — コマンドラインツールでテキストファイルから単語出現頻度を集計し、
上位N件をテーブル形式で表示する。

使い方:
    python wordfreq.py input.txt --top 10
    python wordfreq.py input.txt --top 20 --case-insensitive
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, *, case_insensitive: bool = True) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_insensitive: True の場合、小文字に正規化してから集計する

    Returns:
        単語のリスト（非正規化の単語は小文字化される）
    """
    # 単語は英数字とハイフン、アポストロフィを含む文字列として抽出
    pattern = r"[a-zA-Z0-9'’-]+" if case_insensitive else r"[a-zA-Z0-9'’-]+"
    words = re.findall(pattern, text)
    if case_insensitive:
        words = [w.lower() for w in words]
    return words


def count_words(text: str, *, case_insensitive: bool = True) -> Counter[str]:
    """
    テキスト中の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト
        case_insensitive: True の場合、小文字に正規化してから集計する

    Returns:
        単語 -> 出現回数 の Counter
    """
    tokens = tokenize(text, case_insensitive=case_insensitive)
    return Counter(tokens)
