    """

    Args:
        filepath: 読み込むファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み込みに失敗した場合。
    """
    return filepath.read_text(encoding="utf-8")


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。

    正規化ルール:
        - すべて小文字に変換

        - 空文字列は除外

        text: 入力テキスト。

    Returns:
        単語のリスト。
    """
    # 簡易的なトークン化: 空白・改行・タブなどで分割し、小文字化
    # より厳密なトークン化が必要なら、regex を使うことも可能
    words = text.lower().split()
    # 句読点や記号を含む単語はそのまま残す（例: "hello," → "hello,"）
    # 必要に応じて strip などの前処理を追加可能
    return [w for w in words if w]
