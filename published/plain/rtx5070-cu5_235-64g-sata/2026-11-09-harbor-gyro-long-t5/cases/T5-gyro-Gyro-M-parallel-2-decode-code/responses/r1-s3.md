"""
word_freq.py

A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize the given text into lowercase words.
    
    Words are defined as sequences of alphanumeric characters and underscores.
    All words are converted to lowercase for case-insensitive counting.
    
    Args:
        text: The input text to tokenize.
        
    Returns:
        A list of lowercase tokens (words).
    """
    # Find all sequences of alphanumeric characters and underscores
    tokens = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return [token.lower() for token in tokens]


def count_words(text: str) -> Dict[str, int]:
    """
    Count the occurrence of each word in the given text.
    
    Args:
        text: The input text to analyze.
        
    Returns:
        A dictionary mapping each word to its occurrence count.
    """
    tokens = tokenize(text)
    return dict(Counter(tokens))


def get_top_n(counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N words by frequency.
    
    If multiple words have the same frequency, they are sorted alphabetically.
    
    Args:
        counts: A dictionary mapping words to their counts.
        n: The number of top words to return.
        
    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(counts.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
