"""
コマンドラインツール: テキストファイルから単語出現頻度を上位 N 件表示する。

Usage:
    python wordfreq.py <file> --top N [--min-len M] [--no-case] [--json]
"""

import argparse

import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str, min_len: int, ignore_case: bool) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        min_len: 単語の最小長。これより短い単語は除外。
        ignore_case: True の場合、単語を小文字に変換して集計する。

    Returns:
        抽出された単語のリスト
    """
    # 英数字とアンダースコアのみを文字として扱い、それ以外を区切りとする
    # 日本語等も考慮したい場合は正規表現を調整する
    pattern = r"\b\w+\b"
    matches = re.findall(pattern, text)
    words = []
    for w in matches:
        if len(w) >= min_len:
            if ignore_case:
                w = w.lower()
            words.append(w)
    return words


def count_frequencies(words: List[str]) -> Counter[str]:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数の Counter
    """
    return Counter(words)
