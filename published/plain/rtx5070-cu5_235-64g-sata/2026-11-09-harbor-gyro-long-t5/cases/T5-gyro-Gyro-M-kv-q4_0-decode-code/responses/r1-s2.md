import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple

def tokenize(text: str) -> List[str]:
    """テキストから単語を抽出する。
    
    正規表現を用いて英字を含む単語のみを抽出し、小文字化して返す。
    
    Args:
        text: 入力テキスト
        
    Returns:
        小文字化された単語のリスト
    """
    # 英字（a-z, A-Z）とハイフンを含む連続文字列を単語として抽出
    # 数字や記号は除外する
    words = re.findall(r'[a-zA-Z]+', text)
    return [word.lower() for word in words]

def count_words(text: str) -> Counter:
    """テキスト中の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語をキー、出現回数を値とする Counter
    """
    words = tokenize(text)
    return Counter(words)

def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """上位 N 件の単語と出現回数を取得する。
    
    同数の出現回数がある場合は、辞書順でソートされる。
    
    Args:
        counter: 単語頻度のカウント
        n: 取得する件数
        
    Returns:
        (単語, 出現回数) のタプルのリスト。上位順にソートされている。
    """
    # まず出現回数の降順、次に単語の昇順でソートして上位 N 件を取得
    # Counter.most_common() は出現回数の降順だが、同数の場合は不定順

    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
