"""
word_freq.py

A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a table format.

Usage:
    python word_freq.py <input_file> [-n N] [--ignore-case] [--min-length MIN_LEN]
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, ignore_case: bool = True, min_length: int = 1) -> List[str]:
    """
    Tokenize the input text into words.

    Args:
        text: The input text string.
        ignore_case: If True, convert all words to lowercase before counting.
        min_length: Minimum length of a word to include (default 1).

    Returns:
        A list of tokenized words.
    """
    # Extract words: sequences of alphanumeric characters (including underscores)
    words = re.findall(r'\b\w+\b', text)
    if ignore_case:
        words = [w.lower() for w in words]
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word in the given list.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by frequency descending.
    """
    return counter.most_common(n)
