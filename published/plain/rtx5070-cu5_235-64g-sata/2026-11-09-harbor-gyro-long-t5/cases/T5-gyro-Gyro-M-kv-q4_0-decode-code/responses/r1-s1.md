コマラインツール: テキストファイルから単語の出現頻度を集計し、

使い方:
    python wordfreq.py input.txt

import argparse

import re
from collections import Counter
from typing import List, Tuple, Optional


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合、大文字・小文字を区別する

    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字およびハイフンを含む連続文字列を単語とする
    tokens = re.findall(r"[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*", text)
    return tokens


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト内の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字・小文字の区別フラグ

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    tokens = tokenize(text, case_sensitive=case_sensitive)
    return Counter(tokens)
