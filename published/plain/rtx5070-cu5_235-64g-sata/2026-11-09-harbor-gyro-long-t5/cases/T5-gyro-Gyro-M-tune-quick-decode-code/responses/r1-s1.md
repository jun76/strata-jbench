Usage:
    python word_counter.py <file> [-n TOP_N]

Example:
    python word_counter.py sample.txt -n 5
"""

from __future__ import annotations

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize the text into lowercase words.

    Words are defined as sequences of alphanumeric characters.
    Non-alphanumeric characters are treated as separators.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase words.
    """
    # Match sequences of alphanumeric characters (letters and digits)
    words = re.findall(r"[a-zA-Z0-9]+", text)
    return [word.lower() for word in words]


def count_words(text: str) -> Counter:
    """Count the frequency of each word in the given text.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words from a Counter.

    Args:
        counter: A Counter object containing word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) for the top N words, sorted by
        descending frequency, then by alphabetical order for ties.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
