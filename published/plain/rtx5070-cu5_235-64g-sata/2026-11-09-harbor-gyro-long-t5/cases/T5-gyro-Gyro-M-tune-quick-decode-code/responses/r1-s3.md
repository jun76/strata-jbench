"""
wordfreq: Count word occurrences in a text file and display top N words.
"""

import argparse
import collections
import re
import sys
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words using regex.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens.
    """
    return re.findall(r'\b\w+\b', text.lower())


def count_words(text: str) -> collections.Counter:
    """
    Count word occurrences in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter mapping words to their counts.
    """
    tokens = tokenize(text)
    return collections.Counter(tokens)


def get_top_n(counter: collections.Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter mapping words to counts.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)
