word_freq CLI: Count word frequencies in a text file and display top N.

Usage:
    python word_freq.py [options] <file>

Options:
    --top N       Show top N words (default: 10)
    --case-insensitive  Treat words case-insensitively (default: False)
    --min-length L      Minimum word length to include (default: 1)

Output format:
    Word      Count
    --------  -----
    hello     42
    world     10
    ...
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_insensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text string.
        case_insensitive: If True, convert all words to lowercase.
        min_length: Minimum length of words to include.

    Returns:
        A list of words (strings).
    """
    words: List[str] = []
    # Split on whitespace; strip punctuation-like characters
    raw_tokens = text.split()
    for token in raw_tokens:
        # Strip leading/trailing punctuation characters
        cleaned = token.strip(".,;:!?\"'()[]{}<>-")
        if len(cleaned) >= min_length:
            if case_insensitive:
                cleaned = cleaned.lower()
            words.append(cleaned)
    return words


def count_words(text: str, case_insensitive: bool = False, min_length: int = 1) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text string.
        case_insensitive: If True, treat words case-insensitively.
        min_length: Minimum word length to include.

    Returns:
        A Counter mapping words to their occurrence counts.
    """
    tokens = tokenize(text, case_insensitive=case_insensitive, min_length=min_length)
    return Counter(tokens)
