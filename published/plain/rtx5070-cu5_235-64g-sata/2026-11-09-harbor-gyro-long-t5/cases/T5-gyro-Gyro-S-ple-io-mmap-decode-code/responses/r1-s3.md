def read_file(path: str) -> str:
    """
    Read the contents of a text file.

    Args:
        path: Path to the text file to read.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        OSError: If there is an error reading the file.
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def count_words(text: str) -> Counter:
    """
    Count the frequency of each word in the given text.

    Words are split by whitespace and converted to lowercase for counting.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    words = text.lower().split()
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from a Counter.

    Args:
        counter: A Counter object containing word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count), sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by count descending, then by word ascending for stable ordering
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
