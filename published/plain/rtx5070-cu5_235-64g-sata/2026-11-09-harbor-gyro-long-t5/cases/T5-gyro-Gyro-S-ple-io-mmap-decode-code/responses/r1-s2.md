def tokenize(text: str, ignore_case: bool = True) -> list[str]:
    """テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        ignore_case: True の場合、単語を小文字に変換する

    Returns:
        単語のリスト
    """
    # 英数字を含む連続文字列を単語として抽出
    tokens = re.findall(r'[A-Za-z0-9]+', text)
    if ignore_case:
        tokens = [t.lower() for t in tokens]
    return tokens
