import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count word frequencies from the given text.

    Words are split by whitespace and lowercased.

    Args:
        text: The input text string.

    Returns:
        A Counter mapping each word (lowercase) to its frequency.
    """
    return Counter(text.lower().split())


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most frequent words.

    Args:
        counter: A Counter of word frequencies.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then by word ascending for ties.
    """
    items = counter.most_common()
    # Sort by (-count, word) for stable ordering on ties
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(rows: List[Tuple[str, int]]) -> str:
    """Format rows as a simple text table.

    Args:
        rows: A list of (word, count) tuples.

    Returns:
        A formatted string representing the table.
    """
    if not rows:
        return "No words found."

    max_word_len = max(len(w) for w, _ in rows)
    max_count_len = max(len(str(c)) for _, c in rows)

    header_word = "Word".ljust(max_word_len)
    header_count = "Count".ljust(max_count_len)

    lines = []
    lines.append(f"{header_word}  {header_count}")
    lines.append(f"{'-' * max_word_len}  {'-' * max_count_len}")
    for word, count in rows:
        lines.append(f"{word.ljust(max_word_len)}  {str(count).ljust(max_count_len)}")

    return "\n".join(lines)
