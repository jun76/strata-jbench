"""Command-line tool to read a text file and display the top N most frequent words.

This module provides a simple CLI tool that:
1. Reads a text file.
2. Tokenizes the content into words (case-insensitive, ignoring punctuation).
3. Counts the frequency of each word.
4. Displays the top N most frequent words in a tabular format.

Usage:
    python word_freq.py <input_file> [--top N] [--min-length L]

Example:
    python word_freq.py sample.txt --top 10
"""

import argparse

def tokenize(text: str) -> List[str]:
    """Tokenize the text into lowercase words, ignoring punctuation and whitespace.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens.
    """
    # Normalize to lowercase and split on whitespace, then strip punctuation.
    # A simple approach: replace non-alphanumeric characters with spaces, then split.
    normalized = text.lower()
    # Replace punctuation characters with spaces to ensure clean tokenization
    chars = []
    for ch in normalized:
        if ch.isalnum():
            chars.append(ch)
        elif ch == '-' or ch == '_' or ch == "'":
            # Keep hyphens and underscores and apostrophes as part of words
            # but replace other punctuation with space
            chars.append(ch)
        else:
            chars.append(' ')
    joined = ''.join(chars)
    # Split on whitespace and filter out empty strings
    tokens = joined.split()
    # Remove tokens that are purely punctuation (already handled by replacing non-alnum with space,
    # but hyphens/apostrophes might leave empty-ish tokens; we filter tokens with at least one alnum)
    filtered_tokens = [token for token in tokens if any(c.isalnum() for c in token)]
    return filtered_tokens
