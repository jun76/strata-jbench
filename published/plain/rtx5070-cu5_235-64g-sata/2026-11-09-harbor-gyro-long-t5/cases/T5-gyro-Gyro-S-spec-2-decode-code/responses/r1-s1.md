"""
CLI tool to count word frequencies in a text file and display top N results.

Usage:
    python word_count.py <file_path> [-n N] [--sort-by {count,word}] [--case-sensitive]

Example:
    python word_count.py example.txt -n 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple, Optional


def parse_args(argv: Optional[List[str]] = None) -> argparse.Namespace:
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and display top N results."
    )
    parser.add_argument(
        "file_path",
        type=str,
        help="Path to the text file to analyze",
    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=10,
        help="Number of top words to display (default: 10)",
    )
    parser.add_argument(
        "--sort-by",
        choices=["count", "word"],
        default="count",
        help="Sort order: 'count' (descending frequency) or 'word' (alphabetical) (default: 'count')",
    )
    parser.add_argument(
        "--case-sensitive",
        action="store_true",
        help="Treat words as case-sensitive (default: case-insensitive)",
    )
    return parser.parse_args(argv)


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize text into words.
    
    Args:
        text: Input text string.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.
    
    Returns:
        List of word tokens.
    """
    # Basic tokenization: split on whitespace and punctuation-like boundaries
    # We'll use a simple approach: split on whitespace, then strip non-alphabetic chars from edges
    import re
    
    # Find sequences of alphanumeric characters (including underscores)
    tokens = re.findall(r"[A-Za-z0-9_]+", text)
    
    if not case_sensitive:
        tokens = [token.lower() for token in tokens]
    
    return tokens
