import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count word frequencies from the given text.

    Words are split by whitespace and normalized to lowercase.
    """
    words = text.lower().split()
    return Counter(words)


def top_n_words(counts: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most frequent words.

    Ties are resolved by alphabetical order for deterministic output.
    """
    # Sort by frequency descending, then by word ascending for tie-breaking
    sorted_items = sorted(counts.items(), key=lambda item: (-item[1], item[0]))
    return sorted_items[:n]


def format_table(items: List[Tuple[str, int]]) -> str:
    """Format word-frequency pairs as a simple text table."""
    if not items:
        return ""
    word_width = max(len(word) for word, _ in items)
    freq_width = max(len(str(freq)) for _, freq in items)
    header = f"{'Word':<{word_width}} {'Count':>{freq_width}}"
    separator = f"{'-' * word_width} {'-' * freq_width}"
    rows = []
    for word, freq in items:
        rows.append(f"{word:<{word_width}} {freq:>{freq_width}}")
    return "\n".join([header, separator, *rows])


def main() -> None:
    """Command-line entry point for word frequency counting."""
    parser = argparse.ArgumentParser(
        description="Count word frequencies from a text file and display top N words."
    )
    parser.add_argument(
        "file",
        type=str,
        help="Path to the input text file",
    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=10,
        help="Number of top words to display (default: 10)",
    )

    args = parser.parse_args()

    try:
        with open(args.file, "r", encoding="utf-8") as f:
            text = f.read()
    except FileNotFoundError:
        print(f"Error: File '{args.file}' not found.", file=sys.stderr)
