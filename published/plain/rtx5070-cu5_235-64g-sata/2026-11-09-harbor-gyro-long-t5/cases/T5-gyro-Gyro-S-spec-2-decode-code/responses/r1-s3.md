#!/usr/bin/env python3
"""コマンドラインツール: テキストファイルから単語の出現頻度を上位N件として表形式で表示する."""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def parse_args(args: List[str] | None = None) -> argparse.Namespace:
    """コマライン引数を解析する."""
    parser = argparse.ArgumentParser(
        description="テキストファイルを読み込み、単語の出現頻度を上位N件として表形式で表示します。"
    )
    parser.add_argument(
        "file",
        type=str,
        help="対象のテキストファイルパス",
    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=10,
        help="表示する上位単語の件数 (デフォルト: 10)",
    )
    parser.add_argument(
        "--ignore-case",
        action="store_true",
        help="大文字小文字を区別せず集計する",
    )
    parser.add_argument(
        "--sort",
        choices=["freq", "word"],
        default="freq",
        help="ソート基準: 'freq'(頻度降順) or 'word'(アルファベット順) (デフォルト: freq)",
    )
    return parser.parse_args(args)


def tokenize(text: str, ignore_case: bool = False) -> List[str]:
    """テキストを単語に分割する.

    Args:
        text: 入力テキスト.
        ignore_case: Trueなら小文字に変換する.

    Returns:
        単語のリスト.
    """
    # 非アルファベット文字を区切りとする
    words = []
    current = []
    for ch in text:
        if ch.isalpha():
            current.append(ch)
        else:
            if current:
                words.append("".join(current))
                current = []
    if current:
        words.append("".join(current))
    if ignore_case:
        words = [w.lower() for w in words]
    return words
