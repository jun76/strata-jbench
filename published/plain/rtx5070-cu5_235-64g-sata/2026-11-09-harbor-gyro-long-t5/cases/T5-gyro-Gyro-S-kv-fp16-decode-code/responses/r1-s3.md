"""
text_word_counter.py — コマンドラインツール: テキストファイルから単語出現頻度を
集計し、上位 N 件を表形式で表示する。

使用例:
    python text_word_counter.py input.txt --top 10
    python text_word_counter.py input.txt --top 5 --case-sensitive
"""

def read_file(path: str) -> str:
    """テキストファイルを読み込み、文字列として返す。

    Args:
        path: 読み込み対象のファイルパス。

    Returns:
        ファイル内容の文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み込みエラーの場合。
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストを単語に分割し、必要に応じて小文字化して返す。

    Args:
        text: 分割対象のテキスト。
        case_sensitive: True の場合、小文字化しない。False の場合、
            単語は小文字に統一される。

    Returns:
        単語のリスト。
    """
    # 空白・改行で分割し、空文字列を除去
    words = text.split()
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_words(words: List[str]) -> Counter[str]:
    """単語出現頻度を Counter で集計する。

    Args:
        words: 単語のリスト。

    Returns:
        各単語の出現回数を格納した Counter。
    """
    return Counter(words)
