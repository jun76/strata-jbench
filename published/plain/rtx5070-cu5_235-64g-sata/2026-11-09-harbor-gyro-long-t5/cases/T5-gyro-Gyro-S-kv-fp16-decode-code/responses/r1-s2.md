def normalize_word(word: str) -> str:
    """
    単語を正規化する。
    - 小文字化
    - 句読点や記号を除去
    - 空文字列や数字のみの文字列を除外
    
    Args:
        word: 元の単語文字列
    
    Returns:
        正規化された単語（小文字、句読点除去済み）。無効な場合は None を返す。
    """
    # 句読点や記号を除去
    normalized = word.strip().lower()
    # 句読点、記号を除去
    cleaned = []
    for char in normalized:
        if char.isalpha():
            cleaned.append(char)
        elif char.isdigit():
            # 数字を含む場合は除外（純粋な数字文字列は単語として扱わない）
            # ただし、混在している場合は除去して残す
            continue
        else:
            # 句読点やスペースなどを除去
            continue
    
    result = "".join(cleaned)
    if not result:
        return None
    
    # 数字のみだった場合は除外
    if result.isdigit():
        return None
    
    return result
