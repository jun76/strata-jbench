"""
word_freq.py — コマンドラインツール: テキストファイルから単語出現頻度を集計し、
上位 N 件を表形式で表示する。

使い方:
    python word_freq.py <ファイル> [--top N] [--case-sensitive] [--min-length L]

例:
    python word_freq.py sample.txt --top 10
"""

import argparse
import re
from collections import Counter
from typing import Optional

# ---------------------------------------------------------------------------
# 型別別名
# ---------------------------------------------------------------------------
WordCount = Counter[str]


# ---------------------------------------------------------------------------
# 単語抽出
# ---------------------------------------------------------------------------
def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> list[str]:
    """テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。
            False の場合、全て小文字に変換して集計する。
        min_length: この長さ未満の単語は除外する（デフォルト 1）。

    Returns:
        抽出された単語のリスト（出現順）。
    """
    # 英数字・ハイフン・アポストロフィを含むトークンを抽出
    # （単語の境界は空白・句読点など）
    tokens = re.findall(r"[a-zA-Z0-9'’]+", text)

    if not case_sensitive:
        tokens = [t.lower() for t in tokens]

    # min_length 未満のトークンを除外
    tokens = [t for t in tokens if len(t) >= min_length]

    return tokens


# ---------------------------------------------------------------------------
# 頻度集計
# ---------------------------------------------------------------------------
def count_words(words: list[str]) -> WordCount:
    """単語リストから出現頻度を集計する。

    Args:
        words: 単語のリスト。

    Returns:
        各単語とその出現回数の Counter。
    """
    return Counter(words)
