wordfreq.py - コマンドラインツール：テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する。

使い方:
    python wordfreq.py input.txt --top 10 --case-sensitive --min-len 2

import argparse

def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> Iterator[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するか。False の場合は小文字に変換。
        min_len: 抽出する単語の最小長（文字数）

    Yields:
        単語文字列（case_sensitive=False の場合は小文字）
    """
    # 英字・数字・ハイフン・アポストロフィを含む連続文字列を単語として抽出
    # 通常、単語は英字や数字の連続部分として捉える
    pattern = r"[a-zA-Z0-9]+"
    matches = re.findall(pattern, text)
    for word in matches:
        if not case_sensitive:
            word = word.lower()
        if len(word) >= min_len:
            yield word


def count_words(
    stream: TextIO,
    case_sensitive: bool = False,
    min_len: int = 1,
) -> Counter[str]:
    """
    テキストストリームから単語出現頻度をカウントする。

    Args:
        stream: テキストストリーム（ファイルまたは StringIO など）
        case_sensitive: 大文字小文字を区別するか
        min_len: 単語の最小長

    Returns:
        単語 -> 出現回数の Counter
    """
    counter: Counter[str] = Counter()
    for line in stream:
        for word in tokenize(line, case_sensitive=case_sensitive, min_len=min_len):
            counter[word] += 1
    return counter
