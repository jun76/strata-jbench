wordfreq.py - コマンドラインツール：テキストファイルを読み込み、単語出現頻度を集計し上位 N 件を表形式で表示する。

使い方:
    python wordfreq.py input.txt --top 10
    python wordfreq.py input.txt --top 20 --case-insensitive

import argparse

from collections import Counter

def tokenize(text: str, *, case_insensitive: bool = True) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_insensitive: True の場合、小文字に正規化してからカウントする

    Returns:
        単語のリスト
    """
    # 英数字の連続を単語として抽出する
    words = re.findall(r"[a-zA-Z0-9]+", text)
    if case_insensitive:
        words = [w.lower() for w in words]
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語の出現頻度を Counter で集計する。

    Args:
        words: 単語のリスト

    Returns:
        Counter オブジェクト（単語 -> 出現回数）
    """
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度上位 n 件の (単語, 回数) のリストを返す。

    Args:
        counter: Counter オブジェクト
        n: 上位件数

    Returns:
        [(word, count), ...] のリスト（頻度降順）
    """
    return counter.most_common(n)
