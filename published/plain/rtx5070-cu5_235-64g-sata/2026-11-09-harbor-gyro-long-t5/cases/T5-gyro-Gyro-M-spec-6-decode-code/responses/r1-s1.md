#!/usr/bin/env python3
"""
Word frequency counter CLI tool.

Reads a text file, counts word occurrences (case-insensitive, punctuation stripped),
and prints the top N most frequent words in a tabular format.

Usage:
    python wordfreq.py --file input.txt --top 10
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize a string into lowercase words.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens, with punctuation and non-alphanumeric
        characters stripped. Words are split on whitespace and non-word characters.

    Examples:
        >>> tokenize("Hello, World! hello world")
        ['hello', 'world', 'hello', 'world']
    """
    # Convert to lowercase
    text = text.lower()
    # Extract words: sequences of alphanumeric characters
    # This strips punctuation and treats non-word chars as separators
    words = re.findall(r'\w+', text)
    return words


def count_words(text: str) -> Counter:
    """
    Count word occurrences in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter mapping each word to its occurrence count.

    Examples:
        >>> count_words("apple banana apple cherry") == Counter({'apple': 2, 'banana': 1, 'cherry': 1})
        True
    """
    words = tokenize(text)
    return Counter(words)
