import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    テキストから単語の出現頻度をカウントする。

    Args:
        text: 解析対象のテキスト文字列。

    Returns:
        単語から出現回数を映射する Counter オブジェクト。
    """
    # 単語を抽出する（空白区切りの単純な分割）
    words = text.split()
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    出現頻度が上位 N 件の単語と回数を取り出す。

    Args:
        counter: 単語から出現回数を映射する Counter オブジェクト。
        n: 取得する上位件数。

    Returns:
        (word, count) のタプルを上位順に並べたリスト。
    """
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """
    上位 N 件の単語と回数を表形式の文字列としてフォーマットする。

    Args:
        items: (word, count) のタプルリスト。

    Returns:
        表形式の文字列。
    """
    if not items:
        return ""

    # 列幅を計算
    max_word_len = max(len(word) for word, _ in items)
    max_count_len = max(len(str(count)) for _, count in items)

    # ヘッダー行
    header_word = "Word".ljust(max_word_len)
    header_count = "Count".rjust(max_count_len)
    separator = f"{header_word} {'-' * max_word_len} {' ' * max_count_len} {header_count}"

    lines = [f"{header_word} {header_count}", "-" * (max_word_len + max_count_len + 1)]

    for word, count in items:
        word_part = word.ljust(max_word_len)
        count_part = str(count).rjust(max_count_len)
        lines.append(f"{word_part} {count_part}")

    return "\n".join(lines)
