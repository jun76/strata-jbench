#!/usr/bin/env python3
"""Command-line tool to count word frequencies from a text file.

Usage:
    python word_freq.py <input_file> [-n N] [-c] [-i]

Options:
    -n N: Show top N words (default: 10)
    -c: Case-insensitive counting (default: False)
    -i: Ignore punctuation when tokenizing (default: True)

Example:
    python word_freq.py sample.txt -n 5 -c
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_insensitive: bool = False) -> List[str]:
    """Tokenize text into words, stripping punctuation.

    Args:
        text: The input text to tokenize.
        case_insensitive: If True, convert all words to lowercase.

    Returns:
        A list of word tokens.
    """
    if case_insensitive:
        text = text.lower()
    # Extract words: sequences of alphanumeric characters and underscores
    # This effectively strips punctuation and whitespace
    words = re.findall(r'[a-zA-Z0-9_]+', text)
    return words


def count_word_frequencies(text: str, case_insensitive: bool = False) -> Counter:
    """Count the frequency of each word in the given text.

    Args:
        text: The input text to process.
        case_insensitive: If True, treat words case-insensitively.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    words = tokenize(text, case_insensitive=case_insensitive)
    return Counter(words)
