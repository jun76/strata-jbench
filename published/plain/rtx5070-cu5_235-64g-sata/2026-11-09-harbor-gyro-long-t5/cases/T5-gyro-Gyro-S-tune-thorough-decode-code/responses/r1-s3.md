def tokenize(text: str, *, case_insensitive: bool = True, min_length: int = 1) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト。
        case_insensitive: True なら小文字に変換して集約する。
        min_length: この長さ未満のトークンは除外する。

    Returns:
        トークンリスト（順序保持）。
    """
    # 単語は英数字・ハイフン・アポストロフィを含む連続文字列として抽出
    pattern = r"[A-Za-z0-9'\-]+"
    tokens = re.findall(pattern, text)
    if case_insensitive:
        tokens = [t.lower() for t in tokens]
    if min_length > 1:
        tokens = [t for t in tokens if len(t) >= min_length]
    return tokens


def count_words(text: str, *, case_insensitive: bool = True, min_length: int = 1) -> Counter[str]:
    """
    テキスト内の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト。
        case_insensitive: True なら小文字で集約。
        min_length: 最小トークン長。

    Returns:
        単語 -> 出現回数 の Counter。
    """
    tokens = tokenize(text, case_insensitive=case_insensitive, min_length=min_length)
    return Counter(tokens)
