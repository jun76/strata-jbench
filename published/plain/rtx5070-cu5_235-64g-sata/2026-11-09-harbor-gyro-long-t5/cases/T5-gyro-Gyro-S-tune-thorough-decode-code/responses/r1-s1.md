def tokenize(text: str, case_insensitive: bool = True) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_insensitive: 大文字小文字を無視するかどうか

    Returns:
        単語のリスト
    """
    # 英数字とハイフンを含むトークンを抽出
    # パターン: 英数字、ハイフン、アポストロフィを含む連続文字列
    pattern = r"[a-zA-Z0-9][a-zA-Z0-9'\-]*"
    tokens = re.findall(pattern, text)

    if case_insensitive:
        tokens = [token.lower() for token in tokens]

    return tokens


def count_words(tokens: List[str]) -> Counter[str]:
    """
    単語出現頻度をカウントする。

    Args:
        tokens: 単語のリスト

    Returns:
        単語と出現回数の Counter オブジェクト
    """
    return Counter(tokens)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の (word, count) を取得する。

    Args:
        counter: 単語と出現回数の Counter
        n: 取得する件数

    Returns:
        (word, count) のリスト（出現回数降順、同数時は辞書順）
    """
    # 出現回数降順、同数の場合は辞書順（アルファベット順）
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
