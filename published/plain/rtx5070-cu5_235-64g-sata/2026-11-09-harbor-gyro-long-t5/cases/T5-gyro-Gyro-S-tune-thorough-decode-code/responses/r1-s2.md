オプション:
    --file PATH      入力テキストファイル（必須）
    --top N          表示する上位件数（デフォルト: 10）

    --case-insensitive 大文字小文字を区別しない（デフォルト: False）

import argparse

from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_insensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_insensitive: Trueなら小文字化してから分割

    Returns:
        単語のリスト
    """
    if case_insensitive:
        text = text.lower()
    # 空白・改行で分割し、空文字列を除外
    return [word for word in text.split() if word]


def count_word_frequency(words: List[str]) -> Counter[str]:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数のCounter
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    上位N件の単語とその頻度を取得する。

    Args:
        counter: 単語->頻度のCounter
        n: 上位件数

    Returns:
        (単語, 頻度)のタプルのリスト（頻度降順）
    """
    return counter.most_common(n)
