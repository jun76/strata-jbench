"""
wordfreq.py - コマンドラインツール：テキストファイルの単語出現頻度を上位 N 件で表形式表示

使い方:
    python wordfreq.py input.txt -n 10
    python wordfreq.py input.txt --sort-by-count --format table
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple, Optional


def read_words(filepath: str) -> List[str]:
    """
    指定されたテキストファイルから単語を抽出する。

    Args:
        filepath: テキストファイルのパス

    Returns:
        単語のリスト（小文字化し、非アルファベット文字を除去した単語のみ）

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        IOError: ファイル読み込みエラーの場合
    """
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            content = f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {filepath}")
    except IOError as e:
        raise IOError(f"ファイル読み込みエラー: {e}")

    # 単語を抽出: 空白で分割し、アルファベットのみを含む単語を小文字化して返す
    words = []
    for line in content.split('\n'):
        for word in line.split():
            # 非アルファベット文字を除去
            cleaned = ''.join(ch for ch in word if ch.isalpha())
            if cleaned:
                words.append(cleaned.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数の Counter オブジェクト
    """
    return Counter(words)
