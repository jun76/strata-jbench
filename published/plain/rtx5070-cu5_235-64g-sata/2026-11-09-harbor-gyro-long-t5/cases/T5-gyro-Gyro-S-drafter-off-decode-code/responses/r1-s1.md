#!/usr/bin/env python3
"""
word_counter.py - コマンドラインツールでテキストファイルから単語の出現頻度を集計し、
上位 N 件を表形式で表示する。

使い方:
    python word_counter.py input.txt
    python word_counter.py input.txt -n 10
    python word_counter.py input.txt --case-sensitive
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple, Dict


def count_words(text: str, case_sensitive: bool = False) -> Dict[str, int]:
    """
    テキストから単語を抽出し、出現頻度を返す。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字と小文字を区別する。False の場合、小文字に正規化して集計する。

    Returns:
        単語 -> 出現回数の辞書。
    """
    # 単語は英数字とハイフン、アポストロフィを含む文字列として抽出
    # 正規表現: [a-zA-Z0-9'-]+ で単語を抽出
    words = re.findall(r"[a-zA-Z0-9'-]+", text)
    
    if not case_sensitive:
        words = [w.lower() for w in words]
    
    counter = Counter(words)
    return dict(counter)


def get_top_n(counter: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の (word, count) を返す。
    頻度が同じ場合は、辞書順（アルファベット順）で並び替える。

    Args:
        counter: 単語 -> 出現回数の辞書。
        n: 上位件数。

    Returns:
        (word, count) のリスト。頻度降順、同頻度時は単語の辞書順。
    """
    # 頻度降順、同頻度時は辞書順（アルファベット順）
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
