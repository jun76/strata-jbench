def load_text(filepath: str) -> str:
    """指定されたファイルパスからテキストを読み込む。"""
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。非英数字を区切りとし、小文字化して単語を抽出する。"""
    words: List[str] = []
    current_word: List[str] = []
    for char in text:
        if char.isalnum():
            current_word.append(char.lower())
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))
    return words


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """単語リストから各単語の出現頻度を集計する。"""
    return Counter(words)


def get_top_n(freqs: Counter[str], n: int) -> List[Tuple[str, int]]:
    """上位N件の単語と頻度を取得する。同頻度の場合は辞書順でソートする。"""
    items = list(freqs.items())
    # 頻度降順、同頻度なら辞書順（昇順）
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
