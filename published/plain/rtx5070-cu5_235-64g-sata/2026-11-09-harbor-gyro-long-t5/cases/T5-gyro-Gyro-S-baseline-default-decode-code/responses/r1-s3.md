#!/usr/bin/env python3
"""Command-line tool to read a text file and display the top N most frequent words.

This module provides a simple CLI tool that:
1. Reads a text file specified via command-line arguments.
2. Tokenizes the text into words (lowercased, stripped of punctuation).
3. Counts the frequency of each word.
4. Displays the top N most frequent words in a table format.

Usage:
    python word_freq.py <input_file> [-n TOP_N]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize a string into lowercase words, stripping punctuation.

    Args:
        text: The input string to tokenize.

    Returns:
        A list of lowercase words extracted from the text.
    """
    words = []
    current_word = []
    for char in text:
        if char.isalnum():
            current_word.append(char.lower())
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    if current_word:
        words.append(''.join(current_word))
    return words


def count_word_frequencies(text: str) -> Counter:
    """Count the frequency of each word in the given text.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most frequent words and their counts.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count), sorted by count descending.
        If there are ties in count, words are sorted alphabetically.
    """
    # Sort by count descending, then by word ascending for tie-breaking
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
