"""
A command-line tool to read a text file, count word frequencies, and display
the top N most frequent words in a tabular format.
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def parse_arguments() -> argparse.Namespace:
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and show top N words."
    )
    parser.add_argument(
        "filepath",
        type=str,
        help="Path to the input text file.",
    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=10,
        help="Number of top frequent words to display (default: 10).",
    )
    parser.add_argument(
        "--case-sensitive",
        action="store_true",
        help="If set, word counting is case-sensitive (default: case-insensitive).",
    )
    return parser.parse_args()


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize text into words, splitting on whitespace and punctuation.

    Args:
        text: The input text string.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.

    Returns:
        A list of tokens (words).
    """
    # Simple tokenization: split on whitespace, strip punctuation
    words: List[str] = []
    for token in text.split():
        # Strip leading and trailing punctuation characters
        cleaned = token.strip(".,;:!?\"'()[]{}")
        if cleaned:
            if not case_sensitive:
                cleaned = cleaned.lower()
            words.append(cleaned)
    return words


def count_words(tokens: List[str]) -> Counter:
    """
    Count the frequency of each word in the token list.

    Args:
        tokens: A list of word tokens.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(tokens)
