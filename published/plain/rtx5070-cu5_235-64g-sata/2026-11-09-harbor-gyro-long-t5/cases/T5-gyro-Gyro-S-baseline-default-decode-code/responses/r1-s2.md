import argparse

from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count word frequencies from the given text.

    Words are split by whitespace and normalized to lowercase.
    """
    words = text.lower().split()
    return Counter(words)


def top_n_words(counts: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most frequent words with their counts.

    Args:
        counts: A Counter object mapping words to their frequencies.
        n: The number of top entries to return.

    Returns:
        A list of tuples (word, count), sorted by count descending,
        then alphabetically for ties.
    """
    items = sorted(counts.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(entries: List[Tuple[str, int]]) -> str:
    """Format entries as a table with word and count columns.

    Args:
        entries: A list of (word, count) tuples.

    Returns:
        A formatted string representing the table.
    """
    if not entries:
        return ""
    # Determine column widths
    word_width = max(len(word) for word, _ in entries)
    count_width = max(len(str(count)) for _, count in entries)
    # Header
    header_word = "word".ljust(word_width)
    header_count = "count".rjust(count_width)
    separator = "-" * (word_width + 1 + count_width)
    lines = [f"{header_word} {header_count}", separator]
    for word, count in entries:
        lines.append(f"{word.ljust(word_width)} {str(count).rjust(count_width)}")
    return "\n".join(lines)
