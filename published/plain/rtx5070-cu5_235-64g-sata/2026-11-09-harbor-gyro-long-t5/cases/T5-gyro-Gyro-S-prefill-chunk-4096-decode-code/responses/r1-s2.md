```python
"""Command-line tool to count word frequencies in text files and display top N results.

Usage:
    python word_counter.py [input_file] --top N [--case-sensitive] [--delimiter ...]

Example:
    python word_counter.py sample.txt --top 10
"""

import argparse
import sys
from typing import Dict, List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """Tokenize text into words.

    Args:
        text: Input text string.
        case_sensitive: If False, words are lowercased before counting.

    Returns:
        List of tokens.
    """
    if not case_sensitive:
        text = text.lower()
    # Split on whitespace and strip punctuation-like characters from each token
    tokens = []
    for raw_token in text.split():
        # Remove non-alphanumeric characters from edges to handle punctuation
        token = raw_token.strip(".,!?;:()-\"'`~")
        if token:
            tokens.append(token)
    return tokens


def count_words(tokens: List[str]) -> Dict[str, int]:
    """Count occurrences of each token.

    Args:
        tokens: List of word tokens.

    Returns:
        Dictionary mapping tokens to their counts.
    """
    counts: Dict[str, int] = {}
    for token in tokens:
        counts[token] = counts.get(token, 0) + 1
    return counts


def get_top_n(counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """Return the top N most frequent words.

    Args:
        counts: Dictionary mapping words to counts.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples sorted by count descending, then word ascending.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(counts.items(), key=lambda item: (-item[1], item[0]))
    return sorted_items[:n]
