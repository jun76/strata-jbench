Usage:
    python word_freq.py [options] <file>

Options:
    -n, --top N       Number of top words to display (default: 10)

    --min-length      Minimum word length to include (default: 1)
    --format          Output format: table or json (default: table)
"""

def count_words(
    text: str,
    case_insensitive: bool = True,
    pattern: str = r"\w+",
    min_length: int = 1,
) -> Counter[str]:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text to analyze.
        case_insensitive: If True, normalize words to lowercase before counting.
        pattern: Regular expression pattern to match words.
        min_length: Minimum length of a word to be included in the count.

    Returns:
        A Counter object with word frequencies.
    """
    words = re.findall(pattern, text)
    if case_insensitive:
        words = [w.lower() for w in words]
    # Filter by minimum length
    words = [w for w in words if len(w) >= min_length]
    return Counter(words)
