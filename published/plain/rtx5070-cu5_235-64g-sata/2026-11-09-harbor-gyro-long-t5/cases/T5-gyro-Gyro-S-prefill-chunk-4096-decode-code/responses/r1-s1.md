def read_words_from_file(filepath: str) -> List[str]:
    """
    指定されたファイルから単語を抽出する。

    Args:
        filepath: テキストファイルのパス

    Returns:
        単語のリスト（小文字化された単語）

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        IOError: ファイルの読み込みでエラーが発生した場合
    """
    with open(filepath, "r", encoding="utf-8") as f:
        text = f.read()

    # 単語を抽出：空白や改行で分割し、句読点を除去
    words = []
    for word in text.split():
        # 句読点や特殊文字を除去して純粋な単語を取り出す
        cleaned = "".join(ch for ch in word if ch.isalnum() or ch == "'")
        if cleaned:
            words.append(cleaned.lower())

    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語の出現頻度を集計する。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数 の Counter オブジェクト
    """
    return Counter(words)
