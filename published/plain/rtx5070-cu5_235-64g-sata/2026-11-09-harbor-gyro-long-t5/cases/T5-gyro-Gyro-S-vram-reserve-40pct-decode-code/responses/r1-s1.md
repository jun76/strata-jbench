import argparse
import sys
from collections import Counter
from typing import List, Tuple


def load_text(filepath: str) -> str:
    """Read text from a file and return its content."""
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except UnicodeDecodeError:
        raise UnicodeDecodeError(f"Failed to decode file: {filepath}")


def tokenize(text: str) -> List[str]:
    """Tokenize text into words, splitting on whitespace."""
    return text.split()


def count_words(words: List[str]) -> Counter:
    """Count occurrences of each word."""
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words with their counts."""
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]]) -> str:
    """Format the top entries as a tab-separated table."""
    if not entries:
        return ""
    
    # Find the maximum word length for padding
    max_word_len = max(len(word) for word, _ in entries)
    
    lines = []
    for word, count in entries:
        lines.append(f"{word:<{max_word_len}} {count}")
    
    return "\n".join(lines)


def main():
    """Main function for the command-line tool."""
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and display top N entries."
    )
    parser.add_argument(
        'filepath',
        type=str,
        help="Path to the input text file"
    )
    parser.add_argument(
        '-n',
        '--top',
        type=int,
        default=10,
        help="Number of top entries to display (default: 10)"
    )
    
    args = parser.parse_args()
    
    try:
        text = load_text(args.filepath)
        words = tokenize(text)
        counter = count_words(words)

        output = format_table(top_entries)
