"""
CLI tool: read a text file, aggregate word frequencies, and display the top N entries.
"""

import argparse
import os
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase alphanumeric words.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase word tokens (alphanumeric only).
    """
    # Extract words consisting of letters and digits
    return re.findall(r"[a-zA-Z0-9]+", text.lower())


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        A Counter mapping each word to its occurrence count.
    """
    return Counter(tokenize(text))


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Return the top N most frequent words from a Counter.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples sorted by descending count,
        then lexicographically by word for ties.
    """
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]], title: str = "Top Words") -> str:
    """
    Format a list of (word, count) tuples as a simple table.

    Args:
        entries: List of (word, count) tuples.
        title: Title header for the table.

    Returns:
        A formatted string representing the table.
    """
    if not entries:
        return f"{title}\n(No words found)"

    # Determine column widths
    max_word_len = max(len(word) for word, _ in entries)
    max_count_len = max(len(str(count)) for _, count in entries)

    word_width = max(max_word_len, len("Word"))
    count_width = max(max_count_len, len("Count"))
