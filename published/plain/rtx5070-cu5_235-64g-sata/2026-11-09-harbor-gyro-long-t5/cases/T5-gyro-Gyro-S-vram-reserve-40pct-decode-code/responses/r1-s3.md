import argparse

from typing import Dict, List, Tuple, Optional


def tokenize(text: str, case_insensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、単語を小文字化して集計する。

    Returns:
        単語のリスト（区切り文字で分割）。
    """
    # 区切り文字: 空白類、句読点、特殊文字
    # 単語はアルファベット、数字、ハイフン、アポストロフィを含む
    import re
    # 単語パターン: 文字・数字・ハイフン・アポストロフィで構成される連続文字列
    pattern = r"[a-zA-Z0-9'-]+"
    tokens = re.findall(pattern, text)
    if case_insensitive:
        tokens = [token.lower() for token in tokens]
    return tokens


def count_word_frequencies(
    text: str, case_insensitive: bool = False
) -> Dict[str, int]:
    """
    テキスト中の単語出現頻度を集計する。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、単語を小文字化して集計する。

    Returns:
        単語 -> 出現回数 の辞書。
    """
    tokens = tokenize(text, case_insensitive)
    return dict(Counter(tokens))
