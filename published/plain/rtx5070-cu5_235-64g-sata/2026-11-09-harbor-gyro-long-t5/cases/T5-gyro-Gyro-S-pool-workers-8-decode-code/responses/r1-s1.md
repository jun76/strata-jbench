word_counter.py - テキストファイルの単語出現頻度を上位 N 件で表示する CLI ツール。

使用方法:
    python word_counter.py <ファイル> --top 10
    python word_counter.py <ファイル> --top 5 --format table

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。句読点や数字は除去し、
        英字・数字・ハイフン・アポストロフィを含むトークンを抽出する。
    """
    # 英字、数字、ハイフン、アポストロフィを含むトークンを抽出
    pattern = r"[a-zA-Z][a-zA-Z0-9'-]*|[0-9]+(?:-[0-9]+)*"
    tokens = re.findall(pattern, text)
    # 小文字化
    return [token.lower() for token in tokens]


def count_words(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度を集計する。

    Args:
        text: 入力テキスト。

    Returns:
        単語 -> 出現回数の Counter。
    """
    tokens = tokenize(text)
    return Counter(tokens)
