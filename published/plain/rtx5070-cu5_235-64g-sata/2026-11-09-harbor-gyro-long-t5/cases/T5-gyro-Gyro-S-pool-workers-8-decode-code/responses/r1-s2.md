def parse_args(argv: List[str] | None = None) -> argparse.Namespace:
    """コマンドライン引数をパースする。

    Args:
        argv: 引数リスト（None の場合 sys.argv を使用）

    Returns:
        パースされた引数オブジェクト
    """
    parser = argparse.ArgumentParser(
        description="テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示します。"
    )
    parser.add_argument(
        "file",
        type=str,
        help="読み込むテキストファイルのパス",
    )
    parser.add_argument(
        "--top",
        type=int,
        default=10,
        help="表示する上位単語の数（デフォルト: 10）",
    )
    parser.add_argument(
        "--case-sensitive",
        action="store_true",
        help="大文字小文字を区別する（デフォルト: 区別しない）",
    )
    return parser.parse_args(argv)


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_sensitive: True の場合大文字小文字を区別する

    Returns:
        単語リスト（非空のトークン）
    """
    tokens: List[str] = []
    for line in text.splitlines():
        for token in line.split():
            # 非空のみ
            if token:
                if not case_sensitive:
                    token = token.lower()
                tokens.append(token)
    return tokens
