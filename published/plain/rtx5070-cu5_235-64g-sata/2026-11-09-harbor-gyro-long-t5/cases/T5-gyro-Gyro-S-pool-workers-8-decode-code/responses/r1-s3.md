以下は、テキストファイルを読み込み、単語出現頻度を上位N件で表形式表示するコマンドラインツールです。

**wordfreq.py**
```python
#!/usr/bin/env python3
"""
wordfreq.py — テキストファイルの単語出現頻度を上位N件で表形式表示するCLIツール。

使い方:
    python wordfreq.py input.txt --top 10

依存: argparse, collections, typing (標準ライブラリのみ)
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple, Optional


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    アルファベット・数字・ハイフン・アポストロフィを含む連続文字列を単語として抽出。
    小文字に変換する。
    """
    # 単語パターン: 文字・数字・ハイフン・アポストロフィを含む連続文字列
    pattern = r"[a-zA-Z0-9'\-]+"
    tokens = re.findall(pattern, text)
    return [token.lower() for token in tokens]


def count_words(tokens: List[str]) -> Counter:
    """
    単語の出現頻度をCounterで集計する。
    """
    return Counter(tokens)


def format_table(
    top_words: List[Tuple[str, int]],
    width_word: int = 20,
    width_count: int = 8,
) -> str:
    """
    上位単語と頻度を表形式でフォーマットする。
    """
    header = f"{'Word':<{width_word}} {'Count':>{width_count}}"
    separator = f"{'-' * width_word} {'-' * width_count}"
    lines = [header, separator]
    for word, count in top_words:
        lines.append(f"{word:<{width_word}} {count:>{width_count}}")
    return "\n".join(lines)

    コマンドライン引数をパースする。

    )
