import argparse
import sys
from collections import Counter
from typing import List, Tuple

def count_words(text: str) -> Counter:
    """
    テキストから単語の出現頻度をカウントする。
    
    Args:
        text: カウント対象の文字列
        
    Returns:
        単語とその出現回数をキー・値とするCounterオブジェクト
    """
    # 小文字化して単語に分割（句読点は簡易的に空白・改行で分割）
    words = text.lower().split()
    return Counter(words)

def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counterオブジェクトから上位N件の単語と頻度を取得する。
    
    Args:
        counter: 単語頻度のCounter
        n: 取得する件数
        
    Returns:
        (単語, 頻度)のタプルのリスト（頻度降順）
    """
    return counter.most_common(n)

    """

    """
    if not items:
        return "No words found."

    max_word_len = max(len(word) for word, _ in items)
    
    lines = []
    header = f"{'Rank':<6} {'Word':<{max_word_len}} {'Count':>10}"
    separator = '-' * (6 + max_word_len + 10 + 3)
    
    lines.append(header)
    lines.append(separator)
    
    for rank, (word, count) in enumerate(items, start=1):
        lines.append(f"{rank:<6} {word:<{max_word_len}} {count:>10}")
