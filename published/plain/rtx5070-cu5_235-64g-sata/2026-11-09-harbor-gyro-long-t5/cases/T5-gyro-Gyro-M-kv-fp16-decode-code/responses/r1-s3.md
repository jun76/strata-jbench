#!/usr/bin/env python3
"""
text_freq.py — CLI tool to count word frequencies in a text file
and display the top N results in a table.

Usage:
    python text_freq.py <file> [--top N] [--case-insensitive] [--min-length L]

Example:
    python text_freq.py sample.txt --top 10 --case-insensitive --min-length 3
"""

import argparse

def parse_args(argv: List[str] = None) -> argparse.Namespace:
    """Parse command-line arguments.

    Args:
        argv: List of command-line arguments (defaults to sys.argv[1:]).

    Returns:
        Parsed arguments namespace.
    """
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and display top N results.",
    )
    parser.add_argument(
        "file",
        help="Path to the text file to analyze.",
    )
    parser.add_argument(
        "--top",
        type=int,
        default=10,
        help="Number of top words to display (default: 10).",
    )
    parser.add_argument(
        "--case-insensitive",
        action="store_true",
        help="Perform case-insensitive word counting.",
    )
    parser.add_argument(
        "--min-length",
        type=int,
        default=1,
        help="Minimum word length to include (default: 1).",
    )
    return parser.parse_args(argv)


def read_file(path: str) -> str:
    """Read the entire content of a text file.

    Args:
        path: Path to the text file.

    Returns:
        The file content as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()
