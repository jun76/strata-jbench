    python word_freq.py sample.txt -n 10

def read_text_file(file_path: str) -> str:
    """指定されたファイルを読み込み、テキスト内容を返す。

    Args:
        file_path: 読み込むファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み込み中にエラーが発生した場合。
    """
    with open(file_path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_insensitive: bool = False) -> List[str]:
    """テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_insensitive: Trueの場合、単語を小文字に変換して比較する。

    Returns:
        単語のリスト。
    """
    # 英数字の連続を単語として抽出
    words = re.findall(r"[a-zA-Z0-9]+", text)
    if case_insensitive:
        words = [w.lower() for w in words]
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
