"""
word_count.py - A command-line tool to count word frequencies from a text file.

Usage:
    python word_count.py <input_file> [-n N] [--case-sensitive] [--sort-by {frequency,alphabet}]

This script reads a text file, tokenizes it into words (lowercased by default),
counts the frequency of each word, and prints the top N most frequent words
in a formatted table.
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: Input text string.
        case_sensitive: If True, preserve case; otherwise, lowercase all words.

    Returns:
        List of word tokens.
    """
    # Split on non-alphanumeric characters (including underscores as separators)
    # This regex finds sequences of alphanumeric characters
    tokens = re.findall(r'[a-zA-Z0-9]+', text)
    if not case_sensitive:
        tokens = [token.lower() for token in tokens]
    return tokens


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: Input text string.
        case_sensitive: If True, preserve case; otherwise, lowercase all words.

    Returns:
        Counter object mapping words to their frequencies.
    """
    tokens = tokenize(text, case_sensitive=case_sensitive)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from a Counter.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples, sorted by frequency descending,
        then alphabetically for ties.
    """
    # Sort by frequency descending, then by word ascending for ties
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
