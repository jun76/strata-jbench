"""
Command-line tool to count word frequencies from a text file and display top N results.

Usage:
    python word_freq_counter.py <file> [-n N] [-c CASE]
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def count_word_frequencies(text: str, case_sensitive: bool = False) -> Counter[str]:
    """Count word frequencies in the given text.

    Args:
        text: The input text to analyze.
        case_sensitive: If False, words are lowercased before counting.

    Returns:
        A Counter mapping words to their frequencies.
    """
    if case_sensitive:
        words = re.findall(r"\b\w+\b", text)
    else:
        words = [w.lower() for w in re.findall(r"\b\w+\b", text)]

    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words from a Counter.

    Args:
        counter: A Counter mapping words to frequencies.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending, then by word ascending.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(counter.items(), key=lambda item: (-item[1], item[0]))
    return sorted_items[:n]


def format_table(items: List[Tuple[str, int]]) -> str:
    """Format word frequency items as a table string.

    Args:
        items: A list of (word, count) tuples.

    Returns:
        A formatted table string with columns: Rank, Word, Count.
    """
    if not items:
        return "No words found."

    lines = []
    header = f"{'Rank':<6} {'Word':<20} {'Count':<10}"
    separator = "-" * (len(header))
    lines.append(header)
    lines.append(separator)

    for rank, (word, count) in enumerate(items, start=1):
        line = f"{rank:<6} {word:<20} {count:<10}"
        lines.append(line
