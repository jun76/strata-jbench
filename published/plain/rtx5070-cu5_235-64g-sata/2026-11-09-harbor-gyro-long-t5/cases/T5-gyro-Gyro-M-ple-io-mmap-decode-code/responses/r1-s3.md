テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表示するツール。

def load_text_from_file(file_path: str) -> str:
    """指定されたファイルパスからテキストを読み込む。

    Args:
        file_path: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        UnicodeDecodeError: 文字コードが不正な場合。
    """
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {file_path}")
    except UnicodeDecodeError:
        raise UnicodeDecodeError(f"UTF-8 でデコードできません: {file_path}")


def tokenize_text(text: str) -> List[str]:
    """テキストから単語を抽出する。

    単語は英数字・ハイフン・アポストロフィを含む連続文字列として定義される。
    すべて小文字に変換される。

    Args:
        text: 処理対象のテキスト。

    Returns:
        単語のリスト。
    """
    # 英数字、ハイフン、アポストロフィを含む単語を抽出
    # 例: "don't", "well-known", "123"
    words = re.findall(r"\b[\w'-]+\b", text.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        Counter オブジェクト。
    """
    return Counter(words)
