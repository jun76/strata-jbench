#!/usr/bin/env python3
"""
Command-line tool to count word frequencies in a text file and display top N entries.

Usage:
    python word_count.py input.txt --top 10
    python word_count.py input.txt --top 5 --case-insensitive
"""

import argparse
import re
import sys
from collections import Counter
from typing import Dict, List, Tuple


def load_text(filepath: str) -> str:
    """Load text content from a file.

    Args:
        filepath: Path to the text file.

    Returns:
        The full text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        OSError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_insensitive: bool = True) -> List[str]:
    """Tokenize text into words, splitting on non-alphanumeric characters.

    Args:
        text: The input text string.
        case_insensitive: If True, words are lowercased for counting.

    Returns:
        A list of word tokens (non-empty, alphanumeric sequences).
    """
    if case_insensitive:
        text = text.lower()
    # Match sequences of alphanumeric characters (including underscores)
    words = re.findall(r"[a-zA-Z0-9_]+", text)
    return words


def count_words(text: str, case_insensitive: bool = True) -> Dict[str, int]:
    """Count the frequency of each word in the text.

    Args:
        text: The input text string.
        case_insensitive: If True, words are lowercased before counting.

    Returns:
        A dictionary mapping words to their occurrence counts.
    """
    words = tokenize(text, case_insensitive=case_insensitive)
    counter: Counter = Counter(words)
    return dict(counter)
