テキストファイルを読み込み、単語の出現頻度を集計し、
上位 N 件を表形式で表示するコマンドラインツール。

使用方法:
    python word_freq.py input.txt -n 10

依存関係: なし（標準ライブラリのみ）
"""

import argparse
import re

from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト（任意の文字列）

    Returns:
        小文字化された単語のリスト。非英字・数字は除去される。
    """
    # 英字と数字のみを含む単語に分割し、すべて小文字に変換
    return re.findall(r'[a-zA-Z0-9]+', text.lower())

    """

    Args:
        text: 入力テキスト

    Returns:
        単語 -> 出現回数の Counter
    """
    words = tokenize(text)
    return Counter(words)
