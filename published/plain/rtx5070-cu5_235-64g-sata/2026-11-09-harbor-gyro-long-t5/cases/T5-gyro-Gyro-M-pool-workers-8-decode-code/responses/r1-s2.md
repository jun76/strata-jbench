"""
word_freq: Count word frequencies in a text file and display the top N words.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def read_text_file(filepath: str) -> str:
    """Read the entire content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The file content as a string.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Tokenize text into words, converting to lowercase and stripping punctuation.

    Args:
        text: Input text string.

    Returns:
        List of lowercase word tokens.
    """
    # Extract words (sequences of alphanumeric characters)
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        A Counter object with word frequencies.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)

        entries: List of (word, count) tuples.

        A formatted table string with columns for rank, word, and count.
    """
