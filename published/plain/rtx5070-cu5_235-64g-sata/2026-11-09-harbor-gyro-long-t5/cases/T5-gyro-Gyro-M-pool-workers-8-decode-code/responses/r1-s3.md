def tokenize(text: str, ignore_case: bool = False, delimiters: str = r'\W+') -> List[str]:
    """
    テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト文字列。
        ignore_case: True の場合、単語を小文字に変換して区別しない。
        delimiters: 単語区切りに使う正規表現パターン。デフォルトは非英数字文字。

    Returns:
        抽出された単語のリスト。
    """
    if ignore_case:
        text = text.lower()
    # 正規表現で分割し、空文字列を除く
    words = re.split(delimiters, text)
    return [word for word in words if word]

    """

    Args:
        words: 単語のリスト。

    Returns:
        collections.Counter オブジェクト。
    """
    return Counter(words)
