#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file.

Usage:
    python word_freq.py <file> [-n TOP_N] [--case-sensitive] [--ignore-punctuation]
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple, Optional


def tokenize(text: str, ignore_punctuation: bool = False) -> List[str]:
    """Tokenize text into words.

    Args:
        text: Input text string.
        ignore_punctuation: If True, strip punctuation from tokens.

    Returns:
        List of tokens (lowercase by default).
    """
    # Split by whitespace
    tokens = text.split()

    if ignore_punctuation:
        # Remove punctuation from each token
        tokens = [re.sub(r'[^\w]', '', token) for token in tokens]
        # Filter out empty strings after stripping
        tokens = [t for t in tokens if t]

    return tokens


def count_words(text: str, ignore_punctuation: bool = False) -> Counter:
    """Count word occurrences in text.

    Args:
        text: Input text string.
        ignore_punctuation: If True, strip punctuation from tokens.

    Returns:
        Counter mapping words to their occurrence counts.
    """
    tokens = tokenize(text, ignore_punctuation=ignore_punctuation)
    # Normalize to lowercase
    tokens = [t.lower() for t in tokens]
    return Counter(tokens)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return top N words by frequency.

    Args:
        counter: Counter mapping words to counts.
        n: Number of top words to return.

    Returns:
        List of (word, count) tuples sorted by count descending, then word ascending.
    """
    # Sort by count descending, then by word alphabetically for ties
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
