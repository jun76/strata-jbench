word_freq.py — コマンドラインツール: テキストファイルから単語の出現頻度を
集計し、上位 N 件を表形式で表示する。

使い方:
    python word_freq.py INPUT_FILE [--top N] [--ignore-case] [--min-len M]

例:
    python word_freq.py sample.txt --top 10 --ignore-case --min-len 3
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, ignore_case: bool = False, min_len: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        ignore_case: True の場合、単語を小文字に変換して集計する。
        min_len: 単語の最小文字数。これより短い単語は除外される。

    Returns:
        抽出された単語のリスト（順序は出現順）。
    """
    # 単語を抽出: アルファベット・数字・アポストロフィを含む連続文字列
    # 簡易的なトークナイザー: 非英数字区切りで分割
    tokens = re.findall(r"[a-zA-Z0-9']+", text)
    if ignore_case:
        tokens = [t.lower() for t in tokens]
    # min_len 以上の単語のみ保持
    return [t for t in tokens if len(t) >= min_len]


def count_words(tokens: List[str]) -> Counter:
    """
    単語出現頻度を集計する。

    Args:
        tokens: 単語のリスト。

    Returns:
        各単語の出現回数を保持する Counter。
    """
    return Counter(tokens)
