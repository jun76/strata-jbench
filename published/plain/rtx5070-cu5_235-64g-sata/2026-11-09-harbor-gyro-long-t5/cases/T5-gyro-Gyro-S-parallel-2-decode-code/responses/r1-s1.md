def read_file(path: str) -> str:
    """テキストファイルを読み込み、内容を文字列として返す。

    Args:
        path: 読み込むファイルのパス

    Returns:
        ファイルの内容 (文字列)

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        OSError: ファイル読み込みエラー
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, ignore_case: bool = False) -> List[str]:
    """テキストを単語に分割する。

    Args:
        text: 分割対象のテキスト
        ignore_case: True の場合、単語を小文字に変換する

    Returns:
        単語のリスト
    """
    # 簡易的なトークナイザ: 空白区切り、句読点を除く

    if ignore_case:
        words = [w.lower() for w in words]
    return words
