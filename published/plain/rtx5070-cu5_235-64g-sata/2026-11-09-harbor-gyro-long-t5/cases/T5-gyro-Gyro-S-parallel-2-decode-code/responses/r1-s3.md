#!/usr/bin/env python3
"""
CLI tool to read a text file, count word frequencies, and display top N results.

Usage:
    python word_freq.py input.txt --top 10
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """Extract words from text by splitting on whitespace and stripping punctuation.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    # Convert to lowercase and extract words using regex
    words = re.findall(r'\b[a-zA-Z]+\b', text.lower())
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """Count the frequency of each word.
    
    Args:
        words: A list of words.
        
    Returns:
        A Counter object mapping each word to its occurrence count.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word counts.
        n: The number of top words to retrieve.
        
    Returns:
        A list of tuples containing (word, count), sorted by count descending.
    """
    return counter.most_common(n)


def format_table(results: List[Tuple[str, int]]) -> str:
    """Format the results as a table string.
    
    Args:
        results: A list of tuples containing (word, count).
        
    Returns:
        A formatted string representing the table.
    """
    if not results:
        return "No words found."
    
    # Determine column widths
    word_width = max(len(word) for word, _ in results)
    count_width = max(len(str(count)) for _, count in results)

    separator = f"{'-' * word_width} {'-' * count_width}"
