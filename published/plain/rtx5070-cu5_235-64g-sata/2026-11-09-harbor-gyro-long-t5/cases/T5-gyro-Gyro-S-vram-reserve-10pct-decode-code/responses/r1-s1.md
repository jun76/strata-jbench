import argparse
import sys
from collections import Counter
from typing import List, Tuple


def load_words(path: str) -> List[str]:
    """Load words from a text file.

    Args:
        path: Path to the text file.

    Returns:
        List of words extracted from the file.
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read().split()


def count_word_frequency(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: List of words.

    Returns:
        Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        List of tuples (word, count) sorted by count descending, then by word ascending.
    """
    items = counter.most_common(n)
    # Sort by count descending, then by word ascending for tie-breaking
    return sorted(items, key=lambda x: (-x[1], x[0]))


def format_table(rows: List[Tuple[str, int]]) -> str:
    """Format rows as a simple table.

    Args:
        rows: List of tuples (word, count).

    Returns:
        Formatted table string.
    """
    if not rows:
        return "No words found."

    max_word_len = max(len(word) for word, _ in rows)
    header = f"{'Word':<{max_word_len}}  {'Count':>5}"
    separator = f"{'-' * max_word_len}  {'-' * 5}"
    lines = [header, separator]
    for word, count in rows:
        lines.append(f"{word:<{max_word_len}}  {count:>5}")
    return "\n".join(lines)
