"""
word_freq.py - コマンドラインツール: テキストファイルから単語の出現頻度を集計し、上位N件をテーブル表示する。

使い方:
    python word_freq.py <file> [-n N] [-c] [-s]

オプション:
    -n, --top N       : 上位N件を表示（デフォルト: 10）
    -c, --case        : 大文字/小文字を区別する（デフォルト: 区別しない）
    -s, --sort        : 頻度降順でソート（デフォルト: 頻度降順）
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語リストに分割する。

    Args:
        text: 入力テキスト
        case_sensitive: Trueなら大文字/小文字を区別、Falseなら小文字に統一

    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()
    # 単純な空白区切りで分割し、非アルファベット/数字は除外
    words = []
    for token in text.split():
        # 単語を抽出: 英数字からなる部分のみ
        word = ''.join(ch for ch in token if ch.isalnum())
        if word:
            words.append(word)
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    テキスト中の単語出現頻度をCounterで集計する。

    Args:
        text: 入力テキスト
        case_sensitive: Trueなら大文字/小文字を区別

    Returns:
        単語 -> 出現回数 のCounter
    """
    words = tokenize(text, case_sensitive)
    return Counter(words)
