def read_text_file(path: Path) -> str:
    """
    テキストファイルを読み込み、内容文字列を返す。

    Args:
        path: 読み込むファイルのパス

    Returns:
        ファイルの内容文字列

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        OSError: ファイル読み込み失敗の場合
    """
    return path.read_text(encoding="utf-8")


def tokenize(text: str, case_insensitive: bool = False) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 分割対象のテキスト
        case_insensitive: True の場合、小文字に変換する

    Returns:
        単語のリスト
    """
    if case_insensitive:
        text = text.lower()
    # 英数字・ハイフン・アポストロフィを単語として扱い、それ以外を区切りとする
    # 簡易的なトークナイザ: 空白で分割し、句読点を除去
    import re
    tokens = re.findall(r"[a-zA-Z0-9]+(?:['\-][a-zA-Z0-9]+)*", text)
    return tokens


def count_word_frequencies(tokens: List[str]) -> Counter[str]:
    """
    単語の出現頻度を集計する。

    Args:
        tokens: 単語のリスト

    Returns:
        単語 -> 出現回数 の Counter
    """
    return Counter(tokens)
