```python
#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツール: テキストファイルから単語の出現頻度を集計し、
上位 N 件を表形式で表示する。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Optional


def tokenize(text: str) -> list[str]:
    """テキストから英数字の単語を取り出す。

    Args:
        text: 入力テキスト

    Returns:
        小文字化された単語のリスト
    """
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_words(text: str) -> Counter[str]:
    """テキスト中の単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語と出現回数の Counter
    """
    tokens = tokenize(text)
    return Counter(tokens)


def top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """Counter の上位 n 件を (word, count) のリストとして返す。

    Args:
        counter: 単語と出現回数の Counter
        n: 取得する件数

    Returns:
        (単語, 出現回数) のリスト（出現回数降順、同数なら辞書順）
    """
    # Counter.most_common は出現回数降順だが、同数時の順序は不定。
    # 辞書順で安定させるため、(-count, word) でソートする。
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(items: list[tuple[str, int]]) -> str:
    """(word, count) のリストを簡易表形式の文字列としてフォーマットする。

    Args:
        items: (単語, 出現回数) のリスト

    Returns:
        表形式の文字列
    """
    if not items:
        return ""

    max_word_len = max(len(w) for w, _ in items)
    header = f"{'Word':<{max_word_len}} {'Count':>5}"
    separator = f"{'-' * max_word_len} {'-' * 5
