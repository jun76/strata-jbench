def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: 大文字・小文字を区別するかどうか。
        min_length: 単語の最小長（この長未満の単語は除外）。

    Returns:
        抽出された単語のリスト。
    """
    # 英数字とハイフンを含む単語を抽出（簡易的なトークナイザー）
    # 必要に応じて正規表現を調整可能
    words = re.findall(r"\b[a-zA-Z0-9'-]+\b", text)

    if not case_sensitive:
        words = [w.lower() for w in words]

    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]

    return words


def count_words(words: List[str]) -> Counter[str]:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
