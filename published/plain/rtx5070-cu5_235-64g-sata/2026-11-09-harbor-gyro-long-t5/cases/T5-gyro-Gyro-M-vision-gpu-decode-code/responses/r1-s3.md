"""
word_counter.py - コマンドラインツール：テキストファイルを読み込んで単語の出現頻度を集計し、
上位 N 件を表形式で表示する。

Usage:
    python word_counter.py <file> [--top N]
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple, Dict


def count_words(text: str) -> Dict[str, int]:
    """テキストから単語の出現頻度を集計する。

    Args:
        text: 入力テキスト文字列

    Returns:
        単語をキー、出現回数を値とする辞書
    """
    # 単語を抽出（英字と数字のみを許可し、それ以外で区切る）
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return dict(Counter(words))


def get_top_n(word_counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """出現頻度上位 N 件の単語と回数を取得する。

    Args:
        word_counts: 単語と出現回数の辞書
        n: 取得する件数

    Returns:
        (単語, 回数) のリスト、頻度降順・同頻度では辞書順
    """
    # 頻度降順、同頻度ではアルファベット順（辞書順）でソート
    sorted_items = sorted(word_counts.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
