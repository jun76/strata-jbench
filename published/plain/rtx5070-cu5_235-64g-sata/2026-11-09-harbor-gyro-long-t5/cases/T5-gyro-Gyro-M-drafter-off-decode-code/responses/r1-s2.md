"""
Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words
in a tabular format.
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    Count word occurrences in the given text.

    Words are split by whitespace and normalized to lowercase.
    Punctuation is not stripped; words are taken as-is from split.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter mapping words to their occurrence counts.
    """
    words = text.split()
    # Normalize to lowercase
    normalized = [word.lower() for word in words]
    return Counter(normalized)


def get_top_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a counter.

    Args:
        counter: A Counter object mapping words to counts.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """
    Format word-frequency pairs as a table.

    Args:
        items: A list of (word, count) tuples.

    Returns:
        A formatted string representing the table.
    """
    if not items:
        return "No words found."

    # Determine column widths
    word_width = max(len(word) for word, _ in items)
    # Ensure at least reasonable width
    word_width = max(word_width, 3)

    header = f"{'Word':<{word_width}} | {'Count':>5}"
    separator = f"{'-' * word_width} | {'-' * 5}"

    lines = [header, separator]
    for word, count in items:
        line = f"{word:<{word_width}} | {count:>5}"
        lines.append(line)

    return "\n".join(lines)
