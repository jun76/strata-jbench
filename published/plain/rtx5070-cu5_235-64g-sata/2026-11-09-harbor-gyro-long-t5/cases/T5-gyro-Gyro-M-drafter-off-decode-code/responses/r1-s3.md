wordfreq: A command-line tool to count word frequencies in text files.

Usage:
    python wordfreq.py <file> [-n N] [-c] [-s]

Options:
    -n, --top N       Show top N most frequent words (default: 10)
    -c, --case-sensitive   Case-sensitive counting (default: case-insensitive)

"""

def parse_args(argv: List[str] = None) -> argparse.Namespace:

    Returns:

    """

        "-s",
        "--sort-by-frequency",
        action="store_true",
        default=False,
        help="Sort by frequency descending, then alphabetically for ties.",
    )
    return parser.parse_args(argv)
