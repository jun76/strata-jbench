コマンドラインツール: テキストファイルを読み込み、単語出現頻度の上位 N 件を表形式で表示する。

使い方:
    python wordfreq.py input.txt --top 10
    python wordfreq.py input.txt --top 10 --case-sensitive
    cat input.txt | python wordfreq.py --stdin --top 5

オプション:
    -f, --file        : 入力ファイルパス (省略時は標準入力)
    -t, --top         : 表示する上位件数 (デフォルト: 10)
    -c, --case-sensitive : 大文字・小文字を区別する (デフォルト: 区別しない)
    --stdin           : 標準入力から読み込む (--file と排他)
"""

import argparse

import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。

    Returns:
        抽出された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()

    # 英数字およびアンダースコアを含む連続文字列を単語とみなす
    # 日本語などの CJK 文字は各文字を単語として扱う（簡易版）
    # ここでは、正規表現で英数字の単語を抽出する
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度を集計する。

    Args:
        words: 単語のリスト。

    Returns:
        各単語の出現回数を格納した Counter。
    """
    return Counter(words)
