import argparse

from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    テキストから単語の出現頻度をカウントする。

    Args:
        text: 解析対象のテキスト文字列。

    Returns:
        単語から出現回数を表す Counter オブジェクト。
    """
    # 単語を抽出する（空白区切りに加え、句読点などの基本的な区切りも考慮）
    # 簡単な方法: 空白で分割し、句読点を除去
    import re
    # 英数字とハイフンを含む単語のみを抽出（それ以外は区切りとして扱う）
    words = re.findall(r"[a-zA-Z0-9\-']+", text.lower())
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 N 件の単語とカウントを返す。

    Args:
        counter: 単語から出現回数を表す Counter オブジェクト。
        n: 取得する上位件数。

    Returns:
        (単語, 出現回数) のタプルのリスト。頻度順に降順でソートされている。
        頻度が同じ場合は単語の辞書順で昇順。
    """
    # most_common は頻度降順で返すが、頻度が同じ場合は辞書順ではない
    # 頻度降順、頻度同値なら辞書順昇順でソート
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
