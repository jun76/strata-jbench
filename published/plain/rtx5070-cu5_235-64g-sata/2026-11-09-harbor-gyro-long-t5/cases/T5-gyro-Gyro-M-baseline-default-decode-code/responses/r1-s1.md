"""
text_word_counter.py
====================
A command-line tool that reads a text file, counts word occurrences,
and displays the top N words in a table format.
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    Count word occurrences in the given text.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object mapping words to their occurrence counts.
        Words are lowercased and non-alphanumeric characters are stripped.
    """
    words = []
    for line in text.splitlines():
        for word in line.split():
            # Strip non-alphanumeric characters from the ends
            cleaned = word.strip(".,;:!?\"'()[]{}-")
            if cleaned:
                words.append(cleaned.lower())
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.

    Args:
        counter: A Counter object with word counts.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
        In case of ties, words are sorted alphabetically.
    """
    # Sort by count descending, then by word ascending for ties
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
