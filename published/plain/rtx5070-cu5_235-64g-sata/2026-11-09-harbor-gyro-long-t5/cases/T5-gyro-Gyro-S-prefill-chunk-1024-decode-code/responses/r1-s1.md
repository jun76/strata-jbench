"""

import argparse

from typing import List, Tuple


def read_words_from_file(filepath: str) -> List[str]:
    """
    テキストファイルを読み込み、単語リストを返す。

    Args:
        filepath: 読み込むテキストファイルのパス

    Returns:
        単語のリスト（小文字化、句読点除去）
    """
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            content = f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {filepath}")
    except IOError:
        raise IOError(f"ファイルの読み込み失敗: {filepath}")

    # 単語を抽出: 空白で分割し、句読点・特殊文字を除去して小文字化
    words = []
    for line in content.splitlines():
        # 句読点や特殊文字を除去
        cleaned = ""
        for char in line:
            if char.isalnum() or char == "-":
                cleaned += char
            else:
                cleaned += " "
        # 空白で分割し、空文字を除く
        for word in cleaned.split():
            if word:
                words.append(word.lower())

    return words


def count_word_frequency(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        各単語とその出現回数の Counter
    """
    return Counter(words)
