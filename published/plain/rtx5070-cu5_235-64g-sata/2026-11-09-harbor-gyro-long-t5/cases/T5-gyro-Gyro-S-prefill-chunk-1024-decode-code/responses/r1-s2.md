"""コマンドラインツール: テキストファイルから単語出現頻度を集計し上位 N 件を表示する."""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。正規表現で英数字の連続を抽出する。

    Args:
        text: 入力テキスト文字列。

    Returns:
        単語のリスト。
    """
    # 英数字の連続を抽出（小文字化して集計）
    words = re.findall(r"[a-z0-9]+", text.lower())
    return words


def count_words(text: str) -> Counter[str]:
    """テキスト中の単語出現頻度を Counter で集計する。

    Args:
        text: 入力テキスト文字列。

    Returns:
        単語 -> 出現回数の Counter。
    """
    counter = Counter(tokenize(text))
    return counter


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (単語, 回数) を返す。

    Args:
        counter: 単語の出現回数を含む Counter。
        n: 取得する件数。

    Returns:
        上位 n 件の (単語, 回数) のリスト。回数降順、同点時は単語昇順。
    """
    # most_common は回数降順で返す。同点時の順序は Counter の内部実装依存だが、
    # ここでは安定性のため、回数降順・単語昇順でソートする。
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
