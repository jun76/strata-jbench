#!/usr/bin/env python3
"""Command-line tool to count word frequencies in a text file.

Usage:
    python word_freq.py <file> [-n TOP_N] [--case-sensitive] [--ignore-punctuation]
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple, Optional


def tokenize(text: str, ignore_punctuation: bool = False) -> List[str]:
    """Tokenize text into words.

    Args:
        text: Input text string.
        ignore_punctuation: If True, strip non-alphanumeric characters from tokens.

    Returns:
        List of word tokens.
    """
    # Split on whitespace and remove punctuation if requested
    tokens = text.split()
    if ignore_punctuation:
        # Remove leading/trailing punctuation, keep internal punctuation
        cleaned = []
        for token in tokens:
            # Strip non-alphanumeric characters from start and end
            stripped = token.strip()
            # Remove all non-alphanumeric characters
            cleaned_token = re.sub(r'[^a-zA-Z0-9]', '', stripped)
            if cleaned_token:
                cleaned.append(cleaned_token)
        return cleaned
    return tokens


def count_word_frequencies(
    tokens: List[str],
    case_sensitive: bool = False,
) -> Counter[str]:
    """Count word frequencies in a list of tokens.

    Args:
        tokens: List of word tokens.
        case_sensitive: If False, convert tokens to lowercase before counting.

    Returns:
        Counter mapping words to their frequencies.
    """
    if not case_sensitive:
        tokens = [t.lower() for t in tokens]
    return Counter(tokens)


def get_top_words(
    counter: Counter[str],
    top_n: int,
) -> List[Tuple[str, int]]:
    """Get top N most frequent words.

    Args:
        counter: Word frequency counter.
        top_n: Number of top words to return.

    Returns:
        List of (word, count) tuples, sorted by count descending, then alphabetically.
    """
    # Sort by count descending, then by word alphabetically for stable ordering
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:top_n]
