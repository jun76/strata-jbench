"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a tabular format.

Usage:
    python wordfreq.py [--top N] [--case-sensitive] [--min-length M] FILE
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words, filtering by minimum length and optionally
    normalizing case.

    Args:
        text: Input text to tokenize.
        case_sensitive: If False, words are lowercased for counting.
        min_length: Minimum word length to include (default 1).

    Returns:
        List of tokens that meet the criteria.
    """
    # Split on whitespace and strip punctuation-like characters at edges
    tokens: List[str] = []
    for raw in text.split():
        # Strip leading/trailing punctuation
        stripped = raw.strip(".,!?;:\"'()[]{}")
        if not stripped:
            continue
        if len(stripped) < min_length:
            continue
        if not case_sensitive:
            stripped = stripped.lower()
        tokens.append(stripped)
    return tokens


def count_words(tokens: List[str]) -> Counter:
    """
    Count occurrences of each token.

    Args:
        tokens: List of tokens to count.

    Returns:
        Counter mapping each token to its frequency.
    """
    return Counter(tokens)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words.

    Args:
        counter: Counter of word frequencies.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples sorted by descending count,
        then alphabetically for ties.
    """
    # Sort by (-count, word) to get descending count, ascending word for ties
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
