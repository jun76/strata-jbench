"""word_count.py — コマンドラインツール：テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する。"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Iterable, List, Tuple


def normalize_word(word: str) -> str:
    """単語を小文字に変換し、非英字文字を除去する。

    Args:
        word: 元の単語文字列。

    Returns:
        正規化された単語（小文字、英字のみ）。
    """
    return re.sub(r"[^a-z]+", "", word.lower())


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割し、正規化する。

    Args:
        text: 入力テキスト。

    Returns:
        正規化された単語のリスト（英字のみ、小文字）。
    """
    words = text.split()
    normalized: List[str] = []
    for word in words:
        norm = normalize_word(word)
        if norm:
            normalized.append(norm)
    return normalized


def count_words(words: Iterable[str]) -> Counter[str]:
    """単語の出現頻度を集計する。

    Args:
        words: 単語のイテラブル。

    Returns:
        各単語の出現回数を保持する Counter。
    """
    counter: Counter[str] = Counter()
    for word in words:
        counter[word] += 1
    return counter


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """最も頻繁な N 件の単語を (word, count) のタプルのリストとして返す。

    Args:
        counter: 単語の出現頻度 Counter。
        n: 取得する上位件数。

    Returns:
        (単語, 出現回数) のタプルのリスト。頻度が高い順にソートされる。
    """
    return counter.most_common(n)
