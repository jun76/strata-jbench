def load_text(filepath: str) -> str:
    """Read text from a file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If an error occurs while reading the file.
    """
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except IOError as e:
        raise IOError(f"Error reading file {filepath}: {e}")


def count_words(text: str) -> Counter:
    """Count the frequency of each word in the given text.

    Words are defined as sequences of alphanumeric characters and underscores,
    separated by non-word characters. Case is ignored (all words are lowercased).

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    import re
    # Extract words using a regex that matches sequences of word characters
    words = re.findall(r'\b[a-zA-Z_][a-zA-Z0-9_]*\b', text)
    # Normalize to lowercase for case-insensitive counting
    normalized = [w.lower() for w in words]
    return Counter(normalized)
