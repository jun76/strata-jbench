"""word_count.py - コマンドラインツール：テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する。"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。小文字化し、アルファベット/数字/ハイフンを含むトークンを抽出する。"""
    # 単語は英字、数字、ハイフン、アポストロフィを含むものとする
    pattern = re.compile(r"[a-zA-Z0-9][-a-zA-Z0-9']+")
    return [token.lower() for token in pattern.findall(text)]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を Counter で返す。"""
    return Counter(tokenize(text))


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """頻度上位 N 件の (word, count) のリストを返す。頻度が高い順、同頻度なら辞書順。"""
    items = counter.most_common(n)
    # most_common は頻度順で安定ソートだが、同頻度の場合は辞書順にしたい
    # Counter.most_common は同頻度だと挿入順（ファイル出現順）になるため、
    # 辞書順に再ソートする。
    sorted_items = sorted(items, key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]


def format_table(rows: List[Tuple[str, int]]) -> str:
    """(word, count) のリストを表形式（Markdown-like）で文字列に整形する。"""
    if not rows:
        return ""
    max_word_len = max(len(word) for word, _ in rows)
    max_count_str_len = max(len(str(count)) for _, count in rows)
    header_word = "word".ljust(max_word_len)
    header_count = "count".rjust(max_count_str_len)
    separator = "-" * (max_word_len + max_count_str_len + 1)
    lines = [
        f"{header_word} {header_count}",
        separator,
    ]
