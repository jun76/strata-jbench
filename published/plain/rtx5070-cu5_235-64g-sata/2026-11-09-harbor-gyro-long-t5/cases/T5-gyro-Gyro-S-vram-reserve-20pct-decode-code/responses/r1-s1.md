import argparse
import sys
from collections import Counter
from typing import List, Tuple


def load_words(path: str) -> List[str]:
    """Read a text file and return a list of words.

    Args:
        path: Path to the text file.

    Returns:
        A list of words extracted from the file.
    """
    words: List[str] = []
    try:
        with open(path, "r", encoding="utf-8") as f:
            for line in f:
                words.extend(line.split())
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {path}")
    return words


def count_words(words: List[str]) -> Counter[str]:
    """Count the frequency of each word in a list.

    Args:
        words: A list of words.

    Returns:
        A Counter mapping each word to its occurrence count.
    """
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top n most frequent words with their counts.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then by word ascending for ties.
    """
    items = list(counter.items())
    # Sort by count descending, then by word ascending
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]


def format_table(rows: List[Tuple[str, int]]) -> str:
    """Format rows as a simple table.

    Args:
        rows: A list of (word, count) tuples.

    Returns:
        A formatted string representing the table.
    """
    if not rows:
        return ""
    max_word_len = max(len(word) for word, _ in rows)
    header = f"{'Word'.ljust(max_word_len)} | {'Count'}"
    separator = f"{'-' * max_word_len} | {'-----'}"
    lines = [header, separator]
    for word, count in rows:
        lines.append(f"{word.ljust(max_word_len)} | {count}")
    return "\n
