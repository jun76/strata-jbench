#!/usr/bin/env python3
"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a tabular format.

Usage:
    python word_counter.py -f input.txt -n 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count word occurrences in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping words to their occurrence counts.
        Words are lowercased for case-insensitive counting.
        Only alphanumeric characters and apostrophes/hyphens within words
        are considered part of the word.
    """
    words = []
    current_word = []
    for char in text:
        if char.isalnum() or char in ("'", "-"):
            current_word.append(char)
        else:
            if current_word:
                words.append("".join(current_word).lower())
                current_word = []
    if current_word:
        words.append("".join(current_word).lower())
    return Counter(words)


def format_table(words: List[Tuple[str, int]], top_n: int) -> str:
    """Format word-frequency pairs as a tabular string.

    Args:
        words: A list of (word, count) tuples, sorted by count descending.
        top_n: The number of top entries to display.

    Returns:
        A formatted string representing the table.
    """
    if not words:
        return "No words found."

    top_words = words[:top_n]
    # Determine column widths
    word_width = max(len(word) for word, _ in top_words)
    count_width = max(len(str(count)) for _, count in top_words)

    header = f"{'Word':<{word_width}} {'Count':>{count_width}}"
    separator = f"{'-' * word_width} {'-' * count_width}"

    lines = [header, separator]
    for word, count in top_words:
        lines.append(f"{word:<{word_width}} {count:>{count_width}}")

    return "\n".join(lines)
