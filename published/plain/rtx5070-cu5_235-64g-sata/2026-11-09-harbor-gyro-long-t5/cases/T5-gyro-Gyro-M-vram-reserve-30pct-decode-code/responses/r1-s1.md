def read_text_file(filepath: str) -> str:
    """Read the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        PermissionError: If the file cannot be read.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def count_word_frequencies(text: str) -> Counter[str]:
    """Count the frequency of each word in the given text.

    Words are split by whitespace and lowercased. Non-alphabetic characters
    are stripped from each word before counting.

    Args:
        text: The input text.

    Returns:
        A Counter mapping words to their frequencies.
    """
    words = []
    for raw_word in text.split():
        # Strip non-alphanumeric characters from the start and end
        cleaned_word = raw_word.strip('.,;:!?"\'()[]{}')
        if cleaned_word:
            words.append(cleaned_word.lower())
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.

    Args:
        counter: A Counter mapping words to their frequencies.
        n: The number of top entries to return.

    Returns:
        A list of tuples (word, count) sorted by count descending.
    """
    return counter.most_common(n)
