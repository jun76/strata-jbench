"""
Top-N word frequency counter CLI tool.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.

    Words are defined as sequences of alphanumeric characters.
    The counting is case-insensitive.

    Args:
        text: Input text to analyze.

    Returns:
        Counter object with word frequencies.
    """
    words = re.findall(r'\b\w+\b', text.lower())
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most frequent words.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top entries to return.

    Returns:
        List of tuples (word, count) sorted by frequency descending.
    """
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]]) -> str:
    """Format word frequency entries as a table string.

    Args:
        entries: List of (word, count) tuples.

    Returns:
        Formatted table string.
    """
    if not entries:
        return ""
    max_word_len = max(len(word) for word, _ in entries)
    max_count_len = max(len(str(count)) for _, count in entries)
    word_width = max(max_word_len, len("Word"))
    count_width = max(max_count_len, len("Count"))

    header = f"{'Word'.ljust(word_width)}  {'Count'.rjust(count_width)}"
    separator = f"{'-' * word_width}  {'-' * count_width}"

    lines = [header, separator]
    for word, count in entries:
        lines.append(f"{word.ljust(word_width)}  {str(count).rjust(count_width)}")

    return "\n".join(lines)
