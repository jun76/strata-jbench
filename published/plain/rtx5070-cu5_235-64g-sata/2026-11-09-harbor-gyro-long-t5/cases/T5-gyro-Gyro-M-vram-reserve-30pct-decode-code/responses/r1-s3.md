def count_words(text: str) -> Dict[str, int]:
    """Count occurrences of each word in the given text.

    Words are treated case-insensitively. Non-alphanumeric characters are
    treated as separators. Words must consist of alphanumeric characters.

    Args:
        text: The input text string to analyze.

    Returns:
        A dictionary mapping lowercase words to their occurrence counts.
    """
    words = text.lower().split()
    # Filter out tokens that don't contain alphanumeric characters
    cleaned_words = []
    for word in words:
        # Extract only alphanumeric parts from the token
        filtered = ''.join(c for c in word if c.isalnum())
        if filtered:
            cleaned_words.append(filtered)
    return dict(Counter(cleaned_words))

    """Return the top N most frequent words.

    Args:
        counter: A dictionary mapping words to their counts.
        n: The number of top entries to return.

    Returns:
        A list of tuples (word, count) sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by count descending, then by word ascending for tie-breaking
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
