#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込んで単語の出現頻度を集計し、
上位 N 件を表形式で表示します。

Usage:
    python word_count.py <file> [--top N] [--case-insensitive]
"""

import argparse
import collections
import re
import sys
from typing import List, Tuple


def tokenize(text: str, case_insensitive: bool = True) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト
        case_insensitive: 大文字小文字を区別しない場合 True

    Returns:
        単語のリスト
    """
    if case_insensitive:
        text = text.lower()
    # 英数字とハイフンを単語として扱う（必要に応じて正規表現を調整）
    words = re.findall(r"[a-zA-Z0-9]+(?:[-_][a-zA-Z0-9]+)*", text)
    return words


def count_words(text: str, case_insensitive: bool = True) -> collections.Counter:
    """
    テキスト中の単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_insensitive: 大文字小文字を区別しない場合 True

    Returns:
        単語と出現回数の Counter
    """
    words = tokenize(text, case_insensitive)
    return collections.Counter(words)


def get_top_n(counter: collections.Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (単語, 回数) を返す。

    Args:
        counter: 単語の Counter
        n: 取得する件数

    Returns:
        [(単語, 回数), ...] のリスト
    """
    return counter.most_common(n)
