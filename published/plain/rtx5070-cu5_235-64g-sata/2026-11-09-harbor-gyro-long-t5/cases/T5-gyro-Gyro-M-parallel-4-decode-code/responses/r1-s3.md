"""
word_freq.py

A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize the given text into lowercase words.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words (tokens), ignoring non-alphanumeric characters.
    """
    # Extract words consisting of alphabetic characters and digits
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_words(text: str) -> Counter:
    """
    Count the frequency of each word in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from a Counter.

    Args:
        counter: A Counter object mapping words to their occurrence counts.
        n: The number of top words to retrieve.

    Returns:
        A list of (word, count) tuples sorted by count descending, then by word ascending.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
