テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で表示する CLI ツール。

Usage:
    python word_counter.py <file_path> [--top N] [--case-sensitive]

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize_text(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字・小文字を区別するかどうか。False の場合、すべて小文字に変換する。

    Returns:
        単語のリスト。非英数字は除外される。
    """
    if not case_sensitive:
        text = text.lower()

    # 英数字の連続部分のみを抽出する
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    テキスト内の単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字・小文字を区別するかどうか。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = tokenize_text(text, case_sensitive=case_sensitive)
    return Counter(words)


def get_top_n_counts(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 n 件を取得する。

    Args:
        counter: 単語頻度の Counter。
        n: 取得する件数。

    Returns:
        (単語, 出現回数) のタプルのリスト。
    """
    return counter.most_common(n)
