#!/usr/bin/env python3
"""
Command-line tool to count word frequencies in a text file and display top N.
"""

import argparse
import collections
import sys
from typing import List, Tuple, Dict


def count_words(text: str) -> Dict[str, int]:
    """
    Count the frequency of each word in the given text.
    
    Words are split by whitespace and normalized to lowercase.
    
    Args:
        text: The input text string.
        
    Returns:
        A dictionary mapping words to their counts.
    """
    words = text.lower().split()
    return collections.Counter(words)


def get_top_n(counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words.
    
    Args:
        counts: Dictionary mapping words to their counts.
        n: Number of top items to return.
        
    Returns:
        A list of tuples (word, count) sorted by count descending, then word ascending.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(counts.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]


def format_table(items: List[Tuple[str, int]]) -> str:
    """
    Format the word count items as a table string.
    
    Args:
        items: List of (word, count) tuples.
        
    Returns:
        A formatted table string.
    """
    if not items:
        return "No words found."
    
    # Find max width for word column
    max_word_len = max(len(word) for word, _ in items)
    max_count_len = len(str(max(count for _, count in items)))
    
    header = f"{'Word':<{max_word_len}}  {'Count':>{max_count_len}}"
    separator = "-" * (max_word_len + 2 + max_count_len)
    
    lines = [header, separator]
    for word, count in items:
        lines.append(f"{word:<{max_word_len}}  {count:>{max_count_len}}")
    
    return "\n".join(lines)
