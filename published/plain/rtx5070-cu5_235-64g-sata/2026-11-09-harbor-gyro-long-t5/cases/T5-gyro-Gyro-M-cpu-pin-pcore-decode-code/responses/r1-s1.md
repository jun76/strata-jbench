# Default separators for tokenization: whitespace + common punctuation
_DEFAULT_TOKEN_RE = re.compile(r"[^\w\s]+")


def tokenize(text: str, *, strip_punctuation: bool = False) -> list[str]:
    """Tokenize a text string into words.

    Args:
        text: The raw input text.
        strip_punctuation: If True, non-alphanumeric characters are removed
            before splitting. If False, whitespace is used as delimiter only.

    Returns:
        A list of word tokens (non-empty strings).
    """
    if strip_punctuation:
        # Remove punctuation but keep underscores and alphanumerics
        cleaned = _DEFAULT_TOKEN_RE.sub(" ", text)
        return [token for token in cleaned.split() if token]
    else:
        # Simple whitespace split
        return [token for token in text.split() if token]


def normalize_token(token: str, *, ignore_case: bool = False) -> str:
    """Normalize a single token.

    Args:
        token: The raw token string.
        ignore_case: If True, lowercase the token.

    Returns:
        The normalized token string.
    """
    if ignore_case:
        return token.lower()
    return token
