"""
テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表示するツール。
"""

import argparse

def load_text(file_path: str) -> str:
    """
    ファイルからテキストを読み込む。

    Args:
        file_path (str): 読み込むファイルのパス。

    Returns:
        str: ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み取り中にエラーが発生した場合。
    """
    with open(file_path, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    英数字のみを単語とし、大文字小文字は無視して小文字に変換する。

    Args:
        text (str): 分割対象のテキスト。

    Returns:
        List[str]: 小文字化された単語のリスト。
    """
    # 英数字を抽出し、小文字に変換
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words (List[str]): 単語のリスト。

    Returns:
        Counter: 単語をキー、出現回数を値とするカウント結果。
    """
    return Counter(words)
