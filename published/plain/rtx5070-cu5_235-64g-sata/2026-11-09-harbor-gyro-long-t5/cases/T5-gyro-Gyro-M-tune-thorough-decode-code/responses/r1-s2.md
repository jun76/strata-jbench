"""
Top-N word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format sorted by frequency (descending), then alphabetically.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into words by splitting on whitespace and stripping punctuation.

    Args:
        text: Input text string.

    Returns:
        List of lowercase words (alphabetic characters only).
    """
    # Extract words: sequences of alphabetic characters
    words = re.findall(r'[a-zA-Z]+', text)
    return [word.lower() for word in words]


def count_words(text: str) -> Counter[str]:
    """
    Count word occurrences in the given text.

    Args:
        text: Input text string.

    Returns:
        Counter object mapping words to their frequencies.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words, sorted by frequency descending,
    then by word ascending for ties.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        List of (word, count) tuples.
    """
    # Sort by frequency descending, then by word ascending
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
