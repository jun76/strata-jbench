#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を集計し、
上位 N 件を表形式で表示する。

使い方:
    python word_freq.py [-n N] FILE

オプション:
    -n N : 表示する上位単語数 (デフォルト: 10)
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def load_text(path: Path) -> str:
    """テキストファイルを読み込み、文字列として返す。

    Args:
        path: 読み込むファイルのパス。

    Returns:
        ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        OSError: ファイル読み込みでエラーが発生した場合。
    """
    return path.read_text(encoding="utf-8")


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。

    単語は空白区切りのトークンとし、句読点を除去し、小文字に変換する。

    Args:
        text: 分割するテキスト。

    Returns:
        正規化された単語のリスト。
    """
    # 句読点や特殊文字を除去して、空白で分割
    import re
    # 単語として有効な文字のみ抽出（英数字とハイフン・アポストロフィ等を許可）
    tokens = re.findall(r"[a-zA-Z0-9]+(?:[-']?[a-zA-Z0-9]+)*", text)
    return [token.lower() for token in tokens]


def count_words(tokens: List[str]) -> Dict[str, int]:
    """単語の出現頻度をカウントする。

    Args:
        tokens: 単語のリスト。

    Returns:
        単語 -> 出現回数の辞書。
    """
    return Counter(tokens)
