def tokenize_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字/小文字を区別する。

    Returns:
        単語のリスト。
    """
    words = []
    for line in text.splitlines():
        for token in line.split():
            # 句読点や記号を除去して純粋な単語部分を取り出す
            # ただし、ここでは単純に空白区切りのトークンをそのまま使う
            word = token.strip()
            if not word:
                continue
            if not case_sensitive:
                word = word.lower()
            words.append(word)
    return words


def count_frequencies(words: List[str], min_length: int = 1) -> Dict[str, int]:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。
        min_length: 最小単語長。これより短い単語は除外する。

    Returns:
        単語 -> 出現回数の辞書。
    """
    filtered = [w for w in words if len(w) >= min_length]
    return Counter(filtered)
