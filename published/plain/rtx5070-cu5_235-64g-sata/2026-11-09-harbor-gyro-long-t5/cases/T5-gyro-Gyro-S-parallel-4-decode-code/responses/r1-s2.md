def extract_words(text: str, pattern: str = r"[a-zA-Z]+", case_sensitive: bool = False) -> List[str]:
    """
    与えられたテキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        pattern: 単語を抽出する正規表現パターン。デフォルトは [a-zA-Z]+。
        case_sensitive: Trueなら大文字/小文字を区別する。Falseなら小文字に変換する。

    Returns:
        抽出された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    return re.findall(pattern, text)


def compute_word_frequencies(words: List[str]) -> Counter[str]:
    """

    Args:
        words: 単語のリスト。

    Returns:
        単語とその出現回数をキー・値とするCounterオブジェクト。
    """
    return Counter(words)
