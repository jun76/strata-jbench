#!/usr/bin/env python3
"""
word_freq: Count word frequencies in a text file and display top N words.

Usage:
    python word_freq.py <file> [-n N] [-i] [-s]

Options:
    -n, --top     Number of top words to display (default: 10)
    -i, --ignore-case  Case-insensitive matching (default: False)
    -s, --sort-by  Sort order: 'freq' (frequency) or 'alpha' (alphabetical) (default: freq)
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path


def read_file(filepath: str) -> str:
    """Read a text file and return its content.

    Args:
        filepath: Path to the text file.

    Returns:
        The file content as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    path = Path(filepath)
    if not path.exists():
        raise FileNotFoundError(f"File not found: {filepath}")
    return path.read_text(encoding="utf-8")


def tokenize(text: str, ignore_case: bool = False) -> list[str]:
    """Tokenize text into words.

    Args:
        text: Input text string.
        ignore_case: If True, convert words to lowercase.

    Returns:
        List of word tokens (only alphabetic/numeric words, no punctuation).
    """
    # Extract words: sequences of alphanumeric characters
    words = re.findall(r"\b[a-zA-Z0-9]+\b", text)
    if ignore_case:
        words = [w.lower() for w in words]
    return words


def count_words(text: str, ignore_case: bool = False) -> Counter:
    """Count word frequencies in the given text.

    Args:
        text: Input text string.
        ignore_case: If True, treat words case-insensitively.

    Returns:
        Counter object with word frequencies.
    """
    words = tokenize(text, ignore_case=ignore_case)
    return Counter(words)
