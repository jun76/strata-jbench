"""
Command-line tool to count word frequencies in a text file and display top N words.

Usage:
    python word_frequency.py input.txt -n 10
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def read_file(filepath: str) -> str:
    """Read the contents of a text file.

    Args:
        filepath: Path to the text file to read.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Tokenize text into words, lowercased and stripped of punctuation.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Extract words: sequences of alphanumeric characters
    words = re.findall(r"[a-zA-Z0-9']+", text.lower())
    return words


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.

    Args:
        text: The input text.

    Returns:
        A Counter mapping words to their frequencies.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words from a Counter.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)
