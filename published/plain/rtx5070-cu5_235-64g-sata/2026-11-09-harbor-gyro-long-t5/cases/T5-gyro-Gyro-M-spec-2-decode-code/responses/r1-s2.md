import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in the given text.

    Words are split on whitespace and lowercased.

    Args:
        text: The input text to analyze.

    Returns:
        A Counter mapping words to their occurrence counts.
    """
    words = text.lower().split()
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words from a Counter.

    Args:
        counter: A Counter object mapping words to counts.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then by word ascending for ties.
    """
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]], width: int = 30) -> str:
    """Format entries as a simple text table.

    Args:
        entries: List of (word, count) tuples.
        width: Width of the word column.

    Returns:
        A formatted string representing the table.
    """
    lines = [
        f"{'Word':<{width}} {'Count':>5}",
        "-" * (width + 6),
    ]
    for word, count in entries:
        lines.append(f"{word:<{width}} {count:>5}")
    return "\n".join(lines)
