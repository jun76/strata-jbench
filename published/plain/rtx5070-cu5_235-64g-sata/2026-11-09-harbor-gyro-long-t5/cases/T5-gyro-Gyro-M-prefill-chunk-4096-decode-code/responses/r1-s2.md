"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.
"""

import argparse
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count word occurrences in a given text.

    Args:
        text: Input text string.

    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    # Normalize: lowercase, strip punctuation-like characters by splitting on non-alphanumeric
    # For simplicity, split on whitespace and strip punctuation from each token
    words: List[str] = []
    for token in text.split():
        # Strip leading/trailing punctuation and convert to lowercase
        cleaned = token.lower().strip(".,;:!?'\"()[]{}")
        if cleaned:
            words.append(cleaned)
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.

    Args:
        counter: A Counter object mapping words to counts.
        n: Number of top words to return.

    Returns:
        A list of tuples (word, count) for the top N words.
    """
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """Format a list of (word, count) tuples as a table string.

    Args:
        rows: List of tuples with word and count.

    Returns:
        A formatted table string.
    """
    if not rows:
        return "(no words found)"

    # Determine column widths
    word_width = max(len(word) for word, _ in rows)
    count_width = max(len(str(count)) for _, count in rows)

    header = f"{'Word':<{word_width}} | {'Count':>{count_width}}"
    separator = f"{'-' * word_width} | {'-' * count_width}"

    lines = [header, separator]
    for word, count in rows:
        lines.append(f"{word:<{word_width}} | {count:>{count_width}}")

    return "\n".join(lines)
