"""
word_freq.py - Command-line tool to count word frequencies in text files.

Usage:
    python word_freq.py <file> [--top N] [--case-sensitive] [--min-length L]

Example:
    python word_freq.py sample.txt --top 10
"""

import argparse
import sys
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def read_file(path: str) -> str:
    """Read the content of a text file.

    Args:
        path: Path to the text file.

    Returns:
        The file content as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """Tokenize text into words.

    Args:
        text: The input text.
        case_sensitive: If True, preserve case. If False, lowercase all words.
        min_length: Minimum length of a word to include.

    Returns:
        A list of words.
    """
    if not case_sensitive:
        text = text.lower()
    # Extract words: sequences of alphanumeric characters
    words = re.findall(r"\b\w+\b", text)
    # Filter by minimum length
    words = [w for w in words if len(w) >= min_length]
    return words


def count_words(words: List[str]) -> Counter:
    """Count word frequencies.

    Args:
        words: List of words.

    Returns:
        A Counter mapping words to their frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples sorted by descending frequency.
    """
    return counter.most_common(n)
