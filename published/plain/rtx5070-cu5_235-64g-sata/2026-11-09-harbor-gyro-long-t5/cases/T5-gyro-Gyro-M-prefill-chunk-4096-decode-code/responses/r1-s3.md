"""
A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a table format.

Usage:
    python word_counter.py --file path/to/file.txt --top 10
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """
    Count the frequency of each word in the given text.

    Words are extracted using a regex that matches sequences of alphanumeric
    characters (and underscores). The comparison is case-insensitive.

    Args:
        text: The input text to process.

    Returns:
        A Counter object mapping lowercase words to their occurrence counts.
    """
    # Extract words: sequences of word characters (letters, digits, underscores)
    words = re.findall(r'\b\w+\b', text.lower())
    return Counter(words)


def get_top_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.

    Args:
        counter: A Counter object mapping words to counts.
        n: The number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending, then alphabetically.
    """
    # Counter.most_common returns elements in descending order of count.
    # For ties, we want alphabetical order, so we sort explicitly.
    items = counter.most_common()
    # Sort by count descending, then word ascending for ties
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
