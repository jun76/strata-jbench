#!/usr/bin/env python3
"""
word_freq CLI tool: Read a text file, count word frequencies, and display top N entries as a table.

Usage:
    python word_freq.py --file input.txt --top 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def read_file(filepath: str) -> str:
    """Read the contents of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        UnicodeDecodeError: If the file cannot be decoded as UTF-8.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words, ignoring non-alphanumeric characters.

    Args:
        text: Input text string.

    Returns:
        List of lowercase word tokens.
    """
    # Extract words (sequences of alphanumeric characters) and convert to lowercase
    return re.findall(r"[a-zA-Z0-9]+", text.lower())


def count_word_frequencies(text: str) -> Counter[str]:
    """Count the frequency of each word in the given text.

    Args:
        text: Input text string.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words and their counts.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top entries to return.

    Returns:
        A list of tuples (word, count) sorted by count descending, then by word ascending.
    """
    # Counter.most_common returns items sorted by count descending
    # To break ties by word ascending, we need custom sorting
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
