def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストに分割する。

    - 大文字・小文字を区別しない（すべて小文字に変換）。
    - 英数字とハイフンを含む単語のみを抽出。
    - 1文字以上の単語のみを対象とする。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 小文字化された単語のリスト。
    """
    # 正規表現で「英数字とハイフンを含む連続文字列」を抽出
    words = re.findall(r'[a-z0-9]+(?:-[a-z0-9]+)*', text.lower())
    return words


def count_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度を計算する。

    Args:

    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)
