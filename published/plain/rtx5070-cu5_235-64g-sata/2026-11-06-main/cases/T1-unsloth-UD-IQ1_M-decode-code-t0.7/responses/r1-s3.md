wordfreq: A command-line tool to count word frequencies in text files and display top N words.

Usage:
    python wordfreq.py <file> [-n N] [--case-sensitive] [--output-format {table,csv,json}]
"""

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text string.
        case_sensitive: If False, convert all words to lowercase before counting.

    Returns:
        A list of word tokens (strings).
    """
    # Extract words using regex (sequences of alphanumeric characters)
    words = re.findall(r'\b[a-zA-Z0-9]+(?:\'[a-zA-Z]+)?\b', text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of word tokens.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count) sorted by frequency descending, then alphabetically.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
