"""
word_freq.py - コマンドラインツール: テキストファイルから単語出現頻度を集計し、上位N件を表形式で表示する。

使い方:
    python word_freq.py <file> [-n N] [--case-sensitive] [--min-length L]

例:
    python word_freq.py sample.txt -n 10
    python word_freq.py sample.txt --case-sensitive --min-length 3
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、すべて小文字に変換する。
        min_length: 抽出する単語の最小文字数。

    Returns:
        抽出された単語のリスト。
    """
    # 単語パターン: 英数字とハイフンを含む連続文字列
    pattern = r"[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*"
    words = re.findall(pattern, text)

    if not case_sensitive:
        words = [w.lower() for w in words]

    # 最小長でフィルタ
    words = [w for w in words if len(w) >= min_length]

    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
