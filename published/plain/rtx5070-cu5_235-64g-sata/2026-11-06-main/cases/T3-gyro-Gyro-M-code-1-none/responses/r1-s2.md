def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    """
    Merge all overlapping or touching closed intervals.
    
    Args:
        intervals: A list of intervals, where each interval is [start, end].
                   May be unsorted and may be empty. The input list is not modified.
    
    Returns:
        A new list of merged, non-overlapping (and non-touching) closed intervals,
        sorted by start value.
    """
    if not intervals:
        return []
    
    # Sort by start, then by end
    sorted_intervals = sorted(intervals, key=lambda x: (x[0], x[1]))
    
    merged = []
    current_start, current_end = sorted_intervals[0][0], sorted_intervals[0][1]
    
    for i in range(1, len(sorted_intervals)):
        next_start, next_end = sorted_intervals[i][0], sorted_intervals[i][1]
        # If touching or overlapping (closed intervals, so start <= current_end means overlap/touch)
        if next_start <= current_end:
            current_end = max(current_end, next_end)
        else:
            merged.append([current_start, current_end])
            current_start, current_end = next_start, next_end
    
    merged.append([current_start, current_end])
    return merged
