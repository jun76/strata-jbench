#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツール: テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する。

Usage:
    python word_freq.py <file> [-n N] [--case-sensitive] [--min-length L]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
        min_length: 単語の最小長（この長さ未満の単語は除外）。

    Returns:
        抽出された単語のリスト。
    """
    # 英数字とハイフンを含む単語を抽出（日本語などの文字も考慮）
    # 簡易的に: 英数字、ハイフン、アポストロフィを含む連続文字列
    pattern = r"[a-zA-Z0-9'\-]+"
    words = re.findall(pattern, text)

    if not case_sensitive:
        words = [w.lower() for w in words]

    # 最小長でフィルタ
    words = [w for w in words if len(w) >= min_length]

    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語 -> 出現回数の Counter。
    """
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の (単語, 回数) を返す。

    Args:
        counter: 単語 -> 出現回数の Counter。
        n: 上位件数。

    Returns:
        (単語, 回数) のリスト（回数降順）。
    """
    return counter.most_common(n)
