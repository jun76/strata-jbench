def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するか。False の場合、すべて小文字に変換。
        min_length: 単語の最小長。これ未満の単語は除外。

    Returns:
        抽出された単語のリスト。
    """
    # 単語を抽出: 英数字とハイフンを含む連続文字列
    words = re.findall(r"[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*", text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    words = [w for w in words if len(w) >= min_length]
    return words


def count_frequencies(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
