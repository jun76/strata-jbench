#!/usr/bin/env python3
"""
word_freq.py - 単語出現頻度集計ツール

テキストファイルを読み込み、単語の出現頻度を集計し、上位 N 件を表形式で表示する。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    # 英数字とハイフンを含む単語を抽出（日本語は除外）
    words = re.findall(r"[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*", text)
    return [w.lower() for w in words]


def count_words(text: str) -> Counter:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    words = extract_words(text)
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counter から上位 n 件の (word, count) を返す。

    Args:
        counter: 単語頻度の Counter。
        n: 取得する件数。

    Returns:
        (word, count) のタプルリスト。
    """
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """(word, count) のリストを表形式の文字列に変換する。

    Args:
        rows: (word, count) のタプルリスト。

    Returns:
        表形式の文字列。
    """
    if not rows:
        return "No words found."

    # 列幅を計算
    word_width = max(len(word) for word, _ in rows)
    count_width = max(len(str(count)) for _, count in rows)

    # ヘッダー
    header = f"{'Word':<{word_width}} | {'Count':>{count_width}}"
    separator = f"{'-' * word_width}-+-{'-' * count_width
