```python
"""
word_freq: Command-line tool to count word frequencies in a text file.

Usage:
    python word_freq.py <file> [-n TOP_N] [-e ENCODING] [-c] [--no-sort]

Examples:
    python word_freq.py sample.txt -n 10
    python word_freq.py sample.txt -c -n 5
"""

import argparse
import re
import sys
import unicodedata
from collections import Counter
from typing import Any


def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """Normalize a word by stripping non-alphanumeric characters and optionally lowercasing.

    Args:
        word: The raw word token.
        case_sensitive: If False, convert to lowercase. If True, preserve case.

    Returns:
        The normalized word.
    """
    # Strip surrounding punctuation/whitespace for token boundary handling
    stripped = word.strip()
    if not stripped:
        return stripped

    # Normalize Unicode: NFC form for consistent comparison
    stripped = unicodedata.normalize("NFC", stripped)

    if not case_sensitive:
        stripped = stripped.lower()

    return stripped


def tokenize(text: str) -> list[str]:
    """Tokenize text into words, keeping alphanumeric sequences (including apostrophes within words).

    Args:
        text: The input text.

    Returns:
        A list of word tokens.
    """
    # Match sequences of letters, digits, and apostrophes (for contractions like "don't")
    pattern = re.compile(r"[A-Za-z0-9']+|[\u0080-\uffff]+")
    tokens = pattern.findall(text)
    return tokens


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """Count the frequency of each word in the given text.

    Args:
        text: The input text.
        case_sensitive: Whether to preserve case when counting.

    Returns:
        A Counter mapping normalized words to their occurrence counts.
    """
    tokens = tokenize(text)
    normalized = [normalize_word(w, case_sensitive) for w in tokens]
    # Filter out empty strings
    normalized = [w for w in normalized if w]
    counter = Counter(normalized)
    return counter
