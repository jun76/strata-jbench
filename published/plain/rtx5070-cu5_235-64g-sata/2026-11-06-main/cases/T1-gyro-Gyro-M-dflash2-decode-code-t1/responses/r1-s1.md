"""
テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で出力する CLI ツール。

用法:
    python word_freq.py <file> [-n N] [--case-sensitive]
"""

import argparse
import sys
import re
from collections import Counter
from pathlib import Path


def tokenize(text: str, case_sensitive: bool = False) -> list[str]:
    """テキストを単語のリストに分割する。

    Args:
        text (str): 入力テキスト。
        case_sensitive (bool, optional): 大文字小文字を区別するかどうか。デフォルトは False（区別しない）。

    Returns:
        list[str]: 単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 単語を抽出（アルファベット・数字・ハイフンを含む）
    return re.findall(r"[a-z0-9'-]+|[^\w\s]", text, flags=re.IGNORECASE if not case_sensitive else 0)


def count_words(text: str, top_n: int = 10, case_sensitive: bool = False) -> Counter:
    """テキスト内の単語出現頻度を集計し、上位 top_n 件を返す。

    Args:
        text (str): 入力テキスト。
        top_n (int, optional): 返す上位件数。デフォルトは 10。
        case_sensitive (bool, optional): 大文字小文字を区別するかどうか。デフォルトは False。

    Returns:
        Counter: 単語から出現回数をマッピングした Counter。上位 top_n 件が含まれる。
    """
    words = tokenize(text, case_sensitive=case_sensitive)
    counter = Counter(words)
    # most_common() は Counter を返すので、そのまま使う
    return counter.most_common(top_n)
