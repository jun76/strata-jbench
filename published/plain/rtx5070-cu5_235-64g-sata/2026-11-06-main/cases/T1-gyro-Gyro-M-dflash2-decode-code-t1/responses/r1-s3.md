#!/usr/bin/env python3
"""Word frequency counting CLI tool.

Reads a text file, counts word occurrences, and displays the top N words
in a formatted table.

Usage:
    python main.py input.txt -n 10
    python main.py input.txt --top 5 --ignore-case
"""

import argparse

from collections import Counter

def load_text(filepath: str, encoding: str = "utf-8") -> str:
    """Load the entire text content from a file.

    Args:
        filepath: Path to the input text file.
        encoding: File encoding (default: utf-8).

    Returns:
        The full text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        UnicodeDecodeError: If decoding fails.
    """
    with open(filepath, "r", encoding=encoding) as f:
        return f.read()


def count_words(text: str, ignore_case: bool = False, min_length: int = 1) -> Counter:
    """Count word occurrences in the given text.

    Words are defined as whitespace-separated tokens.
    Only tokens with length >= min_length are counted.

    Args:
        text: Input text string.
        ignore_case: If True, normalize words to lowercase before counting.
        min_length: Minimum token length to include in the count.

    Returns:
        A Counter object mapping each word to its occurrence count.
    """
    words: list[str] = text.split()
    if ignore_case:
        words = [w.lower() for w in words]
    # Filter out tokens shorter than min_length
    words = [w for w in words if len(w) >= min_length]
    return Counter(words)
