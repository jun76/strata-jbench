word_counter.py - Count word frequencies in a text file and display top N entries.

Usage:
    python word_counter.py <input_file> [--top N] [--case-sensitive]

Example:
    python word_counter.py sample.txt --top 10
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple, Optional


def tokenize_text(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize text into words (lowercased by default).

    Args:
        text: The input text to tokenize.
        case_sensitive: If True, preserve case. If False, convert to lowercase.

    Returns:
        A list of word tokens.
    """
    if not case_sensitive:
        text = text.lower()
    # Extract words: sequences of alphanumeric characters (including underscores)
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return words


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word in the list.

    Args:
        words: A list of word tokens.

    Returns:
        A Counter object mapping words to their counts.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
