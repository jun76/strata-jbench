#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込んで単語の出現頻度を集計し、
上位 N 件を表形式で表示するコマンドラインツール。

import argparse

from collections import Counter

from typing import Dict, List, Tuple


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。

    正規表現 [a-zA-Z']+ を使用し、大文字小文字は区別しない(小文字に変換)。
    アポストロフィを含む単語(it's, don't など)も1単語として扱う。

    Args:
        text: 入力テキスト文字列

    Returns:
        小文字化された単語のリスト
    """
    return [word.lower() for word in re.findall(r"[a-zA-Z']+", text)]


def count_words(words: List[str]) -> Dict[str, int]:
    """単語リストから出現頻度を集計する。

    Args:
        words: 単語のリスト

    Returns:
        単語 -> 出現回数 の辞書
    """
    return dict(Counter(words))


def top_n_words(word_counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """最も多く出現した上位 n 件の単語を返す。

        word_counts: 単語 -> 出現回数の辞書
        n: 取得する上位件数

    Returns:
        (単語, 出現回数) のタプルのリスト
    """
    return sorted(word_counts.items(), key=lambda x: (-x[1], x[0]))[:n]
