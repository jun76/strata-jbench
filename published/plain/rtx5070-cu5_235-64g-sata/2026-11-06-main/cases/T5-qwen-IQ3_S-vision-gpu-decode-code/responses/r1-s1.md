"""
word_freq CLI tool: Read a text file, count word frequencies, and display top N words in a table.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def read_file(filepath: str) -> str:
    """Read and return the contents of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The full text content of the file.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """Extract words from text, lowercased, ignoring non-alphanumeric characters.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase words.
    """
    # Find sequences of alphanumeric characters (including underscores) as words
    words = re.findall(r"[a-zA-Z0-9_]+", text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """Count occurrences of each word.

    Args:
        words: List of words.

    Returns:
        A Counter object mapping each word to its count.
    """
    return Counter(words)


def get_top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a counter.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending, then word ascending.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
