#!/usr/bin/env python3
"""
wordfreq.py - テキストファイルから単語の出現頻度を集計し、上位N件を表形式で表示するコマンドラインツール。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    英数字（および日本語などのUnicode文字）を単語として扱い、
    記号や空白を区切りとする。

    Args:
        text: 入力テキスト。

    Returns:
        単語のリスト（小文字化済み）。
    """
    # Unicodeベースの単語抽出: \w は日本語なども含む
    # 英数字と日本語などを単語として抽出
    words = re.findall(r'\w+', text, re.UNICODE)
    # 小文字化して正規化
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    単語リストから頻度カウンターを作成する。

    Args:
        words: 単語のリスト。

    Returns:
        collections.Counter インスタンス。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度カウンターから上位N件の単語とカウントを取得する。
    カウントが同じ場合は単語の辞書順で安定ソート。

    Args:
        counter: 単語頻度カウンター。
        n: 取得する上位件数。

    Returns:
        (単語, カウント) のタプルのリスト。上位順にソート済み。
    """
    # most_common はカウント降順で返すが、同カウント時の順序は不定なので
    # 明示的にカウント降順、単語昇順でソートする
    return sorted(counter.items(), key=lambda x: (-x[1], x[0]))[:n]
