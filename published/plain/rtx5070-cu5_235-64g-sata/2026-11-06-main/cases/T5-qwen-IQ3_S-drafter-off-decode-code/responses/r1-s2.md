#!/usr/bin/env python3
"""wordfreq.py - テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示するツール。"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path

def tokenize(text: str) -> list[str]:
    """テキストから単語を抽出する。

    正規表現を使って単語（アルファベット・数字・ハイフンを含む）を抽出し、
    すべて小文字に変換してリストで返す。

    Args:
        text: 解析対象のテキスト文字列。

    Returns:
        小文字化された単語のリスト。
    """
    # 単語を構成する文字：アルファベット、数字、ハイフン
    return [word.lower() for word in re.findall(r"[a-zA-Z0-9-]+", text)]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 解析対象のテキスト文字列。

    Returns:
        単語 -> 出現回数の Counter オブジェクト。
    """
    return Counter(tokenize(text))


def get_top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """Counter から上位 n 件の (単語, 回数) を取得する。

    回数が同じ場合は、単語の辞書順で安定してソートされる。

    Args:
        counter: 単語出現頻度の Counter。
        n: 取得する件数。

    Returns:
        (単語, 回数) のタプルリスト。
    """
    # most_common(n) は回数の降順で返すが、同順位時の順序は未定義なので
    # 安定ソートとして (-count, word) で再ソートする
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
