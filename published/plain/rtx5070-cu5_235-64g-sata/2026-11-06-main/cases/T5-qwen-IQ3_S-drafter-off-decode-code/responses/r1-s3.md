import argparse

from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。
            False の場合、単語は小文字に正規化される。

    Returns:
        抽出された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 単語は英数字とアンダースコアで構成される文字列とみなす
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        各単語とその出現回数を格納した Counter オブジェクト。
    """
    return Counter(words)
