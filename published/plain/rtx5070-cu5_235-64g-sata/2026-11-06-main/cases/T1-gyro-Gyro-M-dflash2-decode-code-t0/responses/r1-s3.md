"""
Command-line tool to count word frequencies in a text file and display the top N words.

Usage:
    python word_freq.py <file> [-n N] [-c] [-s]

Options:
    -n N: Number of top words to display (default: 10)
    -c: Case-insensitive counting (default: False)
    -s: Sort by frequency descending, then alphabetically (default: True)
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_insensitive: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: Input text string.
        case_insensitive: If True, convert all words to lowercase.

    Returns:
        List of word tokens.
    """
    # Extract words using regex: sequences of alphanumeric characters
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    if case_insensitive:
        words = [w.lower() for w in words]
    return words


def count_words(text: str, case_insensitive: bool = False) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: Input text string.
        case_insensitive: If True, count words case-insensitively.

    Returns:
        Counter object with word frequencies.
    """
    words = tokenize(text, case_insensitive)
    return Counter(words)
