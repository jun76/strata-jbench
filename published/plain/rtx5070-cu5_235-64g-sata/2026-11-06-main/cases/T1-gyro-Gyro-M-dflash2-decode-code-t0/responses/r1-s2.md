def read_file(filepath: str) -> str:
    """Read the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_insensitive: bool = False) -> List[str]:
    """Tokenize text into words.

    Words are defined as sequences of alphanumeric characters.
    Non-alphanumeric characters are treated as separators.

    Args:
        text: The input text to tokenize.
        case_insensitive: If True, convert all words to lowercase.

    Returns:
        A list of word tokens.
    """
    # Extract words using regex: sequences of alphanumeric characters
    words = re.findall(r"[a-zA-Z0-9]+", text)
    if case_insensitive:
        words = [w.lower() for w in words]
    return words


def count_words(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: A list of word tokens.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    return Counter(words)
