#!/usr/bin/env python3
"""
CLI tool to count word frequencies in a text file and display the top N words.

Usage examples:
    # Basic usage
    python word_freq.py --input sample.txt --top 10
    
    # Case-insensitive and ignoring punctuation
    python word_freq.py --input sample.txt --top 5 --case-insensitive --ignore-punctuation
    
    # Sort by value (frequency) descending
    python word_freq.py --input sample.txt --top 3 --sort-by value
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple, Dict


def parse_args() -> argparse.Namespace:
    """Parse command line arguments.

    Returns:
        argparse.Namespace: Parsed arguments object.
    """
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and show top N words.",
        formatter_class=argparse.ArgumentDefaultsHelpFormatter,
    )
    parser.add_argument(
        "--input",
        "-i",
        type=str,
        required=True,
        help="Path to the input text file.",
    )
    parser.add_argument(
        "--top",
        "-n",
        type=int,
        default=10,
        help="Number of top words to display.",
    )
    parser.add_argument(
        "--case-insensitive",
        "-c",
        action="store_true",
        help="Perform case-insensitive counting.",
    )
    parser.add_argument(
        "--ignore-punctuation",
        "-p",
        action="store_true",
        help="Ignore punctuation when counting words.",
    )
    parser.add_argument(
        "--sort-by",
        "-s",
        choices=["key", "value"],
        default="value",
        help="Sort results by word (key) or frequency (value). "
        "For 'value', sorts descending; for 'key', sorts ascending (alphabetical).",
    )
    return parser.parse_args()
