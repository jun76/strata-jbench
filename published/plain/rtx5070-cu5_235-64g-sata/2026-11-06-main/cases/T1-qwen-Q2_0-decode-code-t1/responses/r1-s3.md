import argparse
import sys
from collections import Counter
from typing import Iterable, Tuple

WORD_RE = None  # keep simple: split by whitespace
def tokenize(text: str) -> Iterable[str]:
    """Split text into simple tokens (whitespace-delimited)."""
    return text.split()

def count_words(text: str) -> Counter[str]:
    """
    Count word occurrences (case-insensitive by lowercasing tokens).
    """
    return Counter(t.lower() for t in tokenize(text))

def top_n(counter: Counter[str], n: int) -> list[Tuple[str, int]]:
    """
    Return the top N (word, count) entries sorted by count desc, then word asc.
    """
    # Counter.most_common is stable but ties may be arbitrary.
    # We make ties deterministic by sorting (count desc, word asc).
    items = list(counter.items())
    items.sort(key=lambda item: (-item[1], item[0]))
    return items[:n]

def render_table(rows: Iterable[Tuple[str, int]]) -> str:
    """
    Render a fixed-width table for rows: (word, count).
    """
    if not rows:
        return ""

    words = [w for w, _ in rows]
    counts = [c for _, c in rows]
    max_word_len = max(len(w) for w in words)
    max_count_len = max(len(str(c)) for c in counts)

    header = f"{'word'.ljust(max_word_len)}  {'count'.rjust(max_count_len)}"
    sep = f"{'-' * max_word_len}  {'-' * max_count_len}"

    lines = [header, sep]
    for w, c in rows:
        lines.append(f"{w.ljust(max_word_len)}  {str(c).rjust(max_count_len)}")
    return "\n".join(lines)
