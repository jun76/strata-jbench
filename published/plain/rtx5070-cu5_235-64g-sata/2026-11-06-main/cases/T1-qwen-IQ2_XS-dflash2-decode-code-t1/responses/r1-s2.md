#!/usr/bin/env python3
"""
Text word frequency counting tool.

Reads a text file, counts word occurrences, and displays the top N results
in a table format.
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path


def tokenize(text: str) -> list[str]:
    """
    Tokenize text into lowercase words.

    Uses a regex to extract sequences of alphabetic characters as words.
    Non-alphabetic characters are ignored.

    Args:
        text: Input text string.

    Returns:
        List of lowercase word tokens.
    """
    return [match.lower() for match in re.findall(r'[a-zA-Z]+', text)]


def count_words(text: str) -> Counter[str]:
    """
    Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        Counter mapping each word to its frequency.
    """
    return Counter(tokenize(text))


def format_top_words(counter: Counter[str], top_n: int) -> list[tuple[str, int]]:
    """
    Return the top N most frequent words.

    Args:
        counter: Word frequency counter.
        top_n: Number of top words to return.

    Returns:
        List of (word, count) tuples, ordered by descending frequency.
    """
    return counter.most_common(top_n)


def format_table(rows: list[tuple[str, int]]) -> str:
