```python
#!/usr/bin/env python3
"""
Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words in a table format.
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count word occurrences in the given text."""
    words = text.lower().split()
    return Counter(words)


def format_table(counter: Counter, n: int) -> str:
    """Format the top N entries as a table string."""
    top_items = counter.most_common(n)
    if not top_items:
        return "No words found."

    # Calculate column widths
    max_word_len = max(len(word) for word, _ in top_items)
    max_count_len = max(len(str(count)) for _, count in top_items)

    header = f"{'Word':<{max_word_len}} {'Count':>{max_count_len}}"
    separator = "-" * (max_word_len + 1 + max_count_len)

    lines = [header, separator]
    for word, count in top_items:
        lines.append(f"{word:<{max_word_len}} {count:>{max_count_len}}")

    return "\n".join(lines)


def main() -> None:
    """Main entry point for the CLI tool."""
    parser = argparse.ArgumentParser(
        description="Count word occurrences and display top N results in a table."
    )
    parser.add_argument(
        "file",
        type=str,
        help="Path to the text file",
    )
    parser.add_argument(
        "-n",
        type=int,
        default=10,
        help="Number of top results to display (default: 10)",
    )

    args = parser.parse_args()

    file_path = Path(args.file)
    if not file_path.exists():
        print(f"Error: File not found: {args.file}", file=sys.stderr)
        sys.exit(1)

    text = file_path.read_text(encoding="utf-8")
    counter = count_words(text)

    table = format_table(counter, args.n)
    print(table)


if __name__ == "__main__":
    main()
```
