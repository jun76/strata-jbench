def load_text(filepath: Path) -> str:
    """指定されたテキストファイルを読み込み、内容（文字列）を返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイル内のテキスト内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
    """
    return filepath.read_text(encoding="utf-8")


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストを単語に分割する。

    非英数字を区切りとして扱い、空白・句読点などを除去する。

    Args:
        text: 分割対象のテキスト文字列。
        case_sensitive: True の場合、大文字と小文字を区別する。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()

    words: List[str] = []
    current_word: List[str] = []
    for char in text:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))

    return words


def count_frequencies(words: List[str]) -> Counter[str]:
    """単語の出現頻度を集計する。

    Args:
        words: 単語のリスト。

    Returns:
        各単語と出現回数の Counter。
    """
    return Counter(words)
