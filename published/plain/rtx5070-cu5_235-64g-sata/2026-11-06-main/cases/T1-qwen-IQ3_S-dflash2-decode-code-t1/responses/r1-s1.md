#!/usr/bin/env python3
"""A command-line tool to count word frequencies from a text file and display top N results."""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def parse_args(argv: list[str] | None = None) -> argparse.Namespace:
    """Parse command-line arguments.

    Args:
        argv: Optional list of command-line arguments (defaults to sys.argv if None).

    Returns:
        Namespace with parsed arguments including input file, top N count, and case sensitivity flag.
    """
    parser = argparse.ArgumentParser(
        description="Count word frequencies from a text file and display the top N most frequent words."
    )
    parser.add_argument(
        "file",
        help="Path to the input text file",
    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=10,
        help="Number of top frequent words to display (default: 10)",
    )
    parser.add_argument(
        "--case-sensitive",
        action="store_true",
        help="Case-sensitive word counting (default: case-insensitive)",
    )
    return parser.parse_args(argv)


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """Tokenize text into words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, convert words to lowercase.

    Returns:
        A list of words (tokens) extracted from the text.
    """
    # Extract words using regex to match sequences of alphanumeric characters and underscores
    tokens = re.findall(r"\b\w+\b", text)
    if not case_sensitive:
        tokens = [token.lower() for token in tokens]
    return tokens


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """Count word frequencies in the given text.

    Args:
        text: The input text to count words in.
        case_sensitive: If False, treat words case-insensitively.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    tokens = tokenize(text, case_sensitive)
    return Counter(tokens)
