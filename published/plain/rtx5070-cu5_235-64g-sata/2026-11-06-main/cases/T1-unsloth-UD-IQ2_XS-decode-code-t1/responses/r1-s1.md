word_freq.py — Text file word frequency analysis CLI tool.

Reads a text file, counts word occurrences (case-insensitive, alphanumeric-only),
and displays the top N words in a formatted table.

Usage:
    python word_freq.py <file> [-n N] [--min-len L] [--case-sensitive]

def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """Normalize a word by converting to lowercase and stripping non-alphanumeric characters.

    Args:
        word: The raw word token to normalize.
        case_sensitive: If True, preserve original case. Otherwise, convert to lowercase.

    Returns:
        The normalized word string.
    """
    if not case_sensitive:
        word = word.lower()
    # Strip leading/trailing non-alphanumeric characters
    word = re.sub(r'^[^a-zA-Z0-9]+|[^a-zA-Z0-9]+$', '', word)
    return word


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """Extract normalized words from a text string.

    Args:
        text: The input text.
        case_sensitive: Whether to preserve case.

    Returns:
        A list of normalized, non-empty words.
    """
    # Split on whitespace
    raw_words = text.split()
    words = []
    for w in raw_words:
        normalized = normalize_word(w, case_sensitive)
        if normalized:
            words.append(normalized)
    return words


def count_words(words: List[str]) -> Counter:
    """Count word occurrences.

    Args:
        words: List of normalized words.

    Returns:
        A Counter object mapping words to their counts.
    """
    return Counter(words)
