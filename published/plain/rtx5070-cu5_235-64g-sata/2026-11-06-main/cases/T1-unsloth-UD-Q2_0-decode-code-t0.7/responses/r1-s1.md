#!/usr/bin/env python3
"""
word_freq: コマンドラインツールでテキストファイルを読み込み、
単語の出現頻度を上位 N 件を表形式で表示する。

Usage:
    python word_freq.py -i input.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def load_text(filepath: str) -> str:
    """指定されたパスのテキストファイルを読み込み、内容を返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み取り中にエラーが発生した場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割する。

    単語は、アルファベット・数字・ハイフン・アポストロフィを許容する。
    すべて小文字に変換される。

    Args:
        text: 分割対象のテキスト。

    Returns:
        小文字化された単語のリスト。
    """
    # アルファベット、数字、ハイフン、アポストロフィを含む連続文字列を単語として抽出
    tokens = re.findall(r"[a-zA-Z0-9][a-zA-Z0-9'\-]*", text)
    return [token.lower() for token in tokens]


def count_frequencies(text: str) -> Counter[str]:
    """テキスト中の単語の出現頻度をカウントする。

    Args:
        text: カウント対象のテキスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    tokens = tokenize(text)
    return Counter(tokens)
