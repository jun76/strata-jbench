"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントして
上位 N 件を表形式で表示するコマンドラインツールです。

Usage:
    python word_counter.py <file> [--top N] [--case-insensitive]
"""

import argparse
import re

from collections import Counter

def read_file(filepath: str) -> str:
    """
    テキストファイルを読み込み、その内容を文字列として返す。

    Args:
        filepath: 読み込むファイルのパス。

    Returns:
        ファイルの内容を示す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み取り中にエラーが発生した場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_insensitive: bool = False) -> List[str]:
    """
    テキストから単語のトークンリストを生成する。

    単語は Unicode の英数字およびアンダースコアで構成される連続した文字列として抽出される。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、単語を小文字に変換する。

    Returns:
        単語のトークンリスト。
    """
    # \w は [a-zA-Z0-9_] にマッチする
    tokens = re.findall(r'\w+', text)
    if case_insensitive:
        tokens = [token.lower() for token in tokens]
    return tokens


def count_words(tokens: List[str]) -> Counter:
    """
    トークンリストから単語の出現頻度をカウントする。

    Args:
        tokens: 単語のトークンリスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(tokens)
