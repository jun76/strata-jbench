def normalize_word(word: str) -> str:
    """
    単語を小文字に変換し、前後の空白と句読点を除去する。

    Args:
        word (str): 元の単語。

    Returns:
        str: 正規化された単語。句読点のみなどで空になった場合は空文字列を返す。
    """
    # 句読点や記号を除去して単語を抽出（簡易的な方法）
    # 実際の用途では regex を使うことを推奨
    cleaned = "".join(
        char for char in word if char.isalnum() or char == "_"
    )
    return cleaned.lower()


def count_words(text: str) -> Counter:
    """

    Args:
        text (str): 入力テキスト。

    Returns:
        Counter: 単語とその出現回数をマッピングした Counter オブジェクト。
    """
    words = text.split()
    normalized_words = [normalize_word(w) for w in words]
    # 空文字列を除外
    filtered_words = [w for w in normalized_words if w]
    return Counter(filtered_words)
