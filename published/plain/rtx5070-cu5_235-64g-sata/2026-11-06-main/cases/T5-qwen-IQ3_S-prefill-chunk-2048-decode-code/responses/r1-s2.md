"""Text file word frequency counter command-line tool.

This tool reads a text file, counts the occurrences of each word,
and displays the top N most frequent words in a table format.

def read_file(filepath: str) -> str:
    """Read and return the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """Extract words from text, converting to lowercase.

    Words are defined as sequences of alphanumeric characters.
    Punctuation and other non-alphanumeric characters are stripped.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Match sequences of alphanumeric characters (words)
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_words(words: List[str]) -> Dict[str, int]:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A dictionary mapping each word to its frequency count.
    """
    return dict(Counter(words))
