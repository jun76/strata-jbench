and display the top N most frequent words in a table format.

import argparse

def read_text_file(filepath: str) -> str:
    """
    Read the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        OSError: If there is an error reading the file.
    """
    path = Path(filepath)
    return path.read_text(encoding='utf-8')


def extract_words(text: str) -> List[str]:
    """
    Extract words from the given text. Words are sequences of alphanumeric
    characters, converted to lowercase.

    Args:
        text: The input text.

    Returns:
        A list of lowercase words extracted from the text.
    """
    return re.findall(r'\b[a-zA-Z0-9]+\b', text.lower())


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word in the given list.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to retrieve.

    Returns:
        A list of tuples (word, count) for the top N most common words.
    """
    return counter.most_common(n)
