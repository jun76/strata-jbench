#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位 N 件を表形式で表示するコマンドラインツール。

import argparse
import re

from collections import Counter

def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    大文字小文字を区別せず、アルファベットと数字の連続を単語とみなす。
    
    Args:
        text: 入力テキスト
        
    Returns:
        小文字化された単語のリスト
    """
    # 正規表現で単語（アルファベットと数字の連続）を抽出
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_words(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 N 件を取得する。
    同順位の場合は辞書順でソートする。
    
    Args:
        counter: 単語頻度 Counter
        n: 取得する件数
        
    Returns:
        (単語, 出現回数) のタプルのリスト
    """
    # most_common は頻度降順だが、同頻度時の順序は保証されないため
    # 頻度降順、単語昇順でソートし直す
    sorted_items = sorted(
        counter.items(),
        key=lambda x: (-x[1], x[0])
    )
    return sorted_items[:n]
