def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """
    単語を正規化する。
    
    Args:
        word: 元の単語文字列。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、小文字に変換する。
        
    Returns:
        正規化された単語文字列。
    """
    if not case_sensitive:
        return word.lower()
    return word


def count_words(text: str, case_sensitive: bool = False) -> Dict[str, int]:
    """
    テキストから単語の出現頻度をカウントする。
    
    Args:
        text: 解析対象のテキスト文字列。
        case_sensitive: 大文字小文字を区別するかどうか。
        
    Returns:
        単語をキー、出現回数を値とする辞書。
    """
    # 基本的なトークン化: 空白と改行で分割し、句読点を除去
    # より厳密なトークン化が必要な場合は re を使用することも可能だが、
    # ここではシンプルな実装とする。
    import re
    
    # 単語を抽出する（英数字とハイフンを含む）
    words = re.findall(r'\b[\w-]+\b', text)
    
    normalized_words = [normalize_word(w, case_sensitive) for w in words]
    return dict(Counter(normalized_words))
