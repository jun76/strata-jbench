#!/usr/bin/env python3
"""Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N results
in a formatted table.
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import Optional


def tokenize(text: str) -> list[str]:
    """Split text into lowercase words.

    Args:
        text: Input text string.

    Returns:
        List of lowercase words (tokens).
    """
    # Match sequences of alphanumeric characters as words
    return re.findall(r"[a-z0-9]+", text.lower())


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in text.

    Args:
        text: Input text string.

    Returns:
        Counter mapping each word to its occurrence count.
    """
    return Counter(tokenize(text))


def format_table(counter: Counter[str], top_n: int) -> str:
    """Format word frequency results as a text table.

    Args:
        counter: Word frequency Counter.
        top_n: Number of top entries to display.

    Returns:
        Formatted table string with rank, word, and count columns.
    """
    lines: list[str] = []
    header = f"{'Rank':<6} {'Word':<20} {'Count':>6}"
    separator = "-" * (6 + 1 + 20 + 1 + 6)

    lines.append(header)
    lines.append(separator)

    for rank, (word, count) in enumerate(counter.most_common(top_n), start=1):
        lines.append(f"{rank:<6} {word:<20} {count:>6}")

    return "\n".join(lines)
