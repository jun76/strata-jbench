"""
word_freq.py — 単語出現頻度コマンドラインツール

テキストファイルを読み込み、単語の出現頻度を集計し、上位 N 件を表形式で表示します。

Usage:
    python word_freq.py input.txt --top 10
    python word_freq.py input.txt --top 5 --case-sensitive
"""

import argparse
import sys
from collections import Counter
from typing import Optional


def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """Normalize a word for counting.

    Args:
        word: Raw word token from text.
        case_sensitive: If False (default), lowercase the word.

    Returns:
        Normalized word string.
    """
    if not case_sensitive:
        return word.lower()
    return word


def extract_words(text: str, case_sensitive: bool = False) -> list[str]:
    """Extract word tokens from text.

    Words are defined as sequences of alphanumeric characters.
    Punctuation and whitespace are used as delimiters.

    Args:
        text: Input text string.
        case_sensitive: Whether to preserve case (default False).

    Returns:
        List of normalized word tokens.
    """
    words: list[str] = []
    current: list[str] = []
    for char in text:
        if char.isalnum():
            current.append(char)
        else:
            if current:
                words.append(normalize_word("".join(current), case_sensitive))
                current.clear()
    if current:
        words.append(normalize_word("".join(current), case_sensitive))
    return words


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """Count word frequencies in text.

    Args:
        text: Input text string.
        case_sensitive: Whether to preserve case (default False).

    Returns:
        Counter mapping words to their occurrence counts.
    """
    return Counter(extract_words(text, case_sensitive))
