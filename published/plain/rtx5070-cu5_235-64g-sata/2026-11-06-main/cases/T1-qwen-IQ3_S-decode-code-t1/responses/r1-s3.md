#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位 N 件の単語を表形式で出力するコマンドラインツール。

使い方:
    python word_counter.py input.txt

import argparse
import re
from collections import Counter
from typing import TextIO, List, Tuple


def read_words_from_file(file_path: str) -> List[str]:
    """
    テキストファイルから単語リストを読み取る。

    Args:
        file_path (str): 読み込むテキストファイルのパス。

    Returns:
        List[str]: 正規化された単語のリスト。
    """
    with open(file_path, 'r', encoding='utf-8') as f:
        text = f.read()
    
    # 英数字のみを含む単語を抽出（正規表現で \w+ に一致するトークン）
    words: List[str] = re.findall(r'\w+', text)
    # 小文字に変換
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """
    単語リストから単語の出現頻度をカウントする。

    Args:
        words (List[str]): 単語のリスト。

    Returns:
        Counter[str]: 単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)


def get_top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の単語と回数を取り出す。

    Args:
        counter (Counter[str]): 単語と出現回数の Counter。
        n (int): 上位の件数。

    Returns:
        List[Tuple[str, int]]: (単語, 出現回数) のタプルのリスト。
    """
    return counter.most_common(n)
