"""
word_counter: テキストファイルから単語の出現頻度をカウントし、上位 N 件を表形式で出力する CLI ツール。
"""

import argparse

from collections import Counter
from dataclasses import dataclass

@dataclass(frozen=True)
class WordCount:
    """単語と出現回数を持つデータクラス"""
    word: str
    count: int


def normalize_word(word: str) -> str:
    """
    単語を正規化する。
    
    Args:
        word: 元の単語
        
    Returns:
        正規化された単語（小文字化、句読点の除去）
    """
    import re
    # 英数字以外の文字を除く
    cleaned = re.sub(r'[^a-z0-9]', '', word.lower())
    return cleaned


def load_and_count_words(file_path: str) -> Counter[str]:
    """
    テキストファイルを読み込み、単語の出現頻度をカウントする。
    
    Args:
        file_path: テキストファイルへのパス
        
    Returns:
        単語とその出現回数を含む Counter
        
    Raises:
        FileNotFoundError: ファイルが存在しない場合
        UnicodeDecodeError: ファイルを UTF-8 で読み込めない場合
    """
    counter = Counter()
    with open(file_path, 'r', encoding='utf-8') as f:
        for line in f:
            for word in line.split():
                normalized = normalize_word(word)
                if normalized:
                    counter[normalized] += 1
    return counter
