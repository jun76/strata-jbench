def tokenize_text(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    単語は英数字（およびアンダースコア）で構成される連続部分列として扱う。
    大文字小文字は保持されるが、呼び出し元で case-insensitive な扱いが可能。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語のリスト。空文字列は含まれない。
    """
    import re
    # \w+ は [a-zA-Z0-9_] に一致。必要に応じて拡張可能。
    words = re.findall(r'\w+', text)
    return [word for word in words if word]


def count_words(text: str, case_insensitive: bool = False) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        case_insensitive: True の場合、単語を小文字化して集計する。
        
    Returns:
        単語 -> 頻度 の Counter
    """
    words = tokenize_text(text)
    if case_insensitive:
        words = [w.lower() for w in words]
    return Counter(words)
