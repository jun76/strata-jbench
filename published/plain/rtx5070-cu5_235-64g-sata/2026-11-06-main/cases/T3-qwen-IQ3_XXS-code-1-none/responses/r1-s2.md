def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    if not intervals:
        return []
    
    # Sort intervals by start point (do not modify original input)
    sorted_intervals = sorted(intervals, key=lambda x: x[0])
    
    merged = [sorted_intervals[0][:]]  # Copy to avoid modifying sorted_intervals entries
    
    for i in range(1, len(sorted_intervals)):
        current_start, current_end = sorted_intervals[i]
        last_start, last_end = merged[-1]
        
        # If overlapping or touching (closed intervals touch if end >= next start)
        if current_start <= last_end:
            merged[-1][1] = max(last_end, current_end)
        else:
            merged.append([current_start, current_end])
    
    return merged
