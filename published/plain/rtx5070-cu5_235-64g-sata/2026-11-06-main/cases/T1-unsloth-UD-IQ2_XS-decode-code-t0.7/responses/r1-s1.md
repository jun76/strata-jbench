import argparse

from pathlib import Path
from typing import List, Tuple


def read_file(filepath: Path) -> str:
    """Read the content of a text file and return it as a string.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        OSError: If the file cannot be read.
    """
    return filepath.read_text(encoding="utf-8")


def extract_words(text: str) -> List[str]:
    """Extract words from text, normalizing to lowercase and stripping punctuation.

    Words are defined as sequences of alphanumeric characters and underscores.
    Punctuation and other non-word characters are treated as separators.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words.
    """
    import re
    # Match sequences of word characters (letters, digits, underscores)
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """Count the occurrences of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object with word counts.
    """
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words and their counts.

    Args:
        counter: A Counter object with word counts.
        n: The number of top items to return.

    Returns:
        A list of tuples (word, count), sorted by count descending,
        then by word ascending for ties.
    """
    # most_common returns sorted by count descending, then by insertion order
    # For deterministic tie-breaking, we sort by count desc, then word asc
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
