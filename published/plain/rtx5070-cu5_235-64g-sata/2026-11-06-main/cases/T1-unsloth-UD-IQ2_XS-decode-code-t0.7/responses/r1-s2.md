"""
word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位 N 件表示する。

Usage:
    python word_freq.py <file> [-n TOP_N] [--case-sensitive]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize_text(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字・小文字を区別するかどうか

    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()
    # 単語のみを抽出（英数字、ハイフンでつながれた単語など）
    words = re.findall(r"[a-zA-Z0-9']+", text)
    return words


def count_word_frequencies(text: str, case_sensitive: bool = False) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字・小文字を区別するかどうか

    Returns:
        単語 -> 出現回数の Counter
    """
    words = tokenize_text(text, case_sensitive=case_sensitive)
    return Counter(words)


def get_top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 n 件の単語とその回数を返す。

    Args:
        counter: 単語と出現回数の Counter
        n: 取得する上位の件数

    Returns:
        (単語, 回数) のタプルのリスト
    """
    return counter.most_common(n)
