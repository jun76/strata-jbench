"""テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示するツール."""

import argparse
import re

from collections import Counter

def tokenize(text: str) -> List[str]:
    """テキストから単語を抽出して返す。

    Args:
        text: 解析対象のテキスト文字列。

    Returns:
        抽出された単語のリスト（すべて小文字化）。

    >>> tokenize("Hello, world! Hello World.")
    ['hello', 'world', 'hello', 'world']
    """
    # 日本語対応のため、ひらがな・カタカナ・漢字・英数字・ハイフン・アポストロフィを許容
    # シンプルに単語として「英数字（+内部の'や-）」を抽出
    return re.findall(r"[a-zA-Z0-9']+", text.lower())

    """テキスト中の単語出現頻度をCounterとして返す。

        単語をキー、出現回数を値とする Counter.

    >>> count_words("foo bar foo")
    Counter({'foo': 2, 'bar': 1})
    """
    return Counter(tokenize(text))
