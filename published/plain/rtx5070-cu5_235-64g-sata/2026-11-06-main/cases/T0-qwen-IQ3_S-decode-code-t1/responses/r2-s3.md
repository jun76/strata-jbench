#!/usr/bin/env python3
"""Command-line tool to read a text file and display the top N word frequencies in a table.

Usage:
    python word_freq.py <input_file> [-n TOP_N] [--case-sensitive]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def parse_arguments(argv: List[str] = None) -> argparse.Namespace:
    """Parse command-line arguments.

    Args:
        argv: Command-line arguments to parse. If None, sys.argv is used.

    Returns:
        Parsed arguments namespace.
    """
    parser = argparse.ArgumentParser(
        description="Read a text file and display the top N word frequencies in a table."
    )
    parser.add_argument(
        "input_file",
        help="Path to the input text file.",
    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=10,
        help="Number of top words to display (default: 10).",
    )
    parser.add_argument(
        "--case-sensitive",
        action="store_true",
        help="Preserve case when counting words (default: case-insensitive).",
    )
    return parser.parse_args(argv)


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """Extract words from text.

    Args:
        text: Input text.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.

    Returns:
        List of words.
    """
    if not case_sensitive:
        text = text.lower()
    # Match sequences of alphanumeric characters and apostrophes (to include contractions)
    words = re.findall(r"[a-zA-Z0-9']+", text)
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: List of words.

    Returns:
        Counter object with word frequencies.
    """
    return Counter(words)
