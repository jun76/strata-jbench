import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple

def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出します。
    正規表現を使って、アルファベット、数字、ハイフンで構成される文字列を単語とみなします。
    すべて小文字に変換されます。
    """
    # [a-z0-9'-]+ で一致する部分を取得し、小文字化
    return [word.lower() for word in re.findall(r"[a-zA-Z0-9'-]+", text)]

def count_words(words: List[str]) -> List[Tuple[str, int]]:
    """
    単語リストから頻度を計算し、頻度が高い順に (単語, 頻度) のリストを返します。
    頻度が同じ場合は、アルファベット順（昇順）に並べます。
    """
    counter = Counter(words)
    # key = 単語, value = 頻度
    # sort by -frequency, then by word
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items
