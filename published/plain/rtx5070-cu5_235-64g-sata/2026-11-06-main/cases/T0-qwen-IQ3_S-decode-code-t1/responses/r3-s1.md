def extract_words(text: str) -> List[str]:
    """テキストから単語リストを抽出する。

    正規表現 [A-Za-z0-9']+ を使用し、英数字とアポストロフィを含む文字列を単語として抽出する。
    単語は大文字・小文字を区別せず、すべて小文字に変換する。

    Args:
        text: 解析対象のテキスト文字列

    Returns:
        小文字化された単語リスト (重複あり)
    """
    return [word.lower() for word in re.findall(r"[A-Za-z0-9']+", text)]

        単語 -> 出現回数 の辞書
    """
    return dict(Counter(words))
