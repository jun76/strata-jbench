def read_text(file_path: str) -> str:
    """Read text from a file.

    Args:
        file_path: Path to the input text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
    """
    if not os.path.exists(file_path):
        raise FileNotFoundError(f"File not found: {file_path}")
    with open(file_path, 'r', encoding='utf-8') as f:
        return f.read()


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.

    Words are defined as sequences of alphanumeric characters.
    Words are case-insensitive (converted to lowercase).

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object mapping words to their frequency.
    """
    # Find all words (alphanumeric sequences)
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text.lower())
    return Counter(words)


def get_top_words(word_counts: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words.

    Args:
        word_counts: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        A list of tuples, each containing (word, count), sorted by count descending.
        Ties are broken by alphabetical order.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_words = sorted(
        word_counts.items(),
        key=lambda x: (-x[1], x[0])
    )
    return sorted_words[:n]
