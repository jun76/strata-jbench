#!/usr/bin/env python3
"""A command-line tool that reads a text file, counts word frequencies,
and displays the top N words in a table format."""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple, Dict


def count_words(text: str) -> Counter:
    """Count word occurrences in the given text.
    
    Words are extracted by splitting on non-alphabetic/non-numeric characters
    (case-insensitive). Returns a Counter object with lowercase words as keys.
    """
    words = re.findall(r'[a-z0-9]+', text.lower())
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words and their counts.
    
    Ties are broken alphabetically for consistent ordering.
    """
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]


def format_table(data: List[Tuple[str, int]]) -> str:
    """Format the word count data as a table string."""
    if not data:
        return ""
    
    # Calculate column widths
    word_width = max(len(word) for word, count in data)
    count_width = max(len(str(count)) for word, count in data)
    rank_width = max(len(str(rank)) for rank in range(1, len(data) + 1))
    
    # Header
    header_rank = "Rank".ljust(rank_width)
    header_word = "Word".ljust(word_width)
    header_count = "Count".rjust(count_width)
    header = f"{header_rank} | {header_word} | {header_count}"
    
    # Separator
    separator = f"{'-' * rank_width} | {'-' * word_width} | {'-' * count_width}"
    
    # Rows
    lines = [header, separator]
    for i, (word, count) in enumerate(data, start=1):
        rank_str = str(i).rjust(rank_width)
        word_str = word.ljust(word_width)
        count_str = str(count).rjust(count_width)
        lines.append(f"{rank_str} | {word_str} | {count_str}")
