"""
text_word_counter.py: コマンドラインツールでテキストファイルの単語出現頻度を集計し、
上位 N 件を表形式で表示する。
"""

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from pathlib import Path
from typing import List, Tuple


@dataclass(frozen=True)
class WordCount:
    """単語と出現回数を保持するデータクラス"""
    word: str
    count: int


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    単語は英数字のみを含むものとし、大文字は小文字に変換する。
    """
    # 英数字のみを含む単語を抽出（最小1文字）
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter[str]:
    """
    単語リストから出現頻度をカウントする。
    """
    return Counter(words)


def get_top_n_words(counter: Counter[str], n: int) -> List[WordCount]:
    """
    上位 n 件の単語を返す。
    回数が同じ場合は辞書順で並び替える。
    """
    # (-count, word) でソートすることで、回数の降順 → 単語の昇順に
    sorted_items = sorted(counter.items(), key=lambda item: (-item[1], item[0]))
    return [WordCount(word=word, count=count) for word, count in sorted_items[:n]]
