#!/usr/bin/env python3
"""
word_counter.py - テキストファイルから単語の出現頻度を集計し、上位N件を表形式で出力するツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    与えられたテキストから単語の出現回数をカウントする。
    単語は正規表現 [a-zA-Z]+ で抽出し、大文字小文字を区別せずにカウントする。

    Args:
        text (str): 解析対象のテキスト。

    Returns:
        Counter: 単語(小文字) -> 出現回数のマッピング。
    """
    # 大文字小文字を区別しない単語抽出
    words = re.findall(r'[a-zA-Z]+', text.lower())
    return Counter(words)


def get_top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counterオブジェクトから上位n件の単語とカウントを取得する。

    Args:
        counter (Counter): 単語とカウントのコレクション。
        n (int): 取得する件数。

    Returns:
        List[Tuple[str, int]]: (単語, カウント) のリスト。カウントが同じ場合は単語の辞書順で昇順に並ぶ。
    """
    # most_commonで頻度順に取得。同カウントの場合は単語の昇順にソート
    top = counter.most_common()
    # カウントでソート、同カウントなら単語でソート
    top.sort(key=lambda x: (-x[1], x[0]))
    return top[:n]
