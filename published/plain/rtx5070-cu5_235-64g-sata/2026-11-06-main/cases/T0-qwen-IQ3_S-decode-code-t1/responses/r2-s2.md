import argparse
import sys
import os
import re
from collections import Counter
from typing import List, Tuple


def read_words(file_path: str) -> List[str]:
    """Read a text file and return a list of words (lowercase, alphanumeric only).
    
    Args:
        file_path: Path to the text file.
        
    Returns:
        A list of words extracted from the file, lowercased and consisting only of
        alphanumeric characters.
        
    Raises:
        FileNotFoundError: If the file does not exist.
    """
    with open(file_path, 'r', encoding='utf-8') as f:
        content = f.read()
    # Extract words: sequences of alphanumeric characters, lowercased
    words = re.findall(r'[a-zA-Z0-9]+', content.lower())
    return words


def count_word_frequency(words: List[str]) -> Counter:
    """Count the frequency of each word in the list.
    
    Args:
        words: List of words to count.
        
    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words with their frequencies.
    
    Args:
        counter: A Counter object with word frequencies.
        n: Number of top items to return.
        
    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)

        A formatted string with a table of words and their frequencies.

    if not top_items:
