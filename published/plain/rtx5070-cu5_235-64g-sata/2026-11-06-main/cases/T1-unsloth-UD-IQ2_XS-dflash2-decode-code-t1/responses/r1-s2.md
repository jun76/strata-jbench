コマンドラインツール: テキストファイルを読み込んで単語の出現頻度を集計し、上位 N 件を表形式で表示します。

import argparse

def read_text_file(file_path: str) -> str:
    """テキストファイルを読み込んで、その内容を文字列として返します。

    Args:
        file_path: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容を表す文字列。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        OSError: 読み込み時に他のエラーが発生した場合。
    """
    with open(file_path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """テキストを単語トークンに分割します。

    単語は、英数字または Unicode 文字の連続として定義されます。
    指定された最小長未満の単語は除外されます。

    Args:
        text: トークン化するテキスト。
        case_sensitive: 大文字小文字を区別するかどうか。False の場合は小文字に変換します。
        min_len: 含まれる単語の最小長さ。これより短い単語は除外されます。

    Returns:
        トークン化された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()

    # 英数字および Unicode 文字の連続を単語として抽出
    words = re.findall(r"[^\W_]+", text, re.UNICODE)

    # 最小長未満の単語を除外
    words = [word for word in words if len(word) >= min_len]

    return words
