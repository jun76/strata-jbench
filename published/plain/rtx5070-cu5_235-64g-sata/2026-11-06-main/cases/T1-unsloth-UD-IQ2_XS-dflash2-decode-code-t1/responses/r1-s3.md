#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を上位 N 件を表形式で出力します。

使用方法:
    python word_counter.py <ファイルパス> --top N [--case-insensitive] [--ignore-punctuation]
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple, Optional, TextIO


def tokenize(text: str, case_insensitive: bool = False, ignore_punctuation: bool = True) -> List[str]:
    """
    テキストを単語リストに分割します。

    Args:
        text: 入力テキスト。
        case_insensitive: True の場合、単語を小文字に変換して正規化します。
        ignore_punctuation: True の場合、句読点を除去します。

    Returns:
        単語のリスト。
    """
    if case_insensitive:
        text = text.lower()

    if ignore_punctuation:
        # 英数字とスペース以外の文字をスペースに置換
        text = re.sub(r'[^\w\s]', ' ', text)

    # 空白で分割し、空文字列を除く
    words = [word for word in text.split() if word]
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度を取得します。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
