word_freq.py

テキストファイルを読み込み、単語の出現頻度を計算して上位 N 件を表形式で出力するツール。

def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    正規表現を用いて、アルファベット・数字・ハイフンを含む連続文字列を単語とみなす。
    すべて小文字に変換して返す。
    
    Args:
        text (str): 入力テキスト
        
    Returns:
        List[str]: 抽出された単語のリスト
    """
    # \b\w+\b は単語境界で囲まれた単語を抽出するが、日本語などには対応しない。
    # ここでは英語ベースの単語抽出を想定し、[a-zA-Z0-9'-]+ を使用。
    # 必要に応じて正規表現を調整可能。
    words = re.findall(r"[a-zA-Z0-9'-]+", text)
    return [word.lower() for word in words]


def count_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。
    
    Args:
        words (List[str]): 単語のリスト
        
    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (word, count) タプルを返す。
    
    Args:
        counter (Counter): 頻度カウント
        n (int): 取得する件数
        
    Returns:
        List[Tuple[str, int]]: 上位 n 件の (word, count)
    """
    return counter.most_common(n)
