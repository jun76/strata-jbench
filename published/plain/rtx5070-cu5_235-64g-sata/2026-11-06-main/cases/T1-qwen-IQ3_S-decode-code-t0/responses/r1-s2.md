def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。False の場合、すべて小文字に変換する。
        min_length: 抽出する単語の最小文字数。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 英数字とハイフン、アポストロフィを含む単語を抽出（例: don't, well-known）
    words = re.findall(r"[a-z0-9]+(?:['-][a-z0-9]+)*", text)
    return [w for w in words if len(w) >= min_length]

    """

    Args:

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
