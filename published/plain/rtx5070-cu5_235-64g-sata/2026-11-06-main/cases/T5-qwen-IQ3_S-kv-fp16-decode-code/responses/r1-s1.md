#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度を集計し、
上位 N 件を表形式で表示するコマンドラインツール。

Usage:
    python word_counter.py <file> [--top N] [--sort {freq,word}] [--min-len M]

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def normalize_words(text: str, min_len: int = 1) -> List[str]:
    """
    テキストを正規化し、単語のリストを返す。

    大文字・小文字を区別せず、アルファベットと数字のみを単語として抽出する。
    長さが min_len 未満の単語は除外する。

    Args:
        text (str): 入力テキスト。
        min_len (int): 単語の最小長（デフォルト: 1）。

    Returns:
        List[str]: 正規化された単語のリスト。
    """
    # 英数字のみを含む単語に分割（大文字を小文字に変換）
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return [w for w in words if len(w) >= min_len]


def count_words(text: str, min_len: int = 1) -> Counter:
    """
    テキスト中の単語出現頻度を Counter オブジェクトで返す。

    Args:
        text (str): 入力テキスト。
        min_len (int): 単語の最小長。

    Returns:
        Counter: 単語 -> 出現回数 の辞書型オブジェクト。
    """
    words = normalize_words(text, min_len=min_len)
    return Counter(words)
