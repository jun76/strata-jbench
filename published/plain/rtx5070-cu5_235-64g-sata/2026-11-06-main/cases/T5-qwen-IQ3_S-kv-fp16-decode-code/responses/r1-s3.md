#!/usr/bin/env python3
"""テキストファイルから単語頻度をカウントし、上位N件を表形式で表示するコマンドラインツール。"""

import argparse
import re
import sys
from collections import Counter
from typing import TextIO


def count_words(text: str) -> Counter[str]:
    """テキスト中の単語を抽出し、出現頻度をカウントする。

    Args:
        text: 分析対象のテキスト文字列

    Returns:
        単語をキー、出現回数を値とするCounter
    """
    # 単語を抽出（アルファベット、数字、ハイフン、アポストロフィを含む）
    words = re.findall(r"[a-z0-9]+(?:['-][a-z0-9]+)*", text.lower())
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> list[tuple[str, int]]:
    """Counterから上位n件の単語と回数を取り出す。

    Args:
        counter: 単語頻度のCounter
        n: 取得する件数

    Returns:
        (単語, 回数)のタプルのリスト、回数降順
    """
    return counter.most_common(n)


def format_table(items: list[tuple[str, int]]) -> str:
    """(単語, 回数)のリストを表形式の文字列に変換する。

    Args:
        items: (単語, 回数)のタプルのリスト

    Returns:
        表形式の文字列
    """
    if not items:
        return ""

    # 列幅を計算
    word_width = max(len(word) for word, _ in items)
    count_width = max(len(str(count)) for _, count in items)

    # ヘッダー
    header_word = "単語".ljust(word_width)
    header_count = "回数".rjust(count_width)
    separator = "-" * (word_width + 3 + count_width)
    header_line = f"{header_word} | {header_count}"

    lines = [header_line, separator]
    for word, count in items:
        line = f"{word.ljust(word_width)} | {str(count).rjust(count_width)}"
        lines.append(line)

    return "\n".join(lines)
