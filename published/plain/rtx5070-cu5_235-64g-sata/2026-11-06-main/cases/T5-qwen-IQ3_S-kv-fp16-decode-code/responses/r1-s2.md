#!/usr/bin/env python3
"""
text_stats.py
-------------
テキストファイルを読み込み、単語出現頻度を計算して上位N件を表形式で表示するツール。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple

# デフォルトの出力幅設定
DEFAULT_WIDTH_WORD = 20
DEFAULT_WIDTH_COUNT = 10


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    Args:
        text (str): 入力テキスト
        
    Returns:
        List[str]: 正規化された単語のリスト（小文字化、アルファベット・数字のみ抽出）
    """
    # 正規表現で単語（アルファベット、数字、ハイフンなど）を抽出
    # 簡易的な例: 英数字のみ抽出。日本語などを含める場合は別の正規表現が必要
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    return [word.lower() for word in words]


def count_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。
    
    Args:
        words (List[str]): 単語のリスト
        
    Returns:
        Counter: 単語ごとの出現回数
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位N件を取得する。
    
    Args:
        counter (Counter): 頻度カウント
        n (int): 取得する件数
        
    Returns:
        List[Tuple[str, int]]: (単語, 回数)のリスト、降順ソート済み
    """
    return counter.most_common(n)
