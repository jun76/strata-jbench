"""
Top-N word frequency counter CLI tool.

Reads a text file, counts word occurrences (case-insensitive, alphanumeric-only),
and displays the top N most frequent words in a tabular format.

Usage:
    python word_freq.py input.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.
    
    Words are extracted using a regex that matches sequences of alphanumeric characters.
    Matching is case-insensitive (all words are lowercased).
    
    Args:
        text: The input text to analyze.
        
    Returns:
        A Counter object mapping lowercase words to their occurrence counts.
    """
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.
        
    Returns:
        A list of (word, count) tuples, sorted by count descending, then by word ascending.
    """
    # Counter.most_common() already sorts by count descending.
    # For ties, we want alphabetical order. We can sort manually.
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
