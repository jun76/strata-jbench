#!/usr/bin/env python3
"""
word_freq.py - Text file word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a formatted table.
"""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.

    Extracts sequences of alphanumeric characters and underscores as words,
    converts them to lowercase, and filters out empty tokens.

    Args:
        text: Input text string to tokenize.

    Returns:
        List of lowercase word tokens.
    """
    # Match sequences of word characters (letters, digits, underscores)
    tokens = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return [token.lower() for token in tokens if token]


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.

    Args:
        text: Input text string to analyze.

    Returns:
        Counter mapping words to their occurrence counts.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a counter.

    Args:
        counter: Counter object containing word frequencies.
        n: Number of top items to return.

    Returns:
        List of tuples (word, count) sorted by frequency descending.
        Ties are broken alphabetically for deterministic output.
    """
    # Get most common words, using a secondary sort key for tie-breaking
    items = counter.most_common(len(counter))
    # Sort by count descending, then word ascending for stability
    sorted_items = sorted(items, key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
