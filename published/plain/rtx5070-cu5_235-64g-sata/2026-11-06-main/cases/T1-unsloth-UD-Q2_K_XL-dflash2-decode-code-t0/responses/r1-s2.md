#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツール。

Usage:
    python word_freq.py <file> [-n N] [--case-sensitive] [--min-length LEN]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか
        min_length: 単語の最小長（これ未満の単語は除外）

    Returns:
        抽出された単語のリスト
    """
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    # 例: don't, well-known, abc123
    pattern = r"[a-zA-Z0-9]+(?:['-][a-zA-Z0-9]+)*"
    words = re.findall(pattern, text)

    if not case_sensitive:
        words = [w.lower() for w in words]

    # 最小長でフィルタ
    words = [w for w in words if len(w) >= min_length]

    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 N 件を取得する。

    Args:
        counter: 単語頻度の Counter
        n: 取得する件数

    Returns:
        (単語, 出現回数) のタプルのリスト（頻度降順）
    """
    return counter.most_common(n)
