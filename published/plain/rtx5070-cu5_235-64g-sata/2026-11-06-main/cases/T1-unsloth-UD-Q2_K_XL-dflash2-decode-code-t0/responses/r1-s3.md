word_freq.py - テキストファイルから単語出現頻度を集計し、上位N件を表形式で表示するコマンドラインツール。

Usage:
    python word_freq.py <file> [-n N] [--case-sensitive] [--min-length LEN]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか
        min_length: 単語の最小長（これ未満の単語は除外）

    Returns:
        単語のリスト
    """
    # 英数字とハイフン、アポストロフィを含む単語を抽出
    # 日本語などの非ASCII文字も考慮するため、より汎用的なパターンを使用
    # ここでは一般的な英単語抽出を想定
    pattern = r"[a-zA-Z0-9'-]+"
    words = re.findall(pattern, text)
    
    if not case_sensitive:
        words = [w.lower() for w in words]
    
    # 最小長でフィルタリング
    words = [w for w in words if len(w) >= min_length]
    
    return words


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか
        min_length: 単語の最小長

    Returns:
        単語をキー、出現回数を値とするCounter
    """
    words = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(words)
