#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込み、単語出現頻度の上位 N 件を表形式で表示します。

使い方:
    python word_frequency.py input.txt [-n 10] [--ignore-case] [--no-punctuation]

オプション:
    -n, --top N: 表示する単語の件数 (デフォルト: 10)
    --ignore-case: 大文字/小文字を区別せず集計する
    --no-punctuation: 句読点や記号を除去する
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple, Optional


def extract_words(text: str, ignore_case: bool = False, remove_punctuation: bool = False) -> List[str]:
    """
    テキストから単語のリストを抽出します。

    Args:
        text: 入力テキスト
        ignore_case: True の場合、すべて小文字に変換する
        remove_punctuation: True の場合、英字・数字・空白以外の文字を除去する

    Returns:
        単語のリスト
    """
    if ignore_case:
        text = text.lower()
    if remove_punctuation:
        # 英数字と空白のみを残す
        text = re.sub(r'[^\w\s]', '', text, flags=re.UNICODE)
    # 空白・タブ・改行で分割
    words = text.split()
    return words


def count_word_frequency(words: List[str]) -> Counter[str]:
    """
    単語の出現頻度をカウントします。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    頻度の上位 N 件の単語とカウントを取得します。

    Args:
        counter: 単語頻度の Counter
        n: 取得する件数

    Returns:
        (単語, 頻度) のタプルのリスト、頻度降順
    """
    return counter.most_common(n)
