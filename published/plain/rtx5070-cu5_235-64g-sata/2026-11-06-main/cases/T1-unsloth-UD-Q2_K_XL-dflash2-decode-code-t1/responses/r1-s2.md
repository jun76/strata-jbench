"""
word_freq.py - A command-line tool to count word frequencies in a text file.

Reads a text file, counts word occurrences, and displays the top N most frequent
words in a formatted table.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Dict


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Words are defined as sequences of alphanumeric characters and underscores.
    All tokens are converted to lowercase for case-insensitive counting.

    Args:
        text: The input text string.

    Returns:
        A list of lowercased word tokens.
    """
    # Extract words: sequences of word characters (letters, digits, underscores)
    return re.findall(r'\b\w+\b', text.lower())


def count_words(text: str) -> Dict[str, int]:
    """
    Count the frequency of each word in the given text.

    Args:
        text: The input text string.

    Returns:
        A dictionary mapping each word to its count.
    """
    tokens = tokenize(text)
    return dict(Counter(tokens))


def get_top_n(counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from the counts dictionary.

    Args:
        counts: A dictionary mapping words to their counts.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then by word ascending for ties.
    """
    sorted_items = sorted(
        counts.items(),
        key=lambda item: (-item[1], item[0])
    )
    return sorted_items[:n]
