### 2. `word_freq.py`

```python
#!/usr/bin/env python3
"""
テキストファイルを読み込み、単語の出現頻度を上位N件で表示するツール。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    正規表現を用いて単語（英数字とハイフン、アポストロフィを含む）を抽出し、
    小文字に変換する。

    Args:
        text: 解析対象のテキスト文字列。

    Returns:
        小文字化された単語のリスト。
    """
    # 単語パターン: 英数字、ハイフン、アポストロフィを含む連続文字列
    # 例: "don't", "well-known", "123"
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return [word.lower() for word in words]


def count_word_frequencies(text: str) -> Counter[str]:
    """
    テキスト内の単語頻度をカウントする。

    Args:
        text: 解析対象のテキスト文字列。

    Returns:
        単語をキー、出現回数を値とする collections.Counter オブジェクト。
    """
    words = extract_words(text)
    return Counter(words)


def get_top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    頻度が高い順に上位 N 件の単語と頻度のタプルリストを返す。

    Args:
        counter: 単語頻度を格納した Counter オブジェクト。
        n: 取得する単語の数。

    Returns:
        (単語, 頻度) のタプルのリスト。頻度降順。
    """
    if n <= 0:
        return []
    return counter.most_common(n)
