def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。英数字で構成される連続部分列を単語として扱う。
    大文字小文字は無視する（すべて小文字に変換）。

    Args:
        text: 入力テキスト。

    Returns:
        単語のリスト（小文字化済み）。
    """
    return [word.lower() for word in re.findall(r'[a-zA-Z0-9]+', text)]


def count_words(text: str) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(tokenize(text))


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """

    Args:
        counter: 単語頻度の Counter。

        (単語, 頻度) のタプルのリスト。

    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
