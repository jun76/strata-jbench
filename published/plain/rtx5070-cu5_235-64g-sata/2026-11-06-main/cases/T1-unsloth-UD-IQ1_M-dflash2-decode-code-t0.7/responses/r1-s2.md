#!/usr/bin/env python3
"""テキストファイルから単語出現頻度を解析し、上位 N 件を表形式で表示するコマンドラインツール。"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを単語リストにトークナイズする。小文字化し、英数字のみのトークン抽出。"""
    return re.findall(r'[a-z0-9]+', text.lower())


def count_words(text: str) -> Counter:
    """テキスト内の単語出現頻度を Counter で返す。"""
    return Counter(tokenize(text))


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counter から出現頻度上位 n 件の (word, count) タプルリストを返す。"""
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """(word, count) のリストを ASCII テーブル形式の文字列として返す。"""
    if not rows:
        return "No words found."
    
    word_width = max(len(w) for w, _ in rows)
    word_width = max(word_width, len("Word"))
    count_width = max(len(str(c)) for _, c in rows)
    count_width = max(count_width, len("Count"))
    
    header = f"{'Word'.ljust(word_width)} | {'Count'.rjust(count_width)}"
    separator = '-' * word_width + "-+-" + '-' * count_width
    
    lines = [header, separator]
    for word, count in rows:
        line = f"{word.ljust(word_width)} | {str(count).rjust(count_width)}"
        lines.append(line)
    
    return "\n".join(lines)
