"""
word_freq CLI tool: Read a text file, count word frequencies, and display top N words in a table.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def read_file(filepath: str) -> str:
    """Read and return the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The full text content of the file.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """Extract words from text, lowercased, ignoring non-alphanumeric characters.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase words.
    """
    # Use regex to find sequences of alphanumeric characters (and underscores if desired, but typically words are letters/digits)
    # \w+ includes underscores, so we might want to adjust. Let's use [a-zA-Z0-9]+ for pure word tokens.
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """Count occurrences of each word.

    Args:
        words: List of words.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)
