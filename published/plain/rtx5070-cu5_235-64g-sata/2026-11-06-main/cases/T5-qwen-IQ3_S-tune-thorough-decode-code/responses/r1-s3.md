#!/usr/bin/env python3
"""
word_count.py - Count word frequencies in a text file and display top N.

This module provides a command-line tool that reads a text file, counts
the frequency of each word (case-insensitive, with basic tokenization),
and displays the top N most frequent words in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass
from typing import List, Tuple


@dataclass
class WordCount:
    """Represents a word and its frequency count."""
    word: str
    count: int


def tokenize(text: str) -> List[str]:
    """
    Tokenize the input text into lowercase words.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase word tokens.
    """
    # Match sequences of alphanumeric characters (including underscores)
    return re.findall(r'\b\w+\b', text.lower())


def count_words(text: str) -> Counter:
    """
    Count the frequency of each word in the given text.
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(tokenize(text))


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top entries to return.
        
    Returns:
        A list of tuples (word, count) for the top N entries.
    """
    return counter.most_common(n)


def read_file(filepath: str) -> str:
    """
    Read the contents of a text file.
    
    Args:
        filepath: Path to the text file.
        
    Returns:
        The text content of the file.
        
    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()
