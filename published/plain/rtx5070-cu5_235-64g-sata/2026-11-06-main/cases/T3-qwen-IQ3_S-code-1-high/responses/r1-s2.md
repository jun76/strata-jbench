def merge_intervals(intervals: list[list[int]]) -> list[list[int]]:
    merged: list[list[int]] = []

    for interval in sorted(intervals, key=lambda x: x[0]):
        start, end = interval

        if not merged:
            merged.append([start, end])
        elif start <= merged[-1][1]:
            if end > merged[-1][1]:
                merged[-1][1] = end
        else:
            merged.append([start, end])

    return merged
