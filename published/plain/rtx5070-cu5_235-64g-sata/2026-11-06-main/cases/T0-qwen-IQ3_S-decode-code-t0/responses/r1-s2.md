word_freq.py - コマンドラインツールでテキストファイルの単語出現頻度を上位N件表示する。

使い方:
    python word_freq.py <file> [-n N] [--case-sensitive]

例:
    python word_freq.py sample.txt -n 10
    python word_freq.py sample.txt --case-sensitive
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか

    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()
    # 単語は英数字とハイフン、アポストロフィを含む連続文字列とする
    words = re.findall(r"[a-z0-9]+(?:['-][a-z0-9]+)*", text, re.IGNORECASE if not case_sensitive else 0)
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト

    Returns:
        単語をキー、出現回数を値とするCounter
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位N件の単語と出現回数を取得する。

    Args:
        counter: 単語頻度のCounter
        n: 取得する件数

    Returns:
        (単語, 出現回数) のタプルのリスト
    """
    return counter.most_common(n)
