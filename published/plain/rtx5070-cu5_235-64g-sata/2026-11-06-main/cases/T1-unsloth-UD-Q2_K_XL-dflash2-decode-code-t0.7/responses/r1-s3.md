#!/usr/bin/env python3
"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a formatted table.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Words are defined as sequences of alphanumeric characters.
    Punctuation and whitespace are treated as separators.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase word tokens.
    """
    # Use regex to find all sequences of word characters (alphanumeric + underscore)
    # Convert to lowercase for case-insensitive counting
    return [word.lower() for word in re.findall(r'\b[a-zA-Z0-9]+\b', text)]


def count_words(words: List[str]) -> Counter[str]:
    """
    Count the frequency of each word in the list.

    Args:
        words: A list of word tokens.

    Returns:
        A Counter object mapping each word to its occurrence count.
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to retrieve.

    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then by word ascending for ties.
    """
    # Get most common n items
    top_items = counter.most_common(n)
    # Sort by count descending, then by word ascending for stability
    top_items.sort(key=lambda x: (-x[1], x[0]))
    return top_items
