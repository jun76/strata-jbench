def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    正規表現を用いて英数字、ハイフン、アポストロフィを含むトークンを抽出し、
    小文字に変換する。

    Args:
        text: 入力テキスト文字列

    Returns:

    # 単語パターン: 英数字、ハイフン、アポストロフィを含む連続した文字列

    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return [word.lower() for word in words]
