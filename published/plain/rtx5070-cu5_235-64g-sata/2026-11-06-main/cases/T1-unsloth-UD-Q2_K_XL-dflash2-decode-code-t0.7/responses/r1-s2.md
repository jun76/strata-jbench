#!/usr/bin/env python3
"""
word_count.py

テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で表示するコマンドラインツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    
    Args:
        text (str): 入力テキスト。
        
    Returns:
        List[str]: 小文字化された単語のリスト。
    """
    # 英数字のみを抽出し、小文字に変換
    return [word.lower() for word in re.findall(r'[a-zA-Z0-9]+', text)]


def count_words(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text (str): 入力テキスト。
        
    Returns:
        Counter[str]: 単語と出現回数の対応を表す Counter オブジェクト。
    """
    return Counter(tokenize(text))


def get_top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    出現頻度上位 N 件の単語を取得する。
    
    Args:
        counter (Counter[str]): 単語カウントの Counter オブジェクト。
        n (int): 取得する上位件数。
        
    Returns:
        List[Tuple[str, int]]: (単語, 出現回数) のタプルのリスト。出現頻度順にソートされている。
    """
    return counter.most_common(n)
