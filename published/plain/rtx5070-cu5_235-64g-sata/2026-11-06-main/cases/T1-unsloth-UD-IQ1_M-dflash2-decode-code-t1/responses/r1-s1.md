def read_text(file_path: str) -> str:
    """
    Read text content from a file.

    Args:
        file_path: Path to the input text file.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {file_path}")
    except IOError as e:
        raise IOError(f"Error reading file {file_path}: {e}")


def tokenize(text: str) -> list[str]:
    """
    Tokenize text into lowercase words, removing punctuation.

    Args:
        text: Input text string.

    Returns:
        A list of tokens (words).
    """
    import re
    # Split by whitespace and remove surrounding punctuation, lowercase everything
    tokens = re.findall(r"[a-zA-Z0-9']+", text.lower())
    return tokens


def count_word_frequencies(text: str) -> Counter[str]:
    """
    Count the frequency of each word in the given text.

    Args:
        text: Input text string.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    tokens = tokenize(text)
    return Counter(tokens)
