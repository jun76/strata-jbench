def read_text(filepath: str) -> str:
    """Read the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If an error occurs while reading the file.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.

    Words are defined as sequences of alphanumeric characters, apostrophes,
    and hyphens. Everything else is treated as a delimiter.

    Args:
        text: The input text.

    Returns:
        A list of lowercased word tokens.
    """
    # Extract words consisting of letters, digits, apostrophes, and hyphens
    # This handles contractions like "don't" and hyphenated words like "well-known"
    words = re.findall(r"[A-Za-z0-9][A-Za-z0-9'\-]*", text)
    return [word.lower() for word in words]


def count_words(tokens: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        tokens: A list of word tokens.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(tokens)
