```python
"""word_counter.py - テキストファイルから単語出現頻度を上位N件を表示するCLIツール."""

import argparse
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """テキストを単語に分割し、小文字・アルファベットのみ抽出.

    Args:
        text: 入力テキスト.

    Returns:
        小文字化された英単語のリスト.
    """
    return [w.lower() for w in re.findall(r"[a-zA-Z]+", text)]


def count_words(text: str) -> Counter[str]:
    """テキスト内の単語出現頻度を集計.

    Args:
        text: 入力テキスト.

    Returns:
        単語 -> 出現回数の Counter.
    """
    return Counter(tokenize(text))


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """頻度上位 n 件の (単語, 回数) を返す.

    Args:
        counter: 単語頻度 Counter.
        n: 取得件数.

    Returns:
        頻度降順の (単語, 回数) のリスト.
    """
    if n <= 0:
        return []
    return counter.most_common(n)


def format_table(pairs: List[Tuple[str, int]], total_words: int) -> str:
    """(単語, 回数) のリストを表形式の文字列に変換.

    Args:
        pairs: (単語, 回数) のリスト.
        total_words: 総単語数 (割合計算用).

    Returns:
        表形式の文字列.
    """
    if not pairs:
        return ""

    # 列幅を計算
    max_word_len = max(len(w) for w, _ in pairs)
    max_word_len = max(max_word_len, len("Word"))
    max_count_len = max(len(str(c)) for _, c in pairs)
    max_count_len = max(max_count_len, len("Count"))
    max_pct_len = max(len(f"{c / total_words * 100:.2f}%") for _, c in pairs) if total_words else 0
    max_pct_len
