def load_text_from_file(filepath: str) -> str:
    """
    指定されたテキストファイルを読み込み、内容を文字列として返す。

    Args:
        filepath: テキストファイルへのパス

    Returns:
        読み込まれたテキスト内容

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        UnicodeDecodeError: UTF-8でデコードできない場合
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """
    テキストから英数字の単語を抽出する。
    区切りは非英数字文字とし、単語は小文字に正規化する。

    Args:
        text: 入力テキスト

    Returns:
        抽出された単語のリスト（小文字、順序は出現順）
    """
    words = []
    current_word = []
    for char in text:
        if char.isalnum():
            current_word.append(char.lower())
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストの出現頻度を集計する。

    Args:
        words: 単語のリスト

    Returns:
        単語ごとの出現回数を示すCounterオブジェクト
    """
    return Counter(words)


def get_top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位N件を頻度降順で取得する。

    Args:
        counter: 単語と頻度のCounter
        n: 取得する上位件数

    Returns:
        (単語, 頻度)のタプルのリスト（頻度降順）
    """
    return counter.most_common(n)
