"""
コマンドラインツール: テキストファイルを読み込み、単語出現頻度の上位 N 件を表形式で表示する。

Usage:
    python word_counter.py input.txt --top 10
    python word_counter.py input.txt --top 5 --sort-by count --case-sensitive
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def normalize_word(word: str, case_sensitive: bool) -> str:
    """
    単語を正規化する。

    Args:
        word: 原単語文字列
        case_sensitive: True なら大文字小文字を区別、False なら小文字に変換

    Returns:
        正規化された単語文字列
    """
    return word if case_sensitive else word.lower()


def count_words_from_text(
    text: str,
    case_sensitive: bool = False,
) -> Counter:
    """
    テキストから単語出現頻度を集計する。

    単語は空白文字で区切られ、句読点や記号は除去される。

    Args:
        text: 解析対象のテキスト文字列
        case_sensitive: 大文字小文字を区別するか

    Returns:
        単語: 出現回数 の Counter
    """
    words = []
    for raw_word in text.split():
        # 句読点や記号を除去して純粋な単語部分を取り出す
        word = raw_word.strip(".,;:!\"'()[]{}-_")
        if word:  # 空文字は除外
            normalized = normalize_word(word, case_sensitive)
            words.append(normalized)
    return Counter(words)
