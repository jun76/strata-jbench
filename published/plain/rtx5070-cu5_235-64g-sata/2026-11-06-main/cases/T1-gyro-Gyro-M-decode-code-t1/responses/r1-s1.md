#!/usr/bin/env python3
"""
Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import Dict, List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Extracts words using a simple regex that splits on non-alphanumeric characters.

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase word tokens.
    """
    # Match sequences of alphanumeric characters
    return re.findall(r'\b[a-zA-Z0-9_]+\b', text.lower())


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text.

    Returns:
        A Counter mapping words to their occurrence counts.
    """
    words = tokenize(text)
    return Counter(words)


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top words to retrieve.

    Returns:
        A list of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]]) -> str:
    """
    Format word frequency entries as a text table.

    Args:
        entries: List of (word, count) tuples.

    Returns:
        A formatted string table with aligned columns.
    """
    if not entries:
        return ""

    max_word_len = max(len(word) for word, _ in entries)
    header = f"{'Word':<{max_word_len + 4}}{'Count':>8}"
    separator = '-' * (max_word_len + 12)

    lines = [header, separator]
    for word, count in entries:
        lines.append(f"{word:<{max_word_len + 4}}{count:>8}")

    return "\n".join(lines)
