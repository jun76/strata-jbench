"""
Text word frequency counter.

Command-line tool that reads a text file, counts word occurrences,
and displays the top-N most frequent words in a table format.

Usage:
    python word_freq.py <filename> -n 10
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def tokenize_text(text: str) -> List[str]:
    """Tokenize text into words.

    Args:
        text: Raw text string.

    Returns:
        List of lowercase word tokens (non-empty).
    """
    # Extract sequences of Unicode letters (including CJK)
    words = re.findall(r"\p{L}+", text)
    # Lowercase for consistent counting
    return [w.lower() for w in words if w]


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in text.

    Args:
        text: Raw text string.

    Returns:
        Counter mapping words to their frequencies.
    """
    words = tokenize_text(text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words.

    Args:
        counter: Word frequency counter.
        n: Number of top words to return.

    Returns:
        List of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
