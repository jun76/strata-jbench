def read_words(filename: str) -> List[str]:
    """Read a text file and return a list of raw words.
    
    Args:
        filename: Path to the text file.
        
    Returns:
        List of strings extracted from the file. Words are split by whitespace
        and punctuation is stripped.
    """
    try:
        with open(filename, "r", encoding="utf-8") as f:
            content = f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filename}")

    # Basic tokenization: split by whitespace and strip common punctuation.
    # For a more robust tool, one might use regex, but this keeps it simple and dependency-free.
    raw_tokens = content.split()
    clean_tokens = []
    for token in raw_tokens:
        # Strip leading/trailing punctuation characters
        stripped = token.strip(".,!?:;\"'()[]{}")
        if stripped:
            clean_tokens.append(stripped)
    return clean_tokens
