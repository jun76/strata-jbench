def read_text_file(file_path: str) -> str:
    """
    指定されたパスのテキストファイルを読み込み、その内容を文字列として返す。

    Args:
        file_path (str): 読み込むファイルのパス。

    Returns:
        str: ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイル読み取り中にエラーが発生した場合。
    """
    with open(file_path, "r", encoding="utf-8") as f:
        return f.read()
