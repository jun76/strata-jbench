#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位N件を表形式で出力するコマンドラインツール。

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    大文字小文字を区別せず、英数字の文字列のみを抽出する。
    
    Args:
        text (str): 入力テキスト
        
    Returns:
        List[str]: 小文字化された単語のリスト
    """
    # 英数字のみを抽出し、小文字に変換
    words = re.findall(r'\b[a-z0-9]+\b', text.lower())
    return words


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text (str): 入力テキスト
        
    Returns:
        Counter: 単語とその出現回数の辞書
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度上位N件の単語と回数を取得する。
    同数の場合は辞書順で安定してソートする。
    
    Args:
        counter (Counter): 単語頻度のカウント
        n (int): 取得する件数
        
    Returns:
        List[Tuple[str, int]]: (単語, 回数) のリスト
    """
    # 頻度降順、同数ならアルファベット順で安定ソート
    sorted_items = sorted(
        counter.items(),
        key=lambda x: (-x[1], x[0])
    )
    return sorted_items[:n]
