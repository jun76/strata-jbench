#!/usr/bin/env python3
"""コマンドラインツール: テキストファイルの単語出現頻度を集計し、上位N件を表形式で表示する。

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """単語を正規化する。

    Args:
        word: 元の単語文字列
        case_sensitive: 大文字小文字を区別する場合 True

    Returns:
        正規化された単語文字列
    """
    if not case_sensitive:
        word = word.lower()
    return word


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """テキストから単語トークンを取り出す。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別する場合 True

    Returns:
        正規化された単語トークンのリスト
    """
    words = text.split()
    return [normalize_word(w, case_sensitive) for w in words]


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別する場合 True

    Returns:
        単語と出現回数のCounter
    """
    tokens = tokenize(text, case_sensitive)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Counterから上位N件の単語と出現回数を取り出す。

    Args:
        counter: 単語と出現回数のCounter
        n: 取得する上位件数

    Returns:
        (単語, 出現回数) のタプルのリスト、出現回数の降順
    """
    return counter.most_common(n)
