#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で出力するコマンドラインツール。
"""

import argparse
import re

from collections import Counter

from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出します。
    
    単語は英数字とハイフン、アポストロフィを含む文字列として扱われます。
    大文字・小文字は無視して正規化されます。
    
    Args:
        text (str): 入力テキスト。
        
    Returns:
        List[str]: 抽出された単語のリスト（小文字正規化済み）。
    """
    # 英数字、ハイフン、アポストロフィを含む連続文字列を単語とみなす
    # 例: "don't" -> ["don't"], "hello-world" -> ["hello-world"]
    pattern = r"[a-zA-Z0-9'-]+"
    matches = re.findall(pattern, text, flags=re.IGNORECASE)
    # 大文字小文字を無視して小文字に正規化
    return [word.lower() for word in matches]


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントします。
    
    Args:
        words (List[str]): 単語のリスト。
        
    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の単語とカウントを取得します。
    
    Args:
        counter (Counter): 単語とカウントの Counter。
        n (int): 取得する上位件数。
        
    Returns:
        List[Tuple[str, int]]: (単語, カウント) のタプルのリスト。
    """
    return counter.most_common(n)
