def load_text(filepath: str) -> str:
    """指定されたファイルパスからテキストを読み込む。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイル読み込み中にエラーが発生した場合。
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_sensitive: bool = False) -> list[str]:
    """テキストを単語に分割する。

    単語は、英数字、アンダースコア、および Unicode 文字からなる連続したシーケンスとして定義される。
    大文字・小文字の区別は case_sensitive フラグで制御する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字・小文字を区別する。False（デフォルト）の場合、
                        すべて小文字に変換して区別しない。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    # 単語を抽出: 英数字、アンダースコア、Unicode 文字の連続
    import re
    words = re.findall(r"\w+", text)
    return words


def count_words(words: list[str]) -> Counter:
    """単語リストから各単語の出現回数をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする collections.Counter オブジェクト。
    """
    return Counter(words)
