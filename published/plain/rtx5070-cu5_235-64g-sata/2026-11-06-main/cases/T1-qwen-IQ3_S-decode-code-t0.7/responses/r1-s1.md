"""
word_freq CLI tool: Read a text file, count word frequencies, and display top N words in a table.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def read_file(filepath: str) -> str:
    """Read and return the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The full text content of the file.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str) -> List[str]:
    """Extract words from text, lowercased and stripped of punctuation.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase words (only alphabetic characters).
    """
    # Extract sequences of alphabetic characters only
    words = re.findall(r"[a-zA-Z]+", text)
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter:
    """Count the frequency of each word.

    Args:
        words: List of words to count.

    Returns:
        A Counter object with word frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        A list of (word, frequency) tuples, sorted by frequency descending.
    """
    return counter.most_common(n)


def format_table(data: List[Tuple[str, int]]) -> str:
    """Format word frequency data as a string table.

    Args:
        data: List of (word, frequency) tuples.

    Returns:
        A formatted string table.
    """
    if not data:
        return "No words found."

    # Determine column widths
    word_width = max(len(word) for word, _ in data)
    word_width = max(word_width, len("Word"))
    freq_width = max(len(str(freq)) for _, freq in data)
    freq_width = max(freq_width, len("Frequency"))
