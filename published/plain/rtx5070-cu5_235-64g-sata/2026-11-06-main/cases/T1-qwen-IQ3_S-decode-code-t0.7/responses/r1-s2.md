def extract_words(text: str, case_insensitive: bool = True, min_length: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。
    
    Args:
        text: 解析対象のテキスト。
        case_insensitive: True の場合、単語を小文字に変換して大文字小文字を区別しない。
        min_length: 抽出する単語の最小文字数。1未満の場合、1に補正される。

    Returns:
        抽出された単語のリスト。
    """
    words: List[str] = []
    current_word: List[str] = []
    
    for char in text:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                word = ''.join(current_word)
                if case_insensitive:
                    word = word.lower()
                if len(word) >= min_length:
                    words.append(word)
                current_word = []
    
    # 末尾の単語処理
    if current_word:
        word = ''.join(current_word)
        if case_insensitive:
            word = word.lower()
        if len(word) >= min_length:
            words.append(word)
            
    return words
