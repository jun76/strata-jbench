word_count.py - Count word frequencies in a text file and display top N.

This command-line tool reads a text file, counts the occurrences of each word
(case-insensitive), and prints the top N most frequent words in a table format.
"""

import argparse

from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """Extract words from text, converting to lowercase and removing non-alphanumeric chars.

    Args:
        text: Input text string.

    Returns:
        A list of lowercase alphanumeric words.
    """
    # Match sequences of alphanumeric characters (including underscores? No, just words)
    # We'll use a simple regex to find word-like tokens: letters and digits
    return [word.lower() for word in re.findall(r'[a-zA-Z0-9]+', text)]


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    words = tokenize(text)
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a counter.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples for the top N most common words.
    """
    return counter.most_common(n)
