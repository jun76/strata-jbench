def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。Falseの場合は小文字に変換。
        min_len: 単語の最小長。これより短い単語は除外。

    Returns:
        単語のリスト。
    """
    # 単語を抽出: 英数字とハイフンを含む連続文字列
    words = re.findall(r'[a-zA-Z0-9]+(?:-[a-zA-Z0-9]+)*', text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    if min_len > 1:
        words = [w for w in words if len(w) >= min_len]
    return words


def count_frequencies(words: List[str]) -> Counter:
    """

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とするCounter。
    """
    return Counter(words)
