#!/usr/bin/env python3
"""
word_freq.py - A command-line tool to count word frequencies in text files.

Usage:
    python word_freq.py <file> [-n N] [--case-sensitive] [--min-len LEN]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: Input text string.
        case_sensitive: If False, convert all words to lowercase.
        min_len: Minimum length of a word to include.

    Returns:
        List of tokenized words.
    """
    # Extract words using regex (sequences of alphanumeric characters)
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    if min_len > 1:
        words = [w for w in words if len(w) >= min_len]
    return words


def count_words(text: str, case_sensitive: bool = False, min_len: int = 1) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: Input text string.
        case_sensitive: If False, convert all words to lowercase.
        min_len: Minimum length of a word to include.

    Returns:
        Counter object with word frequencies.
    """
    words = tokenize(text, case_sensitive=case_sensitive, min_len=min_len)
    return Counter(words)
