def tokenize(text: str) -> List[str]:
    """
    Tokenize text into words. Words are sequences of alphanumeric characters
    and apostrophes (for contractions like "don't").

    Args:
        text: The input text to tokenize.

    Returns:
        A list of words extracted from the text.
    """
    # Match words containing alphanumeric characters and apostrophes
    pattern = re.compile(r"[a-zA-Z0-9']+")
    return pattern.findall(text)


def count_words(text: str, case_insensitive: bool = False) -> Counter:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text to analyze.
        case_insensitive: If True, words are lowercased before counting.

    Returns:
        A Counter mapping words to their occurrence counts.
    """
    words = tokenize(text)
    if case_insensitive:
        words = [w.lower() for w in words]
    return Counter(words)
