Options:

def read_file(path: str) -> str:
    """
    テキストファイルを読み込んで内容文字列を返す。

    Args:
        path: 読み込み対象のファイルパス。

    Returns:
        ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        OSError: その他の読み取りエラー。
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def extract_words(text: str, ignore_case: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        ignore_case: True の場合、大文字小文字を区別しない。

    Returns:
        抽出された単語のリスト。
    """
    # 英数字とハイフン・アンダースコアを含む単語を抽出
    pattern = r"[a-zA-Z0-9_\-]+"
    words = re.findall(pattern, text)
    if ignore_case:
        words = [w.lower() for w in words]
    return words
