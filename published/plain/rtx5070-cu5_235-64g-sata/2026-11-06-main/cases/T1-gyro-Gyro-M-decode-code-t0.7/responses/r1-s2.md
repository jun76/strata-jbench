"""
Command-line tool to read a text file, count word frequencies,
and display the top N words in a table format.
"""

import argparse
from collections import Counter
from typing import List, Tuple, Optional


def read_text_file(filepath: str) -> str:
    """
    Read the content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into words (lowercase, alphanumeric only).

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens.
    """
    words: List[str] = []
    current_word: List[str] = []
    for char in text:
        if char.isalnum():
            current_word.append(char.lower())
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    if current_word:
        words.append(''.join(current_word))
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of word tokens.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples sorted by frequency descending,
        then alphabetically ascending for ties.
    """
    items = counter.items()
    # Sort by frequency descending, then word ascending for stable ordering
    sorted_items = sorted(items, key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
