#!/usr/bin/env python3
"""
wordfreq.py – Count word frequencies in a text file and display the top N words.

Usage:
    python wordfreq.py <input_file> [-n N] [-c CASE_INSENSITIVE]

Example:
    python wordfreq.py sample.txt -n 10
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def read_text(filepath: str) -> str:
    """Read the contents of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The full text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str, case_insensitive: bool = True) -> List[str]:
    """Tokenize text into words.

    Words are defined as sequences of alphanumeric characters and underscores
    (similar to ``\\w+``). Non-word characters are treated as separators.

    Args:
        text: The input text string.
        case_insensitive: If True, all words are lowercased before counting.

    Returns:
        A list of word tokens.
    """
    if case_insensitive:
        text = text.lower()
    # Use regex to extract words (alphanumeric + underscore sequences)
    tokens = re.findall(r"\w+", text)
    return tokens


def count_words(tokens: List[str]) -> Counter:
    """Count occurrences of each token.

    Args:
        tokens: A list of word tokens.

    Returns:
        A Counter mapping each word to its frequency.
    """
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words and their counts.

    Args:
        counter: A Counter of word frequencies.
        n: The number of top items to return.

    Returns:
        A list of (word, count) tuples, sorted by frequency descending.
    """
    return counter.most_common(n)
