def normalize_word(word: str) -> str:
    """
    単語を正規化する。
    小文字化し、英数字以外の文字を除去する。

    Args:
        word: 元の単語文字列

    Returns:
        正規化された単語 (空文字列の場合は None 相当だが、ここでは空文字列を返す)
    """
    # 小文字に変換
    lower = word.lower()
    # 英数字以外の文字を除去
    cleaned = re.sub(r'[^a-z0-9]', '', lower)
    return cleaned


def count_words(text: str) -> Counter:
    """
    テキスト中の単語の出現頻度をカウントする。

    Args:
        text: 解析対象のテキスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    # 空白や改行などで分割
    raw_words = text.split()
    normalized_words = [normalize_word(w) for w in raw_words]
    # 空文字列を除外
    filtered_words = [w for w in normalized_words if w]
    return Counter(filtered_words)
