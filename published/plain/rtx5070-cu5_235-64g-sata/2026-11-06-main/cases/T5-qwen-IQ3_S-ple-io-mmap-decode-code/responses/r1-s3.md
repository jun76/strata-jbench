"""
wordcount.py - テキストファイルから単語出現頻度を集計し、上位 N 件を表示するツール。
"""

import argparse
import re

from pathlib import Path

def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出します。
    単語は英数字（およびアンダースコア）の連続として扱われます。
    
    Args:
        text: 入力テキスト。
        
    Returns:
        小文字化された単語のリスト。
    """
    # \w+ は [a-zA-Z0-9_] に一致します。
    # 日本語などのマルチバイト文字は含まれません（このツールは主に英語/ASCII前提）。
    words = re.findall(r'\w+', text.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントします。
    
    Args:
        words: 単語のリスト。
        
    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)


def top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の単語とその頻度を取得します。
    頻度が同じ場合は辞書順でソートされます。
    
    Args:
        counter: 単語頻度カウンター。
        n: 取得する上位件数。
        
    Returns:
        (単語, 頻度) のタプルのリスト。
    """
    # most_common(n) は頻度順に返しますが、頻度が同じ場合の順序は保証されません。
    # 安定した出力のために、頻度降順、単語昇順でソートし直します。
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
