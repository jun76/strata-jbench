word_freq.py - Count word frequencies from a text file and display top N results.

Usage:

Options:
    -n N: Number of top words to display (default: 10)

def read_text_file(filepath: str) -> str:
    """Read text from a file and return its content.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If there is an error reading the file.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def extract_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """Extract words from text, optionally case-insensitive, with minimum length.

    Args:
        text: Input text string.
        case_sensitive: If True, preserve case; otherwise convert to lowercase.
        min_length: Minimum length of words to include.

    Returns:
        A list of extracted words.
    """
    # Extract words using regex: sequences of alphanumeric characters and underscores
    words = re.findall(r'\b\w+\b', text)
    
    if not case_sensitive:
        words = [word.lower() for word in words]
    
    words = [word for word in words if len(word) >= min_length]
    
    return words
