    """単語を正規化する。

        word: 正規化対象の単語。

        min_length: 許可される最小文字数。

    Returns:
        正規化された単語。条件を満たさない場合は空文字列。
    """
    word = word.strip()
    if not case_sensitive:
        word = word.lower()
    if len(word) < min_length:
        return ""
    return word
