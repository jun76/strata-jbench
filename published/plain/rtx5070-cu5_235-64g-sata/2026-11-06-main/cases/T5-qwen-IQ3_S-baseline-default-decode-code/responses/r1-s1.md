wordfreq.py - A command-line tool to count word frequencies in a text file.

Usage:
    python wordfreq.py <input_file> [-n TOP_N] [--case-sensitive]

def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """
    Count word frequencies in the given text.

    Words are split by whitespace and stripped of leading/trailing punctuation.
    By default, words are lowercased unless case_sensitive is True.

    Args:
        text: The input text.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.

    Returns:
        A Counter object with word counts.
    """
    if not case_sensitive:
        text = text.lower()

    words = text.split()
    cleaned_words: List[str] = []
    for word in words:
        # Strip leading and trailing punctuation characters
        stripped = word.strip(".,!?;:'\"()[]{}<>/\\`~!@#$%^&*-=+|")
        if stripped:
            cleaned_words.append(stripped)

    return Counter(cleaned_words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Return the top N most common words from the counter.

    Args:
        counter: A Counter object.
        n: Number of top items to return.

    Returns:
        A list of tuples (word, count), sorted by count descending, then alphabetically.
    """
    # Counter.most_common returns items sorted by count descending.
    # For ties, we want alphabetical order. We can sort manually.
    items = list(counter.items())
    # Sort by count descending, then word ascending
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
