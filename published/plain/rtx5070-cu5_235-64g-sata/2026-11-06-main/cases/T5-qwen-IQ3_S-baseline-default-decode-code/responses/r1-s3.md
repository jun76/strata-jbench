def tokenize(text: str) -> List[str]:
    """
    入力テキストから単語トークンリストを抽出します。
    
    正規表現 [a-zA-Z0-9']+ を使用して、アルファベット・数字・アポストロフィを含む
    単語を抽出します。すべて小文字に変換されます。
    
    Args:
        text: 解析対象のテキスト文字列
        
    Returns:
        小文字化された単語トークンのリスト
    """
    # 単語を抽出: 英数字とアポストロフィを含む連続した文字列
    tokens = re.findall(r"[a-zA-Z0-9']+", text)
    return [token.lower() for token in tokens]
