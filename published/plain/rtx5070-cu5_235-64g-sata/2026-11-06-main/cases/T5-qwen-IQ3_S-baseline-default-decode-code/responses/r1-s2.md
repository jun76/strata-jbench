#!/usr/bin/env python3
"""
word_freq.py - テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示するツール。
"""

import argparse
import re
from collections import Counter

def tokenize(text: str) -> list[str]:
    """
    テキストから単語を抽出する。

    単語は英数字およびアンダースコアで構成されるものとし、
    大文字小文字は区別しない（すべて小文字に変換）。

    Args:
        text: 入力テキスト文字列。

    Returns:
        単語のリスト（重複含む）。
    """
    return re.findall(r"\b[a-zA-Z0-9_]+\b", text.lower())


def count_frequencies(words: list[str]) -> dict[str, int]:
    """
    単語リストから出現頻度を集計する。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする辞書。
    """
    return dict(Counter(words))


def top_n(frequencies: dict[str, int], n: int) -> list[tuple[str, int]]:
    """
    出現頻度辞書から上位 n 件の (単語, 回数) を返す。
    回数が同じ場合は単語の辞書順で昇順に並び替える。

    Args:
        frequencies: 単語と出現回数の辞書。
        n: 取得する件数。

    Returns:
        上位 n 件のタプルのリスト。
    """
    sorted_items = sorted(
        frequencies.items(),
        key=lambda x: (-x[1], x[0])
    )
    return sorted_items[:n]
