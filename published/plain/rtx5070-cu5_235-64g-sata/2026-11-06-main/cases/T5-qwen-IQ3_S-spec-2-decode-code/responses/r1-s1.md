import argparse

def normalize_word(word: str) -> str:
    """
    単語を小文字に変換し、前後の句読点や非英数字文字を除去します。

    Args:
        word (str): 元の単語。

    Returns:
        str: 正規化された単語。空文字列の場合は空文字列を返す。
    """
    # 英数字とアポストロフィのみを残すパターン
    # 'don't' などの略語を考慮し、アポストロフィを含める
    cleaned = re.sub(r"[^a-z0-9']", word.lower())
    return cleaned


def tokenize(text: str) -> List[str]:
    """
    テキストを単語のリストに分割します。
    正規化処理を適用し、空文字列を除外します。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 正規化された単語のリスト。
    """
    # 正規表現で単語らしきものを探す（空白区切りではなく、英数字とアポストロフィの連続）
    raw_words = re.findall(r"[a-z0-9']+", text.lower())
    # 正規化（前後の句読点除去など）を適用し、空でないものだけ残す
    words = [normalize_word(w) for w in raw_words]
    return [w for w in words if w]
