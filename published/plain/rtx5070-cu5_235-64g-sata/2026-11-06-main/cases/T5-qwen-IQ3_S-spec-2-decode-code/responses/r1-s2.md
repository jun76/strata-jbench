Reads a text file, tokenizes words, counts their occurrences, and displays

def read_file(filepath: str) -> str:
    """Read the entire content of a text file.

    Args:
        filepath: Path to the input text file.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        PermissionError: If the file cannot be read due to permissions.
        IOError: For other input/output errors.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Tokenize text into lowercase words.

    Words are defined as sequences of alphanumeric characters and apostrophes
    (to handle contractions like "don't"). Non-word characters are treated as
    separators.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens.
    """
    # Match sequences of word characters including apostrophes within words
    # This handles contractions like "don't", "it's", etc.
    words = re.findall(r"[a-zA-Z0-9]+(?:'[a-zA-Z0-9]+)*", text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter[str]:
    """Count the frequency of each word.

    Args:
        words: A list of word tokens.

    Returns:
        A Counter object mapping each word to its occurrence count.
    """
    return Counter(words)
