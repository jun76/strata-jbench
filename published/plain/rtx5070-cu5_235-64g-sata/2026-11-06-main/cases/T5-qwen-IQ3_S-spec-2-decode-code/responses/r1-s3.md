word_freq_counter.py

A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.

Usage:
    python word_freq_counter.py --file input.txt --top 10 [--case-sensitive] [--min-length 1]
"""

import argparse

def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words based on whitespace and punctuation.
    
    Args:
        text: Input text string.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.
        min_length: Minimum length of a word to include.
    
    Returns:
        List of words matching the criteria.
    """
    # Extract words: sequences of alphanumeric characters and underscores (common word chars)
    # This regex matches typical word boundaries
    words = re.findall(r'\b\w+\b', text)
    
    if not case_sensitive:
        words = [w.lower() for w in words]
    
    words = [w for w in words if len(w) >= min_length]
    
    return words


def count_word_frequencies(words: List[str]) -> Counter[str]:
    """
    Count the frequency of each word in the list.
    
    Args:
        words: List of words.
    
    Returns:
        Counter object with word counts.
    """
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from the counter.
    
    Args:
        counter: Counter object with word frequencies.
        n: Number of top words to return.
    
    Returns:
        List of tuples (word, count) for the top N words.
    """
    return counter.most_common(n)
