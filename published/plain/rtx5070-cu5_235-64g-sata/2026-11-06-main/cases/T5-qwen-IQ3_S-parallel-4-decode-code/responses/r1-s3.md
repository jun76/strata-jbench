#!/usr/bin/env python3
"""
word_freq.py: テキストファイルから単語の出現頻度を計算し、上位N件を表示するツール。
"""

import argparse

from collections import Counter

def read_text_file(filepath: Path) -> str:
    """指定されたパスのテキストファイルを読み込み、内容を文字列として返す。

    Args:
        filepath: 読み込むファイルのパス。

    Returns:
        ファイルのテキスト内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        PermissionError: 読み取り権限がない場合。
    """
    if not filepath.exists():
        raise FileNotFoundError(f"File not found: {filepath}")
    try:
        return filepath.read_text(encoding="utf-8")
    except PermissionError:
        raise PermissionError(f"Permission denied: {filepath}")


def extract_words(text: str) -> List[str]:
    """テキストから単語を抽出する。

    単語は英数字のみを含む文字列として定義されます。
    大文字/小文字は区別せず、すべて小文字に変換して返します。

    Args:
        text: 入力テキスト。

    Returns:
        抽出された単語のリスト（小文字化済み）。
    """
    # 英数字とアンダースコアを単語として扱う (\w)
    words = re.findall(r'\w+', text, flags=re.IGNORECASE)
    return [word.lower() for word in words]


def count_word_frequencies(words: List[str]) -> Counter:
    """単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)
