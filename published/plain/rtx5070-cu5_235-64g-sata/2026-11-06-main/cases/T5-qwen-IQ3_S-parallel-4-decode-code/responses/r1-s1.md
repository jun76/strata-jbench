def read_file(filepath: str) -> str:
    """
    Read the entire content of a text file.

    Args:
        filepath: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        PermissionError: If the file cannot be read due to permissions.
        OSError: For other I/O errors.
    """
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {filepath}")
    except PermissionError:
        raise PermissionError(f"Permission denied: {filepath}")
    except OSError as e:
        raise OSError(f"Error reading file {filepath}: {e}")


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Words are sequences of alphanumeric characters. Non-alphanumeric characters
    are treated as separators.

    Args:
        text: Input text string.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.

    Returns:
        A list of word tokens.
    """
    if not case_sensitive:
        text = text.lower()

    tokens: List[str] = []
    current_token: List[str] = []

    for char in text:
        if char.isalnum():
            current_token.append(char)
        else:
            if current_token:
                tokens.append("".join(current_token))
                current_token = []

    if current_token:
        tokens.append("".join(current_token))

    return tokens

    """

        tokens: List of word tokens.

    Returns:
        A Counter object with word counts.
