#!/usr/bin/env python3

text_frequency.py - Count word frequencies in a text file and display top N words.

Usage:
    python text_frequency.py input.txt -n 10
    python text_frequency.py input.txt --top 20 --ignore-case

import argparse
import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str, ignore_case: bool = True, min_length: int = 1) -> List[str]:
    """
    Extract words from text.
    
    Args:
        text: Input text string.
        ignore_case: If True, convert all words to lowercase.
        min_length: Minimum word length to include.
        
    Returns:
        List of extracted words.
    """
    # Use regex to find word-like tokens (letters, digits, underscores, apostrophes, hyphens)
    words = re.findall(r"[a-zA-Z0-9_\-']+", text)
    
    if ignore_case:
        words = [w.lower() for w in words]
    
    # Filter by minimum length
    words = [w for w in words if len(w) >= min_length]
    
    return words


def count_word_frequencies(text: str, ignore_case: bool = True, min_length: int = 1) -> Counter:
    """
    Count word frequencies in the given text.
    
    Args:
        text: Input text string.
        ignore_case: If True, treat words case-insensitively.
        min_length: Minimum word length to include.
        
    Returns:
        Counter object with word frequencies.
    """
    words = extract_words(text, ignore_case=ignore_case, min_length=min_length)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words and their counts.
    
    Args:
        counter: Counter object with word frequencies.
        n: Number of top items to return.
        
    Returns:
        List of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
