#!/usr/bin/env python3
"""
word_frequency.py - A command-line tool that reads a text file, counts word
frequencies, and displays the top N words in a table format.

Usage:
    python word_frequency.py <file> [-n N] [--case-sensitive] [--min-length L]
"""

from __future__ import annotations

import argparse
import sys
import re
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If False, words are lowercased before counting.
        min_length: Minimum length of a word to include (default 1).

    Returns:
        A list of tokens (words) that meet the criteria.
    """
    # Extract word characters (letters and digits), treating them as words
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    if min_length > 1:
        words = [w for w in words if len(w) >= min_length]
    return words


def count_words(tokens: List[str]) -> Counter:
    """
    Count the frequency of each token.

    Args:
        tokens: A list of tokens.

    Returns:
        A Counter object mapping each token to its count.
    """
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top n most common words from a counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then by word ascending for ties.
    """
    # most_common returns sorted by count descending; we handle tie-breaking
    # by sorting explicitly: primary key = -count, secondary key = word
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n
