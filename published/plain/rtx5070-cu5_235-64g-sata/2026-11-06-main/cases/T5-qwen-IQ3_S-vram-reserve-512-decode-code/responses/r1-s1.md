from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text.
    
    Words are defined as sequences of alphanumeric characters, separated by
    non-alphanumeric characters. The counting is case-insensitive.
    
    Args:
        text: The input text to analyze.
        
    Returns:
        A Counter object mapping words to their occurrence counts.
    """
    import re
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return Counter(words)


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.
        
    Returns:
        A list of tuples, each containing a word and its count, sorted by
        count descending, then by word ascending for ties.
    """
    items = counter.items()
    # Sort by count descending, then by word ascending
    sorted_items = sorted(items, key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
