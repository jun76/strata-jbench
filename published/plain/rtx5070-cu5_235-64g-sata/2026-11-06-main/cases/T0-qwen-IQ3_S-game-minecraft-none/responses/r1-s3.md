function hash2(x,z){const s=Math.sin(x*127.1+z*311.7)*43758.5453;return s-Math.floor(s);}
function smooth(t){return t*t*(3-2*t);}
function vnoise(x,z){const xi=Math.floor(x),zi=Math.floor(z),xf=x-xi,zf=z-zi;
  const a=hash2(xi,zi),b=hash2(xi+1,zi),c=hash2(xi,zi+1),d=hash2(xi+1,zi+1);
  const u=smooth(xf),v=smooth(zf);return a*(1-u)*(1-v)+b*u*(1-v)+c*(1-u)*v+d*u*v;}
function fbm(x,z){let s=0,a=0.5,f=1;for(let i=0;i<3;i++){s+=a*vnoise(x*f,z*f);f*=2;a*=0.5;}return s;}
