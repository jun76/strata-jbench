function mulberry32(a){return function(){a|=0;a=a+0x6D2B79F5|0;let t=Math.imul(a^a>>>15,1|a);t=t+Math.imul(t^t>>>7,61|t)^t;return((t^t>>>14)>>>0)/4294967296;};}

function clamp(v,a,b){return v<a?a:v>b?b:v;}
function smooth(a,b,x){x=clamp((x-a)/(b-a),0,1);return x*x*(3-2*x);}

  const r=mulberry32(seed), grid=[];

  return (x,z)=>{
    const xi=Math.floor(x),zi=Math.floor(z),xf=x-xi,zf=z-zi;
    const a=grid[xi][zi],b=grid[xi+1][zi],c=grid[xi][zi+1],d=grid[xi+1][zi+1];
    const u=xf*xf*(3-2*xf),v=zf*zf*(3-2*zf);
    return a*(1-u)*(1-v)+b*u*(1-v)+c*(1-u)*v+d*u*v;
  };
}

      setVoxel(x,y,z,0);
      if(waterFill<9000){ waterFill++; waterGeo.push({x:x+0.5,y:SEA+0.02,z:z+0.5}); }

  }
}

/* ================= RAYCAST (voxel DDA) ================= */
function raycast(maxDist){
  const dir=new THREE.Vector3(0,0,-1).applyQuaternion(camera.quaternion);
  let x=Math.floor(camera.position.x), y=Math.floor(camera.position.y), z=Math.floor(camera.position.z);
  const stepX=dir.x>0?1:-1, stepY=dir.y>0?1:-1, stepZ=dir.z>0?1:-1;
  const tdx=dir.x!==0?Math.abs(1/dir.x):Infinity;
  const tdy=dir.y!==0?Math.abs(1/dir.y):Infinity;
  const tdz=dir.z!==0?Math.abs(1/dir.z):Infinity;
  let tmx=dir.x>0?(x+1-camera.position.x)/dir.x:(camera.position.x-x)/-dir.x;
  let tmy=dir.y>0?(y+1-camera.position.y)/dir.y:(camera.position.y-y)/-dir.y;
  let tmz=dir.z>0?(z+1-camera.position.z)/dir.z:(camera.position.z-z)/-dir.z;
  if(dir.x===0)tmx=Infinity; if(dir.y===0)tmy=Infinity; if(dir.z===0)tmz=Infinity;
  let t=0, face=null;
  for(let i=0;i<maxDist+2;i++){
    if(getVoxel(x,y,z)>0) return {x,y,z,face};
    if(tmx<tmy&&tmx<tmz){ x+=stepX; t=tmx; tmx+=tdx; face={x:-stepX,y:0,z:0}; }
    else if(tmy<tmz){ y+=stepY; t=tmy; tmy+=tdy; face={x:0,y:-stepY,z:0}; }
    else { z+=stepZ; t=tmz; tmz+=tdz; face={x:0,y:0,z:-stepZ}; }
    if(t>maxDist) break;
  }
  return null;
}

const player={x:WORLD/2+0.5,y:20,z:WORLD/2+0.5,vx:0,vy:0,vz:0,yaw:0,pitch:0,onGround:false};
const eye=1.62, height=1.8, half=0.3;
function collides(px,py,pz){
  const x0=Math.floor(px-half),x1=Math.floor(px+half);
  const y0=Math.floor(py+0.001),y1=Math.floor(py+height-0.001);
  const z0=Math.floor(pz-half),z1=Math.floor(pz+half);
  for(let x=x0;x<=x1;x++)for(let y=y0;y<=y1;y++)for(let z=z0;z<=z1;z++){
    if(x<0||x>=WORLD||z<0||z>=WORLD) return true;
    if(y<0) return true;
    if(y<HEIGHT && getVoxel(x,y,z)>0) return true;
  }
  return false;
}
function updatePlayer(dt){
  const speed=(keys['shiftleft']||keys['shiftright'])?8.2:4.6;
  const fwd=(keys['w']?1:0)-(keys['s']?1:0);
  const str=(keys['d']?1:0)-(keys['a']?1:0);
  let dx=0,dz=0;
  if(fwd||str){
    const fx=-Math.sin(player.yaw),fz=-Math.cos(player.yaw);
    const sx=Math.cos(player.yaw),sz=-Math.sin(player.yaw);
    dx=fx*fwd+sx*str; dz=fz*fwd+sz*str;
    const len=Math.hypot(dx,dz); if(len){dx/=len;dz/=len;}
  }
  player.vx=dx*speed; player.vz=dz*speed;
  if(keys[' ']&&player.onGround){player.vy=JUMP;player.onGround=false;playJump();}
  player.vy-=GRAV*dt; if(player.vy<-40)player.vy=-40;

  // X
  let nx=player.x+player.vx*dt;
  if(!collides(nx,player.y,player.z)) player.x=nx; else player.vx=0;
  // Z
  let nz=player.z+player.vz*dt;
  if(!collides(player.x,player.y,nz)) player.z=nz; else player.vz=0;
  // Y
  let ny=player.y+player.vy*dt;
  player.onGround=false;
  if(!collides(player.x,ny,player.z)){ player.y=ny; }
  else{
    if(player.vy<0){ player.onGround=true; player.vy=0;
      // snap up to a whole cell
      while(collides(player.x,player.y,player.z)) player.y+=0.02;
      player.y=Math.floor(player.y+0.0001)+0.0; // keep
    } else player.vy=0;
  }
  // fell off world
  if(player.y<-10){ player.y=20; player.vy=0; }

  camera.position.set(player.x, player.y+eye, player.z);
  camera.rotation.set(player.pitch, player.yaw, 0);
  sun.target.position.set(player.x,player.y,player.z);
  sun.position.set(player.x+sunDir.x*40, player.y+sunDir.y*40, player.z+sunDir.z*40);
}

## 遊び方メモ
- **開始**：カードの「▶ 遊び始める」→ 画面をクリックで視点ロック（Esc で解除＝メニュー）。
- **壊す**：左クリック（または `Q`）／**置く**：右クリック。中心ブロックに置きたいときはホイール長押し。
- **ブロック選択**：`1〜6`、または下部のホットバーをクリック。
- **移動**：`WASD`、ジャンプ `Space`、走る `Shift`。

世界は自動生成（地形・木・砂浜・水面）で、昼から夜へ時間が流れます。ブロックを積み上げて塔を作ったり、池を埋めたり、砂浜を掘ったりして箱庭を作り込めます。

調整しやすいパラメータを冒頭にまとめてあります：`WORLD`（地形の広さ）、`SEA`（海面の高さ）、`DAY_LENGTH`（1日の秒数）、`REACH`（リーチ距離）。ここを変えると雰囲気が大きく変わります。
