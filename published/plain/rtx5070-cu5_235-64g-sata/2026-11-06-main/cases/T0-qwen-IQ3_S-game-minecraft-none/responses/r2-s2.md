  function raycast(maxDist){
    const dir=camDir(),ox=player.x,oy=player.y+1.62,oz=player.z;
    let x=Math.floor(ox),y=Math.floor(oy),z=Math.floor(oz);
    const sx=Math.sign(dir.x),sy=Math.sign(dir.y),sz=Math.sign(dir.z);
    const tdx=dir.x!==0?Math.abs(1/dir.x):1e30,tdy=dir.y!==0?Math.abs(1/dir.y):1e30,tdz=dir.z!==0?Math.abs(1/dir.z):1e30;
    let tmx=tdx*(dir.x>0?x+1-ox:ox-x),tmy=tdy*(dir.y>0?y+1-oy:oy-y),tmz=tdz*(dir.z>0?z+1-oz:oz-z);
    let face=[0,0,0],t=0;
    while(t<=maxDist){
      if(inW(x,y,z)&&isSolid(blocks[idx(x,y,z)]))return{x,y,z,face,inside:inW(x,y,z)&&blocks[idx(x,y,z)]};

    }
    return null;
  }

## 遊び方のポイント
- **左クリック**＝ブロックを壊す ／ **右クリック**＝設置（壊したブロックは「手」から消えない＝無限の箱庭）
- **F** で飛行、**Space** で上昇、**Shift** で下降／走る。空中から眺めて地形を整えるのがこのゲームの醍醐味
- **水**は半透明でカリングされるので、海・池・水路も作れます。**ガラス**も半透明
- **岩盤**は壊せません（世界が底抜けしないための壁）
- シードを変えると地形ががらっと変わります。編集は自動保存されるので、同じシードなら「いじった世界」を復元できます

## 中身の解説（少しだけ）
- 描画は **WebGL1** を手書きで組んでいます（外部ライブラリなし）。地形は**面カリング**で、外に面したブロックの面だけメッシュ化するので、40×40×40 の箱庭でも軽く回ります
- テクスチャは **1枚のアトラス**に、草・石・レンガ・板材・原木（年輪）・ガラス・水などを**ピクセルアートで直接描画**
- **昼→夕暮れ→夜→夜明け** の 10 分周期で、空の色・環境光・影の強さ・星・太陽／月が連動します
- 破壊／設置は **ボクセルDDA レイキャスト**で、十字照準の先のブロックを判定し、手前のブロックに**ワイヤー枠**を出しています

拡張したければ、`HOTBAR` にブロック追加、`faceSlot` でテクスチャ指定、`generateTerrain` で生成規則を書き換えるだけで、かなり自由に世界をいじれます。
