function makeTexture(base, kind){
  const cv=newCanvas(), x=cv.getContext('2d');
  x.fillStyle=base; x.fillRect(0,0,16,16);
  const px=()=>{x.fillStyle=shade(hexRgb(base),(Math.random()-.5)*40);x.fillRect(Math.random()*16|0,Math.random()*16|0,1,1);};

  else if(kind==='plank'){for(let y=0;y<16;y+=4){line(0,y,15,y,'rgba(0,0,0,.24)');}for(let i=0;i<18;i++)px();}

function hash(x,y){let h=(x*374761393 + y*668265263 + seed*978539287)|0; h=(h^(h>>>13))>>>0; h=(h*1274126177)>>>0; h=(h^(h>>>16))>>>0; return h/4294967295;}
function smooth(t){return t*t*(3-2*t);}
function vnoise(x,y){const xi=Math.floor(x),yi=Math.floor(y),xf=x-xi,yf=y-yi;const a=hash(xi,yi),b=hash(xi+1,yi),c=hash(xi,yi+1),d=hash(xi+1,yi+1);const u=smooth(xf),v=smooth(yf);return (a*(1-u)+b*u)*(1-v)+(c*(1-u)+d*u)*v;}
function fbm(x,y){let s=0,a=.5,f=1;for(let i=0;i<4;i++){s+=a*vnoise(x*f,y*f);a*=.5;f*=2;}return s;}

function raycastBlock(){
  const origin=camera.position.clone();
  const dir=new THREE.Vector3(0,0,-1).applyQuaternion(camera.quaternion);
  let x=Math.floor(origin.x),y=Math.floor(origin.y),z=Math.floor(origin.z);
  const sx=Math.sign(dir.x),sy=Math.sign(dir.y),sz=Math.sign(dir.z);
  const adx=Math.abs(dir.x),ady=Math.abs(dir.y),adz=Math.abs(dir.z);
  const tdx=adx>1e-9?1/adx:Infinity,tdy=ady>1e-9?1/ady:Infinity,tdz=adz>1e-9?1/adz:Infinity;
  let tmx=adx>1e-9?((sx>0?x+1-origin.x:origin.x-x)/adx):Infinity;
  let tmy=ady>1e-9?((sy>0?y+1-origin.y:origin.y-y)/ady):Infinity;
  let tmz=adz>1e-9?((sz>0?z+1-origin.z:origin.z-z)/adz):Infinity;
  let step=0;
  while(step++<600){
    const id=getB(x,y,z);
    if(id!==0&&id!==WATER) return {hit:{x,y,z},face:{x,y,z},id};
    if(tmx<tmy&&tmx<tmz){x+=sx;if(tmx>REACH)return null;tmx+=tdx;}
    else if(tmy<tmz){y+=sy;if(tmy>REACH)return null;tmy+=tdy;}
    else{z+=sz;if(tmz>REACH)return null;tmz+=tdz;}
  }
  return null;
}
function placeBlock(hit){
  const p=hit.face; const id=HOTBAR[selIndex];
  // don't place inside player
  const px=Math.floor(player.x),py=Math.floor(player.y),pz=Math.floor(player.z);
  if((p.x===px&&p.z===pz&&(p.y===py||p.y===py+1))) return;
  if(getB(p.x,p.y,p.z)===0){ setB(p.x,p.y,p.z,id); placedCount++; markDirty(p.x,p.z); flash(); }
}
function mineBlock(hit){
  const id=getB(hit.hit.x,hit.hit.y,hit.hit.z);
  if(id===0||id===WATER) return;
  setB(hit.hit.x,hit.hit.y,hit.hit.z,0); minedCount++; markDirty(hit.hit.x,hit.hit.z); flash();
}

let dayT=0.32; // 0..1 time of day
function updateEnv(t){
  if(dayCycle) dayT=(dayT+0.00025*t)%1;
  const ang=dayT*Math.PI*2 - Math.PI/2;
  const elev=Math.sin(ang);
  sunDir.set(Math.cos(ang),elev,Math.sin(ang)*0.5).normalize();
  skyU.sunPosition.value.copy(sunDir);
  sun.position.copy(camera.position).add(sunDir.clone().multiplyScalar(60));
  sun.target.position.copy(camera.position);
  sun.target.updateMatrixWorld();
  const dayF=Math.max(0,elev);
  sun.intensity=1.15*dayF;
  sun.color.setHSL(0.09,0.6,0.5+0.45*dayF);
  hemi.intensity=0.25+0.55*dayF;
  stars.material.opacity=Math.max(0,-elev*1.4);
  renderer.toneMappingExposure=0.7+0.35*dayF;
  const phase=elev*10;
  timeLabelEl.textContent=dayF>0.15?'昼':dayF>-0.05?'黄昏':'夜';
  // water wave
  for(const w of waterChunks){
    const p=w.geom.attributes.position;
    for(let i=0;i<w.base.length;i+=3){
      p.array[i+1]=w.base[i+1]+Math.sin(w.base[i]*0.7+t*1.6+w.base[i+2]*0.9)*0.035;
    }
    p.needsUpdate=true;
  }
}
