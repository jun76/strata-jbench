#!/usr/bin/env python3
"""
word_freq.py

A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.

Usage:
    python word_freq.py --file <path_to_file> --top <N>
"""

import argparse
import re
import sys
from collections import Counter
from dataclasses import dataclass


@dataclass
class WordCount:
    """Represents a word and its frequency count."""
    word: str
    count: int


def extract_words(text: str) -> list[str]:
    """
    Extract lowercase words from the given text.
    
    Words are defined as sequences of alphanumeric characters.
    All words are converted to lowercase for case-insensitive counting.
    
    Args:
        text: The input text to extract words from.
        
    Returns:
        A list of lowercase words extracted from the text.
    """
    return re.findall(r'\b[a-z0-9]+\b', text.lower())


def count_words(words: list[str]) -> list[WordCount]:
    """
    Count the frequency of each word in the list.
    
    Args:
        words: A list of words to count.
        
    Returns:
        A list of WordCount objects sorted by count descending,
        then by word ascending for ties.
    """
    counter = Counter(words)
    # Sort by count descending, then by word ascending for deterministic ordering
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return [WordCount(word=word, count=count) for word, count in sorted_items]


def read_file(filepath: str) -> str:
    """
    Read the contents of a text file.
    
    Args:
        filepath: Path to the text file to read.
        
    Returns:
        The contents of the file as a string.
        
    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If there is an error reading the file.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()
