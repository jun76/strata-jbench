"""
word_freq.py - 単語出現頻度集計ツール

テキストファイルを読み込み、単語の出現頻度を集計し、
上位 N 件を表形式で表示するコマンドラインツールです。

Usage:
    python word_freq.py <file> [-n TOP_N] [-c] [-i] [-p]

Examples:
    python word_freq.py sample.txt
    python word_freq.py sample.txt -n 10
    python word_freq.py sample.txt -c -i
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple

# 単語を抽出するための正規表現パターン
# 日本語を含む unicode 文字と数字を単語として扱う
WORD_PATTERN = re.compile(r"\b[\w'-]+\b", re.UNICODE)


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
                        False の場合、すべて小文字に変換して区別しない。

    Returns:
        抽出された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    words = WORD_PATTERN.findall(text)
    return words


def count_words(words: List[str]) -> Counter[str]:
    """
    単語リストから頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)
