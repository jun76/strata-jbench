def extract_words(text: str) -> List[str]:
    """
    Extract words from the given text.

    Words are defined as sequences of alphanumeric characters (including underscores).
    The extraction is case-insensitive, meaning all words are lowercased.

    Args:
        text: The input text from which to extract words.

    Returns:
        A list of lowercase words extracted from the text.
    """
    # Use regex to find all sequences of word characters (letters, digits, underscores)
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return [word.lower() for word in words]

        words: A list of words to count.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)
