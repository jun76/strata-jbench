"""
テキストファイルを読み込んで単語の出現頻度を集計し、上位 N 件を表形式で表示するコマンドラインツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語のリスト（小文字化、非英字を除去して分割）
    """
    # 小文字化し、英字のみを含む単語に分割
    words = re.findall(r'[a-zA-Z]+', text.lower())
    return words


def count_words(text: str) -> Counter[str]:
    """
    テキスト中の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語をキー、出現回数を値とする Counter
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の単語と出現頻度を取得する。
    
    Args:
        counter: 単語頻度 Counter
        n: 上位件数
        
    Returns:
        (単語, 出現回数) のリスト（降順）
    """
    return counter.most_common(n)
