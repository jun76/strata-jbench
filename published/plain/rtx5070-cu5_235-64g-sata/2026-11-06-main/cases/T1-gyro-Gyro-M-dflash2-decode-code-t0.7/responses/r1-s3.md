## `wordcount.py`

```python
#!/usr/bin/env python3
"""
wordcount.py — テキストファイルから単語出現頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

使い方:
    python wordcount.py input.txt [-n 10] [-i] [-w]

オプション:
    -n / --top   上位件数（デフォルト: 10）
    -i / --ignore-case  大文字小文字を区別しない
    -w / --words-only   英単語のみを対象にする（非英字を区切りとして扱う）
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, ignore_case: bool = False, words_only: bool = False) -> List[str]:
    """
    テキストから単語リストを抽出する。

    Args:
        text: 入力テキスト
        ignore_case: True の場合、大文字を小文字に変換して正規化
        words_only: True の場合、英単語（[a-zA-Z]+）のみ抽出。False の場合、
                    空白で分割して単語を抽出。

    Returns:
        単語のリスト
    """
    if words_only:
        # 英単語のみ抽出
        tokens = re.findall(r'[a-zA-Z]+', text)
    else:
        # 空白・改行で分割
        tokens = text.split()

    if ignore_case:
        tokens = [tok.lower() for tok in tokens]

    return tokens


def count_words(text: str, ignore_case: bool = False, words_only: bool = False) -> Counter:
    """
    テキスト内の単語出現頻度を Counter で返す。

    Args:
        text: 入力テキスト
        ignore_case: 大文字小文字を区別しない
        words_only: 英単語のみを対象にする

    Returns:
        単語 -> 出現回数の Counter
    """
    tokens = tokenize(text, ignore_case=ignore_case, words_only=words_only)
    return Counter(tokens)
