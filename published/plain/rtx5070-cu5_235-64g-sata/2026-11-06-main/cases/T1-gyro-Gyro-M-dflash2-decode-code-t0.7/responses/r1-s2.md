Command-line tool: Count word frequencies in a text file and display the top N.

Usage:
    python word_count.py <file> [-n N] [--case-sensitive]
"""

import argparse

def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """Normalize a word by stripping punctuation and optionally lowercasing.

    Args:
        word: The raw word string.
        case_sensitive: If False, convert to lowercase.

    Returns:
        The normalized word string.
    """
    # Strip non-alphanumeric characters from both ends
    stripped = word.strip(".,;:!?\"'()[]{}-")
    if not case_sensitive:
        stripped = stripped.lower()
    return stripped


def count_words(text: str, case_sensitive: bool = False) -> Counter:
    """Count word occurrences in the given text.

    Args:
        text: The input text.
        case_sensitive: If False, words are compared case-insensitively.

    Returns:
        A Counter mapping normalized words to their frequencies.
    """
    counter: Counter = Counter()
    for word in text.split():
        normalized = normalize_word(word, case_sensitive)
        if normalized:
            counter[normalized] += 1
    return counter


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending.
    """
    return counter.most_common(n)
