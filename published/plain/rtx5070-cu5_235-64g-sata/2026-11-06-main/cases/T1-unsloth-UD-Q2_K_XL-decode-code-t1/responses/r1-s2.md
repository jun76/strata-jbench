#!/usr/bin/env python3
"""
wordfreq_tool.py: 単語出現頻度の集計と表形式表示を行うコマンドラインツール。
"""

import argparse
import re

from collections import Counter

def tokenize(text: str) -> List[str]:
    """
    与えられたテキストから単語を抽出する。
    英数字のみを単語とし、大文字小文字は無視する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        正規化された単語リスト
    """
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_words(text: str) -> Counter:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語と出現回数の Counter
    """
    words = tokenize(text)
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    カウンターから上位 n 件の単語と頻度を取得する。

    Args:
        counter: 単語頻度の Counter
        n: 取得する件数
        
    Returns:
        (単語, 頻度) のタプルのリスト。頻度降順、辞書順昇順。

        key=lambda x: (-x[1], x[0])

    return items[:n]
