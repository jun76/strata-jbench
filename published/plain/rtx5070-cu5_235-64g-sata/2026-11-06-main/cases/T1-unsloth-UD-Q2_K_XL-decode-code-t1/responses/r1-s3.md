def tokenize(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    Args:
        text: 解析対象のテキスト文字列。

    Returns:
        小文字化し、非英数字を除去した単語のリスト。
    """
    # 正規表現で英数字とハイフン、アポストロフィを含む単語を抽出
    # 例: "don't" -> "don't", "hello-world" -> "hello-world"
    words = re.findall(r"[a-zA-Z0-9][a-zA-Z0-9\-']*", text.lower())
    return words

    """

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(tokenize(text))
