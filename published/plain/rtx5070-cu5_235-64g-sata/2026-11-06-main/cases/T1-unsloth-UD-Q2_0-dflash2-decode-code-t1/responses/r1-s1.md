def tokenize(text: str) -> list[str]:
    """
    テキストを単語リストに分割する。

    単語は英数字（およびハイフン・アポストロフィ）で構成される文字列として抽出する。
    抽出後、すべての単語は小文字に変換される。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語のリスト。
    """
    # 英数字、ハイフン、アポストロフィを含む単語を抽出
    pattern = r"[a-zA-Z0-9']+"
    words = re.findall(pattern, text.lower())
    # 単一のアポストロフィやハイフンだけの「単語」を除外
    return [w for w in words if w.strip("'-")]


def count_words(words: list[str]) -> Counter[str]:
    """
    単語リストから出現頻度を集計する。

    Args:
        words: 単語のリスト。

    Returns:
        単語ごとの出現回数を格納した Counter オブジェクト。
    """
    return Counter(words)
