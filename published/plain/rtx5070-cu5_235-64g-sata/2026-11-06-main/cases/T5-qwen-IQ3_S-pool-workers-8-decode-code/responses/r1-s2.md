#!/usr/bin/env python3
"""
word_counter.py - テキストファイルから単語頻度を集計し、上位 N 件を表形式で表示する CLI ツール。

使用例:
    python word_counter.py input.txt --top 10
    python word_counter.py input.txt --top 5 --sort-by count
    python word_counter.py input.txt --top 20 --sort-by word
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """テキストから単語頻度をカウントする。

    Args:
        text: 解析対象のテキスト文字列。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
        単語は小文字化され、非英数字文字は区切りとして扱われる。
    """
    # 単語を抽出: 英数字およびハイフン・アポストロフィを含む連続文字列
    # 例: "don't" -> "don't", "e-mail" -> "e-mail"
    words = re.findall(r"[a-zA-Z0-9]+(?:['-][a-zA-Z0-9]+)*", text)
    # 小文字化して正規化
    normalized_words = [word.lower() for word in words]
    return Counter(normalized_words)


def read_text_file(filepath: str) -> str:
    """指定されたパスのテキストファイルを読み込む。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイルの内容（文字列）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        PermissionError: ファイルの読み取り権限がない場合。
    """
    path = Path(filepath)
    if not path.exists():
        raise FileNotFoundError(f"ファイルが見つかりません: {filepath}")
    return path.read_text(encoding="utf-8")
