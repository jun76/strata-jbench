"""
コマンドラインツール: テキストファイルを読み込み、単語出現頻度の上位N件を表形式で出力します。
"""

import argparse

def parse_arguments() -> argparse.Namespace:
    """
    コマンドライン引数を解析します。

    Returns:
        argparse.Namespace: 解析された引数オブジェクト
    """
    parser = argparse.ArgumentParser(
        description="テキストファイル内の単語出現頻度を集計し、上位N件を表示します。"
    )
    parser.add_argument(
        "input_file",
        type=str,
        help="読み込むテキストファイルのパス"
    )
    parser.add_argument(
        "-n", "--top",
        type=int,
        default=10,
        help="表示する上位件数 (デフォルト: 10)"
    )
    parser.add_argument(
        "-v", "--verbose",
        action="store_true",
        help="追加の情報（エラーメッセージなど）を表示する"
    )
    return parser.parse_args()


def read_file(filepath: str) -> str:
    """
    指定されたファイルを読み込み、その内容を文字列として返します。

    Args:
        filepath (str): 読み込むファイルのパス

    Returns:
        str: ファイルの内容

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        PermissionError: 読み取り権限がない場合
    """
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        raise
    except PermissionError:
