"""上位 N 件の単語頻度を表示するコマンドラインツール。

使い方:
    python word_freq.py input.txt -n 10 -s
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str, *, strip_punctuation: bool = True) -> List[str]:
    """テキストから単語トークンリストを返す。

    Args:
        text: 入力テキスト。
        strip_punctuation: 単語前後の句読点を取り除くかどうか。

    Returns:
        小文字化された単語トークンのリスト。
    """
    if strip_punctuation:
        # 単語境界で分割し、各トークンから先頭末尾の句読点のみ除去
        tokens = re.findall(r"[A-Za-zÀ-ɏ0-9]+(?:['-][A-Za-zÀ-ɏ0-9]+)*", text)
    else:
        tokens = text.split()
    return [t.lower() for t in tokens if t]


def count_frequencies(text: str, *, strip_punctuation: bool = True) -> Counter[str]:
    """テキスト内の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト。
        strip_punctuation: 句読点を除去するかどうか。

    Returns:
        単語 -> 出現回数の Counter。
    """
    return Counter(tokenize(text, strip_punctuation=strip_punctuation))
