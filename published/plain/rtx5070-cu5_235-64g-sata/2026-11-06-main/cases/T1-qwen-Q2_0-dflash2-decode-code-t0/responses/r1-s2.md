word_freq.py - 単語出現頻度分析ツール

テキストファイルを読み込み、単語の出現頻度を集計し、
上位N件を表形式で表示するコマンドラインツール。

Usage:
    python word_freq.py <file> [--top N] [--case-sensitive] [--min-length L]
"""

def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    文本から単語を抽出する。

    Args:
        text: 分析対象のテキスト
        case_sensitive: 大小写区分（False则统一转小写）
        min_length: 单词最小长度（低于此长度的单词被忽略）

    Returns:
        提取的单词列表
    """
    if not case_sensitive:
        text = text.lower()

    # 提取单词：只保留字母、数字、撇号（如 don't）
    words: List[str] = []
    current_word: List[str] = []

    for char in text:
        if char.isalpha() or char.isdigit() or char == "'":
            current_word.append(char)
        else:
            if current_word:
                word = "".join(current_word)
                # 去除首尾的撇号
                word = word.strip("'")
                if len(word) >= min_length:
                    words.append(word)
                current_word = []

    if current_word:
        word = "".join(current_word).strip("'")
        if len(word) >= min_length:
            words.append(word)

    return words

    Args:
        words: 单词列表

    Returns:
        Counter对象，包含每个单词的出现次数
    """
    return Counter(words)
