#!/usr/bin/env python3
"""
word_freq.py - テキストファイルの単語出現頻度を集計し、上位 N 件を表形式で表示する CLI 工具。

Usage:
    python word_freq.py <file> [--top N] [--case-sensitive] [--min-len L]

Examples:
    python word_freq.py sample.txt --top 10
    python word_freq.py sample.txt --top 5 --case-sensitive
    python word_freq.py sample.txt --top 3 --min-len 4
"""

import argparse

from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大小写を区別するか。False なら小写に統一。
        min_len: 単語の最小長。この値未満の単語は除外。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()

    # 非文字・非数字を区切りとして分割
    import re
    words = re.findall(r"[a-zA-Z0-9]+", text)

    if min_len > 1:
        words = [w for w in words if len(w) >= min_len]

    return words


def count_words(text: str, case_sensitive: bool = False, min_len: int = 1) -> Counter:
    """
    テキスト中の単語出現頻度を Counter として返す。

    Args:
        text: 入力テキスト。
        case_sensitive: 大小写を区別するか。
        min_len: 単語の最小長。

    Returns:
        collections.Counter 实例。
    """
    words = tokenize(text, case_sensitive=case_sensitive, min_len=min_len)
    return Counter(words)
