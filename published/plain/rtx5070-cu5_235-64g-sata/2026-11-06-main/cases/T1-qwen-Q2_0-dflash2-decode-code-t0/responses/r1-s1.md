## 1. 実装 (`word_freq.py`)

#!/usr/bin/env python3
"""
word_freq.py - テキストファイルの単語出現頻度上位 N 件を表形式で表示する CLI 工具。

Usage:
    python word_freq.py <file> [--top N] [--case-sensitive] [--min-len L]

Example:
    python word_freq.py sample.txt --top 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大小写を区別するか。False なら小写に統一。
        min_len: 単語の最小長。この値未満の単語は無視。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()

    # 非文字・数字を区切りとして分割
    # 日本語・Unicode 文字も単語として扱う
    import re
    # \w+ は Unicode 文字を含む
    words = re.findall(r"\w+", text)
    return [w for w in words if len(w) >= min_len]


def count_words(text: str, case_sensitive: bool = False, min_len: int = 1) -> Counter:
    """
    単語出現頻度を Counter として返す。

    Args:
        text: 入力テキスト。
        case_sensitive: 大小写区別。
        min_len: 最小単語長。

    Returns:
        collections.Counter 实例。
    """
    words = tokenize(text, case_sensitive=case_sensitive, min_len=min_len)
    return Counter(words)
