def read_file(path: str) -> str:
    """
    Read the content of a text file.

    Args:
        path: Path to the text file.

    Returns:
        The text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If there is an error reading the file.
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words.

    Words are defined as sequences of alphanumeric characters (including underscores).
    Punctuation and non-word characters are treated as separators.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens.
    """
    # Match sequences of alphanumeric characters or underscores
    tokens = re.findall(r"[a-zA-Z0-9_]+", text)
    return [token.lower() for token in tokens]

        tokens: A list of word tokens.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(tokens)
