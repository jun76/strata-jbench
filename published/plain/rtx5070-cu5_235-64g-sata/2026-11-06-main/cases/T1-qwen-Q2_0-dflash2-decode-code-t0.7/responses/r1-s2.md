#!/usr/bin/env python3
"""
word_freq.py

Command-line tool to count word occurrences in a text file and display
the top N most frequent words in a tabular format.

Usage:
    python word_freq.py <file_path> [--top N] [--case-insensitive]

Example:
    python word_freq.py sample.txt --top 10 --case-insensitive
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize_text(text: str, case_insensitive: bool = False) -> List[str]:
    """
    Tokenize text into words.

    Args:
        text: Raw text content.
        case_insensitive: If True, normalize words to lowercase.

    Returns:
        List of tokenized words (alphabetic characters only, ignoring punctuation/numbers).
    """
    # Extract sequences of alphabetic characters as words
    words = re.findall(r"[a-zA-Z]+", text)
    if case_insensitive:
        words = [w.lower() for w in words]
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: List of tokenized words.

    Returns:
        Counter object mapping each word to its occurrence count.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words from the counter.

    Args:
        counter: Counter object containing word frequencies.
        n: Number of top entries to return.

    Returns:
        List of (word, count) tuples sorted by frequency descending, then by word ascending.
    """
    # Sort by frequency descending, then by word ascending for tie-breaking
    sorted_items = sorted(
        counter.items(),
        key=lambda item: (-item[1], item[0])
    )
    return sorted_items[:n]
