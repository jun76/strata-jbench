wordfreq.py - A command-line tool that reads a text file, counts word frequencies,

Usage:

import argparse
import re

from collections import Counter
from typing import List, Tuple, Optional


def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Tokenize the given text into words.
    
    Words are extracted using a regex pattern that matches sequences of
    alphanumeric characters and underscores. Punctuation and whitespace
    are ignored.
    
    Args:
        text: The input text to tokenize.
        case_sensitive: If False, all words are converted to lowercase
                        before counting. If True, original case is preserved.
    
    Returns:
        A list of tokens (words) extracted from the text.
    """
    # Match words consisting of alphanumeric characters and underscores
    tokens = re.findall(r'\b\w+\b', text)
    if not case_sensitive:
        tokens = [token.lower() for token in tokens]
    return tokens


def count_frequencies(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    Count the frequency of each word in the given text.
    
    Args:
        text: The input text to analyze.
        case_sensitive: If False, all words are converted to lowercase
                        before counting. If True, original case is preserved.
    
    Returns:
        A Counter object mapping each word to its frequency.
    """
    tokens = tokenize(text, case_sensitive=case_sensitive)
    return Counter(tokens)
