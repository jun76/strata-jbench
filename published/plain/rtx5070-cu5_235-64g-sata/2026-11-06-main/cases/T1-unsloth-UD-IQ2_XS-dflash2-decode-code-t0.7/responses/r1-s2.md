### 1. `wordfreq.py` (メインスクリプト)

```python
#!/usr/bin/env python3
"""
wordfreq.py

テキストファイルを読み込み、単語の出現頻度を計算して上位 N 件を表形式で表示するツール。

依存ライブラリ: なし (標準ライブラリのみ)
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def read_file(filepath: str) -> str:
    """
    指定されたファイルパスからテキストを UTF-8 で読み込む。

    Args:
        filepath (str): 読み込むテキストファイルのパス。

    Returns:
        str: ファイルの内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: 読み込み中にエラーが発生した場合。
    """
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"ファイルが見つかりません: {filepath}")
    except IOError as e:
        raise IOError(f"ファイルの読み込みに失敗しました: {e}")


def normalize_text(text: str) -> str:
    """
    テキストを小文字にし、単語区切りを正規化する。
    非アルファベット文字は空白として扱う。

    Args:
        text (str): 元のテキスト。

    Returns:
        str: 正規化されたテキスト。
    """
    import re
    # 英数字以外の文字を空白で置き換える（日本語などの非ASCII文字も考慮）
    # ただし、ここでは一般的な英語ベースのツールとして、英数字と下文字線のみを残すか、
    # より一般的な「単語」の定義として、区切り文字をスペースに変換する。
    # 例: "Hello, world!" -> "hello world"
    normalized = text.lower()
    # 非英数字文字（スペースを除く）をスペースで置換
    normalized = re.sub(r'[^a-z0-9\s]', ' ', normalized)
    # 複数の空白を1つに
    normalized = re.sub(r'\s+', ' ', normalized)
    return normalized
