word_freq.py - テキストファイルの単語頻度を集計し、上位 N 件を表形式で表示するツール。

Usage:
    python word_freq.py <filepath> [--top N] [--case-sensitive] [--min-len LEN]
"""

import argparse

def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """
    単語を正規化する。

    Args:
        word: 正規化する単語。
        case_sensitive: True の場合、大文字小文字を区別しない。False の場合、小文字に変換する。

    Returns:
        正規化された単語。
    """
    if not case_sensitive:
        return word.lower()
    return word
