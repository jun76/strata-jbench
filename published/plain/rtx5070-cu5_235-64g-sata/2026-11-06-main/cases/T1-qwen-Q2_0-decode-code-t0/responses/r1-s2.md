以下は、Pythonでテキストファイルから単語出現頻度を集計し、上位N件を表形式で表示するコマンドラインツールです。argparse、型ヒント、docstring、pytest単体テストを含んでいます。

#!/usr/bin/env python3
"""
word_freq.py - 単語出現頻度分析ツール

テキストファイルを読み込み、単語の出現頻度を集計し、
上位N件を表形式で表示するコマンドラインツール。

Usage:
    python word_freq.py <file> [--top N] [--case-sensitive] [--min-length L]
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    文本を単語に分割する。

    Args:
        text: 分析対象のテキスト
        case_sensitive: 大小写敏感（默认False）
        min_length: 单词最小长度（默认1）

    Returns:
        单词列表
    """
    if not case_sensitive:
        text = text.lower()

    # 提取单词：只保留字母、数字、撇号（如 don't）
    words = []
    current_word = []
    for ch in text:
        if ch.isalnum() or ch == "'":
            current_word.append(ch)
        else:
            if current_word:
                word = "".join(current_word)
                if len(word) >= min_length:
                    words.append(word)
                current_word = []
    if current_word:
        word = "".join(current_word)
        if len(word) >= min_length:
            words.append(word)

    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    单词频率计数。

    Args:
        words: 单词列表

    Returns:
        Counter 对象，包含每个单词的出现次数
    """
    return Counter(words)
