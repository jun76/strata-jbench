from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_len: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: True なら大小文字を区別。False なら小文字化。
        min_len: 単語の最小長。この値未満の単語は除外。

    Returns:
        単語リスト。
    """
    # 非文字・数字を区切りとして分割
    tokens = re.findall(r"[A-Za-z0-9]+", text)
    if not case_sensitive:
        tokens = [t.lower() for t in tokens]
    return [t for t in tokens if len(t) >= min_len]

        text: 入力テキスト。

        min_len: 単語の最小長。

    Returns:
        collections.Counter: 単語 -> 出現回数。
    """
    tokens = tokenize(text, case_sensitive=case_sensitive, min_len=min_len)
    return Counter(tokens)
