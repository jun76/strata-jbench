#!/usr/bin/env python3
"""
word_counter.py - テキストファイルの単語出現頻度集計ツール

コマンドラインから指定したテキストファイルを読み込み、
単語の出現頻度を集計し、上位 N 件を表形式で表示します。

Usage:
    python word_counter.py <file> [--top N] [--case-sensitive] [--min-length L]
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = False, min_length: int = 1) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 分割対象のテキスト
        case_sensitive: 大小文字を区別するか (False なら小文字化)
        min_length: 単語の最小長 (この値未満の単語は除外)

    Returns:
        単語のリスト
    """
    if not case_sensitive:
        text = text.lower()

    # 非文字・非数字を空白で分割
    # 単語 = 文字・数字・ハイphen・apostrophe を含む連続部分
    import re
    words = re.findall(r"[a-zA-Z0-9'-]+", text)

    # 最小長フィルタ
    words = [w for w in words if len(w) >= min_length]

    return words


def count_words(text: str, case_sensitive: bool = False, min_length: int = 1) -> Counter:
    """
    テキスト中の単語出現頻度を Counter として返す。

    Args:
        text: 分析対象テキスト
        case_sensitive: 大小文字を区別するか
        min_length: 単語の最小長

    Returns:
        collections.Counter 形式の頻度
    """
    words = tokenize(text, case_sensitive=case_sensitive, min_length=min_length)
    return Counter(words)
