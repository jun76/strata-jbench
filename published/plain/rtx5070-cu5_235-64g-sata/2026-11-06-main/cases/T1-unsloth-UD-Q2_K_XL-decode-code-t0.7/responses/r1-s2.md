#!/usr/bin/env python3
"""
word_count.py - コマンドラインツールでテキストファイルから単語出現頻度を上位N件集計し、表形式で表示する。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    
    Args:
        text: 入力テキスト
        
    Returns:
        小文字化した単語のリスト
    """
    # 英数字のみの単語を抽出（日本語などの場合はスペース区切りなどに対応可能）
    words = re.findall(r'\b[a-zA-Z0-9]+\b', text.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。
    
    Args:
        words: 単語のリスト
        
    Returns:
        Counter オブジェクト
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    上位 N 件の単語と頻度を取得する。
    
    Args:
        counter: Counter オブジェクト
        n: 上位 N 件
        
    Returns:
        (単語, 頻度) のタプルリスト
    """
    return counter.most_common(n)

    """
    上位単語を表形式でフォーマットする。
    
    Args:
        top_words: (単語, 頻度) のタプルリスト
        
    Returns:
        表形式の文字列
    """
    if not top_words:
        return "単語が見つかりません。"

    word_width = max(len(word) for word, _ in top_words)
    word_width = max(word_width, len("Word"))

    count_width = max(count_width, len("Count"))
