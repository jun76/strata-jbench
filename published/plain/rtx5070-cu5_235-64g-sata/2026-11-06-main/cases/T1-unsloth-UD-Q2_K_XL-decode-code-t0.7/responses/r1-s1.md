def clean_word(word: str) -> str:
    """
    単語から句読点や記号を取り除き、小文字に変換する。

    Args:
        word (str): 元の単語文字列

    Returns:
        str: クリーニングされた単語
    """
    # 句読点や記号を除去（アルファベット、数字、日本語などの文字のみ残す）
    # 簡単な実装のため、isalnum() で判定できない文字を除去する
    # 日本語やUnicode文字も考慮するため、stripではなく各文字をフィルタリング
    cleaned = "".join(char for char in word if char.isalnum())
    return cleaned.lower()


def count_words(text: str) -> Dict[str, int]:
    """
    テキストから単語ごとの出現頻度を計算する。

    Args:
        text (str): 入力テキスト

    Returns:
        Dict[str, int]: 単語をキー、出現回数を値とする辞書
    """
    # 空白や改行で分割
    raw_words = text.split()

    words = [clean_word(w) for w in raw_words]
    words = [w for w in words if w]  # 空文字を除外
    
    return Counter(words)
