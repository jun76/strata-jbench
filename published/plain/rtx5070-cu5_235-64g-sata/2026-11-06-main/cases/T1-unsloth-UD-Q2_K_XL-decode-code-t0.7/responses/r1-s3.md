"""
word_freq.py - テキストファイルから単語出現頻度を算出し、上位N件をテーブル形式で表示するCLIツール。

使用例:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt -n 20 --case-sensitive
    python word_freq.py input.txt --sort-by frequency
"""

import argparse

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストを単語リストにトークン化する。

    Args:
        text: 入力テキスト文字列。
        case_sensitive: Trueの場合、大文字小文字を区別する。Falseの場合、すべて小文字に変換する。

    Returns:
        単語のリスト。各単語は正規表現 \w+ にマッチする文字列。

    Examples:
        >>> tokenize("Hello, World! hello")
        ['hello', 'world', 'hello']
        >>> tokenize("Hello, World! hello", case_sensitive=True)
        ['Hello', 'World', 'hello']
    """
    if not case_sensitive:
        text = text.lower()
    # \w+ にマッチする単語を抽出（英数字とアンダースコア）
    return re.findall(r'\w+', text)


def count_frequencies(tokens: List[str]) -> Counter:
    """
    単語トークンリストから出現頻度をカウントする。

    Args:
        tokens: 単語トークンのリスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。

    Examples:
        >>> count_frequencies(['apple', 'banana', 'apple'])
        Counter({'apple': 2, 'banana': 1})
    """
    return Counter(tokens)
