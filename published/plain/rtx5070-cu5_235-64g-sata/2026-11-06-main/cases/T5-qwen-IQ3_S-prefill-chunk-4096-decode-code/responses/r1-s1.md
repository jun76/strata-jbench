"""
word_freq: A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.

Usage:
    python word_freq.py [options] <file>

Options:
    -n, --top N      Number of top words to display (default: 10)
    -c, --case       Case-sensitive counting (default: case-insensitive)
    -p, --pattern    Regex pattern for word definition (default: \\w+)
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str, case_sensitive: bool = False, pattern: str = r'\w+') -> Counter[str]:
    """
    Count word frequencies in the given text.

    Args:
        text: The input text to analyze.
        case_sensitive: If True, preserve case; otherwise, convert to lowercase.
        pattern: Regular expression pattern to identify words.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    if not case_sensitive:
        text = text.lower()
    words = re.findall(pattern, text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: Number of top items to return.

    Returns:
        A list of (word, count) tuples, sorted by frequency descending,
        then alphabetically for ties.
    """
    # Sort by count descending, then by word ascending for consistent tie-breaking
    sorted_items = sorted(
        counter.items(),
        key=lambda item: (-item[1], item[0])
    )
    return sorted_items[:n]
