"""
word_counter.py

A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a tabular format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    Count word occurrences in the given text.
    
    Words are extracted using a simple regex that matches sequences of
    alphanumeric characters. Words are converted to lowercase for
    case-insensitive counting.
    
    Args:
        text: The input text string.
        
    Returns:
        A Counter object with word counts.
    """
    # Match sequences of alphanumeric characters (words)
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text.lower())
    return Counter(words)


def read_file(filepath: str) -> str:
    """
    Read the content of a text file.
    
    Args:
        filepath: Path to the text file.
        
    Returns:
        The content of the file as a string.
        
    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If there is an error reading the file.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word counts.
        n: The number of top words to return.
        
    Returns:
        A list of tuples, each containing (word, count), sorted by
        count descending, then by word ascending for ties.
    """
    # most_common returns sorted by count descending; for ties, we need
    # to sort further by word ascending. We can do this by manually
    # sorting all items.
    items = list(counter.items())
    # Sort by count descending, then by word ascending
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
