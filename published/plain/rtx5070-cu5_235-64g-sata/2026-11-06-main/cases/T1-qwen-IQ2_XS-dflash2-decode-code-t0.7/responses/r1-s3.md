#!/usr/bin/env python3
"""
Text frequency counter CLI tool.

Reads a text file, counts word frequencies, and displays the top N entries
in a formatted table.
"""

import argparse
import sys
from collections import Counter
from pathlib import Path


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in the given text.

    Args:
        text: Input text string.

    Returns:
        Counter mapping words to their frequencies.
    """
    # Split by whitespace and normalize to lowercase
    words = text.lower().split()
    return Counter(words)


def format_table(counter: Counter[str], top_n: int = 10) -> str:
    """Format the top N word frequencies as a table string.

    Args:
        counter: Counter mapping words to frequencies.
        top_n: Number of top entries to display.

    Returns:
        Formatted table string.
    """
    top_items = counter.most_common(top_n)

    if not top_items:
        return "No words found."

    # Determine column widths
    word_width = max(len(word) for word, _ in top_items)
    count_width = max(len(str(count)) for _, count in top_items)
    rank_width = max(len(str(rank)) for rank in range(1, top_n + 1))

    header = f"{'Rank':>{rank_width}} | {'Word':<{word_width}} | {'Count':>{count_width}}"
    separator = "-" * (rank_width + word_width + count_width + 3)

    lines = [header, separator]
    for rank, (word, count) in enumerate(top_items, start=1):
        line = f"{rank:>{rank_width}} | {word:<{word_width}} | {count:>{count_width}}"
        lines.append(line)

    return "\n".join(lines)
