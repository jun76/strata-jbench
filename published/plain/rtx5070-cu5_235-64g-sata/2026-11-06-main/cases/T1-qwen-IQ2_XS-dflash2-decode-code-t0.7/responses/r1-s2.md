Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.

Usage:
    python wordfreq.py <input_file> [--top N] [--output FILE]

Example:
    python wordfreq.py sample.txt --top 10
"""

def read_file(filepath: str) -> str:
    """Read the contents of a text file.

    Args:
        filepath: Path to the input text file.

    Returns:
        The full text content of the file.

    Raises:
        FileNotFoundError: If the file does not exist.
        PermissionError: If the file cannot be read.
    """
    path = Path(filepath)
    if not path.exists():
        raise FileNotFoundError(f"File not found: {filepath}")
    if not path.is_file():
        raise ValueError(f"Path is not a file: {filepath}")
    try:
        return path.read_text(encoding="utf-8")
    except PermissionError:
        raise PermissionError(f"Cannot read file: {filepath}")


def extract_words(text: str) -> List[str]:
    """Extract words from text, converting to lowercase and removing punctuation.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words (letters only, no punctuation).
    """
    words: List[str] = []
    current_word: List[str] = []
    for char in text:
        if char.isalpha():
            current_word.append(char.lower())
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))
    return words


def count_words(text: str) -> Dict[str, int]:
    """Count the frequency of each word in the given text.

    Args:
        text: The input text string.

    Returns:
        A dictionary mapping each word to its occurrence count.
    """
    words = extract_words(text)
    return Counter(words)
