import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple


def read_text_file(filepath: Path) -> str:
    """Read text from a file.

    Args:
        filepath: Path to the text file.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        OSError: If the file cannot be read.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def count_words(text: str) -> Dict[str, int]:
    """Count word occurrences in a text string.

    Args:
        text: The input text string.

    Returns:
        A dictionary mapping words to their occurrence counts.
    """
    words = text.lower().split()
    return dict(Counter(words))


def top_n_words(word_counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words.

    Args:
        word_counts: Dictionary mapping words to their counts.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples sorted by count descending, then word ascending.
    """
    sorted_items = sorted(word_counts.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]


def format_table(rows: List[Tuple[str, int]]) -> str:
    """Format rows as a table.

    Args:
        rows: List of (word, count) tuples.

    Returns:
        A formatted table string.
    """
    if not rows:
        return ""

    # Calculate column widths
    word_width = max(len(word) for word, _ in rows)
    count_width = max(len(str(count)) for _, count in rows)

    header = f"{'Word':<{word_width}} | {'Count':>{count_width}}"
    separator = f"{'-' * word_width}-|-{'-' * count_width}"
    lines = [header, separator]
    for word, count in rows:
        lines.append(f"{word:<{word_width}} | {count:>{count_width}}")
