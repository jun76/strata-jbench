Usage:
    python word_freq.py <input_file> [-n TOP_N] [--case-sensitive]

def tokenize(text: str, case_sensitive: bool = False) -> List[str]:
    """Tokenize text into words.

    Args:
        text: The input text to tokenize.
        case_sensitive: If True, preserve case. Otherwise, lowercase all tokens.

    Returns:
        A list of word tokens (alphabetic sequences).
    """
    if not case_sensitive:
        text = text.lower()
    # Extract words: sequences of letters (Unicode-aware)
    tokens = re.findall(r'\b\p{L}+\b' if False else r'[^\W\d_]+', text, flags=re.UNICODE)
    return tokens


def count_words(text: str, top_n: int = 10, case_sensitive: bool = False) -> List[Tuple[str, int]]:
    """Count word frequencies and return the top-N most frequent words.

    Args:
        text: The input text.
        top_n: Number of top entries to return.
        case_sensitive: If True, treat upper/lower case as distinct.

    Returns:
        A list of (word, count) tuples sorted by count descending, then alphabetically.
    """
    tokens = tokenize(text, case_sensitive=case_sensitive)
    counter = Counter(tokens)
    # Sort by count descending, then by word ascending for stability
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:top_n]
