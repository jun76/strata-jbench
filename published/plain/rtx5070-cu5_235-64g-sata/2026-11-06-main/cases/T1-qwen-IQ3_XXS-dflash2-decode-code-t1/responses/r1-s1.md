@dataclass(frozen=True)
class WordFrequencyEntry:
    """1つの単語とその出現回数を表すデータクラス。"""

    word: str
    count: int

    def __repr__(self) -> str:
        return f"WordFrequencyEntry(word={self.word!r}, count={self.count})"


def tokenize(text: str, *, case_sensitive: bool = False, min_length: int = 1) -> list[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別する。
        min_length: 含まれる最小単語長（この長さ未満のトークンは無視）。

    Returns:
        単語のリスト（順序保持）。
    """
    if not case_sensitive:
        text = text.lower()

    # 単語を抽出: 英数字とハイフン、アポストロフィを含む連続文字列
    tokens = re.findall(r"[a-zA-Z0-9'-]+", text)

    # 最小長フィルタ
    if min_length > 1:
        tokens = [t for t in tokens if len(t) >= min_length]

    return tokens
