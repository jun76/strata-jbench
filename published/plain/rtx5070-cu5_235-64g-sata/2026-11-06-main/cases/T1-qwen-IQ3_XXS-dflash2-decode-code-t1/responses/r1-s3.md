"""word_freq: Command-line tool to count word frequencies in text files and display top N entries."""

def read_text_file(path: Path) -> str:
    """Read a text file and return its content as a string.

    Args:
        path: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        OSError: If the file cannot be read.
    """
    return path.read_text(encoding="utf-8")


def extract_words(text: str) -> List[str]:
    """Extract words from text by splitting on non-alphanumeric characters.

    Words are lowercased for case-insensitive counting.

    Args:
        text: Input text string.

    Returns:
        A list of lowercased words extracted from the text.
    """
    # Split on any non-alphanumeric character, filter out empty strings
    words = re.split(r"[^a-zA-Z0-9']+", text)
    return [word.lower() for word in words if word]


def count_frequencies(words: List[str]) -> Counter:
    """Count the frequency of each word in a list.

    Args:
        words: A list of words to count.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most frequent words.

    Args:
        counter: Counter object with word frequencies.
        n: Number of top entries to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending, then alphabetically.
    """
    return counter.most_common(n)
