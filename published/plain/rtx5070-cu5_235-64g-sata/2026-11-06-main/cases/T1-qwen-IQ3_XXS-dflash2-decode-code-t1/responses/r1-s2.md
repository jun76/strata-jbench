#!/usr/bin/env python3
"""
word_freq_counter.py

A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    Extract lowercase alphanumeric words from text.
    
    Args:
        text: Input text string.
        
    Returns:
        List of lowercase words found in the text.
    """
    # Match sequences of letters and digits (words), convert to lowercase
    return re.findall(r"[a-zA-Z0-9']+", text.lower())


def count_word_frequencies(text: str) -> Counter:
    """
    Count the frequency of each word in the given text.
    
    Args:
        text: Input text string.
        
    Returns:
        Counter mapping each word to its frequency.
    """
    words = extract_words(text)
    return Counter(words)


def get_top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.
    
    Args:
        counter: A Counter object with word frequencies.
        n: Number of top words to return.
        
    Returns:
        List of tuples (word, count) sorted by count descending.
        For ties, words are sorted alphabetically.
    """
    # most_common returns items sorted by count descending;
    # for ties, Python's Counter preserves insertion order which is not guaranteed
    # alphabetical, so we sort explicitly for deterministic output.
    common = counter.most_common()
    # Sort by (-count, word) to handle ties alphabetically
    sorted_items = sorted(common, key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
