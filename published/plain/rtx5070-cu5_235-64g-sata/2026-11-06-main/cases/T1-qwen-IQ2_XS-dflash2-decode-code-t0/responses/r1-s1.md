#!/usr/bin/env python3
"""
Word frequency counter command-line tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """Count word occurrences in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter mapping each word to its occurrence count.
    """
    words = text.lower().split()
    # Strip punctuation from each word
    cleaned_words = []
    for word in words:
        # Remove leading/trailing punctuation
        stripped = word.strip(".,!?;:'\"()[]{}")
        if stripped:
            cleaned_words.append(stripped)
    return Counter(cleaned_words)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words.

    Args:
        counter: A Counter of word frequencies.
        n: Number of top words to return.

    Returns:
        A list of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """Format word-frequency pairs as a table string.

    Args:
        rows: List of (word, count) tuples.

    Returns:
        A formatted table string with header and rows.
    """
    if not rows:
        return "No words found."

    # Determine column widths
    word_width = max(len(word) for word, _ in rows)
    count_width = max(len(str(count)) for _, count in rows)

    # Header
    header_word = "Word".ljust(word_width)
    header_count = "Count".rjust(count_width)
    separator = "-" * (word_width + count_width + 3)

    lines = [
        f"{header_word} | {header_count}",
        separator,
    ]

    for word, count in rows:
        lines.append(f"{word.ljust(word_width)} | {str(count).rjust(count_width)}")

    return "\n".
