## `word_counter.py`

```python
#!/usr/bin/env python3
"""Word frequency counter CLI.

Reads one or more text files, counts word occurrences (case-insensitive),
and prints the top N words in a formatted table.

Usage:
    python word_counter.py --input FILE1 [FILE2 ...] [--top N] [--case-sensitive]
"""

from __future__ import annotations

import argparse
import re
import sys
from collections import Counter
from pathlib import Path


def extract_words(text: str, case_sensitive: bool = False) -> list[str]:
    """Extract words from text.

    Words are sequences of alphanumeric characters (including underscores).
    By default, matching is case-insensitive (words are lowercased).

    Args:
        text: Input text string.
        case_sensitive: If True, preserve original casing.

    Returns:
        List of extracted words.
    """
    if case_sensitive:
        # Extract words preserving case
        return re.findall(r"\b[a-zA-Z0-9_]+\b", text)
    else:
        # Extract words and lowercase them
        return [w.lower() for w in re.findall(r"\b[a-zA-Z0-9_]+\b", text)]


def count_words(files: list[Path], case_sensitive: bool = False) -> Counter[str]:
    """Count word frequencies across multiple text files.

    Args:
        files: List of Path objects pointing to text files.
        case_sensitive: If True, preserve word case when counting.

    Returns:
        Counter mapping words to their occurrence counts.
    """
    counter: Counter[str] = Counter()
    for file_path in files:
        try:
            text = file_path.read_text(encoding="utf-8")
        except FileNotFoundError:
            raise FileNotFoundError(f"File not found: {file_path}")
        except UnicodeDecodeError:
            raise UnicodeDecodeError(
                "utf-8",
                file_path.read_bytes(),
                0,
                1,
