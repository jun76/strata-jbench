word_count.py - A command-line tool to count word frequencies in a text file.

This script reads a text file, counts the frequency of each word, and displays

Usage:
    python word_count.py <input_file> [-n N] [--case-sensitive]
"""

def read_file(file_path: str) -> str:
    """
    Read the content of a text file.

    Args:
        file_path: Path to the input text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If there is an error reading the file.
    """
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        raise FileNotFoundError(f"File not found: {file_path}")
    except IOError as e:
        raise IOError(f"Error reading file {file_path}: {e}")


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Extract words from the given text.

    Words are defined as sequences of alphanumeric characters (including underscores).
    Punctuation and other non-word characters are treated as separators.

    Args:
        text: The input text string.
        case_sensitive: If True, preserve case. If False, convert to lowercase.

    Returns:
        A list of extracted words.
    """
    # Define a regex pattern to match words (alphanumeric + underscore)
    pattern = r'\b[a-zA-Z0-9_]+\b'
    if not case_sensitive:
        text = text.lower()
    words = re.findall(pattern, text)
    return words


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word in the list.

    Args:
        words: A list of words.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)
