#!/usr/bin/env python3
"""
word_freq.py - A command-line tool to count word frequencies in text files.

Reads a text file, counts occurrences of each word (case-insensitive,
punctuation stripped), and displays the top N most frequent words in a
formatted table.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def normalize_words(text: str) -> List[str]:
    """
    Extract words from text, converting to lowercase and stripping punctuation.

    Words are defined as sequences of alphanumeric characters (including
    underscores for simplicity). All other characters are treated as delimiters.

    Args:
        text: The input text string.

    Returns:
        A list of normalized (lowercase) words.
    """
    # Find all sequences of word characters (letters, digits, underscores)
    words = re.findall(r'\b[a-zA-Z0-9_]+\b', text)
    return [w.lower() for w in words]


def count_words(words: List[str]) -> Counter:
    """
    Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of tuples (word, count), sorted by count descending,
        then by word ascending for ties.
    """
    # most_common returns in order of frequency (descending),
    # and for ties, it preserves original order. To ensure deterministic
    # sorting for ties (alphabetical), we sort explicitly.
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
