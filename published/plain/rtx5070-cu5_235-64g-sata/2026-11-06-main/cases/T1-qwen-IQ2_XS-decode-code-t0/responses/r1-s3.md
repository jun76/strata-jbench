#!/usr/bin/env python3
"""
Word frequency counter CLI tool.

Reads a text file, counts word occurrences, and displays the top N words
in a table format.
"""

import argparse
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """Count word occurrences in the given text.

    Args:
        text: Input text string.

    Returns:
        Counter mapping words to their occurrence counts.
    """
    # Normalize: lowercase, strip punctuation
    words = []
    for token in text.split():
        # Strip leading/trailing punctuation from each token
        cleaned = token.strip(".,;:!?\"'()[]{}<>")
        if cleaned:
            words.append(cleaned.lower())
    return Counter(words)


def top_n_words(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return the top N most common words.

    Args:
        counter: Counter of word frequencies.
        n: Number of top words to return.

    Returns:
        List of (word, count) tuples sorted by count descending.
    """
    return counter.most_common(n)


def format_table(rows: List[Tuple[str, int]]) -> str:
    """Format rows as a simple text table.

    Args:
        rows: List of (word, count) tuples.

    Returns:
        Formatted table string.
    """
    if not rows:
        return "No words found."

    # Determine column widths
    word_width = max(len(word) for word, _ in rows)
    count_width = max(len(str(count)) for _, count in rows)

    header = f"{'Word':<{word_width}}  {'Count':>{count_width}}"
    separator = f"{'-' * word_width}  {'-' * count_width}"

    lines = [header, separator]
    for word, count in rows:
        lines.append(f"{word:<{word_width}}  {count:>{count_width}}")

    return "\n".join(lines)
