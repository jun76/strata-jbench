def read_text_file(path: Path) -> str:
    """指定されたパスのテキストファイルを読み、その内容を文字列で返す。

    Args:
        path: 読み込むファイルのパス。

    Returns:
        ファイルの全文（UTF-8）。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IsADirectoryError: パスがディレクトリの場合。
        UnicodeDecodeError: UTF-8 でデコードできない場合。
    """
    with path.open("r", encoding="utf-8") as f:
        return f.read()
