#!/usr/bin/env python3
"""
text_word_counter: A command-line tool to count word frequencies in text files.

Usage:
    python text_word_counter.py <file1> [file2 ...] [--top N] [--min-length L]

Examples:
    python text_word_counter.py input.txt
    python text_word_counter.py input.txt --top 10
    python text_word_counter.py file1.txt file2.txt --top 5 --min-length 3
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def extract_words(text: str, min_length: int = 1) -> List[str]:
    """
    Extract words from text, filtering by minimum length.

    Args:
        text: Input text string.
        min_length: Minimum word length to include (inclusive).

    Returns:
        List of lowercase words meeting the length criterion.
    """
    # Normalize to lowercase and extract word-like tokens (letters only)
    words = re.findall(r"[a-zA-Z]+", text.lower())
    return [w for w in words if len(w) >= min_length]


def count_words(files: List[Path], min_length: int = 1) -> Counter:
    """
    Read text from multiple files and count word frequencies.

    Args:
        files: List of Path objects to text files.
        min_length: Minimum word length to include.

    Returns:
        Counter object mapping words to their occurrence counts.

    Raises:
        FileNotFoundError: If a file does not exist.
        OSError: If a file cannot be read.
    """
    counter: Counter = Counter()
    for file_path in files:
        if not file_path.exists():
            raise FileNotFoundError(f"File not found: {file_path}")
        try:
            text = file_path.read_text(encoding="utf-8")
        except OSError as e:
            raise OSError(f"Cannot read file {file_path}: {e}") from e
        words = extract_words(text, min_length)
        counter.update(words)
    return counter
