"""
top_words.py - A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a formatted table.

Usage:
    python top_words.py input.txt --top 10
    python top_words.py input.txt -n 5 --case-insensitive
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple, Dict, Optional


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Extract words from text. Words are sequences of alphanumeric characters and underscores.
    
    Args:
        text: The input text to tokenize.
        case_sensitive: If False, convert all words to lowercase before counting.
        
    Returns:
        A list of extracted words.
    """
    words = re.findall(r'\w+', text)
    if not case_sensitive:
        words = [word.lower() for word in words]
    return words


def count_frequencies(words: List[str]) -> Dict[str, int]:
    """
    Count the frequency of each word.
    
    Args:
        words: A list of words.
        
    Returns:
        A dictionary mapping each word to its count.
    """
    return dict(Counter(words))


def get_top_n_words(frequency_dict: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most frequent words.
    
    Args:
        frequency_dict: A dictionary mapping words to their counts.
        n: The number of top words to return.
        
    Returns:
        A list of (word, count) tuples sorted by count descending, then by word ascending.
    """
    # Sort by count descending, then by word ascending for ties
    sorted_items = sorted(frequency_dict.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
