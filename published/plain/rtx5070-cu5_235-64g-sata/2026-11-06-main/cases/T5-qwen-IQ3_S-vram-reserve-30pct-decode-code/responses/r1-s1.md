```python
#!/usr/bin/env python3
"""
word_freq.py

A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a table format.

Usage:
    python word_freq.py --file <path_to_text_file> [--top N] [--min-len MIN_LEN]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize the given text into lowercase words.

    A word is defined as a sequence of alphanumeric characters (including underscores).

    Args:
        text: The input text to tokenize.

    Returns:
        A list of lowercase word tokens.
    """
    # Match sequences of word characters (letters, digits, underscores)
    tokens = re.findall(r'\b\w+\b', text.lower())
    return tokens


def count_words(text: str, min_len: int = 1) -> Counter:
    """
    Count the frequency of each word in the text.

    Args:
        text: The input text to analyze.
        min_len: Minimum length of words to include in the count.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    tokens = tokenize(text)
    # Filter tokens by minimum length
    filtered_tokens = [token for token in tokens if len(token) >= min_len]
    return Counter(filtered_tokens)


def get_top_words(counter: Counter, top_n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        top_n: The number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending,
        then by word ascending for ties.
    """
    # most_common returns a list of (word, count) sorted by count descending.
    # For ties, we want alphabetical order, so we sort explicitly.
    most_common = counter.most_common(top_n)
    # Re-sort to ensure alphabetical order for equal counts
    sorted_items = sorted(most_common, key=lambda x: (-x[1], x[0]))
    return sorted_items
