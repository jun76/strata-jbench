#!/usr/bin/env python3
"""
wordfreq: テキストファイルから単語の出現頻度を数え、上位 N 件を表示するツール。
"""

import argparse
import re
import sys
from collections import Counter

def tokenize(text: str) -> List[str]:
    """
    テキストを単語リストに分割する。
    英数字の連続を単語とみなし、その他を区切りとする。
    単語は小文字化される。
    
    Args:
        text: 入力テキスト
    
    Returns:
        単語のリスト
    """
    return re.findall(r'[a-zA-Z0-9]+', text.lower())


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。
    
    Args:
        words: 単語のリスト
    
    Returns:
        単語 -> 出現回数 の Counter
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から出現回数の多い順に上位 n 件を取り出す。
    回数が同じ場合は単語の辞書順で安定してソートする。
    
    Args:
        counter: 単語 -> 出現回数 の Counter
        n: 取り出す件数
    
    Returns:
        [(word, count), ...] のリスト
    """
    if n <= 0:
        return []

    items = sorted(
        counter.items(),
        key=lambda x: (-x[1], x[0])
    )
    return items[:n]
