#!/usr/bin/env python3
"""
word_freq.py - コマンドラインツールでテキストファイルから単語出現頻度を集計し、
上位N件を表形式で表示する。

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def count_words(text: str, case_sensitive: bool = False) -> Counter[str]:
    """
    テキスト中の単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するかどうか

    Returns:
        単語をキー、出現回数を値とするCounterオブジェクト
    """
    if not case_sensitive:
        text = text.lower()

    # 単語の抽出: アルファベット、数字、アポストロフィ、ハイフンを含むトークン
    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return Counter(words)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Counterから上位n件の単語とカウントを返す。

    Args:
        counter: 単語と出現回数を保持するCounter
        n: 取得する上位件数

    Returns:
        (単語, 出現回数)のタプルのリスト
    """
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]]) -> str:
    """
    単語と出現回数を表形式の文字列にフォーマットする。

    Args:
        entries: (単語, 出現回数)のリスト

    Returns:
        表形式の文字列
    """
    if not entries:
        return "No words found."

    # 列幅を計算
    word_width = max(len(word) for word, _ in entries)
    count_width = max(len(str(count)) for _, count in entries)

    header = f"{'Word':<{word_width}} | {'Count':>{count_width}}"
