"""
word_freq CLI tool: reads a text file, counts word frequencies, and prints the top N words in a table.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def normalize_word(word: str) -> str:
    """Normalize a word by lowercasing and stripping non-alphanumeric characters."""
    return re.sub(r'[^a-z0-9]+', '', word.lower())


def extract_words(text: str) -> List[str]:
    """Extract normalized words from the given text."""
    # Split by whitespace, then normalize each token
    tokens = text.split()
    return [normalize_word(w) for w in tokens if normalize_word(w)]


def count_words(text: str) -> Counter:
    """Count word frequencies in the given text."""
    words = extract_words(text)
    return Counter(words)


def format_table(counts: Counter, top_n: int) -> str:
    """Format the top N word counts as a table string."""
    top_items = counts.most_common(top_n)
    if not top_items:
        return "No words found."
    
    # Determine column widths
    word_width = max(len(word) for word, _ in top_items)
    count_width = max(len(str(count)) for _, count in top_items)
    
    # Build header
    header = f"{'Word':<{word_width}}  {'Count':>{count_width}}"
    separator = '-' * (word_width + count_width + 2)
    
    lines = [header, separator]
    for word, count in top_items:
        lines.append(f"{word:<{word_width}}  {count:>{count_width}}")
    
    return '\n'.join(lines)


def main(argv: List[str] = None) -> None:
    """Main entry point for the CLI tool."""
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and print the top N words."
    )
    parser.add_argument(
        "file",
        help="Path to the input text file."
    )
    parser.add_argument(
        "-n", "--top",
        type=int,
        default=10,
        help="Number of top words to display (default: 10)."
    )
