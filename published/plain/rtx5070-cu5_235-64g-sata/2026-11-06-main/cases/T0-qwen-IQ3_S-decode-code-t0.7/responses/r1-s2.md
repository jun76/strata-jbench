def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字小文字を区別。False の場合、すべて小文字に変換。

    Returns:
        単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    
    # 単語を抽出: 英数字とアンダースコアの連続を単語とみなす
    import re
    words = re.findall(r'[a-z0-9_]+', text.lower()) if not case_sensitive else re.findall(r'[A-Za-z0-9_]+', text)
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語頻度の Counter オブジェクト。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (word, count) タプルを返す。

    Args:
        counter: 単語頻度 Counter。
        n: 取得する件数。

    Returns:
        (word, count) のタプルのリスト。
    """
    return counter.most_common(n)
