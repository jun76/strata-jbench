#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を算出し、
上位 N 件を表形式で表示する。
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def extract_words(text: str) -> List[str]:
    """
    入力テキストから単語リストを抽出する。
    
    Args:
        text: 入力テキスト文字列
        
    Returns:
        小文字に変換された単語のリスト
    """
    # 単語はアルファベット、数字、ハイフン、アポストロフィで構成されると仮定
    # 日本語等には対応していないが、一般的な英語テキスト向け
    words = re.findall(r"[a-zA-Z0-9'-]+", text)
    return [word.lower() for word in words]


def count_words(words: List[str]) -> Counter:
    """
    単語リストから単語の出現頻度をカウントする。
    
    Args:
        words: 単語のリスト
        
    Returns:
        単語と出現回数をキー・値とする Counter オブジェクト
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の単語と出現回数を取り出す。
    出現回数が同じ場合は辞書順で安定ソートする。
    
    Args:
        counter: 単語の出現頻度 Counter
        n: 取得する上位件数
        
    Returns:
        (単語, 出現回数) のタプルのリスト
    """
    # counter.most_common(n) は頻度順に返すが、同頻度時の順序は不定
    # 安定性を保証するため、頻度降順、単語昇順でソート
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
