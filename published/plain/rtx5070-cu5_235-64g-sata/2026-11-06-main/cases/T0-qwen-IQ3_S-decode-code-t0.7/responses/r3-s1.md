#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で出力するコマンドラインツール。
"""

import argparse

from collections import Counter

def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出し、正規化してリストで返す。
    
    Args:
        text (str): 入力テキスト
        
    Returns:
        List[str]: 小文字化された単語のリスト
    """
    # 単語を抽出（アルファベット、数字、ハイフン、アポストロフィを含む）
    # 例: "don't" -> "don't", "hello-world" -> "hello-world"
    words = re.findall(r"[a-zA-Z0-9'\-]+", text.lower())
    return words


def count_words(words: List[str]) -> Counter:
    """
    単語リストから出現頻度をカウントする。
    
    Args:
        words (List[str]): 単語のリスト
        
    Returns:
        Counter: 単語をキー、出現回数を値とする Counter オブジェクト
    """
    return Counter(words)


def get_top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度が高い上位 N 個の単語を取得する。
    
    Args:
        counter (Counter): 単語頻度の Counter
        n (int): 取得する単語の数
        
    Returns:
        List[Tuple[str, int]]: (単語, 出現回数) のタプルのリスト
    """
    return counter.most_common(n)
