#!/usr/bin/env python3
"""
Word Frequency Counter

A command-line tool that reads a text file, counts word frequencies,
and displays the top N most frequent words in a tabular format.
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple


def normalize_word(word: str) -> str:
    """
    Normalize a word by lowercasing and removing punctuation.
    
    Args:
        word: The raw word string.
        
    Returns:
        The normalized word string.
    """
    # Convert to lowercase
    word = word.lower()
    # Remove leading and trailing punctuation
    word = word.strip(".,!?;:'\"()[]{}-")
    return word


def extract_words(text: str) -> List[str]:
    """
    Extract words from text, handling common punctuation.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of normalized words.
    """
    # Split by whitespace, then normalize each token
    raw_tokens = text.split()
    words = []
    for token in raw_tokens:
        normalized = normalize_word(token)
        if normalized:  # Skip empty strings after normalization
            words.append(normalized)
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    Count the frequency of each word.
    
    Args:
        words: A list of words.
        
    Returns:
        A Counter object mapping words to their frequencies.
    """
    return Counter(words)
