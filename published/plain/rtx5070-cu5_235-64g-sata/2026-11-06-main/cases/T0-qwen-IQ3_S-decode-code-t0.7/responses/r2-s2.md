def read_file(filepath: str) -> str:
    """
    テキストファイルを読み込み、その内容を文字列として返す。

    Args:
        filepath: 読み込むテキストファイルのパス。

    Returns:
        ファイル内のテキスト内容。

    Raises:
        FileNotFoundError: ファイルが存在しない場合。
        IOError: ファイルの読み込み中にエラーが発生した場合。
    """
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        raise
    except Exception as e:
        raise IOError(f"ファイルの読み込み中にエラーが発生しました: {e}")


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    単語は、アルファベット、数字、および日本語の文字（漢字、ひらがな、カタカナ）で構成される
    連続した文字列として定義される。

    Args:
        text: 入力テキスト。
        case_sensitive: True の場合、大文字と小文字を区別する。False の場合、小文字に変換する。

    Returns:
        抽出された単語のリスト。
    """
    # 正規表現で単語を抽出: 英数字、日本語のひらがな・カタカナ・漢字
    # パターン: 1つ以上の英数字または日本語文字
    pattern = r'[a-zA-Z0-9\u3040-\u309f\u30a0-\u30ff\u4e00-\u9fff]+'
    words = re.findall(pattern, text)
    
    if not case_sensitive:
        words = [word.lower() for word in words]
    
    return words
