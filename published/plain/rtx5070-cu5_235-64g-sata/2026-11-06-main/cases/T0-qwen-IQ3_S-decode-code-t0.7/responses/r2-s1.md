#!/usr/bin/env python3
"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度を計算し、
上位 N 件の単語を表形式で出力するコマンドラインツール。

Usage:
    python word_freq.py <input_file> [-n N] [--case-sensitive]

Example:
    python word_freq.py sample.txt -n 10
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple, Dict

# 単語を抽出するための正規表現は使わず、標準的な文字列操作で分割する。
# 句読点や記号を除去するため、str.translate を用いる。
import string

def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語のリストを抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。

    Returns:
        単語のリスト。句読点や空白は除去される。
    """
    # 大文字小文字を無視する場合は小文字に変換
    if not case_sensitive:
        text = text.lower()

    # 句読点や記号を除去するための翻訳テーブルを作成
    # string.punctuation を使わず、Unicode の句読点も考慮するため、
    # ここでは簡易的に string.punctuation を使用する（必要に応じて拡張可能）
    translator = str.maketrans('', '', string.punctuation)
    cleaned_text = text.translate(translator)

    # 空白で分割し、空文字列を除く
    words = cleaned_text.split()
    return [word for word in words if word]

def count_word_frequencies(words: List[str]) -> Dict[str, int]:
    """
    単語リストから頻度を集計する。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする辞書。
    """
    return dict(Counter(words))
