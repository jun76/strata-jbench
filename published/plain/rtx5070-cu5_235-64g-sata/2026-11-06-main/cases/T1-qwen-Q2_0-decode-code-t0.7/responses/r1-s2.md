テキストファイルから単語の出現頻度を集計し、上位 N 件を表形式で表示する CLI 工具。

用法:

from __future__ import annotations

import argparse

from collections import Counter

from typing import List, Dict, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト（小写化、非文字を分離）
    """
    # 非文字を空白で分離（Unicode 文字のみを単語として扱う）
    import re
    tokens = re.findall(r"[a-zA-Z]+", text)
    return [t.lower() for t in tokens]


def count_words(text: str) -> Dict[str, int]:
    """
    文本中の各単語の出現頻度を集計する。

    Args:
        text: 入力テキスト

    Returns:
        単語 -> 出現回数 の dict
    """
    tokens = tokenize(text)
    return dict(Counter(tokens))


def top_n(counter: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    上位 n 件の (word, count) を頻度降順で返す。

    Args:
        counter: word -> count の dict
        n: 上位件数

    Returns:
        [(word, count), ...] 頻度降順
    """
    return sorted(counter.items(), key=lambda x: x[1], reverse=True)[:n]
