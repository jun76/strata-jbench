def read_file(path: str) -> str:
    """Read the entire content of a text file.

    Args:
        path: Path to the text file.

    Returns:
        The content of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
    """
    with open(path, "r", encoding="utf-8") as f:
        return f.read()


def tokenize(text: str) -> List[str]:
    """Tokenize text into words.

    Args:
        text: The input text.

    Returns:
        A list of words, lowercased and stripped of punctuation.

    Notes:
        Words are defined as sequences of alphanumeric characters.
        Non-alphanumeric characters are treated as separators.
    """
    words: List[str] = []
    current_word: List[str] = []
    for ch in text:
        if ch.isalnum():
            current_word.append(ch.lower())
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))
    return words


def count_words(text: str) -> Dict[str, int]:
    """Count the frequency of each word in the text.

    Args:
        text: The input text.

    Returns:
        A dictionary mapping each word to its count.
    """
    return Counter(tokenize(text))


def top_n(counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """Return the top N most frequent words.

    Args:
        counts: A dictionary mapping words to their counts.
        n: The number of top entries to return.

    Returns:
        A list of tuples (word, count), sorted by count descending,
        then by word ascending for ties.
    """
    return sorted(counts.items(), key=lambda item: (-item[1], item[0]))[:n]
