"""
Command-line tool to count word frequencies in a text file and display the top N results in a table.

Usage:
    python word_freq.py <input_file> --top 10
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Optional


def tokenize(text: str) -> List[str]:
    """
    Tokenize a string into words.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase words, stripped of punctuation and non-alphanumeric characters.
    """
    # Normalize: lowercase, replace non-alphanumeric (except apostrophes within words) with spaces
    # This handles contractions like "don't" -> "don't"
    normalized = re.sub(r"[^a-z0-9']+|[^\x00-\x7f]", " ", text.lower())
    # Remove leading/trailing apostrophes from tokens
    tokens = re.findall(r"[a-z0-9]+(?:'[a-z0-9]+)?", normalized)
    return tokens


def count_frequencies(tokens: List[str]) -> Counter[str]:
    """
    Count the frequency of each token.

    Args:
        tokens: List of word tokens.

    Returns:
        A Counter object mapping each word to its count.
    """
    return Counter(tokens)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    Return the top N most frequent words.

    Args:
        counter: A Counter object of word frequencies.
        n: Number of top results to return.

    Returns:
        A list of tuples (word, count), sorted by count descending, then alphabetically.
    """
    # Sort by count descending, then by word ascending for stable ordering
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
