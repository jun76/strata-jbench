def normalize_word(word: str, case_sensitive: bool = False) -> str:
    """単語を正規化する。

    大文字・小文字の区別をなくす場合（デフォルト）、単語を小文字に変換する。
    非英数字は単語の境界として扱い、単語のみを抽出する。

    Args:
        word: 正規化する前の文字列。
        case_sensitive: True の場合、大文字・小文字を区別する。

    Returns:
        正規化された単語。空文字列の場合は空文字列を返す。
    """
    if not case_sensitive:
        word = word.lower()
    # 英数字以外を単語の区切りとして扱う
    cleaned = []
    for char in word:
        if char.isalnum():
            cleaned.append(char)
        else:
            if cleaned:
                break
            # 先頭に非英数字がある場合は無視
    return "".join(cleaned)
