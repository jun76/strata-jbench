"""
word_freq.py - A command-line tool to count word frequencies in a text file.

Usage:
    python word_freq.py <input_file> [-n N] [--min-length LEN] [-v]
    
    - input_file: Path to the text file to analyze.
    - -n, --top: Number of top words to display (default: 10).
    - --min-length: Minimum length of words to include (default: 1).
    - -v, --verbose: Show additional statistics.

Example:
    python word_freq.py sample.txt -n 20
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple, Dict


def normalize_text(text: str) -> str:
    """Normalize text by converting to lowercase and replacing non-alphanumeric characters with spaces.
    
    Args:
        text: The input text string.
        
    Returns:
        Normalized text string with only alphanumeric characters and spaces.
    """
    return re.sub(r'[^a-z0-9\s]', ' ', text.lower())


def tokenize(text: str, min_length: int = 1) -> List[str]:
    """Tokenize normalized text into words.
    
    Args:
        text: The input text string.
        min_length: Minimum length of words to include.
        
    Returns:
        List of words (tokens) meeting the minimum length criterion.
    """
    normalized = normalize_text(text)
    words = normalized.split()
    return [word for word in words if len(word) >= min_length]


def count_frequencies(words: List[str]) -> Dict[str, int]:
    """Count the frequency of each word in the list.
    
    Args:
        words: List of words.
        
    Returns:
        Dictionary mapping each word to its frequency count.
    """
    return dict(Counter(words))
