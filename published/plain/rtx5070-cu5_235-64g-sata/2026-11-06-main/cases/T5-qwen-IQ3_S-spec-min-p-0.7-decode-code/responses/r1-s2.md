#!/usr/bin/env python3
"""
word_freq.py — テキストファイルから単語出現頻度を上位 N 件まで表形式で出力する CLI ツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストから単語リストを抽出する。

    正規表現 \\b[a-zA-Z]+\\b を用いて英単語のみ抽出し、小文字化して返す。
    日本語など非 ASCII 文字は除外される（必要に応じて正規表現を調整）。

    Args:
        text: 入力テキスト文字列。

    Returns:
        小文字化された単語のリスト。
    """
    return re.findall(r'\b[a-zA-Z]+\b', text.lower())


def count_words(text: str) -> Counter:
    """
    テキスト中の単語出現回数を Counter で返す。

    Args:
        text: 入力テキスト文字列。

    Returns:
        単語 -> 出現回数の Counter オブジェクト。
    """
    return Counter(tokenize(text))


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Counter から上位 n 件の (word, count) を頻度順に返す。
    頻度が同じ場合はアルファベット順にソートされる。

    Args:
        counter: 単語と出現回数の Counter。
        n: 取得する上位件数。

    Returns:
        (単語, 出現回数) のタプルのリスト。
    """
    # most_common(n) は頻度降順で返すが、同頻度時の順序は保証しないため、
    # 明示的に (-count, word) でソートする。
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items
