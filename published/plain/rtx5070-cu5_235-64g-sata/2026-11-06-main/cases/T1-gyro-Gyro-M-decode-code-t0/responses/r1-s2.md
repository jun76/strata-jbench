"""
word_freq.py - コマンドラインツール: テキストファイルから単語出現頻度を上位N件表示
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    英数字の連続を単語とし、それ以外を区切りとする。
    大文字・小文字は区別しない（すべて小文字に変換）。

    Args:
        text: 入力テキスト

    Returns:
        単語のリスト（すべて小文字）
    """
    # 英数字の連続を抽出し、小文字に変換
    words = re.findall(r'[a-zA-Z0-9]+', text)
    return [w.lower() for w in words]


def count_words(text: str) -> Counter:
    """
    テキスト中の単語出現頻度をカウントする。

    Args:
        text: 入力テキスト

    Returns:
        単語をキー、出現回数を値とする Counter
    """
    words = tokenize(text)
    return Counter(words)


def top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 N 件を返す。
    同頻度の場合は辞書順（アルファベット順）でソートする。

    Args:
        counter: 単語出現頻度の Counter
        n: 上位件数

    Returns:
        (単語, 出現回数) のタプルのリスト（上位順）
    """
    # 頻度降順、同頻度なら単語のアルファベット順
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
