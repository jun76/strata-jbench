"""
word_counter.py

A command-line tool that reads a text file, counts word occurrences,
and displays the top N most frequent words in a tabular format.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    Tokenize the given text into lowercase words.

    Words are defined as sequences of alphanumeric characters and underscores.
    All tokens are converted to lowercase for case-insensitive counting.

    Args:
        text: The input text string.

    Returns:
        A list of lowercase word tokens.
    """
    return re.findall(r'\b\w+\b', text.lower())


def count_words(text: str) -> Counter:
    """
    Count the frequency of each word in the given text.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top entries to return.

    Returns:
        A list of tuples (word, count), sorted by count descending,
        then by word ascending for ties.
    """
    # Sort by count descending, then by word ascending for stable ordering
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
