#!/usr/bin/env python3
"""wordfreq: Count word frequencies in a text file and display top N results.

Usage:
    python wordfreq.py <file> [-n N] [-i/--ignore-case]
"""

import argparse
import re
from collections import Counter
from typing import List, Tuple


def parse_args(argv: List[str] | None = None) -> argparse.Namespace:
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(
        description="Count word frequencies in a text file and display top N results."
    )
    parser.add_argument(
        "file",
        help="Path to the input text file",
    )
    parser.add_argument(
        "-n",
        "--top",
        type=int,
        default=10,
        help="Number of top words to display (default: 10)",
    )
    parser.add_argument(
        "-i",
        "--ignore-case",
        action="store_true",
        help="Ignore case when counting words",
    )
    return parser.parse_args(argv)


def extract_words(text: str, ignore_case: bool = False) -> List[str]:
    """Extract words from text using a simple regex pattern.

    Args:
        text: Input text string.
        ignore_case: If True, convert all words to lowercase before extracting.

    Returns:
        A list of extracted words.
    """
    if ignore_case:
        text = text.lower()
    # Match sequences of alphanumeric characters and underscores as words
    return re.findall(r"\b\w+\b", text)


def count_frequencies(words: List[str]) -> Counter[str]:
    """Count the frequency of each word.

    Args:
        words: A list of words.

    Returns:
        A Counter object mapping each word to its frequency.
    """
    return Counter(words)
