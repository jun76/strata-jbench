import argparse

def normalize_word(word: str) -> str:
    """
    単語を正規化する。
    小文字に変換し、アルファベットと数字のみを抽出する。
    """
    return "".join(c for c in word.lower() if c.isalnum())


def count_words(text: str) -> Counter:
    """
    テキストから単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    words = text.split()
    normalized_words = [normalize_word(w) for w in words]
    # 空文字列を除外
    filtered_words = [w for w in normalized_words if w]
    return Counter(filtered_words)


def get_top_n_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の高い順に上位 n 件の単語と回数を取得する。

    Args:
        counter: 単語と出現回数の Counter。
        n: 取得する件数。

    Returns:
        (単語, 出現回数) のリスト。頻度が同じ場合は辞書順でソートされる。
    """
    # most_common は頻度順だが、同点の順序は不定。
    # 安定した出力のために、(-count, word) でソートする
    sorted_items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return sorted_items[:n]
