#!/usr/bin/env python3
"""
wordfreq: Command-line tool to count word frequencies in a text file and display top N results.
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter:
    """
    Count word frequencies in the given text.

    Words are case-insensitive and consist of alphanumeric characters (including underscores).

    Args:
        text: The input text to analyze.

    Returns:
        A Counter object mapping each word (lowercase) to its frequency.
    """
    # Match words composed of alphanumeric characters and underscores
    words = re.findall(r'\b\w+\b', text.lower())
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples, sorted by count descending, then word ascending.
    """
    # Counter.most_common already sorts by count descending; for ties, we sort by word ascending
    common = counter.most_common()
    # Sort by count descending, then by word ascending for stable ordering
    common.sort(key=lambda x: (-x[1], x[0]))
    return common[:n]
