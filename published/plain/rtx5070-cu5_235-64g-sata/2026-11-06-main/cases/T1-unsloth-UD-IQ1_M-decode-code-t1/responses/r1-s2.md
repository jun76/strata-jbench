#!/usr/bin/env python3
"""
word_freq.py

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位 N 件を表形式で出力するコマンドラインツール。

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def normalize_text(text: str) -> str:
    """
    テキストを小文字化し、非アルファベット文字を空白に置き換える。
    
    Args:
        text: 入力テキスト文字列。
        
    Returns:
        正規化されたテキスト文字列。
    """
    # 英数字以外の文字を空白に置き換え、連続する空白を単一にまとめ、先頭末尾をトリム
    normalized = re.sub(r'[^a-z0-9\s]', ' ', text.lower())
    normalized = re.sub(r'\s+', ' ', normalized)
    return normalized.strip()


def count_word_frequencies(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト文字列。
        
    Returns:
        単語をキー、出現回数を値とする collections.Counter オブジェクト。
    """
    normalized_text = normalize_text(text)
    if not normalized_text:
        return Counter()
    words = normalized_text.split()
    return Counter(words)
