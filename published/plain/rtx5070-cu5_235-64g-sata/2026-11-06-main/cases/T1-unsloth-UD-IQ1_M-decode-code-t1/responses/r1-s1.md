#!/usr/bin/env python3
"""
コマンドラインツール: テキストファイルを読み込み、単語の出現頻度を上位N件を
表形式で出力します。
"""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def read_file(path: str) -> str:
    """
    指定されたパスからUTF-8でテキストを読み込む。

    Args:
        path: テキストファイルのパス

    Returns:
        str: 読み込まれたテキスト内容

    Raises:
        FileNotFoundError: ファイルが存在しない場合
        IOError: その他読み込みエラー
    """
    try:
        with open(path, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        print(f"Error: File not found: {path}", file=sys.stderr)
        raise
    except IOError as e:
        print(f"Error reading file {path}: {e}", file=sys.stderr)
        raise


def count_words(text: str) -> Counter[str]:
    """
    テキストから単語の出現頻度をカウントする。
    大文字小文字を区別せず、単語はスペースや句読点などで区切られるとみなす。
    """
    # 簡単なトークナイザー: 改行、スペース、タブ、句読点(.,;:!?'()[]{}<>/"等)で分割
    # より正確な処理ならnltkやspacyを使うべきだが、標準ライブラリのみで実装
    import re
    # 単語を抽出する正規表現: 文字、数字、ハイフン、アポストロフィで構成されるシーケンス
    words = re.findall(r"\b[a-zA-Z0-9'-]+\b", text)
    return Counter(word.lower() for word in words)
