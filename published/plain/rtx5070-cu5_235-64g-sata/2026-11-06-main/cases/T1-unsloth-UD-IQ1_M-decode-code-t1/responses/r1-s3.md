Text Word Frequency Analyzer CLI Tool

This module provides a command-line tool that reads a text file, counts word
frequencies, and displays the top N most frequent words in a table format.

Usage examples:
    python word_freq.py input.txt -n 10
    python word_freq.py input.txt -n 20 --case-sensitive
"""

import argparse

from collections import Counter
from typing import List, Tuple, Dict


def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    Extract words from text.
    
    Words are defined as sequences of alphanumeric characters, underscores,
    or apostrophes (for contractions like don't). Punctuation and whitespace
    are treated as delimiters.
    
    Args:
        text: Input text string.
        case_sensitive: If True, preserve case. If False, convert to lowercase.
        
    Returns:
        List of extracted words.
    """
    if not case_sensitive:
        text = text.lower()
    
    words: List[str] = []
    current_word: List[str] = []
    
    for char in text:
        if char.isalnum() or char == '_' or char == "'":
            current_word.append(char)
        else:
            if current_word:
                word = ''.join(current_word)
                # Strip leading/trailing apostrophes from the word
                word = word.strip("'")
                if word:
                    words.append(word)
                current_word = []
    
    # Don't forget the last word
    if current_word:
        word = ''.join(current_word)
        word = word.strip("'")
        if word:
            words.append(word)
    
    return words


def count_word_frequencies(words: List[str]) -> Dict[str, int]:
    """
    Count the frequency of each word.
    
    Args:
        words: List of words.
        
    Returns:
        Dictionary mapping words to their counts.
    """
    return dict(Counter(words))
