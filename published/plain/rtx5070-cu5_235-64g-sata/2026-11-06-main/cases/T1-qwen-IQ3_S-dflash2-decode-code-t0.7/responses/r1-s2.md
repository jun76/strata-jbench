#!/usr/bin/env python3
"""
word_freq - コマンドラインツールでテキストファイルの単語頻度を上位N件表示する

使い方:
    python word_freq.py <file_path> [--top N] [--case-sensitive] [--min-length L]
"""

import argparse
import re
import sys
from collections import Counter
from typing import List, Tuple


def tokenize(text: str, case_sensitive: bool = True, min_length: int = 1) -> List[str]:
    """
    テキストから単語トークンを抽出する。

    Args:
        text: 入力テキスト
        case_sensitive: 大文字小文字を区別するか
        min_length: 単語の最小文字数

    Returns:
        単語トークンのリスト
    """
    # 英数字とアンダースコアを含む単語を抽出
    tokens = re.findall(r'[a-zA-Z0-9_]+', text)
    if not case_sensitive:
        tokens = [token.lower() for token in tokens]
    if min_length > 1:
        tokens = [token for token in tokens if len(token) >= min_length]
    return tokens


def count_frequencies(tokens: List[str]) -> Counter:
    """
    トークンリストから頻度をカウントする。

    Args:
        tokens: 単語トークンのリスト

    Returns:
        単語 -> 出現回数のCounter
    """
    return Counter(tokens)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度上位N件の単語と回数を取得する。

    Args:
        counter: 頻度カウンター
        n: 上位件数

    Returns:
        [(word, count), ...] のリスト（頻度高い順）
    """
    return counter.most_common(n)


def format_table(items: List[Tuple[str, int]]) -> str:
    """
    単語と回数を表形式でフォーマットする。

    Args:
        items: [(word, count), ...] のリスト

    Returns:
        表形式の文字列
    """
    if not items:
        return ""

    # 列の幅を計算
    word_width = max(len(word) for word, _ in items
