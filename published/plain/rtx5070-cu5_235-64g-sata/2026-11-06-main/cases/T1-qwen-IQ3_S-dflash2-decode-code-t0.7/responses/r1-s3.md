#!/usr/bin/env python3
"""
wordfreq.py

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位 N 件を表形式で出力するコマンドラインツール。

使い方:
    python wordfreq.py input.txt -n 10
    python wordfreq.py input.txt --top 20 --ignore-case
"""

import argparse

from typing import List, Tuple


def count_words(
    text: str,
    ignore_case: bool = True,
    min_length: int = 1,
) -> Counter[str]:
    """
    テキスト中の単語の出現頻度をカウントする。

    Args:
        text: 入力テキスト文字列。
        ignore_case: True の場合、単語を小文字に変換してカウントする。
        min_length: この長さ未満の単語は無視する。

    Returns:
        単語 -> 出現回数 の Counter オブジェクト。
    """
    # 単語の抽出: アルファベット、数字、ハイフン、アポストロフィを含む
    # 簡易的なトークナイザーとして str.split() + 正規表現なしで対応する場合、
    # 非英数字は区切りとみなす。ここでは str.translate を使う簡易実装。
    import re

    # 単語を抽出する正規表現: 英数字、ハイフン、アポストロフィを含む
    words = re.findall(r"[a-zA-Z0-9']+", text)

    if ignore_case:
        words = [w.lower() for w in words]

    # 最小長未満の単語を除外
    words = [w for w in words if len(w) >= min_length]

    return Counter(words)
