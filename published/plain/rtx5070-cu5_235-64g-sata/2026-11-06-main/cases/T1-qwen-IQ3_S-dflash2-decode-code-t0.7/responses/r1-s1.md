word_freq.py - A command-line tool to count word frequencies in a text file.

Usage:
    python word_freq.py <input_file> [-n N] [--sort-by {count,alpha}]

Args:
    input_file: Path to the text file to analyze.
    -n, --top: Number of top words to display (default: 10).
    --sort-by: Sort by 'count' (default) or 'alpha' (alphabetical).

def tokenize(text: str) -> List[str]:
    """
    Tokenize text into lowercase words, ignoring punctuation and non-alphanumeric characters.
    
    Args:
        text: The input text string.
        
    Returns:
        A list of lowercase word tokens.
    """
    # Match sequences of alphanumeric characters (words)
    return re.findall(r'\b[a-zA-Z]+\b', text.lower())


def count_words(words: List[str]) -> Dict[str, int]:
    """
    Count occurrences of each word.
    
    Args:
        words: A list of word tokens.
        
    Returns:
        A dictionary mapping words to their counts.
    """
    return dict(Counter(words))
