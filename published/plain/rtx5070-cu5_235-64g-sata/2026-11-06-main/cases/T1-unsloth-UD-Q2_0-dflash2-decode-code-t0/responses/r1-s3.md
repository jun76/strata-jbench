#!/usr/bin/env python3
"""
word_counter.py

テキストファイルを読み込み、単語の出現頻度をカウントし、
上位 N 件を表形式で表示するコマンドラインツール。
"""

import argparse
import sys
import re
from collections import Counter
from typing import List, Tuple, Dict


def tokenize(text: str) -> List[str]:
    """
    テキストを単語に分割する。
    大文字小文字を区別せず、アルファベットと数字の連続を単語とみなす。
    
    Args:
        text: 入力テキスト
        
    Returns:
        小文字化された単語のリスト
    """
    # 正規表現で単語（アルファベットと数字の連続）を抽出
    words = re.findall(r'[a-zA-Z0-9]+', text.lower())
    return words


def count_words(text: str) -> Dict[str, int]:
    """
    テキスト内の単語出現頻度をカウントする。
    
    Args:
        text: 入力テキスト
        
    Returns:
        単語をキー、出現回数を値とする辞書
    """
    words = tokenize(text)
    return dict(Counter(words))


def get_top_n(counts: Dict[str, int], n: int) -> List[Tuple[str, int]]:
    """
    出現頻度の上位 N 件を取得する。
    頻度が同じ場合は辞書順（アルファベット順）でソートする。
    
    Args:
        counts: 単語と出現回数の辞書
        n: 取得する件数
        
    Returns:
        (単語, 出現回数) のタプルのリスト（降順）
    """
    # 頻度降順、同頻度なら単語の昇順でソート
    sorted_items = sorted(
        counts.items(),
        key=lambda item: (-item[1], item[0])
    )
    return sorted_items[:n]
