def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。
    
    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。False の場合、すべて小文字に変換する。
        
    Returns:
        抽出された単語のリスト。
    """
    if not case_sensitive:
        text = text.lower()
    
    # 単語を抽出する正規表現: 英数字とハイフン、アポストロフィを含む連続文字列

    words = re.findall(r"[a-zA-Z0-9'\-]+", text)
    return words

        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    return Counter(words)
