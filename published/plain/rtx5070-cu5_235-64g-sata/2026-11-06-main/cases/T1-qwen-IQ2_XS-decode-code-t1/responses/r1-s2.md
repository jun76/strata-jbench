#!/usr/bin/env python3
"""Command-line tool to count word frequencies from a text file and display top N entries."""

from __future__ import annotations

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def count_words(text: str) -> Counter[str]:
    """Count word frequencies in text.

    Words are extracted by splitting on whitespace and stripping punctuation.
    """
    words: List[str] = []
    for token in text.split():
        # Strip punctuation from start and end
        stripped = token.strip(".,!?;:'\"()[]{}")
        if stripped:
            words.append(stripped.lower())
    return Counter(words)


def top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """Return top N most common words."""
    return counter.most_common(n)


def format_table(entries: List[Tuple[str, int]]) -> str:
    """Format entries as a text table."""
    if not entries:
        return ""
    
    # Find max word length for alignment
    max_word_len = max(len(word) for word, _ in entries)
    max_count_len = max(len(str(count)) for _, count in entries)
    
    header = f"{'Word':<{max_word_len + 2}} {'Count':>{max_count_len + 3}}"
    separator = '-' * (max_word_len + max_count_len + 6)
    
    lines = [header, separator]
    for word, count in entries:
        lines.append(f"{word:<{max_word_len + 2}} {count:>{max_count_len + 3}}")
    
    return "\n".join(lines)


def main(argv: List[str] | None = None) -> int:
    """Parse arguments, read file, and display top N word frequencies."""
    parser = argparse.ArgumentParser(
        description="Count word frequencies from a text file and display top N entries as a table."
    )
    parser.add_argument("file", help="Path to the text file to analyze")
    parser.add_argument(
        "-n", "--top",
        type=int,
        default=10,
        help="Number of top entries to display (default: 10)"
    )
    
    args = parser.parse_args(argv
