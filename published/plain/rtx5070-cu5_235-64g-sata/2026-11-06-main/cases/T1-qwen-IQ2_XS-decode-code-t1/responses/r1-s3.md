Command-line tool that reads a text file, counts word occurrences,
and displays the top N words in a table format.

Usage:
    python word_counter.py <input_file> [--top N] [--encoding UTF-8]

Example:
    python word_counter.py sample.txt --top 10
"""

def read_file(path: Path, encoding: str = "utf-8") -> str:
    """Read the contents of a text file.

    Args:
        path: Path to the text file.
        encoding: Encoding to use when reading the file. Defaults to 'utf-8'.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If the file cannot be read.
    """
    if not path.exists():
        raise FileNotFoundError(f"File not found: {path}")
    try:
        with open(path, "r", encoding=encoding) as f:
            return f.read()
    except OSError as e:
        raise IOError(f"Error reading file '{path}': {e}")


def count_words(text: str) -> Dict[str, int]:
    """Count the occurrences of each word in the text.

    Words are defined as sequences of alphanumeric characters.
    Punctuation and whitespace are ignored. Words are lowercased.

    Args:
        text: The input text string.

    Returns:
        A dictionary mapping each word to its occurrence count.
    """
    import re
    words = re.findall(r'\b[a-z0-9]+\b', text.lower())
    return dict(Counter(words))
