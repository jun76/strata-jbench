import argparse

def extract_words(text: str, case_sensitive: bool = False) -> List[str]:
    """
    テキストから単語を抽出する。

    Args:
        text: 入力テキスト。
        case_sensitive: 大文字小文字を区別するかどうか。False の場合は小文字に変換。

    Returns:
        単語のリスト。
    """
    # 単語を抽出: 英数字とアンダースコアを含む連続文字列
    words = re.findall(r'[a-zA-Z0-9_]+', text)
    if not case_sensitive:
        words = [w.lower() for w in words]
    return words


def count_word_frequencies(words: List[str]) -> Counter:
    """
    単語の出現頻度をカウントする。

    Args:
        words: 単語のリスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    return Counter(words)


def get_top_n(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """
    頻度上位 N 件の単語と回数を取得する。

    Args:
        counter: 単語頻度の Counter。
        n: 取得する件数。

    Returns:
        (単語, 回数) のタプルのリスト。頻度降順、同頻度の場合は辞書順。
    """
    # most_common は頻度降順。同頻度の場合は挿入順なので、辞書順にソートし直す
    items = list(counter.items())
    items.sort(key=lambda x: (-x[1], x[0]))
    return items[:n]
