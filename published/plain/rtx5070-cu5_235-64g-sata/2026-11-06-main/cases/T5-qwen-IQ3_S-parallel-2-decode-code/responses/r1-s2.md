    """
    テキストから単語を抽出する。
    正規表現を用いて、アルファベット、数字、ハイフンを含む単語を抽出し、
    すべて小文字に変換する。

    Args:
        text (str): 入力テキスト。

    Returns:
        List[str]: 抽出された単語のリスト。
    """
    # 単語パターン: 英数字、ハイフン、アポストロフィを含む

    matches = re.findall(pattern, text)
    return [word.lower() for word in matches]
