#!/usr/bin/env python3
"""A command-line tool that reads a text file, counts word frequencies, and displays the top N words in a table format."""

import argparse
import sys
from collections import Counter
from typing import List, Tuple


def read_file(filepath: str) -> str:
    """Read the contents of a text file.

    Args:
        filepath: Path to the text file to read.

    Returns:
        The contents of the file as a string.

    Raises:
        FileNotFoundError: If the file does not exist.
        IOError: If there is an error reading the file.
    """
    with open(filepath, 'r', encoding='utf-8') as f:
        return f.read()


def count_words(text: str) -> Counter:
    """Count the frequency of each word in the given text.

    Words are converted to lowercase and non-alphanumeric characters are stripped.

    Args:
        text: The input text string.

    Returns:
        A Counter object mapping words to their frequencies.
    """
    words = []
    for word in text.lower().split():
        # Strip non-alphanumeric characters from both ends
        cleaned = word.strip('.,;:!?\'"()[]{}<>/\\|`~@#$%^&*-=+_')
        if cleaned:
            words.append(cleaned)
    return Counter(words)


def get_top_words(counter: Counter, n: int) -> List[Tuple[str, int]]:
    """Get the top N most common words from a Counter.

    Args:
        counter: A Counter object with word frequencies.
        n: The number of top words to return.

    Returns:
        A list of (word, count) tuples for the top N words.
    """
    return counter.most_common(n)
