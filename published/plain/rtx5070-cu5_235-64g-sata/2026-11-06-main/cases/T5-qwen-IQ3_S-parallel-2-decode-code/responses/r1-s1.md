    Args:
        text: 入力テキスト文字列

    Returns:

    """
    # \b\w+\b は英数字とアンダースコアを単語とみなす。
    # より正確な単語抽出のため、アポストロフィを含む場合を考慮する場合はパターンを調整。

    words = re.findall(r"[a-zA-Z0-9']+", text)
    return [word.lower() for word in words]

    """

    """

    return Counter(words)
