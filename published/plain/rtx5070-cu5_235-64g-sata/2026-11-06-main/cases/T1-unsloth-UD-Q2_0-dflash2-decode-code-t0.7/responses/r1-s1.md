## 1. メインツール (`word_counter.py`)

```python
#!/usr/bin/env python3
"""
word_counter.py - テキストファイルを読み込み、単語の出現頻度を上位 N 件まで表形式で表示する CLI ツール。
"""

import argparse
import re
import sys
from collections import Counter
from pathlib import Path
from typing import List, Tuple


def tokenize(text: str) -> List[str]:
    """
    テキストを単語トークンに分割する。
    小文字化し、単語境界（英数字・アンダースコア）で分割する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化された単語トークンのリスト。
    """
    # 正規表現で単語（英数字とアンダースコア）を抽出
    tokens = re.findall(r'\b[\w]+\b', text.lower())
    return tokens


def count_words(text: str) -> Counter[str]:
    """
    テキストから単語頻度を計算する。

    Args:
        text: 入力テキスト。

    Returns:
        単語をキー、出現回数を値とする Counter オブジェクト。
    """
    tokens = tokenize(text)
    return Counter(tokens)


def get_top_n(counter: Counter[str], n: int) -> List[Tuple[str, int]]:
    """
    頻度上位 N 件の単語を (単語, 頻度) のタプルリストとして返す。
    頻度が同じ場合は辞書順でソートされる。

    Args:
        counter: 単語頻度を格納した Counter。
        n: 取得する上位件数。

    Returns:
        上位 N 件の (単語, 頻度) タプルのリスト。
    """
    # most_common は頻度降順だが、同頻度の順序は保証しないため
    # 頻度降順 → 辞書順で安定ソートする
    items = sorted(counter.items(), key=lambda x: (-x[1], x[0]))
    return items[:n]
