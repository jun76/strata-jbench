"""
テキストファイルを読み込み、単語の出現頻度を集計して上位 N 件を表形式で表示するツール。
"""

import argparse

from pathlib import Path
from collections import Counter
from typing import List, Tuple, Dict


def extract_words(text: str) -> List[str]:
    """
    テキストから単語を抽出する。
    大文字・小文字を区別せず、句読点や記号を除き、単語のみを抽出する。
    
    Args:
        text (str): 入力テキスト。
        
    Returns:
        List[str]: 小文字化された単語のリスト。
    """
    # 正規表現で単語（アルファベット、数字、ハイフンなど）を抽出
    # \b で単語の境界を指定
    words = re.findall(r'\b[a-zA-Z0-9_-]+\b', text.lower())
    return words


def count_words(words: List[str]) -> Dict[str, int]:
    """
    単語リストから出現頻度をカウントする。
    
    Args:
        words (List[str]): 単語のリスト。
        
    Returns:
        Dict[str, int]: 単語をキー、頻度を値とする辞書。
    """
    return dict(Counter(words))
