#!/usr/bin/env python3
"""
wordfreq.py — テキストファイルから単語出現頻度を集計し、上位N件を表形式で表示するCLIツール。

使用方法:
    python wordfreq.py input.txt -n 10

def normalize_text(text: str) -> str:
    """
    テキストを単語集計用に正規化する。

    Args:
        text: 入力テキスト。

    Returns:
        小文字化し、非英数字（Unicode基準）で分割された単語リストを返すための前処理済み文字列。
    """
    # Unicode正規化 (NFC) を適用して、アクセント付き文字などを統一
    text = unicodedata.normalize("NFC", text)
    # 小文字に変換
    text = text.lower()
    return text


def extract_words(text: str) -> list[str]:
    """
    テキストから単語（英数字の連続）を抽出する。

    Args:
        text: 正規化済みのテキスト。

    Returns:
        単語のリスト。
    """
    words: list[str] = []
    current: list[str] = []
    for char in text:
        if char.isalnum():
            current.append(char)
        else:
            if current:
                words.append("".join(current))
                current = []
    if current:
        words.append("".join(current))
    return words


def count_words(text: str) -> Counter[str]:
    """
    テキスト内の単語出現頻度をカウントする。

    Args:
        text: 生テキスト。

    Returns:
        単語をキー、出現回数を値とする Counter。
    """
    normalized = normalize_text(text)
    words = extract_words(normalized)
    return Counter(words)
